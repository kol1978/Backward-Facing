48Foam::Istream& Foam::operator>>(Istream& is, wordRe& w)
49{
50 token t(is);
51
52 if (!t.good())
53 {
54 is.setBad();
55 return is;
56 }
57
58 if (t.isWord())
59 {
60 w = t.wordToken();
61 }
62 else if (t.isString())
63 {
64 // Auto-tests for regular expression
65 w = t.stringToken();
66
67 // flag empty strings as an error
68 if (w.empty())
69 {
70 is.setBad();
71 FatalIOErrorInFunction(is)
72 << "empty word/expression "
73 << exit(FatalIOError);
74 return is;
75 }
76 }
77 else
78 {
79 is.setBad();
80 FatalIOErrorInFunction(is)
81 << "wrong token type - expected word or string, found "
82 << t.info()
83 << exit(FatalIOError);
84
85 return is;
86 }
87
88 // Check state of IOstream
89 is.check("Istream& operator>>(Istream&, wordRe&)");
90
91 return is;
92}
