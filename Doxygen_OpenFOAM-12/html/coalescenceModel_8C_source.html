42Foam::autoPtr<Foam::diameterModels::coalescenceModel>
43Foam::diameterModels::coalescenceModel::New
44(
45 const word& type,
46 const populationBalanceModel& popBal,
47 const dictionary& dict
48)
49{
50 Info<< "Selecting coalescence model for "
51 << popBal.name() << ": " << type << endl;
52
53 dictionaryConstructorTable::iterator cstrIter =
54 dictionaryConstructorTablePtr_->find(type);
55
56 if (cstrIter == dictionaryConstructorTablePtr_->end())
57 {
58 FatalErrorInFunction
59 << "Unknown coalescence model type "
60 << type << nl << nl
61 << "Valid coalescence model types : " << endl
62 << dictionaryConstructorTablePtr_->sortedToc()
63 << exit(FatalError);
64 }
65
66 return autoPtr<coalescenceModel>(cstrIter()(popBal, dict));
67}
