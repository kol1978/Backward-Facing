173 const phaseModel& phase

178 FatalErrorInFunction

371template<class Type, template<class> class PatchField, class GeoMesh>
372PtrList<GeometricField<Type, PatchField, GeoMesh>> operator&
373(
374 const PtrList<PtrList<GeometricField<scalar, PatchField, GeoMesh>>>& As,
375 const PtrList<GeometricField<Type, PatchField, GeoMesh>>& fs
376)
377{
378 PtrList<GeometricField<Type, PatchField, GeoMesh >> Afs(fs.size());
379
380 forAll(Afs, i)
381 {
382 Afs.set(i, As[i][i]*fs[i]);
383
384 forAll(Afs, j)
385 {
386 if (j != i)
387 {
388 Afs[i] += As[i][j]*fs[j];
389 }
390 }
391 }
392
393 return Afs;
394}
