59Foam::CompactListList<T>::CompactListList
60(
61 const labelUList& rowSizes,
62 const T& t
63)
64:
65 UCompactListList<T>(),
66 offsets_(rowSizes.size() + 1),
67 m_()
68{

85template<class T>
86void Foam::CompactListList<T>::setSize(const label mRows)
87{
88 if (mRows == 0)
89 {
90 clear();
91 }
92 if (mRows < this->size())
93 {
94 offsets_.setSize(mRows + 1);
95 m_.setSize(offsets_[mRows]);
96
97 UCompactListList<T>::shallowCopy(UCompactListList<T>(offsets_, m_));
98 }
99 else if (mRows > this->size())
100 {
101 FatalErrorInFunction
102 << "Cannot be used to extend the list from " << offsets_.size()
103 << " to " << mRows << nl
104 << " Please use one of the other setSize member functions"
105 << abort(FatalError);
106 }
107}
