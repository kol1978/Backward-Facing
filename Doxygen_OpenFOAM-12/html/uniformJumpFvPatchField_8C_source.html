30template<class Type>
31Foam::uniformJumpFvPatchField<Type>::uniformJumpFvPatchField
32(
33 const fvPatch& p,
34 const DimensionedField<Type, volMesh>& iF,
35 const dictionary& dict
36)
37:
38 fixedJumpFvPatchField<Type>(p, iF, dict, false),
39 jumpTable_()
40{
41 if (this->cyclicPatch().owner())
42 {
43 jumpTable_ =
44 Function1<Type>::New
45 (
46 "jumpTable",
47 this->db().time().userUnits(),
48 iF.dimensions(),
49 dict
50 );
51
52 this->jumpRef() =
53 Field<Type>(p.size(), jumpTable_->value(this->db().time().value()));
54 }
55
56 this->evaluateNoUpdateCoeffs();
57}
