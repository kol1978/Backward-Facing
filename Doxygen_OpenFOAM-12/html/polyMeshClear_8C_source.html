53void Foam::polyMesh::clearGeom()
54{
55 if (debug)
56 {
57 InfoInFunction << "Clearing geometric data" << endl;
58 }
59
60 // Clear all geometric mesh objects
61 meshObjects::clear<pointMesh, DeletableMeshObject>(*this);
62 meshObjects::clear<polyMesh, DeletableMeshObject>(*this);
63
64 primitiveMesh::clearGeom();
65
66 boundary_.clearGeom();
67
68 // Reset valid directions (could change with rotation)
69 geometricD_ = Zero;
70 solutionD_ = Zero;
71
72 // Remove the cell tree
73 cellTreePtr_.clear();
74}

77void Foam::polyMesh::clearAddressing(const bool isMeshUpdate)
78{
79 if (debug)
80 {
81 InfoInFunction
82 << "Clearing topology isMeshUpdate:" << isMeshUpdate << endl;
83 }
84
85 if (isMeshUpdate)
86 {
87 // Part of a mesh update. Keep meshObjects that have an topoChange
88 // callback
89 meshObjects::clearUpto
90 <
91 pointMesh,
92 DeletableMeshObject,
93 TopoChangeableMeshObject
94 >
95 (
96 *this
97 );
98 meshObjects::clearUpto
99 <
100 polyMesh,
101 DeletableMeshObject,
102 TopoChangeableMeshObject
103 >
104 (
105 *this
106 );
107 }
108 else
109 {
110 meshObjects::clear<pointMesh, DeletableMeshObject>(*this);
111 meshObjects::clear<polyMesh, DeletableMeshObject>(*this);
112 }
113
114 primitiveMesh::clearAddressing();
115
116 // parallelData depends on the processorPatch ordering so force
117 // recalculation
118 globalMeshDataPtr_.clear();
119
120 // Reset valid directions
121 geometricD_ = Zero;
122 solutionD_ = Zero;
123
124 // Update zones
125 pointZones_.clearAddressing();
126 faceZones_.clearAddressing();
127 cellZones_.clearAddressing();
128
129 // Remove the stored tet base points
130 tetBasePtIsPtr_.clear();
131
132 // Remove the cell tree
133 cellTreePtr_.clear();
134}
