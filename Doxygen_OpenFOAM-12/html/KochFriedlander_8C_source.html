76Foam::tmp<Foam::volScalarField::Internal>
77Foam::diameterModels::shapeModels::sinteringModels::KochFriedlander::tau() const
78{
79 tmp<volScalarField::Internal> tTau
80 (
81 volScalarField::Internal::New
82 (
83 "tau",
84 fractal_.group().mesh(),
85 dimensionedScalar(dimTime, Zero)
86 )
87 );
88
89 volScalarField::Internal& tau = tTau.ref();
90
91 const sizeGroup& fi = fractal_.group();
92 const volScalarField& kappai = fractal_.fld();
93 const volScalarField& T = fractal_.group().phase().thermo().T();
94
95 forAll(tau, celli)
96 {
97 const scalar dp = 6/max(6/fi.dSph().value(), kappai[celli]);
98
99 tau[celli] =
100 Cs_*pow(dp, n_)*pow(T[celli], m_)*exp(Ta_/T[celli]*(1 - dpMin_/dp));
101 }
102
103 return tTau;
104}

107Foam::tmp<Foam::fvScalarMatrix>
108Foam::diameterModels::shapeModels::sinteringModels::KochFriedlander::R() const
109{
110 const sizeGroup& fi = fractal_.group();
111 const volScalarField& kappai = fractal_.fld();
112 const volScalarField& alpha = fi.phase();
113
114 volScalarField::Internal R
115 (
116 IOobject
117 (
118 typedName("R"),
119 fi.time().name(),
120 fi.mesh()
121 ),
122 fi.mesh(),
123 dimensionedScalar(inv(dimTime), Zero)
124 );
125
126 volScalarField::Internal tau(this->tau());
127
128 forAll(R, celli)
129 {
130 R[celli] = fi[celli]*alpha[celli]/tau[celli];
131 }
132
133 return fvm::Sp(R, kappai) - 6/fi.dSph()*R;
134}
