55 const ScalarFieldType& alphaLiquid
56) const
57{
58 return
59 pos0(alphaLiquid1_ - alphaLiquid)
60 *(
61 neg(alphaLiquid0_ - alphaLiquid)
62 *(
63 0.5
64 *(
65 1 - cos
66 (
67 constant::mathematical::pi
68 *(alphaLiquid - alphaLiquid0_)

71 )
72 )
73 )
