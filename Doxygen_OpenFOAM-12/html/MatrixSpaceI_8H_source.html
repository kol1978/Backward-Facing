572>
573inline typename typeOfInnerProduct<Cmpt, Form1, Form2>::type operator&
574(

576 const MatrixSpace<Form2, Cmpt, Mrows2, Ncols2>& matrix2
577)
578{

586
587 for (direction i=0; i<Mrows1; ++i)
588 {
