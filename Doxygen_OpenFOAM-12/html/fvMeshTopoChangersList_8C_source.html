45Foam::fvMeshTopoChangers::list::list(fvMesh& mesh, const dictionary& dict)
46:
47 fvMeshTopoChanger(mesh)
48{
49 const dictionary& solversDict = dict.subDict("topoChangers");
50
51 forAllConstIter(dictionary, solversDict, iter)
52 {
53 if (iter().isDict())
54 {
55 const word& name = iter().keyword();
56 const dictionary& dict = iter().dict();
57
58 list_.insert
59 (
60 name,
61 fvMeshTopoChanger::New(mesh, dict).ptr()
62 );
63 }
64 }
65}
