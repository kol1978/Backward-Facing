131template<class BasicMomentumTransportModel>
132Foam::autoPtr<Foam::LESModel<BasicMomentumTransportModel>>
133Foam::LESModel<BasicMomentumTransportModel>::New
134(
135 const alphaField& alpha,
136 const rhoField& rho,
137 const volVectorField& U,
138 const surfaceScalarField& alphaRhoPhi,
139 const surfaceScalarField& phi,
140 const viscosity& viscosity
141)
142{
143 const IOdictionary modelDict
144 (
145 momentumTransportModel::readModelDict
146 (
147 U.db(),
148 alphaRhoPhi.group()
149 )
150 );
151
152 const word modelType =
153 modelDict.subDict("LES").lookupBackwardsCompatible<word>
154 (
155 {"model", "LESModel"}
156 );
157
158 Info<< "Selecting LES turbulence model " << modelType << endl;
159
160 typename dictionaryConstructorTable::iterator cstrIter =
161 dictionaryConstructorTablePtr_->find(modelType);
162
163 if (cstrIter == dictionaryConstructorTablePtr_->end())
164 {
165 FatalErrorInFunction
166 << "Unknown LESModel type "
167 << modelType << nl << nl
168 << "Valid LESModel types:" << endl
169 << dictionaryConstructorTablePtr_->sortedToc()
170 << exit(FatalError);
171 }
172
173 return autoPtr<LESModel>
174 (
175 cstrIter()(alpha, rho, U, alphaRhoPhi, phi, viscosity)
176 );
177}

182template<class BasicMomentumTransportModel>
183bool Foam::LESModel<BasicMomentumTransportModel>::read()
184{
185 if (BasicMomentumTransportModel::read())
186 {
187 LESDict_ <<= this->subDict("LES");
188 LESDict_.lookup("turbulence") >> turbulence_;
189
190 coeffDict_ <<= LESDict_.optionalSubDict(type() + "Coeffs");
191
192 delta_().read(LESDict_);
193
194 kMin_.readIfPresent(LESDict_);
195 nutMaxCoeff_.readIfPresent(LESDict_);
196
197 return true;
198 }
199 else
200 {
201 return false;
202 }
203}
