49// Trait for converting the ThermoModel's thermo type to the thermo type needed
50// for the thermophysical transport model type; i.e., from rho-type thermo to
51// fluid-type thermo.
52
53template<class ThermoModel>
54struct MovingPhaseModelTransportThermoModel;
55
56template<>
57struct MovingPhaseModelTransportThermoModel<rhoFluidThermo>
58{
59 typedef fluidThermo type;
60};

62template<>
63struct MovingPhaseModelTransportThermoModel<rhoFluidMulticomponentThermo>
64{
65 typedef fluidMulticomponentThermo type;
66};
