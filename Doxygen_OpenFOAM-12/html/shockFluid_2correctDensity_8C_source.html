32void Foam::solvers::shockFluid::correctDensity()
33{
34 volScalarField& rho(rho_);
35
36 fvScalarMatrix rhoEqn
37 (
38 fvm::ddt(rho) + fvc::div(phi)
39 ==
40 fvModels().source(rho)
41 );
42
43 fvConstraints().constrain(rhoEqn);
44
45 rhoEqn.solve();
46
47 fvConstraints().constrain(rho);
