69bool Foam::sampledSurfaces::sampledIsoSurfaceSurface::expire()
70{
71 // Clear data
72 sampledSurface::clearGeom();
73 isoSurfPtr_.clear();
74
75 // Already marked as expired
76 if (isoSurfTimeIndex_ == -1)
77 {
78 return false;
79 }
80
81 // Force update
82 isoSurfTimeIndex_ = -1;
83 return true;
84}

87bool Foam::sampledSurfaces::sampledIsoSurfaceSurface::update() const
88{
89 // Quick return if no update needed
90 if (!needsUpdate())
91 {
92 return false;
93 }
94
95 // Clear any information in the base class
96 sampledSurface::clearGeom();
97
98 // Update the iso surface
99 isoSurfPtr_.reset(calcIsoSurf().ptr());
100
101 // Set the time index
102 isoSurfTimeIndex_ = mesh().time().timeIndex();
103
104 return true;
105}
