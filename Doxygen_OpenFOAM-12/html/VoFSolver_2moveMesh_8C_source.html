33void Foam::solvers::VoFSolver::moveMesh()
34{
35 if (pimple.firstIter() || pimple.moveMeshOuterCorrectors())
36 {
37 if
38 (
39 (correctPhi || mesh.topoChanged())
40 && divergent()
41 && !divU.valid()
42 )
43 {
44 // Construct and register divU for correctPhi
45 divU = new volScalarField
46 (
47 "divU0",
48 fvc::div(fvc::absolute(phi, U))
49 );
50 }
51
52 // Move the mesh
53 mesh_.move();
54 }
55}

58void Foam::solvers::VoFSolver::motionCorrector()
59{
60 if (pimple.firstIter() || pimple.moveMeshOuterCorrectors())
61 {
62 if (mesh.changing())
63 {
64 buoyancy.moveMesh();
65
66 MRF.update();
67
68 if (correctPhi || mesh.topoChanged())
69 {
70 // Calculate absolute flux
71 // from the mapped surface velocity
72 phi_ = mesh.Sf() & Uf();
73
74 correctUphiBCs(U_, phi_, true);
75
76 if (incompressible())
77 {
78 fv::correctPhi
79 (
80 phi_,
81 U,
82 p_rgh,
83 rAU,
84 divU,
85 pressureReference(),
86 pimple
87 );
88 }
89 else
90 {
91 fv::correctPhi
92 (
93 phi_,
94 p_rgh,
95 psiByRho(),
96 rAU,
97 divU(),
98 pimple
99 );
100 }
101
102 // Make the fluxes relative to the mesh motion
103 MRF.makeRelative(phi_);
104 fvc::makeRelative(phi_, U);
105 }
106
107 meshCourantNo();
108
109 correctInterface();
110 }
111
112 divU.clear();
113 }
114}
