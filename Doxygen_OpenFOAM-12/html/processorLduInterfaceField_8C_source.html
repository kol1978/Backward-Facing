45void Foam::processorLduInterfaceField::transformCoupleField
46(
47 scalarField& f,
48 const direction cmpt
49) const
50{
51 if (transforms())
52 {
53 const vector diagV(diag(transform().T()));
54
55 if (rank() == 1)
56 {
57 f *= diagV.component(cmpt);
58 }
59 else if (rank() == 2)
60 {
61 f *= sqr(diagV).component(cmpt);
62 }
63 else
64 {
65 FatalErrorInFunction

67 }
68 }
