31Foam::barycentric2D Foam::barycentric2D01(randomGenerator& rndGen)
32{
33 scalar s = rndGen.scalar01();
34 scalar t = rndGen.scalar01();
35
36 // Transform the random point in the unit square to a random point in the
37 // unit tri by reflecting across the diagonal
38
39 if (s + t > 1)
40 {
41 s = 1 - s;
42 t = 1 - t;
43 }
44
45 return Foam::barycentric2D(1 - s - t, s, t);
46}
