27Description
28 Calculates and outputs the mean and maximum Courant Numbers.
29
30\*---------------------------------------------------------------------------*/
31
32scalar alphaCoNum = 0;
33
34{
35 const scalarField sumPhi
36 (
37 mixture.nearInterface()().primitiveField()
38 *fvc::surfaceSum(mag(phi))().primitiveField()
39 );
40
41 alphaCoNum =
42 0.5*gMax(sumPhi/mesh.V().primitiveField())*runTime.deltaTValue();
43
44 const scalar meanAlphaCoNum =
45 0.5
46 *(gSum(sumPhi)/gSum(mesh.V().primitiveField()))
47 *runTime.deltaTValue();
48
49 Info<< "Interface Courant Number mean: " << meanAlphaCoNum
50 << " max: " << alphaCoNum << endl;
51}
