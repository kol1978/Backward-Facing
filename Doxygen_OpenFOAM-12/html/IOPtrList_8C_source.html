32Foam::IOPtrList<Type>::IOPtrList(const IOobject& io, const INew& iNew)
33:
34 regIOobject(io)
35{
36 if
37 (
38 (
39 io.readOpt() == IOobject::MUST_READ
40 || io.readOpt() == IOobject::MUST_READ_IF_MODIFIED
41 )
42 || (io.readOpt() == IOobject::READ_IF_PRESENT && this->headerOk())
43 )
44 {
45 // For if MUST_READ_IF_MODIFIED
46 this->addWatch();
47
48 PtrList<Type>::read(this->readStream(IOPtrList<Type>::typeName), iNew);

51}
