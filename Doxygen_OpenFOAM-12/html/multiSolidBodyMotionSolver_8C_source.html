49Foam::multiSolidBodyMotionSolver::multiSolidBodyMotionSolver
50(
51 const word& name,
52 const polyMesh& mesh,
53 const dictionary& dict
54)
55:
56 points0MotionSolver(name, mesh, dict, typeName)
57{
58 zoneIndices_.setSize(coeffDict().size());
59 SBMFs_.setSize(coeffDict().size());
60 pointIndices_.setSize(coeffDict().size());
61 label zonei = 0;
62
63 forAllConstIter(dictionary, coeffDict(), iter)
64 {
65 if (iter().isDict())
66 {
67 zoneIndices_[zonei] = mesh.cellZones().findIndex(iter().keyword());
68
69 if (zoneIndices_[zonei] == -1)
70 {
71 FatalIOErrorInFunction
72 (
73 coeffDict()
74 ) << "Cannot find cellZone named " << iter().keyword()
75 << ". Valid zones are " << mesh.cellZones().toc()
76 << exit(FatalIOError);
77 }
78
79 const dictionary& subDict = iter().dict();
80
81 SBMFs_.set
82 (
83 zonei,
84 solidBodyMotionFunction::New(subDict, mesh.time())
85 );
86
87 // Collect points of cell zone.
88 const cellZone& cz = mesh.cellZones()[zoneIndices_[zonei]];
89
90 boolList movePts(mesh.nPoints(), false);
91
92 forAll(cz, i)
93 {
94 label celli = cz[i];
95 const cell& c = mesh.cells()[celli];
96 forAll(c, j)
97 {
98 const face& f = mesh.faces()[c[j]];
99 forAll(f, k)
100 {
101 label pointi = f[k];
102 movePts[pointi] = true;
103 }
104 }
105 }
106
107 syncTools::syncPointList(mesh, movePts, orEqOp<bool>(), false);
108
109 DynamicList<label> ptIDs(mesh.nPoints());
110 forAll(movePts, i)
111 {
112 if (movePts[i])
113 {
114 ptIDs.append(i);
115 }
116 }
117
118 pointIndices_[zonei].transfer(ptIDs);
119
120 Info<< "Applying solid body motion " << SBMFs_[zonei].type()
121 << " to " << pointIndices_[zonei].size()
122 << " points of cellZone " << iter().keyword()
123 << endl;
124
125 zonei++;
126 }
127 }
128 zoneIndices_.setSize(zonei);
129 SBMFs_.setSize(zonei);
130 pointIndices_.setSize(zonei);
131}

142Foam::tmp<Foam::pointField> Foam::multiSolidBodyMotionSolver::curPoints() const
143{

145 pointField& transformedPts = ttransformedPts.ref();
146
147 forAll(zoneIndices_, i)
148 {
149 const labelList& zonePoints = pointIndices_[i];
150
151 UIndirectList<point>(transformedPts, zonePoints) = transformPoints

155 );
156 }
157
158 return ttransformedPts;
159}
