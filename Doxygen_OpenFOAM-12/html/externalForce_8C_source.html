77void Foam::RBD::restraints::externalForce::restrain
78(
79 scalarField& tau,
80 Field<spatialVector>& fx,
81 const rigidBodyModelState& state
82) const
83{
84 const vector force = externalForce_().value(state.t());
85 const vector moment(location_ ^ force);
86
87 if (model_.debug)
88 {
89 Info<< " location " << location_
90 << " force " << force
91 << " moment " << moment
92 << endl;
93 }
94
95 // Accumulate the force for the restrained body
96 fx[masterBodyIndex_] += spatialVector(moment, force);
97}
