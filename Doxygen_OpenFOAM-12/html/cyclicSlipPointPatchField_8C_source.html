82template<class Type>
83void Foam::cyclicSlipPointPatchField<Type>::evaluate(const Pstream::commsTypes)
84{
85 const vectorField& nHat = this->patch().pointNormals();
86
87 tmp<Field<Type>> tvalues =
88 (
89 (
90 this->patchInternalField()
91 + transform(I - 2.0*sqr(nHat), this->patchInternalField())
92 )/2.0
93 );
94
95 // Get internal field to insert values into
96 Field<Type>& iF = const_cast<Field<Type>&>(this->primitiveField());
97
98 this->setInternalField(iF, tvalues());
99}
