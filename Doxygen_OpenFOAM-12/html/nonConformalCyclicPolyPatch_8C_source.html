248const Foam::patchToPatches::intersection&
249Foam::nonConformalCyclicPolyPatch::intersection() const
250{
251 if (!owner())
252 {
253 FatalErrorInFunction
254 << "The non-conformal cyclic intersection is only available to "
255 << "the owner patch" << abort(FatalError);
256 }
257
258 if (!intersectionIsValid_)
259 {
260 const polyMesh& mesh = boundaryMesh().mesh();
261
262 const nonConformalBoundary& ncb = nonConformalBoundary::New(mesh);
263
264 intersection_.update
265 (
266 origPatch(),
267 ncb.patchPointNormals(origPatchIndex()),
268 nbrPatch().origPatch(),
269 transform()
270 );
271
272 intersectionIsValid_ = true;
273 }
274
275 return intersection_;
276}

279const Foam::patchToPatches::rays&
280Foam::nonConformalCyclicPolyPatch::rays() const
281{
282 if (!owner())
283 {
284 FatalErrorInFunction
285 << "The non-conformal cyclic rays is only available to "
286 << "the owner patch" << abort(FatalError);
287 }
288
289 if (!raysIsValid_)
290 {
291 const polyMesh& mesh = boundaryMesh().mesh();
292
293 const nonConformalBoundary& ncb = nonConformalBoundary::New(mesh);
294
295 rays_.update
296 (
297 primitiveOldTimePatch
298 (
299 origPatch(),
300 mesh.points(),
301 mesh.oldPoints()
302 ),
303 ncb.patchPointNormals(origPatchIndex()),
304 ncb.patchPointNormals0(origPatchIndex()),
305 primitiveOldTimePatch
306 (
307 nbrPatch().origPatch(),
308 mesh.points(),
309 mesh.oldPoints()
310 ),
311 transform()
312 );
313
314 raysIsValid_ = true;
315 }
316
317 return rays_;
318}

321Foam::remote Foam::nonConformalCyclicPolyPatch::ray
322(
323 const scalar fraction,
324 const label origFacei,
325 const vector& p,
326 const vector& n,
327 point& nbrP
328) const
329{
330 const polyMesh& mesh = boundaryMesh().mesh();
331
332 const nonConformalCyclicPolyPatch& ownerPatch =
333 owner() ? *this : nbrPatch();
334
335 auto ownerRaysMethod =
336 owner()
337 ? &patchToPatches::rays::srcToTgtRay
338 : &patchToPatches::rays::tgtToSrcRay;
339
340 return
341 (ownerPatch.rays().*ownerRaysMethod)
342 (
343 primitiveOldTimePatch
344 (
345 nbrPatch().origPatch(),
346 mesh.points(),
347 mesh.oldPoints()
348 ),
349 fraction,
350 origFacei,
351 transform().invTransformPosition(p),
352 transform().invTransform(n),
353 nbrP
354 );
355}
