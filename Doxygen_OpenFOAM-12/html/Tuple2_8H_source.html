162template<class Type1, class Type2>
163template<class HashT1, class HashT2>
164inline unsigned Tuple2<Type1, Type2>::Hash<HashT1, HashT2>::operator()
165(
166 const Tuple2<Type1, Type2>& t,
167 unsigned seed
168) const
169{
170 // Hash incrementally
171 unsigned val = seed;
172 val = HashT1()(t.first(), val);
173 val = HashT2()(t.second(), val);
