143template<class Cmpt>
144inline typename innerProduct<Vector<Cmpt>, Vector<Cmpt>>::type
145operator&(const Vector<Cmpt>& v1, const Vector<Cmpt>& v2)
146{
147 return Cmpt(v1.x()*v2.x() + v1.y()*v2.y() + v1.z()*v2.z());
148}

151template<class Cmpt>
152inline Vector<Cmpt> operator^(const Vector<Cmpt>& v1, const Vector<Cmpt>& v2)
153{
154 return Vector<Cmpt>
155 (
156 (v1.y()*v2.z() - v1.z()*v2.y()),
157 (v1.z()*v2.x() - v1.x()*v2.z()),
158 (v1.x()*v2.y() - v1.y()*v2.x())
159 );
160}

165template<class Cmpt>
166inline Vector<Cmpt> perpendicular(const Vector<Cmpt>& v)
167{
168 Vector<Cmpt> u(Zero);
169 u[findMin(cmptMag(v))] = 1;
170 return u ^ v;
171}
