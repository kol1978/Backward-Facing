35void Foam::solvers::incompressibleDenseParticleFluid::momentumPredictor()
36{
37 volVectorField& Uc(Uc_);
38
39 tUcEqn =
40 (
41 fvm::ddt(alphac, Uc) + fvm::div(alphaPhic, Uc)
42 - fvm::Sp(fvc::ddt(alphac) + fvc::div(alphaPhic), Uc)
43 + momentumTransport->divDevSigma(Uc)
44 ==
45 fvModels().source(alphac, Uc)
46 );
47 fvVectorMatrix& UcEqn = tUcEqn.ref();
48
49 UcEqn.relax();
50
51 fvConstraints().constrain(UcEqn);
52
53 if (pimple.momentumPredictor())
54 {
55 // Face buoyancy force
56 const surfaceScalarField Fgf(g & mesh.Sf());
57
58 solve
59 (
60 UcEqn
61 ==
62 fvc::reconstruct
63 (
64 Fgf - fvc::snGrad(p)*mesh.magSf()
65 - Dcf()*(phic - phid())
66 )
67 + Dc()*fvc::reconstruct(phic - phid())
68 + Fd() - fvm::Sp(Dc(), Uc)
69 );
70
71 fvConstraints().constrain(Uc);
72 }
73}
