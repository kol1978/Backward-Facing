48void Foam::fv::heatTransferCoefficientModels::constant::readCoeffs
49(
50 const dictionary& dict
51)
52{
53 typeIOobject<volScalarField> htcIO
54 (
55 "htc",
56 mesh_.time().constant(),
57 mesh_,
58 IOobject::MUST_READ,
59 IOobject::NO_WRITE
60 );
61
62 if (dict.found("htc"))
63 {
64 htc_ =
65 dimensionedScalar
66 (
67 "htc",
68 dimPower/dimTemperature/dimArea,
69 dict
70 );
71 htcPtr_.clear();
72 }
73 else if (htcIO.headerOk())
74 {
75 htc_ = dimensionedScalar("htc", dimPower/dimTemperature/dimArea, NaN);
76 htcPtr_.set(new volScalarField(htcIO, mesh_));
77 }
78 else
79 {
80 FatalIOErrorInFunction(dict)
81 << "Heat transfer coefficient (htc) not found. A uniform htc "
82 << "value should be specified, or a non-uniform field should "
83 << "exist at " << htcIO.objectPath()
84 << exit(FatalIOError);
85 }
86}
