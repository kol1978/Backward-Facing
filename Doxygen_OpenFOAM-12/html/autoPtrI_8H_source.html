47template<class T>
48inline Foam::autoPtr<T>::autoPtr(const autoPtr<T>& ap, const bool reuse)
49{
50 if (reuse)
51 {
52 ptr_ = ap.ptr_;
53 ap.ptr_ = nullptr;
54 }
55 else if (ap.valid())
56 {
57 ptr_ = ap().clone().ptr();
58 }
59 else
60 {
61 ptr_ = nullptr;
62 }
63}
