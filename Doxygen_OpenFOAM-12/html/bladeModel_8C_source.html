40void Foam::bladeModel::interpolateWeights
41(
42 const scalar& xIn,
43 const List<scalar>& values,
44 label& i1,
45 label& i2,
46 scalar& ddx
47) const
48{
49 i2 = 0;
50 label nElem = values.size();
51
52 if (nElem == 1)
53 {
54 i1 = i2;
55 ddx = 0.0;
56 return;
57 }
58 else
59 {
60 while ((i2 < nElem) && (values[i2] < xIn))
61 {
62 i2++;
63 }
64
65 if (i2 == 0)
66 {
67 i1 = i2;
68 ddx = 0.0;
69 return;
70 }
71 else if (i2 == nElem)
72 {
73 i2 = nElem - 1;
74 i1 = i2;
75 ddx = 0.0;
76 return;
77 }
78 else
79 {
80 i1 = i2 - 1;
81 ddx = (xIn - values[i1])/(values[i2] - values[i1]);
82 }
83 }
84}

89Foam::bladeModel::bladeModel(const dictionary& dict)
90:
91 profileName_(),
92 profileIndex_(),
93 radius_(),
94 twist_(),
95 chord_(),
96 fName_(fileName::null)
97{
98 List<Tuple2<word, vector>> data;
99 if (readFromFile())
100 {
101 IFstream is(fName_);
102 is >> data;
103 }
104 else
105 {
106 dict.lookup("data") >> data;
107 }
108
109 if (data.size() > 0)
110 {
111 profileName_.setSize(data.size());
112 profileIndex_.setSize(data.size());
113 radius_.setSize(data.size());
114 twist_.setSize(data.size());
115 chord_.setSize(data.size());
116
117 forAll(data, i)
118 {
119 profileName_[i] = data[i].first();
120 profileIndex_[i] = -1;
121 radius_[i] = data[i].second()[0];
122 twist_[i] = degToRad(data[i].second()[1]);
123 chord_[i] = data[i].second()[2];
124 }
125 }
126 else
127 {
128 FatalErrorInFunction
129 << "No blade data specified" << exit(FatalError);
130 }
131}

177void Foam::bladeModel::interpolate
178(
179 const scalar radius,
180 scalar& twist,
181 scalar& chord,
182 label& i1,
183 label& i2,
184 scalar& invDr
185) const
186{
187 interpolateWeights(radius, radius_, i1, i2, invDr);
188
189 twist = invDr*(twist_[i2] - twist_[i1]) + twist_[i1];
190 chord = invDr*(chord_[i2] - chord_[i1]) + chord_[i1];
191}
