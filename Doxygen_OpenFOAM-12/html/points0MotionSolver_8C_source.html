37Foam::pointVectorField Foam::points0MotionSolver::readPoints0
38(
39 const polyMesh& mesh
40)
41{
42 const word instance
43 (
44 mesh.time().findInstance
45 (
46 ".",
47 "points0",
48 IOobject::READ_IF_PRESENT
49 )
50 );
51
52 if (instance != mesh.time().constant())
53 {
54 // Points0 written to a time folder
55
56 return pointVectorField
57 (
58 IOobject
59 (
60 "points0",
61 instance,
62 mesh,
63 IOobject::MUST_READ,
64 IOobject::NO_WRITE,
65 false
66 ),
67 pointMesh::New(mesh)
68 );
69 }
70 else
71 {
72 // Return copy of original mesh points
73
74 pointIOField points
75 (
76 IOobject
77 (
78 "points",
79 mesh.time().constant(),
80 polyMesh::meshSubDir,
81 mesh,
82 IOobject::MUST_READ,
83 IOobject::NO_WRITE,
84 false
85 )
86 );
87
88 pointVectorField points0
89 (
90 IOobject
91 (
92 "points0",
93 instance,
94 mesh,
95 IOobject::NO_READ,
96 IOobject::NO_WRITE,
97 false
98 ),
99 pointMesh::New(mesh),
100 dimensionedVector(dimLength, Zero)
101 );
102
103 points0.primitiveFieldRef() = points;
104
105 return points0;
106 }
107}

162void Foam::points0MotionSolver::mapMesh(const polyMeshMap& map)
163{
164 points0_.primitiveFieldRef() = mesh().points();
165
166 // The processor boundaries may have changed, so we need to update the
167 // boundary field. There is no data in this field, so we don't need to map
168 // anything. We can just reset it to a freshly created calculated field.
169 points0_.boundaryFieldRef().reset
170 (
171 pointVectorField::Boundary
172 (
173 points0_.mesh().boundary(),
174 points0_.internalField(),
175 calculatedPointPatchVectorField::typeName
176 )
177 );
178}
