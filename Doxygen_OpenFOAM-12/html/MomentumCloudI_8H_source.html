310template<class CloudType>
311inline Foam::scalar Foam::MomentumCloud<CloudType>::Dij
312(
313 const label i,
314 const label j
315) const
316{
317 scalar si = 0.0;
318 scalar sj = 0.0;
319 forAllConstIter(typename MomentumCloud<CloudType>, *this, iter)
320 {
321 const parcelType& p = iter();
322 si += p.nParticle()*pow(p.d(), i);
323 sj += p.nParticle()*pow(p.d(), j);
324 }
325
326 reduce(si, sumOp<scalar>());
327 reduce(sj, sumOp<scalar>());
328 sj = max(sj, vSmall);
329
330 return si/sj;
331}

430template<class CloudType>
431inline Foam::tmp<Foam::fvVectorMatrix>
432Foam::MomentumCloud<CloudType>::SU(const volVectorField& U) const
433{
434 if (debug)
435 {
436 Info<< "UTrans min/max = " << min(UTrans()).value() << ", "
437 << max(UTrans()).value() << nl
438 << "UCoeff min/max = " << min(UCoeff()).value() << ", "
439 << max(UCoeff()).value() << endl;
440 }
441
442 if (solution_.coupled())
443 {
444 if (solution_.semiImplicit("U"))
445 {
446 const volScalarField::Internal
447 Vdt(this->mesh().V()*this->db().time().deltaT());
448
449 return UTrans()/Vdt - fvm::Sp(UCoeff()/Vdt, U) + UCoeff()/Vdt*U;
450 }
451 else
452 {
453 tmp<fvVectorMatrix> tfvm(new fvVectorMatrix(U, dimForce));
454 fvVectorMatrix& fvm = tfvm.ref();
455
456 fvm.source() = -UTrans()/(this->db().time().deltaT());
457
458 return tfvm;
459 }
460 }
461
462 return tmp<fvVectorMatrix>(new fvVectorMatrix(U, dimForce));
463}

466template<class CloudType>
467inline const Foam::tmp<Foam::volScalarField>
468Foam::MomentumCloud<CloudType>::vDotSweep() const
469{
470 tmp<volScalarField> tvDotSweep
471 (
472 volScalarField::New
473 (
474 this->name() + ":vDotSweep",
475 this->mesh(),
476 dimensionedScalar(dimless/dimTime, 0),
477 extrapolatedCalculatedFvPatchScalarField::typeName
478 )
479 );
480
481 volScalarField& vDotSweep = tvDotSweep.ref();
482 forAllConstIter(typename MomentumCloud<CloudType>, *this, iter)
483 {
484 const parcelType& p = iter();
485 const label celli = p.cell();
486
487 vDotSweep[celli] += p.nParticle()*p.areaP()*mag(p.U() - U_[celli]);
488 }
489
490 vDotSweep.primitiveFieldRef() /= this->mesh().V();
491 vDotSweep.correctBoundaryConditions();
492
493 return tvDotSweep;
494}
