67Foam::tmp<Foam::surfaceScalarField>
68Foam::motionDirectionalDiffusivity::operator()() const
69{
70 if (mesh().foundObject<volVectorField>("cellMotionU"))
71 {
72 const volVectorField& cellMotionU =
73 mesh().lookupObject<volVectorField>("cellMotionU");
74
75 volVectorField D
76 (
77 IOobject
78 (
79 "D",
80 mesh().time().name(),
81 mesh()
82 ),
83 diffusivityVector_.y()*vector::one
84 + (diffusivityVector_.x() - diffusivityVector_.y())*cellMotionU
85 /(mag(cellMotionU) + dimensionedScalar(dimVelocity, small)),
86 zeroGradientFvPatchVectorField::typeName
87 );
88 D.correctBoundaryConditions();
89
90 const surfaceVectorField n(mesh().Sf()/mesh().magSf());
91
92 return surfaceScalarField::New
93 (
94 "faceDiffusivity",
95 (n & cmptMultiply(fvc::interpolate(D), n))
96 );
97 }
98 else
99 {
100 return tmp<surfaceScalarField>
101 (
102 new surfaceScalarField
103 (
104 IOobject
105 (
106 "faceDiffusivity",
107 mesh().time().name(),
108 mesh(),
109 IOobject::NO_READ,
110 IOobject::NO_WRITE
111 ),
112 mesh(),
113 dimensionedScalar(dimless, 1.0)
114 )
115 );
116 }
117}
