36inline Foam::scalar Foam::Function1s::squarePulse::integral
37(
38 const scalar t1,
39 const scalar t2
40) const
41{
42 scalar sign = t1 < t2 ? +1 : -1;
43
44 const scalar tL = min(t1, t2);
45 const scalar tR = max(t1, t2);
46
47 const scalar tLStar = min(max(tL, start_), tR);
