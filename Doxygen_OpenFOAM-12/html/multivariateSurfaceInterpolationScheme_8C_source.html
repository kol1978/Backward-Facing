53template<class Type>
54Foam::tmp<Foam::multivariateSurfaceInterpolationScheme<Type>>
55Foam::multivariateSurfaceInterpolationScheme<Type>::New
56(
57 const fvMesh& mesh,
58 const multivariateSurfaceInterpolationScheme<Type>::fieldTable& vtfs,
59 const surfaceScalarField& faceFlux,
60 Istream& schemeData
61)
62{
63 if (fv::debug)
64 {
65 InfoInFunction
66 << "Constructing surfaceInterpolationScheme<Type>" << endl;
67 }
68
69 const word schemeName(schemeData);
70
71 typename IstreamConstructorTable::iterator constructorIter =
72 IstreamConstructorTablePtr_->find(schemeName);
73
74 if (constructorIter == IstreamConstructorTablePtr_->end())
75 {
76 FatalIOErrorInFunction
77 (
78 schemeData
79 ) << "Unknown discretisation scheme " << schemeName << nl << nl
80 << "Valid schemes are :" << endl
81 << IstreamConstructorTablePtr_->sortedToc()
82 << exit(FatalIOError);
83 }
84
85 return constructorIter()(mesh, vtfs, faceFlux, schemeData);
86}
