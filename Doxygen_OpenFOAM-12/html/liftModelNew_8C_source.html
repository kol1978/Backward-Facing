31Foam::autoPtr<Foam::liftModel> Foam::liftModel::New
32(
33 const dictionary& dict,
34 const phaseInterface& interface,
35 const bool outer
36)
37{
38 const dictionary& modelDict =
39 outer ? interface.fluid().modelSubDict<liftModel>(dict) : dict;
40
41 const word liftModelType(modelDict.lookup("type"));
42
43 Info<< "Selecting liftModel for "
44 << interface.name() << ": " << liftModelType << endl;
45
46 dictionaryConstructorTable::iterator cstrIter =
47 dictionaryConstructorTablePtr_->find(liftModelType);
48
49 if (cstrIter == dictionaryConstructorTablePtr_->end())
50 {
51 FatalErrorInFunction
52 << "Unknown liftModel type "
53 << liftModelType << endl << endl
54 << "Valid liftModel types are : " << endl
55 << dictionaryConstructorTablePtr_->sortedToc()
56 << exit(FatalError);
57 }
58
59 return cstrIter()(modelDict, interface);
60}
