39Foam::tmp<Foam::volScalarField>
40Foam::incompressibleMultiphaseVoFMixture::mu() const
41{
42 tmp<volScalarField> tmu
43 (
44 phases_[0]*phases_[0].rho()*phases_[0].nu()
45 );

48 for (label phasei=1; phasei<phases_.size(); phasei++)
49 {
50 mu += phases_[phasei]*phases_[phasei].rho()*phases_[phasei].nu();
51 }
52

115void Foam::incompressibleMultiphaseVoFMixture::correct()
116{
117 forAll(phases_, phasei)
118 {
119 phases_[phasei].correct();
120 }
121
122 rho_ = phases_[0]*phases_[0].rho();
123
124 for (label phasei=1; phasei<phases_.size(); phasei++)
125 {
126 rho_ += phases_[phasei]*phases_[phasei].rho();
127 }
128
129 // Update the mixture kinematic viscosity
130 nu_ = mu()/rho_;
131
132 calcAlphas();
133}
