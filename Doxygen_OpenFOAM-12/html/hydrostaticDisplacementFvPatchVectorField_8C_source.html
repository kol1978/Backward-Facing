82void Foam::hydrostaticDisplacementFvPatchVectorField::updateCoeffs()
83{
84 if (updated())
85 {
86 return;
87 }
88
89 const uniformDimensionedVectorField& g =
90 db().lookupObject<uniformDimensionedVectorField>("g");
91
92 tractionDisplacementFvPatchVectorField::updateCoeffs
93 (
94 liquidSurfacePressure_
95 + rhoLiquid_
96 *max(g.value() & (patch().Cf() - liquidSurfacePoint_), scalar(0))
97 );
98}
