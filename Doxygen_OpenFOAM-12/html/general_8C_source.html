66void Foam::relativeVelocityModels::general::correct()
67{
68 const volScalarField& alphad = mixture_.alphad();
69
70 Udm_ =
71 (mixture_.rhoc()/mixture_.rho())*Vc_*acceleration()
72 *(
73 exp(-a_*max(alphad - residualAlpha_, scalar(0)))
74 - exp(-a1_*max(alphad - residualAlpha_, scalar(0)))
75 );
76}
