37void Foam::radiationModels::absorptionCoeffs::checkT(const scalar T) const
38{
39 if (T < Tlow_ || T > Thigh_)
40 {
41 WarningInFunction
42 << "using absorptionCoeffs out of temperature range:" << nl
43 << " " << Tlow_ << " -> " << Thigh_ << "; T = " << T
44 << nl << endl;
45 }
46}

50Foam::radiationModels::absorptionCoeffs::coeffs
