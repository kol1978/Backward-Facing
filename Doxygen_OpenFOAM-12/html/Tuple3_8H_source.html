174template<class Type1, class Type2, class Type3>
175template<class HashT1, class HashT2, class HashT3>
176inline unsigned
177Tuple3<Type1, Type2, Type3>::Hash<HashT1, HashT2, HashT3>::operator()
178(
179 const Tuple3<Type1, Type2, Type3>& t,
180 unsigned seed
181) const
182{
183 // Hash incrementally
184 unsigned val = seed;
185 val = HashT1()(t.first(), val);
186 val = HashT2()(t.second(), val);
187 val = HashT3()(t.third(), val);
188 return val;
189}
