31template<class Type>
32Foam::tmp<Foam::Field<Type>> Foam::coordSet::gather
33(
34 const Field<Type>& values,
35 const labelList& order
36)
37{
38 // Collect data from all processors
39 List<List<Type>> gatheredValues(Pstream::nProcs());
40 gatheredValues[Pstream::myProcNo()] = values;
41 Pstream::gatherList(gatheredValues);
42
43 // Combine processor lists into one big list
44 List<Type> allValues
45 (
46 ListListOps::combine<List<Type>>
47 (
48 gatheredValues,
49 accessOp<List<Type>>()
50 )
51 );
52
53 // Sort by order and overwrite
54 return tmp<Field<Type>>(new Field<Type>(allValues, order));
55}
