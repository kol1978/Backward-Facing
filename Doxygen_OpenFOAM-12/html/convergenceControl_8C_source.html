89bool Foam::convergenceControl::converged()
90{
91 const convergenceData cd(criteriaSatisfied());
92
93 if
94 (
95 control_.time().timeIndex() != control_.time().startTimeIndex()
96 && cd.checked && cd.satisfied
97 )
98 {
99 Info<< nl << control_.algorithmName() << " solution converged in "
100 << control_.time().name() << " iterations" << nl << endl;
101
102 return true;
103 }
104
105 return false;
106}

109bool Foam::convergenceControl::endIfConverged(Time& time)
110{
111 if (converged())
112 {
113 if (time.writeTime())
114 {
115 time.stopAt(Time::stopAtControl::noWriteNow);
116 time.setEndTime(time);
117 }
118 else
119 {
120 time.writeAndEnd();
121 }
122
123 return true;
124 }
125 else
126 {
127 return false;
128 }
129}
