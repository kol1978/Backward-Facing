30Foam::autoPtr<Foam::fvMeshStitcher> Foam::fvMeshStitcher::New
31(
32 fvMesh& mesh,
33 bool changing
34)
35{
36 // Determine if the mesh is actually changing and load the
37 // fvMeshStitchers library if so
38 if (changing)
39 {
40 typeIOobject<IOdictionary> dictHeader
41 (
42 IOobject
43 (
44 "dynamicMeshDict",
45 mesh.time().constant(),
46 mesh,
47 IOobject::READ_IF_PRESENT,
48 IOobject::NO_WRITE,
49 false
50 )
51 );
52
53 changing = changing && dictHeader.headerOk();
54
55 if (changing)
56 {
57 libs.open("lib" + typeName + "s.so");
58 }
59 }
60
61 // Select a non-changing or changing stitcher as appropriate
62 forAllConstIter
63 (
64 fvMeshConstructorTable,
65 *fvMeshConstructorTablePtr_,
66 cstrIter
67 )
68 {
69 autoPtr<fvMeshStitcher> stitcherPtr(cstrIter()(mesh));
70
71 if (stitcherPtr->changing() == changing)
72 {
73 return stitcherPtr;
74 }
75 }
76
77 // Error if an appropriate stitcher was not found
78 FatalErrorInFunction
79 << typeName << " for " << (changing ? "" : "non-")
80 << "changing mesh not found " << nl << nl
81 << "Valid " << typeName << "s are : " << endl
82 << fvMeshConstructorTablePtr_->toc()
83 << exit(FatalError);
84
85 return autoPtr<fvMeshStitcher>(nullptr);
86}
