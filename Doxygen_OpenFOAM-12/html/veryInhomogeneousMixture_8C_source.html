46template<class ThermoType>
47const ThermoType& Foam::veryInhomogeneousMixture<ThermoType>::mixture
48(
49 const scalar ft,
50 const scalar fu
51) const
52{
53 if (ft < 0.0001)
54 {
55 return oxidant_;
56 }
57 else
58 {
59 const scalar ox = 1 - ft - (ft - fu)*stoicRatio_.value();
60 const scalar pr = 1 - fu - ox;
61
62 mixture_ = fu*fuel_;
63 mixture_ += ox*oxidant_;
64 mixture_ += pr*products_;
65
66 return mixture_;
67 }
68}
