182Foam::scalar Foam::solvers::VoFSolver::maxDeltaT() const
183{
184 const scalar maxAlphaCo =
185 runTime.controlDict().lookup<scalar>("maxAlphaCo");
186
187 scalar deltaT = fluidSolver::maxDeltaT();
188
189 if (alphaCoNum > small)
190 {
191 deltaT = min(deltaT, maxAlphaCo/alphaCoNum*runTime.deltaTValue());
192 }
193
194 return deltaT;
195}

198void Foam::solvers::VoFSolver::preSolve()
199{
200 if ((mesh.dynamic() || MRF.size()) && !Uf.valid())
201 {
202 Info<< "Constructing face momentum Uf" << endl;
203
204 // Ensure the U BCs are up-to-date before constructing Uf
205 U_.correctBoundaryConditions();
206
207 Uf = new surfaceVectorField
208 (
209 IOobject
210 (
211 "Uf",
212 runTime.name(),
213 mesh,
214 IOobject::READ_IF_PRESENT,
215 IOobject::AUTO_WRITE
216 ),
217 fvc::interpolate(U_)
218 );
219 }
220
221 if (transient())
222 {
223 correctCoNum();
224 }
225 else if (LTS)
226 {
227 setRDeltaT();
228 }
229
230 fvModels().preUpdateMesh();
231
232 // Store divU from the previous mesh so that it can be mapped
233 // and used in correctPhi to ensure the corrected phi has the
234 // same divergence
235 if (correctPhi && divergent())
236 {
237 // Construct and register divU for mapping
238 divU = new volScalarField
239 (
240 "divU0",
241 fvc::div(fvc::absolute(phi, U))
242 );
243 }
244
245 // Update the mesh for topology change, mesh to mesh mapping
246 mesh_.update();
247}
