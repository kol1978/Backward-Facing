82tmp<scalarField> nutLowReWallFunctionFvPatchScalarField::yPlus() const
83{
84 const label patchi = patch().index();
85
86 const momentumTransportModel& turbModel =
87 db().lookupType<momentumTransportModel>(internalField().group());
88
89 const scalarField& y = turbModel.yb()[patchi];
90 const tmp<scalarField> tnuw = turbModel.nu(patchi);
91 const scalarField& nuw = tnuw();
92 const fvPatchVectorField& Uw = turbModel.U().boundaryField()[patchi];
93
94 return y*sqrt(nuw*mag(Uw.snGrad()))/nuw;
95}
