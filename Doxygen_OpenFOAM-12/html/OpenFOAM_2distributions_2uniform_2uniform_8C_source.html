43Foam::scalar Foam::distributions::uniform::Phi(const scalar x) const
44{
45 if (q() == -1)
46 {
47 return log(x);
48 }
49 else
50 {
51 return integerPow(x, 1 + q())/(1 + q());
52 }
53}

102Foam::scalar Foam::distributions::uniform::sample() const
103{
104 const scalar s = rndGen_.sample01<scalar>();
105
106 if (q() == -1)
107 {
108 return min_*pow(max_/min_, s);
109 }
110 else
111 {
112 const scalar PhiS = (1 - s)*Phi0_ + s*Phi1_;
113
114 return integerRoot((1 + q())*PhiS, 1 + q());
115 }
116}

131Foam::scalar Foam::distributions::uniform::mean() const
132{
133 if (q() == -2)
134 {
135 return Foam::log(max_/min_)/(Phi1_ - Phi0_);
136 }
137 else
138 {
139 const scalar Mu0 = integerPow(min_, 2 + q())/(2 + q());
140 const scalar Mu1 = integerPow(max_, 2 + q())/(2 + q());
141
142 return (Mu1 - Mu0)/(Phi1_ - Phi0_);
143 }
144}

161Foam::distributions::uniform::PDF(const scalarField& x) const
162{
163 if (q() == -1)
164 {
165 return clipPDF(x, 1/x/Foam::log(max_/min_));
166 }
167 else
168 {
