42template<class Type>
43tmp<VolField<Type>>
44curl
45(
46 const VolField<Type>& vf
47)
48{
49 word nameCurlVf = "curl(" + vf.name() + ')';
50
51 // Gausses theorem curl
52 // tmp<VolField<Type>> tcurlVf =
53 // fvc::surfaceIntegrate(vf.mesh().Sf() ^ fvc::interpolate(vf));
54
55 // Calculate curl as the Hodge dual of the skew-symmetric part of grad
56 tmp<VolField<Type>> tcurlVf =
57 2.0*(*skew(fvc::grad(vf, nameCurlVf)));
58
59 tcurlVf.ref().rename(nameCurlVf);
60
61 return tcurlVf;
62}
