36template
37<
38 class Type,
39 template<class> class PatchField,
40 class GeoMesh
41>
42bool reusable(const tmp<GeometricField<Type, PatchField, GeoMesh>>& tgf)
43{
44 if (tgf.isTmp())
45 {
46 if (GeometricField<Type, PatchField, GeoMesh>::debug)
47 {
48 const GeometricField<Type, PatchField, GeoMesh>& gf = tgf();
49 const typename GeometricField<Type, PatchField, GeoMesh>::
50 Boundary& gbf = gf.boundaryField();
51
52 forAll(gbf, patchi)
53 {
54 if
55 (
56 !polyPatch::constraintType(gbf[patchi].patch().type())
57 && !isA<typename PatchField<Type>::Calculated>(gbf[patchi])
58 )
59 {
60 WarningInFunction
61 << "Attempt to reuse temporary with non-reusable BC "
62 << gbf[patchi].type() << endl;
63
64 return false;
65 }
66 }
67 }
68
69 return true;
70 }
71 else
72 {
73 return false;
74 }
75}

78template<class TypeR, template<class> class PatchField, class GeoMesh>
79tmp<GeometricField<TypeR, PatchField, GeoMesh>> New
80(
81 const tmp<GeometricField<TypeR, PatchField, GeoMesh>>& tgf1,
82 const word& name,
83 const dimensionSet& dimensions,
84 const bool initRet = false
85)
86{
87 GeometricField<TypeR, PatchField, GeoMesh>& gf1 =
88 const_cast<GeometricField<TypeR, PatchField, GeoMesh>& >(tgf1());
89
90 if (reusable(tgf1))
91 {
92 gf1.rename(name);
93 gf1.dimensions().reset(dimensions);
94 return tgf1;
95 }
96 else
97 {
98 tmp<GeometricField<TypeR, PatchField, GeoMesh>> rtgf
99 (
100 GeometricField<TypeR, PatchField, GeoMesh>::New
101 (
102 name,
103 gf1.mesh(),
104 dimensions
105 )
106 );
107
108 if (initRet)
109 {
110 rtgf.ref() == tgf1();
111 }
112
113 return rtgf;
114 }
115}
