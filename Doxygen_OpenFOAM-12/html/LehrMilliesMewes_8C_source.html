65void Foam::diameterModels::binaryBreakupModels::LehrMilliesMewes::
66addToBinaryBreakupRate
67(
68 volScalarField& binaryBreakupRate,
69 const label i,
70 const label j
71)
72{
73 const phaseModel& continuousPhase = popBal_.continuousPhase();
74 const sizeGroup& fi = popBal_.sizeGroups()[i];
75 const sizeGroup& fj = popBal_.sizeGroups()[j];
76
77 volScalarField L
78 (
79 pow
80 (
81 popBal_.sigmaWithContinuousPhase(fj.phase())/continuousPhase.rho(),
82 3.0/5.0
83 )
84 /pow(popBal_.continuousTurbulence().epsilon(), 2.0/5.0)
85 );
86
87 // Reset of dimension to pure length to avoid problems in transcendental
88 // functions due to small exponents
89 L.dimensions().reset(dimLength);
90
91 const volScalarField T
92 (
93 pow
94 (
95 popBal_.sigmaWithContinuousPhase(fj.phase())/continuousPhase.rho(),
96 2.0/5.0
97 )
98 /pow(popBal_.continuousTurbulence().epsilon(), 3.0/5.0)
99 );
100
101 binaryBreakupRate +=
102 0.5*pow(fj.dSph()/L, 5.0/3.0)
103 *exp(-sqrt(2.0)/pow3(fj.dSph()/L))
104 *6/pow(pi, 1.5)/pow3(fi.dSph()/L)
105 *exp(-9.0/4.0*sqr(log(pow(2.0, 0.4)*fi.dSph()/L)))
106 /max(1 + erf(1.5*log(pow(2.0, 1.0/15.0)*fj.dSph()/L)), small)
107 /(T*pow3(L));
108}
