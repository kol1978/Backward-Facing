42Foam::label Foam::pow(label a, label b)
43{
44 label ans = 1;
45 for (label i=0; i<b; i++)
46 {
47 ans *= a;
48 }
49
50 #ifdef FULLDEBUG
51 if (b < 0)
52 {
53 FatalErrorInFunction
54 << "negative value for b is not supported"
55 << abort(FatalError);
56 }
57 #endif
58
59 return ans;
60}
