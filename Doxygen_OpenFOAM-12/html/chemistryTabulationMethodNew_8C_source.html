32Foam::autoPtr<Foam::chemistryTabulationMethod>
33Foam::chemistryTabulationMethod::New
34(
35 const IOdictionary& dict,
36 const odeChemistryModel& chemistry
37)
38{
39 if (dict.found("tabulation"))
40 {
41 const dictionary& tabulationDict(dict.subDict("tabulation"));
42
43 const word methodName(tabulationDict.lookup("method"));
44
45 Info<< "Selecting chemistry tabulation method " << methodName << endl;
46
47 typename dictionaryConstructorTable::iterator cstrIter =
48 dictionaryConstructorTablePtr_->find(methodName);
49
50 if (cstrIter == dictionaryConstructorTablePtr_->end())
51 {
52 FatalErrorInFunction
53 << "Unknown " << typeName_() << " type " << methodName << endl
54 << "Valid " << typeName_() << " types are:"
55 << dictionaryConstructorTablePtr_->sortedToc() << endl
56 << exit(FatalError);
57 }
58
59 return autoPtr<chemistryTabulationMethod>
60 (
61 cstrIter()(dict, chemistry)
62 );
63 }
64 else
65 {
66 return autoPtr<chemistryTabulationMethod>
67 (
68 new chemistryTabulationMethods::none(dict, chemistry)
69 );
70 }
71}
