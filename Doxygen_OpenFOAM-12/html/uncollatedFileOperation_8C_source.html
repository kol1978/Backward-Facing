59Foam::fileName Foam::fileOperations::uncollatedFileOperation::filePathInfo
60(
61 const bool globalFile,
62 const bool isFile,
63 const IOobject& io
64) const
65{
66 if (io.instance().isAbsolute())
67 {
68 fileName objectPath = io.instance()/io.name();
69
70 if (isFileOrDir(isFile, objectPath))
71 {
72 return objectPath;
73 }
74 else
75 {
76 return fileName::null;
77 }
78 }
79 else
80 {
81 fileName path = io.path(globalFile);
82 fileName objectPath = path/io.name();
83
84 if (isFileOrDir(isFile, objectPath))
85 {
86 return objectPath;
87 }
88 else
89 {
90 if
91 (
92 globalFile
93 && io.time().processorCase()
94 && (
95 io.instance() == io.time().system()
96 || io.instance() == io.time().constant()
97 )
98 )
99 {
100 // Constant & system can come from global case
101
102 fileName parentObjectPath =
103 io.rootPath()/io.time().globalCaseName()
104 /io.instance()/io.db().dbDir()/io.local()/io.name();
105
106 if (isFileOrDir(isFile, parentObjectPath))
107 {
108 return parentObjectPath;
109 }
110 }
111
112 // Check if parallel "processors" directory
113 if (io.time().processorCase())
114 {
115 tmpNrc<dirIndexList> pDirs
116 (
117 lookupProcessorsPath(io.objectPath(globalFile))
118 );
119 forAll(pDirs(), i)
120 {
121 const fileName& pDir = pDirs()[i].first();
122 fileName objPath =
123 processorsPath(io, io.instance(), pDir)
124 /io.name();
125 if (objPath != objectPath && isFileOrDir(isFile, objPath))
126 {
127 return objPath;
128 }
129 }
130 }
131
132
133 // Check for approximately same time. E.g. if time = 1e-2 and
134 // directory is 0.01 (due to different time formats)
135 if (!Foam::isDir(path))
136 {
137 word newInstancePath = io.time().findInstancePath
138 (
139 instant(io.instance())
140 );
141
142 if (newInstancePath.size())
143 {
144 fileName fName
145 (
146 io.rootPath()/io.caseName(globalFile)
147 /newInstancePath/io.db().dbDir()/io.local()/io.name()
148 );
149
150 if (isFileOrDir(isFile, fName))
151 {
152 return fName;
153 }
154 }
155 }
156 }
157
158 return fileName::null;
159 }
160}

468bool Foam::fileOperations::uncollatedFileOperation::readHeader
469(
470 IOobject& io,
471 const fileName& fName,
472 const word& typeName
473) const
474{
475 if (debug)
476 {
477 Pout<< "uncollatedFileOperation::readHeader :"
478 << " fName:" << fName
479 << " typeName:" << typeName << endl;
480 }
481 if (fName.empty())
482 {
483 if (IOobject::debug)
484 {
485 InfoInFunction
486 << "file for object " << io.name() << " could not be opened"
487 << endl;
488 }
489
490 return false;
491 }
492
493 autoPtr<ISstream> isPtr(NewIFstream(fName));
494
495 if (!isPtr.valid() || !isPtr->good())
496 {
497 return false;
498 }
499
500 bool ok = io.readHeader(isPtr());
501
502 if (io.headerClassName() == decomposedBlockData::typeName)
503 {
504 // Read the header inside the container (master data)
505 ok = decomposedBlockData::readMasterHeader(io, isPtr());
506 }
507
508 if (debug)
509 {
510 Pout<< "uncollatedFileOperation::readHeader :"
511 << " for fName:" << fName
512 << " ok:" << ok
513 << " headerClassName:" << io.headerClassName() << endl;
514 }
515
516 return ok;
517}

520Foam::autoPtr<Foam::ISstream>
521Foam::fileOperations::uncollatedFileOperation::readStream
522(
523 regIOobject& io,
524 const fileName& fName,
525 const word& typeName,
526 const bool read
527) const
528{
529 autoPtr<ISstream> isPtr;
530
531 if (!read)
532 {
533 isPtr = autoPtr<ISstream>(new dummyISstream());
534 return isPtr;
535 }
536
537 if (fName.empty())
538 {
539 FatalErrorInFunction
540 << "cannot find file " << io.objectPath()
541 << exit(FatalError);
542 }
543
544 isPtr = NewIFstream(fName);
545
546 if (!isPtr.valid() || !isPtr->good())
547 {
548 FatalIOError
549 (
550 "uncollatedFileOperation::readStream()",
551 __FILE__,
552 __LINE__,
553 IOerrorLocation(fName, 0)
554 ) << "cannot open file"
555 << exit(FatalIOError);
556 }
557 else if (!io.readHeader(isPtr()))
558 {
559 FatalIOErrorInFunction(isPtr())
560 << "problem while reading header for object " << io.name()
561 << exit(FatalIOError);
562 }
563
564 if (io.headerClassName() != decomposedBlockData::typeName)
565 {
566 return isPtr;
567 }
568 else
569 {
570 // Analyse the objectpath to find out the processor we're trying
571 // to access
572 label proci = detectProcessorPath(io.objectPath());
573
574 if (proci == -1)
575 {
576 FatalIOErrorInFunction(isPtr())
577 << "could not detect processor number"
578 << " from objectPath:" << io.objectPath()
579 << " fName:" << fName
580 << exit(FatalIOError);
581 }
582
583 // Analyse the fileName for any processor subset. Note: this
584 // should really be part of filePath() which should return
585 // both file and index in file.
586 fileName path, procDir, local;
587 label groupStart, groupSize, nProcs;
588 splitProcessorPath
589 (
590 fName,
591 path,
592 procDir,
593 local,
594 groupStart,
595 groupSize,
596 nProcs
597 );
598 if (groupStart != -1 && groupSize > 0)
599 {
600 proci = proci-groupStart;
601 }
602
603 // Read data and return as stream
604 return decomposedBlockData::readBlock(proci, isPtr(), io);
605 }
606}

609bool Foam::fileOperations::uncollatedFileOperation::read
610(
611 regIOobject& io,
612 const bool masterOnly,
613 const IOstream::streamFormat defaultFormat,
614 const word& typeName
615) const
616{
617 bool ok = true;
618
619 // Initialise format to the defaultFormat
620 // but reset to ASCII if defaultFormat and file format are ASCII
621 IOstream::streamFormat format = defaultFormat;
622
623 if (Pstream::master() || !masterOnly)
624 {
625 if (debug)
626 {
627 Pout<< "uncollatedFileOperation::read :"
628 << " Reading object " << io.objectPath()
629 << " from file " << endl;
630 }
631
632 // Open file and read header
633 Istream& is = io.readStream(typeName);
634
635 // Set format to ASCII if defaultFormat and file format are ASCII
636 if (defaultFormat == IOstream::ASCII)
637 {
638 format = is.format();
639 }
640
641 // Read the data from the file
642 ok = io.readData(is);
643
644 // Close the file
645 io.close();
646
647 if (debug)
648 {
649 Pout<< "uncollatedFileOperation::read :"
650 << " Done reading object " << io.objectPath()
651 << " from file " << endl;
652 }
653 }
654
655 if (masterOnly && Pstream::parRun())
656 {
657 // Master reads headerclassname from file. Make sure this gets
658 // transferred as well as contents.
659 Pstream::scatter(io.headerClassName());
660 Pstream::scatter(io.note());
661
662 if (defaultFormat == IOstream::ASCII)
663 {
664 std::underlying_type_t<IOstream::streamFormat> formatValue(format);
665 Pstream::scatter(formatValue);
666 format = IOstream::streamFormat(formatValue);
667 }
668
669 // Get my communication order
670 const List<Pstream::commsStruct>& comms =
671 (
672 (Pstream::nProcs() < Pstream::nProcsSimpleSum)
673 ? Pstream::linearCommunication()
674 : Pstream::treeCommunication()
675 );
676 const Pstream::commsStruct& myComm = comms[Pstream::myProcNo()];
677
678 // Receive from up
679 if (myComm.above() != -1)
680 {
681 IPstream fromAbove
682 (
683 Pstream::commsTypes::scheduled,
684 myComm.above(),
685 0,
686 Pstream::msgType(),
687 Pstream::worldComm,
688 format
689 );
690 ok = io.readData(fromAbove);
691 }
692
693 // Send to my downstairs neighbours
694 forAll(myComm.below(), belowI)
695 {
696 OPstream toBelow
697 (
698 Pstream::commsTypes::scheduled,
699 myComm.below()[belowI],
700 0,
701 Pstream::msgType(),
702 Pstream::worldComm,
703 format
704 );
705 bool okWrite = io.writeData(toBelow);
706 ok = ok && okWrite;
707 }
708 }
709 return ok;
710}
