31template<class Type>
32Foam::timeVaryingMappedFixedValueFvPatchField<Type>::
33timeVaryingMappedFixedValueFvPatchField
34(
35 const fvPatch& p,
36 const DimensionedField<Type, volMesh>& iF,
37 const dictionary& dict
38)
39:
40 fixedValueFvPatchField<Type>(p, iF, dict, false),
41 fieldMapper_(p, iF, dict)
42{
43 if (dict.found("value"))
44 {
45 fvPatchField<Type>::operator==
46 (
47 Field<Type>("value", iF.dimensions(), dict, p.size())
48 );
49 }
50 else
51 {
52 // Note: we use evaluate() here to trigger updateCoeffs followed
53 // by re-setting of fvatchfield::updated_ flag. This is
54 // so if first use is in the next time step it retriggers
55 // a new update.
56 this->evaluate(Pstream::commsTypes::blocking);
57 }
58}

123template<class Type>
124void Foam::timeVaryingMappedFixedValueFvPatchField<Type>::updateCoeffs()
125{
126 if (this->updated())
127 {
128 return;
129 }
130
131 this->operator==(fieldMapper_.map());
132
133 fixedValueFvPatchField<Type>::updateCoeffs();
134}
