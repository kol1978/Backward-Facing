39void Foam::solvers::incompressibleFluid::correctPressure()
40{
41 volScalarField& p(p_);
42 volVectorField& U(U_);
43 surfaceScalarField& phi(phi_);
44
45 fvVectorMatrix& UEqn = tUEqn.ref();
46
47 volScalarField rAU(1.0/UEqn.A());
48 volVectorField HbyA(constrainHbyA(rAU*UEqn.H(), U, p));
49 surfaceScalarField phiHbyA
50 (
51 "phiHbyA",
52 fvc::flux(HbyA)
53 + fvc::interpolate(rAU)*fvc::ddtCorr(U, phi, Uf)
54 );
55
56 MRF.makeRelative(phiHbyA);
57
58 if (p.needReference())
59 {
60 fvc::makeRelative(phiHbyA, U);
61 adjustPhi(phiHbyA, U, p);
62 fvc::makeAbsolute(phiHbyA, U);
63 }
64
65 tmp<volScalarField> rAtU(rAU);
66
67 if (pimple.consistent())
68 {
69 rAtU = 1.0/max(1.0/rAU - UEqn.H1(), 0.1/rAU);
70 phiHbyA +=
71 fvc::interpolate(rAtU() - rAU)*fvc::snGrad(p)*mesh.magSf();
72 HbyA -= (rAU - rAtU())*fvc::grad(p);
73 }
74
75 if (pimple.nCorrPiso() <= 1)
76 {
77 tUEqn.clear();
78 }
79
80 // Update the pressure BCs to ensure flux consistency
81 constrainPressure(p, U, phiHbyA, rAtU(), MRF);
82
83 // Evaluate any volume sources
84 fvScalarMatrix p_rghEqnSource(fvModels().sourceProxy(p));
85
86 // Non-orthogonal pressure corrector loop
87 while (pimple.correctNonOrthogonal())
88 {
89 fvScalarMatrix pEqn
90 (
91 fvm::laplacian(rAtU(), p)
92 ==
93 fvc::div(phiHbyA)
94 - p_rghEqnSource
95 );
96
97 pEqn.setReference
98 (
99 pressureReference.refCell(),
100 pressureReference.refValue()
101 );
102
103 pEqn.solve();
104
105 if (pimple.finalNonOrthogonalIter())
106 {
107 phi = phiHbyA - pEqn.flux();
108 }
109 }
110
111 continuityErrors();
112
113 // Explicitly relax pressure for momentum corrector
114 p.relax();
115
116 U = HbyA - rAtU*fvc::grad(p);
117 U.correctBoundaryConditions();
118 fvConstraints().constrain(U);
119
120 // Correct Uf if the mesh is moving
121 fvc::correctUf(Uf, U, phi, MRF);
122
123 // Make the fluxes relative to the mesh motion
124 fvc::makeRelative(phi, U);
125}
