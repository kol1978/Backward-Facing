30Foam::decompositionMethods::geometric::geometric
31(
32 const dictionary& decompositionDict,
33 const word& derivedType
34)
35:
36 decompositionMethod(decompositionDict),
37 geometricDict_(decompositionDict.optionalSubDict(derivedType + "Coeffs")),
38 n_(geometricDict_.lookup("n")),
39 delta_(geometricDict_.lookupOrDefault<scalar>("delta", 0.001)),
40 rotDelta_(I)
41{
42 // Check that the decomposition specification makes sense:
43 if (nProcessors_ != n_.x()*n_.y()*n_.z())
44 {
45 FatalErrorInFunction
46 << "Wrong number of processor divisions in geometric:" << nl
47 << "Number of domains : " << nProcessors_ << nl
48 << "Wanted decomposition : " << n_
49 << exit(FatalError);
50 }
51
52 const scalar d = 1 - 0.5*delta_*delta_;
53 const scalar d2 = sqr(d);
54
55 const scalar a = delta_;
56 const scalar a2 = sqr(a);
57
58 rotDelta_ = tensor
59 (
60 d2, -a*d, a,
61 a*d - a2*d, a*a2 + d2, -2*a*d,
62 a*d2 + a2, a*d - a2*d, d2 - a2
63 );
64}
