66template<class IntType>
67class StrideRange
68{
69 static_assert(std::is_integral_v<IntType>, "Integral required");
70
71 // Private Data
72
73 //- The start of the interval
74 IntType start_;
75
76 //- The length (number of items) for the interval
77 IntType size_;
78
79 //- The stride of the interval
80 IntType stride_;
81
82
83 // Private Member Functions
84
85 //- The value "at" the local index \p idx, with bounds checking.
86 // If index is out-of-bounds, returns the end_value()
87 inline IntType at_value(IntType idx) const noexcept;
88
89 //- Compare start/size/stride in that order.
90 // \return -1,0,+1
91 template<class IntT2>
92 inline constexpr int
93 compare(const StrideRange<IntT2>& other) const noexcept;
94
95
96public:
97
98 // STL type definitions
99
100 //- Type of values the range contains
101 typedef IntType value_type;
102
103 //- The type that can represent the size of the range
104 typedef IntType size_type;
105
106 //- Input iterator with const access
107 using const_iterator = StrideIterator<IntType>;
108
109 //- Reverse input iterator with const access
110 using const_reverse_iterator = RevStrideIterator<IntType>;
111
112
113 // Generated Methods: copy/move construct, copy/move assignment
114
115 // Constructors
116
117 //- Default construct an empty stride range (0,0,0)
118 inline constexpr StrideRange() noexcept;
119
120 // //- Construct a range starting at zero with the specified length
121 // //- and stride (0,len,stride)
122 // inline constexpr StrideRange(IntType len, stride) noexcept;
123
124 //- Construct start/size/stride, no checks
125 inline constexpr StrideRange
126 (
127 IntType beg,
128 IntType len,
129 IntType stride
130 ) noexcept;
131
132
133 // Member Functions
134
135 // Access
136
137 //- True if range is empty (zero-sized)
138 constexpr bool empty() const noexcept { return !size_; }
139
140 //- The size of the range (number of items)
141 constexpr IntType size() const noexcept { return size_; }
142
143 //- Non-const access to size of the range (number of items)
144 constexpr IntType& size() noexcept { return size_; }
145
146 //- The (inclusive) lower value of the range
147 constexpr IntType start() const noexcept { return start_; }
148
149 //- Non-const access to start of the range
150 constexpr IntType& start() noexcept { return start_; }
151
152 //- The stride of the range
153 constexpr IntType stride() const noexcept { return stride_; }
154
155 //- Non-const access to the stride
156 constexpr IntType& stride() noexcept { return stride_; }
157
158 //- The (inclusive) lower value of the range,
159 //- same as start(), begin_value()
160 constexpr IntType min() const noexcept { return start_; }
161
162 //- The (inclusive) upper value of the range,
163 //- same as rbegin_value(). Ill-defined for an empty range
164 constexpr IntType max() const noexcept
165 {
166 return (size_ > 0) ? rbegin_value() : start_;
167 }
168
169
170 // Member Operators
171
172 //- Offset dereference, without bounds checking
173 //- Return element in the range, without bounds checking
174 inline constexpr IntType operator[](IntType i) const noexcept;
175
176
177 // Comparison
178
179 //- Test for equality of start/size/stride values
180 template<class IntT2>
181 constexpr bool
182 operator==(const StrideRange<IntT2>& other) const noexcept;
183
184 //- Compare lexicographically: compares start, size, stride
185 template<class IntT2>
186 constexpr bool
187 operator<(const StrideRange<IntT2>& other) const noexcept
188 {
189 return (this->compare(other) < 0);
190 }
191
192 // Derived comparisons
193 template<class IntT2>
194 constexpr bool
195 operator!=(const StrideRange<IntT2>& other) const noexcept
196 {
197 return !(*this == other);
198 }
199
200 template<class IntT2>
201 constexpr bool
202 operator<=(const StrideRange<IntT2>& other) const noexcept
203 {
204 return !(other < *this);
205 }
206
207 template<class IntT2>
208 constexpr bool
209 operator>(const StrideRange<IntT2>& other) const noexcept
210 {
211 return (other < *this);
212 }
213
214 template<class IntT2>
215 constexpr bool
216 operator>=(const StrideRange<IntT2>& other) const noexcept
217 {
218 return !(*this < other);
219 }
220
221
222 // Iterator range values
223
224 //- The value at the beginning of the range - same as start()
225 inline constexpr IntType begin_value() const noexcept;
226
227 //- The value that is one stride beyond the end of the range
228 inline constexpr IntType end_value() const noexcept;
229
230 //- The max value of the end of start/size range
231 inline constexpr IntType rbegin_value() const noexcept;
232
233 //- The value that is one stride before the begin of start/size range
234 inline constexpr IntType rend_value() const noexcept;
235
236
237 // Iteration / Generation
238
239 //- Return a forward values generator
240 StrideIndexer<IntType> generator() const noexcept
241 {
242 return { start_, stride_ };
243 }
244
245 //- Return const_iterator to a position within the range,
246 //- with bounds checking.
247 // \return iterator at the requested position, or end() for
248 // out-of-bounds
249 const_iterator at(IntType i) const
250 {
251 return { at_value(i), stride_ };
252 }
253
254 //- A const_iterator set to the beginning of the range
255 const_iterator begin() const noexcept
256 {
257 return { begin_value(), stride_ };
258 }
259
260 //- A const_iterator set to the beginning of the range
261 const_iterator cbegin() const noexcept
262 {
263 return { begin_value(), stride_ };
264 }
265
266 //- A const_iterator set to 1 stride beyond the end of the range.
267 const_iterator cend() const noexcept
268 {
269 return { end_value(), stride_ };
270 }
271
272 //- A const_iterator set to 1 stride beyond the end of the range.
273 const_iterator end() const noexcept
274 {
275 return { end_value(), stride_ };
276 }
277
278 //- A const_reverse_iterator set to 1 stride before the end of range
279 const_reverse_iterator rbegin() const noexcept
280 {
281 return { rbegin_value(), stride_ };
282 }
283
284 //- A const_reverse_iterator set to 1 stride before the end of range
285 const_reverse_iterator crbegin() const noexcept
286 {
287 return { rbegin_value(), stride_ };
288 }
289
290 //- A const_reverse_iterator set to 1 stride before the begin of range
291 const_reverse_iterator rend() const noexcept
292 {
293 return { rend_value(), stride_ };
294 }
295
296 //- A const_reverse_iterator set to 1 stride before the begin of range
297 const_reverse_iterator crend() const noexcept
298 {
299 return { rend_value(), stride_ };
300 }
301};
