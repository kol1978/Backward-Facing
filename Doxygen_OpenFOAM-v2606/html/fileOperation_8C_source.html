89// Need to parse the numbers
90// from "processors(\d+)" and
91// from "processors(\d+)_(\d+)-(\d+)"
92//
93// Receive the string matching "^(\d+)(?:_(\d+)-(\d+))?/?$"
94//
95// \1 = numProcs
96// \2 = firstProc
97// \3 = lastProc
98//
99// Return true on success and set parameters numProcs and group (start,size)
100//
101// Use low-level C-string to integer parsing to drive the sequence.
102//
103// For simplicity, also skip INT_MAX checks everywhere but check for
104// - (errno) for success
105// - (nptr == endptr) for leading junk
106// - (*endptr != endChar) for trailing junk
107// - skip INT_MAX checks as being too pessimistic
108
109template<class IntType>
110bool parseProcsNumRange
111(
112 std::string str,
113 int& numProcs,
114 Foam::IntRange<IntType>& group
115)
116{
117 const char * nptr = str.c_str();
118 char *endptr = nullptr;
119
120 // 1. numProcs
121 errno = 0;
122 intmax_t parsed = std::strtoimax(nptr, &endptr, 10);
123 if (errno || nptr == endptr) return false; // bad parse
124
125 const int nProcs = int(parsed);
126
127 // End of string? Then no range and we are done.
128 if (*endptr == '\0')
129 {
130 numProcs = nProcs;
131 return true;
132 }
133
134 // Parse point at start of range ('_' character)?
135 if (*endptr != '_') return false;
136 nptr = ++endptr;
137
138
139 // 2. Min rank in the range
140 errno = 0;
141 parsed = std::strtoimax(nptr, &endptr, 10);
142 if (errno || nptr == endptr) return false; // bad parse
143
144 const int minProc = int(parsed);
145
146 // Parse point at range separator ('-' character)?
147 if (*endptr != '-') return false;
148 nptr = ++endptr;
149
150
151 // 3. Max rank in the range (inclusive)
152 errno = 0;
153 parsed = std::strtoimax(nptr, &endptr, 10);
154 if (errno || nptr == endptr) return false; // bad parse
155
156 const int maxProc = int(parsed);
157
158
159 if
160 (
161 // Parse point at end of string
162 (*endptr == '\0')
163
164 // Input plausibility
165 // Accept nProcs == 0 in case that becomes useful in the future
166 && (nProcs >= 0 && minProc >= 0 && minProc <= maxProc)
167 )
168 {
169 numProcs = nProcs;
170
171 // Convert min/max to start/size
172 group.reset(minProc, maxProc-minProc+1);
173
174 return true;
175 }
176
177 return false;
178}

212Foam::instantList
213Foam::fileOperation::sortTimes
214(
215 const fileNameList& dirEntries,
216 const word& constantName
217)
218{
219 // Check for "constant"
220 bool haveConstant = false;
221
222 if (!constantName.empty())
223 {
224 for (const fileName& dirName : dirEntries)
225 {
226 if (dirName == constantName)
227 {
228 haveConstant = true;
229 break;
230 }
231 }
232 }
233
234 instantList times(dirEntries.size() + 1);
235 label nTimes = 0;
236
237 if (haveConstant)
238 {
239 times[nTimes].value() = 0;
240 times[nTimes].name() = constantName;
241 ++nTimes;
242 }
243
244 // Parse directory entries for scalar values
245 for (const fileName& dirName : dirEntries)
246 {
247 if (Foam::readScalar(dirName, times[nTimes].value()))
248 {
249 times[nTimes].name() = dirName;
250 ++nTimes;
251 }
252 }
253
254 times.resize(nTimes);
255
256 if (haveConstant)
257 {
258 if (nTimes > 2)
259 {
260 std::sort(times.begin()+1, times.end(), instant::less());
261 }
262 }
263 else if (nTimes > 1)
264 {
265 std::sort(times.begin(), times.end(), instant::less());
266 }
267
268 return times;
269}

327void Foam::fileOperation::mergeTimes
328(
329 const UList<instant>& extraTimes,
330 const word& constantName,
331 instantList& times
332)
333{
334 if (extraTimes.empty())
335 {
336 // Nothing to do
337 }
338 else
339 {
340 // Combined times
341 DynamicList<instant> combinedTimes(times.size()+extraTimes.size());
342
343 const bool haveConstant =
344 (
345 times.size()
346 && times[0].name() == constantName
347 );
348
349 const bool haveExtraConstant =
350 (
351 extraTimes.size()
352 && extraTimes[0].name() == constantName
353 );
354
355 if (!haveConstant && haveExtraConstant)
356 {
357 // No constant in times, but have one in extraTimes
358 combinedTimes.push_back(extraTimes[0]);
359 }
360
361 combinedTimes.push_back(std::move(times));
362
363 for (label i = (haveExtraConstant ? 1 : 0); i < extraTimes.size(); ++i)
364 {
365 combinedTimes.push_back(extraTimes[i]);
366 }
367
368 times.transfer(combinedTimes);
369
370 // Sort
371 if (times.size() > 1)
372 {
373 label count = 0;
374
375 if (times[0].name() == constantName)
376 {
377 // Exclude "constant" from sorting and duplicate checks
378 ++count;
379 }
380 std::sort(times.begin()+count, times.end(), instant::less());
381
382 // Filter any subsequent duplicates
383 ++count;
384
385 for (label i = count; i < times.size(); ++i)
386 {
387 if (times[i].value() != times[i-1].value())
388 {
389 if (count != i)
390 {
391 // copy (not move) - still needed for later comparison
392 times[count] = times[i];
393 }
394 ++count;
395 }
396 }
397
398 times.resize(count);
399 }
400 }

410Foam::refPtr<Foam::fileOperation::dirIndexList>
411Foam::fileOperation::lookupAndCacheProcessorsPath
412(
413 const fileName& fName,
414 const bool syncPar
415) const
416{
417 // If path is local to a processor (e.g. contains 'processor2')
418 // find the corresponding actual processor directory (e.g. 'processors4')
419 // and index (2)
420
421 // Behaviour affected by
422 // - UPstream::parRun()
423 // - syncPar : usually true, only uncollated does false. Determines
424 // if directory status gets synchronised
425 // - distributed() : different processors have different roots
426 // - fileModificationChecking : (uncollated only) do IO on master only
427 // - nProcsFilter_ : if set to
428 // - 0 : no filtering
429 // - >0 : accept the exact specified number of prcessors
430 // - -1 : accept only processorsDDD where DDD is nProcs(worldComm)
431
432 // Collated : check whether/how to filter processorsXXX directory names
433 const int targetNProcs
434 (
435 (UPstream::parRun() && nProcsFilter_ < 0)
436 ? static_cast<int>(UPstream::nProcs(UPstream::worldComm))
437 : nProcsFilter_
438 );
439
440
441 fileName path, pDir, local;
442 procRangeType group;
443 label numProcs(-1);
444 const label proci =
445 splitProcessorPath(fName, path, pDir, local, group, numProcs);
446
447 if (proci != -1)
448 {
449 const fileName procPath(path/pDir);
450
451 if (cacheLevel() > 0)
452 {
453 if (const auto iter = procsDirs_.cfind(procPath); iter.good())
454 {
455 return iter.val();
456 }
457 }
458
459 // Read all directories to see any beginning with processor
460 // ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
461
462 // Note: use parallel synchronised reading so cache will be same
463 // order on all processors
464
465 const bool readDirMasterOnly
466 (
467 UPstream::parRun() && !distributed()
468 && IOobject::fileModificationChecking_masterOnly()
469 );
470
471 // The above selection excludes masterUncollated, which uses inotify or
472 // timeStamp but provides its own internals for readDir() anyhow.
473
474 fileNameList dirEntries;
475 if (readDirMasterOnly)
476 {
477 // Parallel and non-distributed
478 // Read on master only and send to subProcs
479
480 if (UPstream::master(UPstream::worldComm))
481 {
482 dirEntries = Foam::readDir(path, fileName::Type::DIRECTORY);
483
484 DebugInfo
485 << "readDir on master: send " << dirEntries.size()
486 << " names to sub-processes" << endl;
487 }
488
489 Pstream::broadcast(dirEntries, UPstream::worldComm);
490 }
491 else
492 {
493 // Serial or distributed roots.
494 // Handle readDir() with virtual method
495
496 if (debug)
497 {
498 Pout<< "readDir without special master/send treatment"
499 << endl;
500 }
501
502 dirEntries = readDir(path, fileName::Type::DIRECTORY);
503 }
504
505 // Extract info from processorN or processorsNN
506 // - highest processor number
507 // - directory+offset containing data for proci
508
509 label nProcs = 0;
510 DynamicList<dirIndex> procDirs;
511
512 for (const fileName& dirN : dirEntries)
513 {
514 // Analyse directory name
515 label rNum(-1);
516 const label readProci =
517 fileOperation::detectProcessorPath(dirN, group, &rNum);
518
519 nProcs = Foam::max(nProcs, readProci+1);
520
521 Tuple2<pathType, int> pathTypeIdx(pathType::NOTFOUND, 0);
522
523 if (proci == readProci)
524 {
525 // Found "processorN"
526 pathTypeIdx.first() = pathType::PROCUNCOLLATED;
527 }
528 else if (rNum != -1)
529 {
530 if (targetNProcs > 1 && (targetNProcs != rNum))
531 {
532 // Current directory can never contain wanted proc
533 //Pout<< "For fName:" << fName
534 // << "Ignoring directory " << dirN
535 // << " since parsed rNum:" << rNum
536 // << " targetNProcs:" << targetNProcs
537 // << endl;
538 //error::printStack(Pout);
539
540 continue;
541 }
542
543 // "processorsNN" or "processorsNN_min-max"
544 nProcs = Foam::max(nProcs, rNum);
545
546 if (group.empty())
547 {
548 // "processorsNN"
549
550 if (proci < rNum)
551 {
552 // And it is also in range.
553 // Eg for "processors4": 3 is ok, 10 is not
554
555 pathTypeIdx.first() = pathType::PROCBASEOBJECT;
556 pathTypeIdx.second() = proci;
557 }
558 }
559 else if (group.contains(proci))
560 {
561 // "processorsNN_min-max"
562 // - save the local proc offset
563
564 pathTypeIdx.first() = pathType::PROCOBJECT;
565 pathTypeIdx.second() = (proci - group.start());
566 }
567 }
568
569 if (pathTypeIdx.first() != pathType::NOTFOUND)
570 {
571 procDirs.emplace_back(dirN, pathTypeIdx);
572 }
573 }
574
575 // Sort processor directory names (natural order)
576 sortProcessorDirs(procDirs);
577
578
579 // Global check of empty/exists.
580 // 1 : empty directory
581 // 2 : non-empty directory
582 // 3 : mixed empty/non-empty directory (after reduce)
583 // Combines andOp<bool>() and orOp<bool>() in single operation
584
585 unsigned int procDirsStatus = (procDirs.empty() ? 1u : 2u);
586
587 if (debug)
588 {
589 Pout<< "fileOperation::lookupProcessorsPath " << procPath
590 << " detected:" << flatOutput(procDirs) << endl;
591 }
592
593 if (UPstream::parRun() && (!distributed() || syncPar))
594 {
595 reduce(procDirsStatus, bitOrOp<unsigned int>()); // worldComm
596
597 if (procDirsStatus == 3u)
598 {
599 // Mixed empty/exists for procDirs.
600 // Synthesize missing directory name (consistency in cache
601 // existence).
602 // Cannot reliably synthesize RANK-COLLATED, only COLLATED or
603 // UNCOLLATED.
604 //
605 // RANK-COLLATED should have been read from its corresponding
606 // master anyhow
607
608 int flavour(pathType::PROCUNCOLLATED);
609 for (const dirIndex& pDir : procDirs)
610 {
611 flavour = Foam::max(flavour, int(pDir.second().first()));
612 }
613
614 reduce(nProcs, maxOp<label>()); // worldComm
615 reduce(flavour, maxOp<int>()); // worldComm
616
617 if (procDirs.empty())
618 {
619 Tuple2<pathType, int> pathTypeIdx(pathType(flavour), 0);
620
621 if
622 (
623 pathTypeIdx.first() == pathType::PROCBASEOBJECT
624 // Do not restrict to currently used processors
625 // && proci < nProcs
626 )
627 {
628 pathTypeIdx.second() = proci;
629
630 procDirs.emplace_back
631 (
632 processorsBaseDir + Foam::name(nProcs),
633 pathTypeIdx
634 );
635 }
636 else
637 {
638 // - pathType::PROCUNCOLLATED
639 // - poor fallback for pathType::PROCOBJECT
640 // - out-of-range pathType::PROCBASEOBJECT
641
642 procDirs.emplace_back
643 (
644 "processor" + Foam::name(proci),
645 pathTypeIdx
646 );
647 }
648
649 if (debug)
650 {
651 Pout<< "fileOperation::lookupProcessorsPath "
652 << procPath
653 << " synthetic:" << procDirs << endl;
654 }
655 }
656 }
657 }
658 else if (!UPstream::parRun())
659 {
660 // Serial: use the number of decompositions (if found)
661 if (nProcs)
662 {
663 this->constCast().nProcs(nProcs);
664 }
665 }
666
667 if (procDirsStatus & 2u)
668 {
669 if (cacheLevel() > 0)
670 {
671 procsDirs_(procPath) = std::move(procDirs);
672
673 // Make sure to return a reference
674 return refPtr<dirIndexList>(procsDirs_[procPath]);
675 }
676 else
677 {
678 return refPtr<dirIndexList>::New(std::move(procDirs));
679 }
680 }
681 }
682
683 return refPtr<dirIndexList>(dirIndexList::null());
684}

694bool Foam::fileOperation::exists(IOobject& io) const
695{
696 // Generate output filename for object
697 fileName objPath(objectPath(io, word::null));
698
699 // Test for either directory or a (valid) file & IOobject
700 bool ok;
701 if (io.name().empty())
702 {
703 ok = isDir(objPath);
704 }
705 else
706 {
707 ok =
708 (
709 isFile(objPath)
710 // object with local scope
711 && io.typeHeaderOk<regIOobject>(false)
712 );
713 }
714
715 if (!ok)
716 {
717 // Re-test with searched for objectPath. This is for backwards
718 // compatibility
719 fileName originalPath(filePath(io.objectPath()));
720 if (originalPath != objPath)
721 {
722 // Test for either directory or a (valid) file & IOobject
723 if (io.name().empty())
724 {
725 ok = isDir(originalPath);
726 }
727 else
728 {
729 ok =
730 (
731 isFile(originalPath)
732 // object with local scope
733 && io.typeHeaderOk<regIOobject>(false)
734 );
735 }
736 }
737 }
738
739 return ok;
740}

831Foam::fileName Foam::fileOperation::filePath
832(
833 const fileName& fName,
834 const bool checkGzip,
835 const bool followLink
836) const
837{
838 if (debug)
839 {
840 Pout<< "fileOperation::filePath :" << " fName:" << fName << endl;
841 }
842
843 fileName path, pDir, local;
844 procRangeType group;
845 label numProcs(-1);
846 label proci =
847 splitProcessorPath(fName, path, pDir, local, group, numProcs);
848
849 if (numProcs != -1)
850 {
851 WarningInFunction << "Filename is already adapted:" << fName << endl;
852 }
853
854 // Give preference to processors variant
855 fileName foundName;
856 if (proci != -1)
857 {
858 // Get all processor directories
859 const refPtr<dirIndexList> procDirs(lookupProcessorsPath(fName));
860 for (const dirIndex& dirIdx : procDirs())
861 {
862 const fileName& procDir = dirIdx.first();
863
864 fileName collatedName(path/procDir/local);
865 if (exists(collatedName, checkGzip, followLink))
866 {
867 if (debug)
868 {
869 Pout<< "fileOperation::filePath : " << collatedName << endl;
870 }
871 foundName = collatedName;
872 }
873 }
874 }
875
876 //if (returnReduceOr(foundName.empty())) // worldComm
877 if (foundName.empty())
878 {
879 // There is at least one processor that cannot find the processors
880 // directory. Re-do with straight supplied filename
881 if (exists(fName, checkGzip, followLink))
882 {
883 if (foundName.empty())
884 {
885 foundName = fName;
886 }
887 }
888 }
889
890 if (!foundName.empty())
891 {
892 if (debug)
893 {
894 Pout<< "fileOperation::filePath : " << foundName << endl;
895 }
896 }
897 else
898 {
899 if (debug)
900 {
901 Pout<< "fileOperation::filePath : Not found" << endl;
902 }
903 }
904 return foundName;
905}

1061Foam::IOobject Foam::fileOperation::findInstance
1062(
1063 const IOobject& startIO,
1064 const scalar startValue,
1065 const word& stopInstance,
1066 const bool constant_fallback
1067) const
1068{
1069 const Time& time = startIO.time();
1070 IOobject io(startIO);
1071
1072 // Note: - if name is empty, just check the directory itself
1073 // - check both for isFile and headerOk since the latter does a
1074 // filePath so searches for the file.
1075 // - check for an object with local file scope (so no looking up in
1076 // parent directory in case of parallel)
1077
1078 if (exists(io))
1079 {
1080 DebugInFunction
1081 << "Found exact match for \"" << io.name()
1082 << "\" in " << io.instance()/io.local()
1083 << endl;
1084
1085 return io;
1086 }
1087
1088 // Handling failures afterwards
1089 const bool exitIfMissing = startIO.isReadRequired();
1090
1091 enum failureCodes { FAILED_STOPINST = 1, FAILED_CONSTINST = 2 };
1092 int failed(0);
1093
1094 // The timeDirs
1095 instantList ts
1096 (
1097 this->findTimes
1098 (
1099 time.path(), // time.path(layout)
1100 time.constant()
1101 )
1102 );
1103
1104 {
1105 label instIndex = ts.size()-1;
1106
1107 // Backward search for first time that is <= startValue
1108 for (; instIndex >= 0; --instIndex)
1109 {
1110 if (ts[instIndex] <= startValue)
1111 {
1112 break;
1113 }
1114 }
1115
1116 // Continue (forward) searching from here
1117 for (; instIndex >= 0; --instIndex)
1118 {
1119 io.instance() = ts[instIndex].name();
1120
1121 // Shortcut: if actual directory is the timeName we've
1122 // already tested it
1123 if
1124 (
1125 io.instance() == startIO.instance()
1126 && io.instance() != stopInstance
1127 )
1128 {
1129 continue;
1130 }
1131
1132 if (exists(io))
1133 {
1134 DebugInFunction
1135 << "Found exact match for \"" << io.name()
1136 << "\" in " << io.instance()/io.local()
1137 << endl;
1138
1139 return io;
1140 }
1141
1142 // Check if hit minimum instance
1143 if (io.instance() == stopInstance)
1144 {
1145 DebugInFunction
1146 << "Hit stopInstance " << stopInstance << endl;
1147
1148 if (exitIfMissing)
1149 {
1150 failed = failureCodes::FAILED_STOPINST;
1151 }
1152 else
1153 {
1154 // At the stopInstance
1155 return io;
1156 }
1157 break;
1158 }
1159 }
1160
1161
1162 // times() usually already includes the constant() so would
1163 // have been checked above. However, re-test under these conditions:
1164 // - times() is empty. Sometimes this can happen (e.g. decomposePar
1165 // with collated)
1166 // - times()[0] is not constant
1167 // - Times is empty.
1168 // Sometimes this can happen (eg, decomposePar with collated)
1169 // - Times[0] is not constant
1170 // - The startValue is negative (eg, kivaTest).
1171 // This plays havoc with the reverse search, causing it to miss
1172 // 'constant'
1173
1174 if
1175 (
1176 !failed
1177 && (ts.empty() || ts[0].name() != time.constant() || startValue < 0)
1178 )
1179 {
1180 io.instance() = time.constant();
1181
1182 if (exists(io))
1183 {
1184 DebugInFunction
1185 << "Found constant match for \"" << io.name()
1186 << "\" in " << io.instance()/io.local()
1187 << endl;
1188
1189 return io;
1190 }
1191 }
1192 }
1193
1194
1195 if (!failed)
1196 {
1197 if (exitIfMissing)
1198 {
1199 failed = failureCodes::FAILED_CONSTINST;
1200 }
1201 else if (constant_fallback)
1202 {
1203 io.instance() = time.constant();
1204 }
1205 else
1206 {
1207 io.instance().clear();
1208 }
1209 }
1210
1211
1212 // Handle failures
1213 // ~~~~~~~~~~~~~~~
1214
1215 if (failed)
1216 {
1217 FatalErrorInFunction << "Cannot find";
1218
1219 if (!io.name().empty())
1220 {
1221 FatalError
1222 << " file \"" << io.name() << "\" in";
1223 }
1224
1225 FatalError
1226 << " directory "
1227 << io.local() << " in times "
1228 << startIO.instance() << " down to ";
1229
1230 if (failed == failureCodes::FAILED_STOPINST)
1231 {
1232 FatalError << stopInstance;
1233 }
1234 else // FAILED_CONSTINST
1235 {
1236 FatalError << "constant";
1237 }
1238 FatalError << exit(FatalError);
1239 }
1240
1241 return io;
1242}

1306Foam::label Foam::fileOperation::nProcs
1307(
1308 const fileName& dir,
1309 const fileName& local,
1310 const label wanted // expected nProcs. 0 if not supplied
1311) const
1312{
1313 label nProcs = 0;
1314 if (UPstream::master(comm_))
1315 {
1316 fileNameList dirNames(Foam::readDir(dir, fileName::Type::DIRECTORY));
1317
1318 // E.g. (maybe left over)
1319 // processor0
1320 // processor1
1321 // processors5
1322 // processor90 // gap!
1323 // processors10
1324 //
1325 //- if wanted is 2 then processor0,1 is enough
1326 //- if wanted is not set then return highest valid range
1327 // (processors10 in above case)
1328 //- if wanted cannot be matched (e.g. 37) return 0
1329
1330 // For marking off contiguous ranges
1331 bitSet foundDirs;
1332 if (wanted > 0)
1333 {
1334 foundDirs.resize(wanted);
1335 }
1336
1337 // Detect any processorsDDD or processorDDD
1338 label maxProc = -1;
1339 for (const fileName& dirN : dirNames)
1340 {
1341 // Analyse directory name
1342 label rNum(-1);
1343 procRangeType group;
1344 auto readProci =
1345 fileOperation::detectProcessorPath(dirN, group, &rNum);
1346
1347 maxProc = Foam::max(maxProc, readProci);
1348
1349 if (rNum > 0) // processorsDDD where DDD>0
1350 {
1351 // Use processors number
1352 maxProc = Foam::max(maxProc, rNum-1);
1353 // Mark in cache. TBD separate handling for groups?
1354 foundDirs.set(labelRange(0, rNum));
1355
1356 if (wanted == rNum)
1357 {
1358 // Exact match of processorsDDD. Direct exit.
1359 maxProc = rNum-1;
1360 foundDirs.resize(rNum);
1361 break;
1362 }
1363 }
1364 else if (readProci >= 0)
1365 {
1366 // Mark in cache
1367 foundDirs.set(readProci);
1368 }
1369 }
1370 nProcs = maxProc+1;
1371
1372 // Override with any gaps in processorDDD numbering (can never happen
1373 // with collated)
1374 if (label gapIndex = foundDirs.find_first_not(); gapIndex > 0)
1375 {
1376 nProcs = gapIndex-1;
1377 }
1378
1379 if (nProcs == 0 && Foam::isDir(dir/processorsBaseDir))
1380 {
1381 WarningInFunction
1382 << "Defunct collated naming: " << processorsBaseDir << nl
1383 << "Manually rename with the decomposition number. Eg,"
1384 << nl << nl
1385 << " mv processors processors16" << nl << nl
1386 << "...returning 1" << endl;
1387
1388 nProcs = 1;
1389 }
1390 }
1391 Pstream::broadcast(nProcs, comm_);
1392 return nProcs;
1393}

1475Foam::label Foam::fileOperation::splitProcessorPath
1476(
1477 const fileName& objPath,
1478 fileName& path,
1479 fileName& procDir,
1480 fileName& local,
1481
1482 procRangeType& group,
1483 label& nProcs
1484)
1485{
1486 // Return value
1487 label returnProci = -1;
1488
1489 // Clear out the return parameters
1490
1491 path.clear();
1492 procDir.clear();
1493 local.clear();
1494 group.clear();
1495
1496 // Invalidate detected number of processors
1497 nProcs = -1;
1498
1499 // The local processor group is read as first/last, but stored as
1500 // start/size. Empty with start=0, size=0 if no range is detected
1501
1502
1503 // Start of 'processor..' directory name (the procDir)
1504 size_t pos = 0;
1505
1506 // The slash starting the trailing (local) directory
1507 size_t slashLocal = std::string::npos;
1508
1509
1510 // Search for processor at start of string or after /processor
1511 //
1512 // 'processor(\d+)'
1513 // 'processors(\d+)'
1514 // 'processors(\d+)_(\d+)-(\d+)'
1515
1516 for
1517 (
1518 /*nil*/;
1519 (pos = objPath.find("processor", pos)) != std::string::npos;
1520 pos += 9
1521 )
1522 {
1523 if (pos > 0 && objPath[pos-1] != '/')
1524 {
1525 // Not start of string or after /processor
1526 continue;
1527 }
1528
1529 // The parse point. One past 'processor'
1530 size_t firstp = pos + 9;
1531
1532 // normal: 'processor(\d+)'
1533 // plural: 'processors(\d+)'
1534
1535 const bool plural = (objPath[firstp] == 's');
1536
1537 if (plural)
1538 {
1539 ++firstp; // Skip over the 's'
1540 }
1541 else if (!std::isdigit(objPath[firstp]))
1542 {
1543 // Non-plural version (uncollated) requires digits only
1544 continue;
1545 }
1546
1547 // The next slash indicates there is a local directory
1548 slashLocal = objPath.find('/', firstp);
1549
1550 // The last parse point is the slash, or end of string
1551 const size_t lastp =
1552 (slashLocal == std::string::npos ? objPath.size() : slashLocal);
1553
1554 if (!std::isdigit(objPath[lastp-1]))
1555 {
1556 // Must end in a digit!
1557 // This traps entries that are too short or look quite wrong
1558 // and avoid a string to int conversion that will fail anyhow
1559 continue;
1560 }
1561
1562
1563 // Match: '^processors(\d+)$' -> nProcs
1564
1565 // Match: '^processors(\d+)_(\d+)-(\d+)$'
1566 // \1 = nProcs
1567 // \2 = beg processor group
1568 // \3 = end processor group (inclusive)
1569
1570 if (plural)
1571 {
1572 int nProcsRead = 0;
1573
1574 if
1575 (
1576 parseProcsNumRange
1577 (
1578 objPath.substr(firstp, lastp-firstp),
1579 nProcsRead,
1580 group
1581 )
1582 )
1583 {
1584 // Total number of processors
1585 nProcs = nProcsRead;
1586
1587 // We are done!
1588 break;
1589 }
1590 }
1591
1592 // Single
1593 // Match: '^processor(\d+)$' -> proci
1594
1595 if
1596 (
1597 label proci = 0;
1598 Foam::read(objPath.substr(firstp, lastp-firstp), proci)
1599 && (proci >= 0)
1600 )
1601 {
1602 // Capture value of an individual processor
1603 returnProci = proci;
1604
1605 // We are done!
1606 break;
1607 }
1608 }
1609
1610 if (pos != std::string::npos)
1611 {
1612 // The split succeeded, extract the components.
1613
1614 // The leading directory
1615 if (pos > 0)
1616 {
1617 path = objPath.substr(0, pos-1);
1618 }
1619
1620 // The slash starting the trailing (local) directory
1621 if (slashLocal != std::string::npos)
1622 {
1623 procDir = objPath.substr(pos, slashLocal-pos);
1624 local = objPath.substr(slashLocal+1);
1625 }
1626 else
1627 {
1628 procDir = objPath.substr(pos);
1629 }
1630 }
1631
1632 return returnProci;
1633}
