68template<class T>
69Foam::Ostream& Foam::CompactListList<T>::writeMatrix
70(
71 Ostream& os,
72 const label shortLen
73) const
74{
75 const CompactListList<T>& mat = *this;
76
77 const auto oldFmt = os.format(IOstreamOption::ASCII);
78
79 // Write like multi-line matrix of rows (ASCII)
80 // TBD: with/without indenting?
81
82 const label nRows = mat.length();
83
84 os << nRows << nl << token::BEGIN_LIST << nl;
85
86 for (label i = 0; i < nRows; ++i)
87 {
88 mat.localList(i).writeList(os, shortLen) << nl;
89 }
90
91 os << token::END_LIST << nl;
92 os.format(oldFmt);
93
94 return os;
95}
