30template<class Type>
31Foam::tmp<Foam::Field<Type>>
32Foam::expressions::fieldExpr::parseDriver::getField
33(
34 const word& name
35) const
36{
37 bool hasPointData = false;
38
39 refPtr<expressions::exprResult> tvar;
40
41 if (hasVariable(name) && variable(name).isType<Type>())
42 {
43 tvar.cref(variable(name));
44 hasPointData = tvar().isPointData();
45 }
46
47
48 if (tvar)
49 {
50 const auto& var = tvar.cref();
51 const Field<Type>& vals = var.cref<Type>();
52
53 const label len = (hasPointData ? this->pointSize() : this->size());
54
55 if (returnReduceAnd(vals.size() == len))
56 {
57 // Return a copy of the field
58 return tmp<Field<Type>>::New(vals);
59 }
60
61 if (!var.isUniform())
62 {
63 WarningInFunction
64 << "Variable " << name
65 << " is nonuniform and does not fit the size"
66 << ". Using average" << endl;
67 }
68
69 return tmp<Field<Type>>::New(this->size(), gAverage(vals));
70 }
71
72 return nullptr;
73}
