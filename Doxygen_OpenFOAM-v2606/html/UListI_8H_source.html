65template<class T>
66inline void Foam::UList<T>::fill_uniform(Foam::zero)
67{
68 if constexpr (std::is_arithmetic_v<T>)
69 {
70 // Can dispatch with
71 // - std::execution::par_unseq
72 // - std::execution::unseq
73 std::fill_n
74 (
75 this->v_, this->size_, T(0)
76 );
77 }
78 else if constexpr (is_contiguous_v<T>)
79 {
80 // Can dispatch with
81 // - std::execution::par_unseq
82 // - std::execution::unseq
83 std::fill_n
84 (
85 this->data_bytes(), this->size_bytes(), char(0)
86 );
87 }
88 else
89 {
90 // May also have special triggers when assigning non-contiguous
91 // from zero...
92
93 const auto last = (this->v_ + this->size_);
94
95 for (auto first = this->v_; (first != last); (void)++first)
96 {
97 *first = Foam::zero{};
98 }
99 }
100}

174template<class T>
175inline void Foam::UList<T>::checkRange
176(
177 const label start,
178 const label len
179) const
180{
181 // Artificially allow the start of a zero-sized subList to be
182 // one past the end of the original list.
183 if (len)
184 {
185 if (len < 0)
186 {
187 FatalErrorInFunction
188 << "size " << len << " is negative, out of range [0,"
189 << size_ << "]\n"
190 << abort(FatalError);
191 }
192 this->checkStart(start);
193 this->checkSize(start + len);
194 }
195 else
196 {
197 // Start index needs to fall between 0 and size. One position
198 // behind the last element is allowed
199 this->checkSize(start);
200 }
201}

224inline bool Foam::UList<T>::uniform() const
225{
226 // This convenience method is mostly just used for compact output
227 // notation in the writeList() method.
228 // - requires 'operator!=' to be defined for the type.
229
230 return
231 (
232 Foam::ListPolicy::uniformity::UNIFORM
233 == Foam::ListPolicy::check_uniformity(this->v_, this->v_ + size_)
234 );

362template<class T>
363inline T& Foam::UList<T>::operator[](const label i)
364{
365 if constexpr (std::is_same_v<bool, std::remove_cv_t<T>>)
366 {
367 // Lazy evaluation - return false for out-of-range
368 // Note: strictly speaking should not be modifiable but we cannot
369 // alway control which signature (const or non-const) is called
370 if (i < 0 || i >= size_)
371 {
372 return const_cast<bool&>(Foam::pTraits<bool>::null());
373 }
374 }
375 else
376 {
377 #ifdef FULLDEBUG
378 checkIndex(i);
379 #endif
380 }
381 return v_[i];
382}

525{

540{
541 const label nBy2 = n/2;

544 {
545 Foam::Swap(list[i], list[n-1-i]);
