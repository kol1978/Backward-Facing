75void incompressibleKEpsilon::correctNut()
76{
77 //this->nut_ = Cmu_*sqr(k_)/epsilon_;
78
79 // Create 'oneField' from Cmu
80 const volConstant Cmu(this->nut_, Cmu_);
81
82 volEvaluator(this->nut_).evaluate
83 (
84 Cmu
85 * sqr(volExpr(k_))
86 / volExpr(epsilon_)
87 );
88
89 if (twoLayerTreatment_)
90 {
91 const volScalarField& y = wallDist::New(this->mesh_).y();
92
93 // Using the TKE of previous time-step to freeze the turbulent Reynolds
94 // number during the current time-step and hence freeze the separation
95 // of inner and outer layer.
96 const volScalarField Rey(mag(y*sqrt(k_.oldTime())/this->nu()));
97
98 const dimensionedScalar ClStar(0.41*pow(Cmu_, -0.75));
99 auto& lEps = *lEpsPtr_;
100 lEps = y*ClStar*(1.0 - exp(-1.0*Rey/(2*ClStar)));
101
102 const dimensionedScalar A(ReyFactor_*ReyStar_/atanh(0.98));
103 auto& lambdaEps = *lambdaEpsPtr_;
104 lambdaEps = 0.5*(1.0 + tanh((Rey - ReyStar_)/A));
105
106 // High-Re part
107 this->nut_ *= lambdaEps;
108
109 // Low-Re part
110 const scalar Amu = 70.0;
111 const volScalarField lMu(y*ClStar*(1.0 - exp(-1.0*Rey/Amu)));
112 this->nut_ += (1.0-lambdaEps)*Cmu_*lMu*sqrt(k_);
113 }
114
115 this->nut_.correctBoundaryConditions();
116 fv::options::New(this->mesh_).correct(this->nut_);
117
118 IncompressibleTurbulenceModel<Foam::transportModel>::correctNut();
119}

122tmp<fvScalarMatrix> incompressibleKEpsilon::kSource() const
123{
124 // Source term for k equation (added to RHS)
125
126 if (twoLayerTreatment_)
127 {
128 // Local references
129 const alphaField& alpha = this->alpha_;
130 const rhoField& rho = this->rho_;
131 const volScalarField::Internal& lEps = *lEpsPtr_;
132 const volScalarField& lambdaEps = *lambdaEpsPtr_;
133
134 return
135 -fvm::Sp
136 (
137 alpha()*rho()*(1.0-lambdaEps())
138 *(sqrt(k_())/lEps - epsilon_()/k_()),
139 k_
140 );
141 }
142
143 return tmp<fvScalarMatrix>::New
144 (
145 k_,
146 dimVolume*this->rho_.dimensions()*k_.dimensions()/dimTime
147 );
148}

360void incompressibleKEpsilon::correct()
361{
362 if (!this->turbulence_)
363 {
364 return;
365 }
366 // const std::chrono::high_resolution_clock::time_point t1 =
367 // std::chrono::high_resolution_clock::now();
368
369 // Sizing
370 const label nCells = this->nut_.size();
371
372 // Create some expression constants
373 const constant alpha(nCells, 1.0);
374 const constant rho(nCells, 1.0);
375 const constant Cmu(nCells, Cmu_);
376 const constant C1(nCells, C1_);
377 const constant C2(nCells, C2_);
378 const constant C3(nCells, C3_);
379 const constant C1C3(nCells, (2.0/3.0)*C1_ - C3_);
380 const constant twoThird(nCells, (2.0/3.0));
381 const constant minOne(nCells, -1.0);
382 //const constant one(nCells, 1.0);
383
384 const surfaceScalarField& alphaRhoPhi = this->alphaRhoPhi_;
385 const volVectorField& U = this->U_;
386 const volScalarField& nut = this->nut_;
387
388 fv::options& fvOptions(fv::options::New(this->mesh_));
389
390 eddyViscosity<incompressible::RASModel>::correct();
391
392 const volScalarField::Internal divU
393 (
394 fvc::div(fvc::absolute(this->phi(), U))().v()
395 );
396
397 tmp<volTensorField> tgradU = fvc::grad(U);
398 const volScalarField::Internal GbyNu
399 (
400 IOobject::scopedName(this->type(), "GbyNu"),
401 tgradU().v() && devTwoSymm(tgradU().v())
402 );
403 const volScalarField::Internal G(this->GName(), nut()*GbyNu);
404 tgradU.clear();
405
406 // Update epsilon and G at the wall
407 epsilon_.boundaryFieldRef().updateCoeffs();
408 // Push any changed cell values to coupled neighbours
409 epsilon_.boundaryFieldRef().template evaluateCoupled<coupledFvPatch>();
410
411
412 // Wrap internal fields as list expressions for convenience
413 const expr wepsilon(epsilon_.internalField());
414 const expr wk(k_.internalField());
415 const expr V(k_.mesh().V());
416
417
418 // Dissipation equation
419 // ~~~~~~~~~~~~~~~~~~~~
420
421 // Wrap zero
422 const constant constantZero(V.size(), 0.0);
423
424 // Dissipation equation
425 tmp<fvScalarMatrix> epsEqn
426 (
427 fvm::ddt(epsilon_)
428 + fvm::div(alphaRhoPhi, epsilon_)
429 - fvm::laplacian(DepsilonEff(), epsilon_)
430 ==
431 // C1_*alpha()*rho()*GbyNu*Cmu_*k_()
432 // - fvm::SuSp(((2.0/3.0)*C1_ - C3_)*alpha()*rho()*divU, epsilon_)
433 // - fvm::Sp(C2_*alpha()*rho()*epsilon_()/k_(), epsilon_)
434 epsilonSource()
435 + fvOptions(epsilon_)
436 );
437 {
438 // Implement source terms as expression templates.
439 auto& m = epsEqn.ref();
440
441 m =
442 (
443 m.expr() // Get matrix in fvMatrixExpression form
444 ==
445 Expression::Su(C1*alpha*rho*GbyNu*Cmu*k_, m)
446 - Expression::SuSp(C1C3*alpha*rho*divU, m)
447 - Expression::Sp(C2*alpha*rho*epsilon_/k_, m)
448 );
449
450 // Alternative using field. Assumes dummy dimensions.
451 // epsEqn.ref() =
452 // (
453 // epsEqn().expr()
454 // ==
455 // Expression::Su(C1*alpha*rho*GbyNu*Cmu*k_, epsilon_)
456 // - Expression::SuSp(C1C3*alpha*rho*divU, epsilon_)
457 // - Expression::Sp(C2*alpha*rho*epsilon_/k_, epsilon_)
458 // );
459 }
460
461
462
463 epsEqn.ref().relax();
464 fvOptions.constrain(epsEqn.ref());
465 epsEqn.ref().boundaryManipulate(epsilon_.boundaryFieldRef());
466 solve(epsEqn);
467 fvOptions.correct(epsilon_);
468 bound(epsilon_, this->epsilonMin_);
469
470
471 // Turbulent kinetic energy equation
472 // ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
473
474 tmp<fvScalarMatrix> kEqn
475 (
476 fvm::ddt(k_)
477 + fvm::div(alphaRhoPhi, k_)
478 - fvm::laplacian(DkEff(), k_)
479 ==
480 // G
481 // - fvm::SuSp((2.0/3.0)*divU, k_)
482 // - fvm::Sp(epsilon_()/k_(), k_)
483 kSource()
484 + fvOptions(k_)
485 );
486 {
487 // Implement source terms as expression templates.
488 auto& m = kEqn.ref();
489
490 m =
491 (
492 m.expr()
493 ==
494 Expression::Su(alpha*rho*G, m)
495 - Expression::SuSp(twoThird*alpha*rho*divU, m)
496 - Expression::Sp(alpha*rho*epsilon_/k_, m)
497 );
498 }
499
500 kEqn.ref().relax();
501 fvOptions.constrain(kEqn.ref());
502 solve(kEqn);
503 fvOptions.correct(k_);
504 bound(k_, this->kMin_);
505
506 correctNut();
507
508 // const std::chrono::high_resolution_clock::time_point t2 =
509 // std::chrono::high_resolution_clock::now();
510 // const std::chrono::duration<double> time_span = t2 - t1;
511 // time_total += time_span.count();
512 // Info<< "incompressibleKEpsilon: spent:" << time_span.count()
513 // << " total:" << time_total << endl;
514}
