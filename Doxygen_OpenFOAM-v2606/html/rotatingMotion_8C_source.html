93Foam::septernion
94Foam::solidBodyMotionFunctions::rotatingMotion::transformation() const
95{
96 scalar t = time_.value();
97
98 // Rotation around axis
99 scalar angle = omega_->integrate(0, t);
100
101 quaternion R(axis_, angle);
102 septernion TR(septernion(-origin_)*R*septernion(origin_));
103
104 DebugInFunction << "Time = " << t << " transformation: " << TR << endl;
105
106 return TR;
107}
