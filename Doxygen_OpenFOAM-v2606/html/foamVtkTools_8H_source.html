29Description
30 A collection of static methods to assist converting OpenFOAM data
31 structures into VTK internal data structures.
32
33 Remapping of the symmTensor order is required in input or output
34 directions. OpenFOAM uses (XX, XY, XZ, YY, YZ, ZZ) order,
35 VTK uses (XX, YY, ZZ, XY, YZ, XZ) order.
36
37Note
38 This file is implemented as headers-only.

82//- Bookkeeping for internal caching.
83// Retain an original copy of the geometry as well as a shallow copy
84// with the output fields.
85// The original copy is reused for different timesteps etc.
86template<class DataType>
87struct Caching
88{
89 //- The VTK geometry data type
90 typedef DataType dataType;
91
92 //- The geometry, without any cell/point data
93 vtkSmartPointer<dataType> vtkgeom;
94
95 //- The shallow-copy of geometry, plus additional data
96 vtkSmartPointer<dataType> dataset;
97
98 //- Number of points associated with the geometry
99 inline uint64_t nPoints() const
100 {
101 return vtkgeom ? vtkgeom->GetNumberOfPoints() : 0;
102 }
103
104 //- Clear geometry and dataset
105 void clearGeom()
106 {
107 vtkgeom = nullptr;
108 dataset = nullptr;
109 }
110
111 //- Return a shallow copy of vtkgeom for manipulation
112 vtkSmartPointer<dataType> getCopy() const
113 {
114 auto copy = vtkSmartPointer<dataType>::New();
115 if (vtkgeom)
116 {
117 copy->ShallowCopy(vtkgeom);
118 }
119 return copy;
120 }
121
122 //- Make a shallow copy of vtkgeom into dataset
123 void reuse()
124 {
125 dataset = vtkSmartPointer<dataType>::New();
126 if (vtkgeom)
127 {
128 dataset->ShallowCopy(vtkgeom);
129 }
130 }
131
132 //- Set the geometry and make a shallow copy to dataset
133 void set(vtkSmartPointer<dataType> geom)
134 {
135 vtkgeom = geom;
136 reuse();
137 }
138
139 //- Report basic information to output
140 void PrintSelf(std::ostream& os) const
141 {
142 os << "geom" << nl;
143 if (vtkgeom)
144 {
145 vtkgeom->PrintSelf(std::cout, vtkIndent(2));
146 }
147 else
148 {
149 os << "nullptr";
150 }
151 os << nl;
152
153 os << "copy" << nl;
154 if (dataset)
155 {
156 dataset->PrintSelf(std::cout, vtkIndent(2));
157 }
158 else
159 {
160 os << "nullptr";
161 }
162 os << nl;
163 }
164};
