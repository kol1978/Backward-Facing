100template<class T>
101template<class UnaryCondition>
102Foam::IndirectList<T> Foam::IndirectList<T>::subset
103(
104 const UList<T>& values,
105 UnaryCondition&& select,
106 const bool invert
107)
108{
109 const label len = values.size();
110
111 IndirectList<T> result(values, Foam::zero{});
112 labelList& addr = result.addressing();
113
114 addr.resize_nocopy(len);
115
116 label count = 0;
117 for (label i=0; i < len; ++i)
118 {
119 // Test on position
120 if (select(i) ? !invert : invert)
121 {
122 addr[count] = i;
123 ++count;
124 }
125 }
126 addr.resize(count);
127
128 return result;
129}

164template<class T>
165Foam::IndirectList<T> Foam::IndirectList<T>::uniq
166(
167 const UList<T>& values,
168 const bool sorted
169)
170{
171 const label len = values.size();
172
173 IndirectList<T> result(values, Foam::zero{});
174 labelList& order = result.addressing();
175
176 // Start from sorted order
177 Foam::sortedOrder(values, order);
178
179 if (len > 1)
180 {
181 label count = 0;
182 for (label i = 1; i < len; ++i)
183 {
184 // Eliminate duplicates
185 if (values[order[count]] != values[order[i]])
186 {
187 ++count;
188 order[count] = order[i];
189 }
190 }
191 ++count;
192 order.resize(count);
193
194 // Recover the original input order
195 if (!sorted)
196 {
197 Foam::sort(order);
198 }
199 }
200
201 return result;
202}
