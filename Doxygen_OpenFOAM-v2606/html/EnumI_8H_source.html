142template<class EnumType>
143template<class OS>
144inline OS& Foam::Enum<EnumType>::writeList(OS& os, const label) const
145{
146 unsigned i = 0;
147
148 os << '(';
149 for (const word& k : keys_)
150 {
151 if (i++) os << ' ';
152 os << k;
153 }
154 os << ')';
155
