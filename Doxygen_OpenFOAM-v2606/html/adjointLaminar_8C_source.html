78tmp<volSymmTensorField> adjointLaminar::devReff
79(
80 const volVectorField& U
81) const
82{

84 (
85 "devRhoReff",

94tmp<fvVectorMatrix> adjointLaminar::divDevReff(volVectorField& U) const
95{
96 return
97 (
98 - fvm::laplacian(nuEff(), U)
99 - fvc::div(nuEff()*dev(T(fvc::grad(U))))
100 );
101}
