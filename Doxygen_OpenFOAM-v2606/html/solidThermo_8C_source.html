55//- Compute harmonic-averaged face-normal alpha and kappa for
56// anisotropic zone mixtures.
57// Returns (αf, κf) where:
58// αf[face] = 1 / ((1-w)/a_own + w/a_nei) (harmonic of n·aniAlpha·n)
59// κf[face] = 1 / ((1-w)/k_own + w/k_nei) (harmonic of n·aniAlpha·Cp·n)
60std::pair<tmp<surfaceScalarField>, tmp<surfaceScalarField>>
61calcHarmonicCoeffs
62(
63 const fvMesh& mesh,
64 const volSymmTensorField& aniAlpha,
65 const volScalarField& cp
66)
67{
68 IOobject ioAlpha
69 (
70 "alphaf",
71 mesh.time().timeName(),
72 mesh,
73 IOobject::NO_READ,
74 IOobject::NO_WRITE
75 );
76
77 IOobject ioKappa
78 (
79 "kappaf",
80 mesh.time().timeName(),
81 mesh,
82 IOobject::NO_READ,
83 IOobject::NO_WRITE
84 );
85
86 // Kappa = aniAlpha * cp (for dimension checking)
87 auto tKappa = aniAlpha*cp;
88 const dimensionSet dimKappa = tKappa().dimensions();
89 const dimensionSet dimAlpha = aniAlpha.dimensions();
90
91 auto talphaf = tmp<surfaceScalarField>::New
92 (
93 ioAlpha,
94 mesh,
95 dimAlpha
96 );
97 auto tkappaf = tmp<surfaceScalarField>::New
98 (
99 ioKappa,
100 mesh,
101 dimKappa
102 );
103
104 auto& alphaf = talphaf.ref();
105 auto& kappaf = tkappaf.ref();
106
107 // Standard linear interpolation weights from the mesh
108 const surfaceScalarField& w = mesh.surfaceInterpolation::weights();
109
110 const labelUList& owner = mesh.owner();
111 const labelUList& neighbour = mesh.neighbour();
112 const surfaceVectorField& Sf = mesh.Sf();
113 const surfaceScalarField& magSf = mesh.magSf();
114
115 // Internal faces: distance-weighted harmonic mean
116 for (label facei = 0; facei < mesh.nInternalFaces(); ++facei)
117 {
118 const vector n = Sf[facei] / magSf[facei];
119
120 const scalar aOwn = n & aniAlpha[owner[facei]] & n;
121 const scalar aNei = n & aniAlpha[neighbour[facei]] & n;
122 const scalar kOwn = aOwn * cp[owner[facei]];
123 const scalar kNei = aNei * cp[neighbour[facei]];
124
125 const scalar wf = w[facei]; // weight toward neighbour
126
127 alphaf[facei] = 1.0 / ((1.0 - wf)/aOwn + wf/aNei + VSMALL);
128 kappaf[facei] = 1.0 / ((1.0 - wf)/kOwn + wf/kNei + VSMALL);
129 }
130
131 // Boundary faces
132 forAll(alphaf.boundaryField(), patchi)
133 {
134 const fvPatch& patch = mesh.boundary()[patchi];
135 const vectorField n(patch.nf());
136
137 auto& alphab = alphaf.boundaryFieldRef()[patchi];
138 auto& kappab = kappaf.boundaryFieldRef()[patchi];
139
140 if (aniAlpha.boundaryField()[patchi].coupled())
141 {
142 // Coupled patch (processor, cyclic, etc.): harmonic average
143 // using both sides, consistent with the internal-face treatment.
144 const tmp<symmTensorField> taniOwn =
145 aniAlpha.boundaryField()[patchi].patchInternalField();
146 const tmp<symmTensorField> taniNei =
147 aniAlpha.boundaryField()[patchi].patchNeighbourField();
148 const tmp<scalarField> tcpOwn =
149 cp.boundaryField()[patchi].patchInternalField();
150 const tmp<scalarField> tcpNei =
151 cp.boundaryField()[patchi].patchNeighbourField();
152 const scalarField& pw = patch.weights();
153
154 forAll(patch, facei)
155 {
156 const scalar aOwn = n[facei] & taniOwn()[facei] & n[facei];
157 const scalar aNei = n[facei] & taniNei()[facei] & n[facei];
158 const scalar kOwn = aOwn * tcpOwn()[facei];
159 const scalar kNei = aNei * tcpNei()[facei];
160 const scalar wf = pw[facei];
161
162 alphab[facei] = 1.0 / ((1.0 - wf)/aOwn + wf/aNei + VSMALL);
163 kappab[facei] = 1.0 / ((1.0 - wf)/kOwn + wf/kNei + VSMALL);
164 }
165 }
166 else
167 {
168 // Physical boundary: one-sided projection from the adjacent cell
169 const labelUList& faceCells = patch.faceCells();
170
171 forAll(patch, facei)
172 {
173 const label celli = faceCells[facei];
174 const scalar an = n[facei] & aniAlpha[celli] & n[facei];
175
176 alphab[facei] = an;
177 kappab[facei] = an * cp[celli];
178 }
179 }
180 }
181
182 return {std::move(talphaf), std::move(tkappaf)};
183}
184
185} // End anonymous namespace
186} // End namespace Foam

290Foam::tmp<Foam::fvScalarMatrix> Foam::solidThermo::heatDiffusion
291(
292 const volScalarField& betav,
293 const volScalarField& h
294) const
295{
296 if (!isZoneMixture())
297 {
298 // Standard case: single implicit laplacian
299 if (isotropic())
300 {
301 return fvm::laplacian(betav*alpha(), h, "laplacian(alpha,h)");
302 }
303 else
304 {
305 return fvm::laplacian(betav*aniAlpha(), h, "laplacian(alpha,h)");
306 }
307 }
308 else
309 {
310 // Zone mixture: add explicit correction terms.
311 // At convergence, fvm::laplacian(alpha, h) and
312 // fvc::laplacian(alpha, h) cancel, leaving
313 // -fvc::laplacian(kappa, T), which is the correct diffusion.
314 if (isotropic())
315 {
316 return
317 fvm::laplacian(betav*alpha(), h, "laplacian(alpha,h)")
318 - fvc::laplacian(betav*alpha(), h, "laplacian(alpha,h)")
319 + fvc::laplacian(betav*kappa(), T(), "laplacian(kappa,T)");
320 }
321 else
322 {
323 // Hybrid approach: harmonic normal-component + full tensor coupling
324 //
325 // The orthogonal face flux uses the harmonic mean of n·A·n for
326 // correct behaviour at material interfaces. The non-orthogonal
327 // correction uses the full face tensor (linearly interpolated) to
328 // retain off-diagonal tensor coupling.
329 //
330 // Construction of the corrected face tensor:
331 // A_f_corr = A_f_lin + (alphaf_harm - n·A_f_lin·n) · (n ⊗ n)
332 //
333 // This replaces only the normal-normal component with the harmonic
334 // value while preserving all off-diagonal entries.
335
336 const fvMesh& mesh = h.mesh();
337 const volSymmTensorField& aniAlpha = this->aniAlpha();
338
339 tmp<volScalarField> tCp = this->Cp();
340 const volScalarField& cp = tCp();
341
342 // 1. Harmonic scalar for n·A·n (correct at interfaces)
343 auto [alphaf_harm, kappaf_harm] =
344 calcHarmonicCoeffs(mesh, aniAlpha, cp);
345
346 // 2. Linearly-interpolated full tensors at faces
347 const surfaceSymmTensorField A_f_lin(fvc::interpolate(aniAlpha));
348
349 const surfaceSymmTensorField kappa_f_lin
350 (
351 fvc::interpolate(aniAlpha*cp)
352 );
353
354 // 3. Face unit normal
355 const surfaceVectorField n(mesh.Sf() / mesh.magSf());
356
357 // 4. Normal-normal component from linear interpolation
358 const surfaceScalarField nnAlpha(n & A_f_lin & n);
359
360 // 5. Corrected face tensors
361 surfaceSymmTensorField A_f_corr(A_f_lin);
362 A_f_corr += (alphaf_harm - nnAlpha) * symm(n * n);
363
364 surfaceSymmTensorField kappa_f_corr(kappa_f_lin);
365 const surfaceScalarField nnKappa(n & kappa_f_lin & n);
366 kappa_f_corr += (kappaf_harm - nnKappa) * symm(n * n);
367
368 // 6. Apply porosity factor
369 const surfaceScalarField betav_f(fvc::interpolate(betav));
370
371 // 7. Full tensor laplacians — the standard
372 // gaussLaplacianScheme now sees SfGammaCorr ≠ 0,
373 // so the non-orthogonal correction correctly includes
374 // off-diagonal tensor coupling. The nNonOrthogonalCorrectors
375 // loop in the solver operates as usual.
376 return
377 fvm::laplacian(betav_f * A_f_corr, h, "laplacian(alpha,h)")
378 - fvc::laplacian(betav_f * A_f_corr, h, "laplacian(alpha,h)")
379 + fvc::laplacian(betav_f * kappa_f_corr, T(), "laplacian(kappa,T)");
380 }
381 }
382}
