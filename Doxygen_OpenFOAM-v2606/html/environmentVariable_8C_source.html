50Foam::substitutionModels::environmentVariable::environmentVariable
51(
52 const dictionary& dict,
53 const Time& time
54)
55:
56 substitutionModel(dict, time),
57 entries_()
58{
59 // Populate entries
60 const dictionary& entriesDict = dict.subDict("entries");

79bool Foam::substitutionModels::environmentVariable::apply
80(
81 const word& key,
82 string& buffer
83) const
84{
85 if (!valid(key)) return false;
86
87 const string env(Foam::getEnv(entries_[key]));
88
89 buffer.replaceAll(keyify(key), env);
90
91 return true;
