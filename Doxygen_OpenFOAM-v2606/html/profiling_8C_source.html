46Foam::profilingInformation* Foam::profiling::create()
47{
48 // Top-level entry: reset everything
49 pool_.clear();
50 children_.clear();
51 stack_.clear();
52 times_.clear();
53
54 Information* info = new Information;
55
56 pool_.push_back(info);
57 children_.resize(pool_.size());
58 children_.back().clear(); // safety
59
60 return info;
61}

64Foam::profilingInformation* Foam::profiling::create
65(
66 profilingInformation *parent,
67 const std::string& descr
68)
69{
70 const label parentId = parent->id();
71
72 for (Information* child : children_[parentId])
73 {
74 if (descr == child->description())
75 {
76 return child; // Found existing
77 }
78 }
79
80 Information* info = new Information(parent, descr, pool_.size());
81
82 pool_.push_back(info);
83 children_.resize(pool_.size());
84 children_.back().clear(); // safety
85 children_[parentId].push_back(info);
86
87 return info;
88}

99Foam::profilingInformation* Foam::profiling::endTimer()
100{
101 Information *info = stack_.back();
102 clockValue clockval = times_.back();
103 stack_.pop_back();
104 times_.pop_back();
105
106 info->update(clockval.elapsed()); // Update elapsed time
107 info->setActive(false); // Mark as off stack
108
109 return info;
110}

185Foam::profilingInformation* Foam::profiling::New(const std::string& descr)
186{
187 Information *info = nullptr;
188
189 if (active())
190 {
191 Information *parent = singleton_->stack_.back();
192
193 info = singleton_->create(parent, descr);
194 singleton_->beginTimer(info);
195
196 if (singleton_->memInfo_)
197 {
198 singleton_->memInfo_->update();
199
200 info->maxMem_ = Foam::max
201 (
202 info->maxMem_,
203 singleton_->memInfo_->size()
204 );
205 }
206 }
207
208 return info;
209}

212void Foam::profiling::unstack(const profilingInformation *info)
213{
214 if (active() && info)
215 {
216 Information *top = singleton_->endTimer();
217
218 if (info->id() != top->id())
219 {
220 FatalErrorInFunction
221 << "Profiling information to unstack has different id than"
222 << " the top of the profiling stack" << nl
223 << " info: " << info->id() << " (" << info->description()
224 << ")\n"
225 << " top: " << top->id() << " (" << top->description()
226 << ")\n" << endl
227 << abort(FatalError);
228 }
229 }
230}

308bool Foam::profiling::writeData(Ostream& os) const
309{
310 static DynamicList<scalar> elapsed;
311
312 const clockValue now(clockValue::now());
313
314 const label nstack = stack_.size();
315
316 elapsed.resize(nstack+1); // extend for last entry, which has no child.
317
318 for (label stacki=0; stacki < nstack; ++stacki)
319 {
320 elapsed[stacki] = (now - times_[stacki]);
321 }
322 elapsed.back() = 0;
323
324 os.beginBlock("profiling");
325
326 // Active items
327 for (label stacki=0; stacki < nstack; ++stacki)
328 {
329 if (stacki) os << nl; // Extra line between entries
330
331 stack_[stacki]->write
332 (
333 os,
334 true,
335 elapsed[stacki], // elapsedTime
336 elapsed[stacki+1] // childTimes
337 );
338 }
339
340 // Non-active items
341 for (const Information& info : pool_)
342 {
343 if (!info.active())
344 {
345 os << nl;
346 info.write(os);
347 }
348 }
349
350 os.endBlock();
351
352 if (sysInfo_)
353 {
354 os << nl;
355 sysInfo_->writeEntry("sysInfo", os);
356 }
357 else
358 {
359 // Ensure that host information is available somewhere
360 os << nl;
361 os.writeEntry("host", Foam::hostName());
362 }
363
364 if (cpuInfo_)
365 {
366 os << nl;
367 cpuInfo_->writeEntry("cpuInfo", os);
368 }
369
370 if (memInfo_)
371 {
372 memInfo_->update();
373 os << nl;
374 memInfo_->writeEntry("memInfo", os);
375 }
376
377 return os.good();
378}
