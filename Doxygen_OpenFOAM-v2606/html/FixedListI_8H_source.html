232inline void Foam::FixedList<T, N>::checkStart(const label start) const
233{
234 if (start < 0 || (start && unsigned(start) >= N))
235 {
236 // Note: always accept start=0, even for zero-sized lists
237 FatalErrorInFunction
238 << "start " << start << " out of range [0," << N << ")"
239 << abort(FatalError);

268template<class T, unsigned N>
269inline bool Foam::FixedList<T, N>::uniform() const
270{
271 // This convenience method is not even used for compact output
272 // notation in the writeList() method.
273 // - requires 'operator!=' to be defined for the type.
274
275 return
276 (
277 Foam::ListPolicy::uniformity::UNIFORM
278 == Foam::ListPolicy::check_uniformity(this->v_, this->v_ + N)
279 );

339template<class T, unsigned N>
340inline void Foam::FixedList<T, N>::fill(Foam::zero)
341{
342 if constexpr (std::is_arithmetic_v<T>)
343 {
344 // Usually small enough that parallel execution is pointless...
345 std::fill_n(v_, N, T(0));
346 }
347 else
348 {
349 for (unsigned i = 0; i < N; ++i)
350 {
351 v_[i] = Foam::zero{};
352 }
353 }
354}

357template<class T, unsigned N>
358inline void Foam::FixedList<T, N>::swap(FixedList<T, N>& other)
359{
360 if (this == &other)
361 {
362 return; // Self-swap is a no-op
363 }
364
365 // Essentially std::swap_ranges
366 for (unsigned i=0; i<N; ++i)
367 {
368 Foam::Swap(v_[i], other.v_[i]);
369 }
370}
