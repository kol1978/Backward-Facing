389 // Hashing
390
391 //- The (commutative) hash value for triFace, hashes lower value first
392 inline unsigned hash_code(unsigned seed=0) const noexcept
393 {
394 // Sort smallest-first for commutative incremental hashing
395 label x = a(), y = b(), z = c();
396 if (x > y) std::swap(x, y);
397 if (y > z) std::swap(y, z);
398 if (x > y) std::swap(x, y);
399
400 Foam::Hash<label> op;
401 return op(z, op(y, op(x, seed)));
402 }
403
