39void Foam::zoneMotion::selectZonePoints(const polyMesh& mesh)
40{
41 const auto& allCellZones = mesh.cellZones();
42
43 if (!cellZoneIDs_.empty())
44 {
45 bitSet movePts(mesh.nPoints());
46
47 // Markup points associated with cell zone(s)
48 for (const label zonei : cellZoneIDs_)
49 {
50 for (const label celli : allCellZones[zonei])
51 {
52 for (const label facei : mesh.cells()[celli])
53 {
54 movePts.set(mesh.faces()[facei]);
55 }
56 }
57 }
58
59 syncTools::syncPointList(mesh, movePts, orEqOp<unsigned int>(), 0u);
60
61 pointIDs_ = movePts.sortedToc();
62
63 // No cell points selected => move all points
64 if (returnReduceAnd(pointIDs_.empty()))
65 {
66 cellZoneIDs_.clear(); // consistency
67 }
68 }
69
70 selectionMode_ =
71 (
72 cellZoneIDs_.empty()
73 ? selectionModes::smAll
74 : selectionModes::smCellZone
75 );
76}

145void Foam::zoneMotion::resetZone
146(
147 const polyMesh& mesh,
148 const wordRe& cellZoneSelection
149)
150{
151 pointIDs_.clear();
152 cellZoneIDs_.clear();
153 selectionMode_ = selectionModes::smAll;
154
155 if (!cellZoneSelection.empty())
156 {
157 const auto& allCellZones = mesh.cellZones();
158
159 // Also handles groups, multiple zones (as wordRe match) ...
160 cellZoneIDs_ = allCellZones.indices(cellZoneSelection);
161
162 if (cellZoneIDs_.empty())
163 {
164 WarningInFunction
165 << "No matching cellZones: " << cellZoneSelection << nl
166 << " Valid zones : "
167 << flatOutput(allCellZones.names()) << nl
168 << " Valid groups: "
169 << flatOutput(allCellZones.groupNames()) << nl;
170 }
171 else
172 {
173 selectZonePoints(mesh);
174 }
175 }
176}

179void Foam::zoneMotion::resetZone
180(
181 const polyMesh& mesh,
182 const dictionary& dict
183)
184{
185 pointIDs_.clear();
186 cellZoneIDs_.clear();
187 selectionMode_ = selectionModes::smAll;
188
189 int useSubset(0);
190 const auto& allCellZones = mesh.cellZones();
191
192 constexpr bool verbose = true;
193
194 // Specified cellSet?
195 labelHashSet cellIDs;
196 if
197 (
198 word cellSetName;
199 (
200 dict.readIfPresent("cellSet", cellSetName)
201 && !cellSetName.empty()
202 && cellSetName != "none" // Compat: ignore 'none' placeholder
203 )
204 )
205 {
206 if (verbose)
207 {
208 Info<< "Applying motion to cellSet: " << cellSetName << endl;
209 }
210 useSubset |= int(selectionModes::smCellSet);
211 cellIDs = cellSet::readContents(mesh, cellSetName);
212 }
213
214 // Specified cellZone(s) ?
215 if
216 (
217 wordRe cellZoneSelection;
218 (
219 dict.readIfPresent("cellZone", cellZoneSelection)
220 && !cellZoneSelection.empty()
221 && cellZoneSelection != "none" // Compat: ignore 'none' placeholder
222 )
223 )
224 {
225 if (verbose)
226 {
227 Info<< "Applying motion to cellZone: " << cellZoneSelection << endl;
228 }
229 useSubset |= int(selectionModes::smCellZone);
230
231 // Also handles groups, multiple zones (as wordRe match) ...
232 cellZoneIDs_ = allCellZones.indices(cellZoneSelection);
233
234 if (cellZoneIDs_.empty())
235 {
236 FatalIOErrorInFunction(dict)
237 << "No matching cellZones: " << cellZoneSelection << nl
238 << " Valid zones : "
239 << flatOutput(allCellZones.names()) << nl
240 << " Valid groups: "
241 << flatOutput(allCellZones.groupNames()) << nl
242 << exit(FatalIOError);
243 }
244 }
245
246 if (useSubset)
247 {
248 bitSet movePts(mesh.nPoints());
249
250 // Markup points associated with cell zones
251 for (const label zonei : cellZoneIDs_)
252 {
253 for (const label celli : mesh.cellZones()[zonei])
254 {
255 for (const label facei : mesh.cells()[celli])
256 {
257 movePts.set(mesh.faces()[facei]);
258 }
259 }
260 }
261
262 // Markup points associated with cellSet
263 for (const label celli : cellIDs)
264 {
265 for (const label facei : mesh.cells()[celli])
266 {
267 movePts.set(mesh.faces()[facei]);
268 }
269 }
270
271 syncTools::syncPointList(mesh, movePts, orEqOp<unsigned int>(), 0u);
272
273 pointIDs_ = movePts.sortedToc();
274 }
275
276
277 // No cell points selected => move all points
278 if (returnReduceAnd(pointIDs_.empty()))
279 {
280 if (verbose)
281 {
282 Info<< "Applying motion to entire mesh" << endl;
283 }
284 useSubset = 0;
285 cellZoneIDs_.clear(); // consistency
286 }
287
288 selectionMode_ = selectionModes(useSubset);
289}
