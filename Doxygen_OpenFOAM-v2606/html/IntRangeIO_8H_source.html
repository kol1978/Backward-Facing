41// Format is identical to Tuple2
42template<class T>
43Foam::Istream& Foam::operator>>
44(
45 Foam::Istream& is,
46 Foam::IntRange<T>& range
47)
48{
49 is.readBegin("IntRange");
50 is >> range.start() >> range.size();
51 is.readEnd("IntRange");
52
53 is.check(FUNCTION_NAME);
54 return is;
55}
56
57
58// Format is identical to Tuple2
59template<class T>
60Foam::Ostream& Foam::operator<<
61(
62 Foam::Ostream& os,
63 const Foam::IntRange<T>& range
64)
65{
66 os << Foam::token::BEGIN_LIST
67 << range.start() << Foam::token::SPACE
68 << range.size()
69 << Foam::token::END_LIST;
70
71 os.check(FUNCTION_NAME);
72 return os;
73}
