260 //- Helper to evaluate (=construct) an fvMatrix
261 template<class Matrix>
262 Matrix& evaluate(Matrix& m) const
263 {
264 {
265 // Assign to source.
266 source().evaluate(m.source());
267 }
268 if (hasDiag())
269 {
270 // Assign to diag. Also sets hasDiag.
271 diag().evaluate(m.diag());
272 }
273 if (hasUpper())
274 {
275 upper().evaluate(m.upper());
276 }
277 if (hasLower())
278 {
279 lower().evaluate(m.lower());
280 }
281 if (hasFaceFluxCorrection())
282 {
283 faceFluxCorrection().evaluate(*m.faceFluxCorrectionPtr());
284 }
285
286 // Do boundary
287 if (hasUpper() || hasLower())
288 {
289 auto& intCoeffs = m.internalCoeffs();
290 auto& bouCoeffs = m.boundaryCoeffs();
291 const label n = intCoeffs.size();
292
293 typedef
294 typename std::remove_reference_t<decltype(m.source())>::
295 value_type value_type;
296 typedef Field<value_type> FieldType;
297
298#if defined(__STDPAR__)
299 constexpr bool foam_offload = true;
300#else
301 constexpr bool foam_offload = false;
302#endif
303 if constexpr (foam_offload)
304 {
305 //- Parallel via ListsRefWrap. See GeometricFieldExpression.
306
307 // Split into internal and boundary expressions
308 std::vector<UpperExpr> intExprs;
309 std::vector<UpperExpr> bouExprs;
310 DynamicList<label> intID;
311
312 intExprs.reserve(n);
313 bouExprs.reserve(n);
314 intID.reserve(n);
315 for (label i = 0; i < n; ++i)
316 {
317 const auto intExpr = internalCoeffs(i);
318 if (intExpr.size())
319 {
320 intExprs.emplace_back(intExpr);
321 intID.emplace_back(i);
322 if (!intCoeffs.set(i))
323 {
324 intCoeffs.set(i, new FieldType(intExpr.size()));
325 }
326 const auto bouExpr = boundaryCoeffs(i);
327 bouExprs.emplace_back(bouExpr);
328 if (!bouCoeffs.set(i))
329 {
330 bouCoeffs.set(i, new FieldType(bouExpr.size()));
331 }
332 }
333 }
334
335 // Internal : register 'chunks' for destination container
336 ListsRefWrap<FieldType> intExpr
337 (
338 intCoeffs,
339 static_cast<const labelUList&>(intID)
340 );
341 // Internal : evaluate all at once
342 ListsRefWrap<UpperExpr>(intExprs).evaluate(intExpr);
343
344 // Boundary : register 'chunks' for destination container
345 ListsRefWrap<FieldType> bouExpr
346 (
347 bouCoeffs,
348 static_cast<const labelUList&>(intID)
349 );
350 // Boundary : evaluate all at once
351 ListsRefWrap<UpperExpr>(bouExprs).evaluate(bouExpr);
352 }
353 else
354 {
355 //- Sequential
356 for (label i = 0; i < n; ++i)
357 {
358 const auto intExpr = internalCoeffs(i);
359 if (intExpr.size())
360 {
361 if (!intCoeffs.set(i))
362 {
363 intCoeffs.set(i, new FieldType(intExpr.size()));
364 }
365 intExpr.evaluate(intCoeffs[i]);
366 const auto bouExpr = boundaryCoeffs(i);
367 if (!bouCoeffs.set(i))
368 {
369 bouCoeffs.set(i, new FieldType(bouExpr.size()));
370 }
371 bouExpr.evaluate(bouCoeffs[i]);
372 }
373 }
374 }
375 }
376
377 const_cast<dimensionSet&>(m.dimensions()) = dimensions();
378
379 return m;
380 }
