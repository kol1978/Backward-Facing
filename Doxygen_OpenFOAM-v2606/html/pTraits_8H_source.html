65// The base implementation is a pass-through to the base class.
66// Accordingly it inherits all static methods (eg, typeName etc).
67template<class Base>
68class pTraits
69:
70 public Base
71{
72public:
73
74 // Constructors
75
76 //- Copy construct from base class
77 explicit pTraits(const Base& obj)
78 :
79 Base(obj)
80 {}
81
82 //- Construct from Istream
83 explicit pTraits(Istream& is)
84 :
85 Base(is)
86 {}
87};

111//- The underlying component data type: default is pass-through.

115//- The underlying component data type for vector-space (or complex).
116// Enabled when pTraits<T>:zero has a defined type (ie, exists),
117// such as for arithmetic primitives, vector-space etc. where the concept
118// of a component type also makes sense.
119template<class T>
120struct pTraits_cmptType
121<
122 T,
123 std::void_t<decltype(pTraits<std::remove_cv_t<T>>::zero)>
124>
125{
126 using type = typename pTraits<std::remove_cv_t<T>>::cmptType;
127};

171//- Test for VectorSpace : default is false
172template<class T, class = void>
173struct is_vectorspace : std::false_type {};

175//- Test for VectorSpace : test for T::rank != 0 static member directly
176// Do not need pTraits layer since rank is defined via VectorSpace etc
177template<class T>
178struct is_vectorspace
179<
180 T,
181 std::void_t<decltype(std::remove_cv_t<T>::rank)>
182>
183:
184 std::bool_constant<std::remove_cv_t<T>::rank != 0>
185{};

188//- Test for VectorSpace and multiple components (eg, not sphericalTensor)
189//- which means it is not rotationally invariant
190template<class T, class = void>
191struct is_rotational_vectorspace : std::false_type {};
192
193//- Test for VectorSpace and multiple components (eg, not sphericalTensor)
194// Do not need pTraits layer since rank and nComponents are both
195// defined via VectorSpace
196template<class T>
197struct is_rotational_vectorspace
198<
199 T,
200 std::void_t
201 <
202 decltype(std::remove_cv_t<T>::rank),
203 decltype(std::remove_cv_t<T>::nComponents)
204 >
205>
206:
207 std::bool_constant
208 <
209 (std::remove_cv_t<T>::rank != 0)
210 && (std::remove_cv_t<T>::nComponents > 1)
211 >
212{};
213
214
215//- The is_vectorspace value of Type
216template<class T>
217inline constexpr bool is_vectorspace_v
218 = is_vectorspace<T>::value;
219
220//- The is_rotational_vectorspace value of Type
221template<class T>
222inline constexpr bool is_rotational_vectorspace_v
223 = is_rotational_vectorspace<T>::value;
224

226//- Pointer to component data (VectorSpace) or the address (regular types)
227template<class T>
228inline constexpr const auto* vectorspace_cdata(const T& obj) noexcept
229{
230 if constexpr (is_vectorspace_v<T>)
231 {
232 return obj.cdata();
233 }
234 else
235 {
236 return std::addressof(obj);
237 }
238}
239
240//- Pointer to component data (VectorSpace) or the address (regular types)
241template<class T>
242inline constexpr auto* vectorspace_data(T& obj) noexcept
243{
244 if constexpr (is_vectorspace_v<T>)
245 {
246 return obj.data();
247 }
248 else
249 {
250 return std::addressof(obj);
251 }
252}
