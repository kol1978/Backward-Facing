213

218 virtual Istream& readRaw(char* data, std::streamsize count) override;
219
220 //- Start of low-level raw binary read
221 virtual bool beginRawRead() override;
222
223 //- End of low-level raw binary read
224 virtual bool endRawRead() override { return true; }
225
226 //- Read contiguous content as raw binary with 8-byte alignment,
227 //- a no-op (also no alignment) if the \p count == 0.
228 //- If \p buffer a null pointer:
229 //- an aligned forward seek of \p count items.
230 template<class T>
231 Istream& read_binary(T* buffer, std::streamsize count);
232
233 //- Read single contiguous value as raw binary with 8-byte alignment.
234 template<class T>
235 Istream& read_binary(T& value);

237

243
244 //- Align the input position (byte boundary) for the data type.
245 //- [ADVANCED USAGE]
246 template<class T>
247 void aligng() { aligng(sizeof(T)); }
248
249 //- Set input position pointer to the absolute location, without
250 //- any EOF checks.
251 //- Does not modify the buffer content. [ADVANCED USAGE]
252 void seekg(int64_t pos);
253
254 //- The current input position (in the buffer).
255 int64_t tellg() const noexcept { return recvBufPos_; }
256
257 //- The number of characters remaining in the input buffer
258 int64_t remaining() const noexcept;
259
260 //- The total number of characters in the input buffer
261 int64_t size() const noexcept;
262
263 //- Rewind the receive stream position so that it may be read again
264 virtual void rewind() override;
265
266
267 // Print
268
269 //- Print stream description to Ostream
270 void print(Ostream& os) const override;
271};
