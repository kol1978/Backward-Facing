46static int is_supported(-1);
47
48bool Foam::memInfo::supported()
49{
50 if (is_supported < 0)
51 {
52 #ifdef __linux__
53 // This is Linux-specific!
54 std::ifstream is("/proc/meminfo");
55 is_supported = is.good();
56 #else
57 // POSIX getrusage is universally available
58 struct rusage usage;
59 is_supported = (::getrusage(RUSAGE_SELF, &usage) == 0);
60 #endif
61 }
62
63 return is_supported;
64}

67// Wrapper to get process memory using POSIX getrusage()
68// and normalize the units.
69inline static int64_t get_rusage_maxrss()
70{
71 if
72 (
73 struct rusage usage;
74 ::getrusage(RUSAGE_SELF, &usage) == 0
75 )
76 {
77 // Normalize values to kB for consistency
78 #ifdef __linux__
79 return (usage.ru_maxrss); // Linux: ru_maxrss already in kB
80 #else
81 return (usage.ru_maxrss / 1024); // BSD/MacOS: ru_maxrss is bytes
82 #endif
83 }
84 else
85 {
86 return 0;
87 }
88}

125void Foam::memInfo::populate()
126{
127 #ifdef __linux__
128 std::string line;
129
130 // This is all Linux-specific!
131
132 // "/proc/meminfo"
133 // ===========================
134 // MemTotal: 65879268 kB
135 // MemFree: 51544256 kB
136 // MemAvailable: 58999636 kB
137 // Buffers: 2116 kB
138 // ...
139 // Stop parsing when known keys have been extracted
140
141 if (std::ifstream is("/proc/meminfo"); is.good())
142 {
143 for
144 (
145 unsigned nkeys = 1;
146 nkeys && is.good() && std::getline(is, line);
147 /*nil*/
148 )
149 {
150 const auto delim = line.find(':');
151 if (delim == std::string::npos)
152 {
153 continue;
154 }
155
156 // Compare initial part of line to "Key"
157 #undef isKeyEqual
158 #define isKeyEqual(Key) (!line.compare(0, delim, Key))
159
160 // std::stol() and std::strtol()
161 // both skip whitespace before using as many digits as possible.
162 // So just skip over the ':' and let those do the rest
163
164 const char * value = (line.data() + (delim+1));
165 char *endptr = nullptr;
166
167 #undef parseValue
168 #define parseValue (std::strtol(value, &endptr, 10))
169 // Could also check for 'kB' etc ending
170
171 // ------------------
172 // Extract key: value
173 // ------------------
174
175 if (isKeyEqual("MemFree"))
176 {
177 free_ = parseValue;
178 --nkeys;
179 }
180
181 #undef isKeyEqual
182 #undef parseValue
183 }
184 }
185
186 // "/proc/PID/status"
187 // ===========================
188 // VmPeak: 15920 kB
189 // VmSize: 15916 kB
190 // VmLck: 0 kB
191 // VmPin: 0 kB
192 // VmHWM: 6972 kB
193 // VmRSS: 6972 kB
194 // ...
195 // Stop parsing when known keys have been extracted
196
197 // These units are kibi-btyes (1024)
198 if
199 (
200 std::ifstream is("/proc/" + std::to_string(Foam::pid()) + "/status");
201 is.good()
202 )
203 {
204 for
205 (
206 unsigned nkeys = 4;
207 nkeys && is.good() && std::getline(is, line);
208 /*nil*/
209 )
210 {
211 const auto delim = line.find(':');
212 if (delim == std::string::npos)
213 {
214 continue;
215 }
216
217 // Compare initial part of line to "Key"
218 #undef isKeyEqual
219 #define isKeyEqual(Key) (!line.compare(0, delim, Key))
220
221 // std::stol() and std::strtol()
222 // both skip whitespace before using as many digits as possible.
223 // So just skip over the ':' and let those do the rest
224
225 const char * value = (line.data() + (delim+1));
226 char *endptr = nullptr;
227
228 #undef parseValue
229 #define parseValue (std::strtol(value, &endptr, 10))
230 // Could also check for 'kB' etc ending
231
232
233 // ------------------
234 // Extract key: value
235 // ------------------
236
237 if (isKeyEqual("VmHWM"))
238 {
239 hwm_ = parseValue;
240 --nkeys;
241 }
242 else if (isKeyEqual("VmPeak"))
243 {
244 peak_ = parseValue;
245 --nkeys;
246 }
247 else if (isKeyEqual("VmSize"))
248 {
249 size_ = parseValue;
250 --nkeys;
251 }
252 else if (isKeyEqual("VmRSS"))
253 {
254 rss_ = parseValue;
255 --nkeys;
256 }
257
258 #undef isKeyEqual
259 #undef parseValue
260 }
261 }
262
263 #else /* __linux__ */
264
265 // Get process memory using POSIX getrusage() and also use that
266 // for the baseline information
267 {
268 hwm_ = get_rusage_maxrss();
269
270 peak_ = hwm_; // ru_maxrss is the peak memory used
271 rss_ = hwm_; //<- getrusage doesn't distinguish current RSS from peak
272 size_ = hwm_; // Virtual size estimate from peak
273 }
274
275 // Get system free memory - platform dependent
276 #ifdef __APPLE__
277 {
278 // // MacOS: use mach API to get memory information
279 // mach_port_t host = mach_host_self();
280 // mach_msg_type_number_t count = HOST_VM_INFO64_COUNT;
281 // vm_statistics64_data_t vm_stat;
282
283 //- Tends to break with Mach include headers inside c++. Might be
284 //- fixed maybe in updates.
285 // if
286 // (
287 // host_statistics64
288 // (
289 // host,
290 // HOST_VM_INFO64,
291 // reinterpret_cast<host_info64_t>(&vm_stat),
292 // &count
293 // ) == KERN_SUCCESS
294 // )
295 // {
296 // vm_page_size is in bytes, convert to kB
297 // free_ = (vm_stat.free_count * vm_page_size) / 1024;
298 // }
299
300 // Use sysctlbyname() to get the page size and free page count
301 int64_t vm_page_size = 0;
302 size_t len = sizeof(vm_page_size);
303 sysctlbyname("hw.pagesize", &vm_page_size, &len, NULL, 0);
304 len = sizeof(free_);
305 sysctlbyname("vm.page_free_count", &free_, &len, NULL, 0);
306 free_ = (free_ * vm_page_size) / 1024;
307 }
308 #endif /* __APPLE__ */
309 #endif /* __linux__ */
310}
