50Foam::fileName Foam::functionObjects::writeFile::baseFileDir() const
51{
52 // Put in undecomposed case
53 // (Note: gives problems for distributed data running)
54
55 fileName baseDir
56 (
57 fileObr_.time().globalPath()
58 / functionObject::outputPrefix
59 );
60
61 // Append mesh region name if not default region
62 if (const auto* meshPtr = isA<polyMesh>(fileObr_); meshPtr)
63 {
64 baseDir /= meshPtr->regionName();
65 }
66 baseDir.clean(); // Remove unneeded ".."
67
68 return baseDir;
69}

87Foam::autoPtr<Foam::OFstream> Foam::functionObjects::writeFile::newFile
88(
89 const fileName& fName
90) const
91{
92 autoPtr<OFstream> osPtr;
93
94 if (Pstream::master() && writeToFile_)
95 {
96 fileName outputDir(filePath(fName).path());
97
98 mkDir(outputDir);
99
100 osPtr.reset(new OFstream(outputDir/(fName.name() + ext_)));
101
102 if (!osPtr->good())
103 {
104 FatalIOErrorInFunction(osPtr()) << "Cannot open file"
105 << exit(FatalIOError);
106 }
107
108 initStream(osPtr());
109 }
110
111 return osPtr;
112}

115Foam::autoPtr<Foam::OFstream> Foam::functionObjects::writeFile::newFileAtTime
116(
117 const word& name,
118 scalar timeValue
119) const
120{
121 autoPtr<OFstream> osPtr;
122
123 if (Pstream::master() && writeToFile_)
124 {
125 if (useUserTime_)
126 {
127 timeValue = fileObr_.time().timeToUserTime(timeValue);
128 }
129
130 const word timeName = Time::timeName(timeValue);
131
132 fileName outputDir(baseFileDir()/prefix_/timeName);
133
134 mkDir(outputDir);
135
136 word fName(name);
137
138 // Check if file already exists
139 IFstream is(outputDir/(fName + ext_));
140 if (is.good())
141 {
142 fName = fName + "_" + timeName;
143 }
144
145 osPtr.reset(new OFstream(outputDir/(fName + ext_)));
146
147 if (!osPtr->good())
148 {
149 FatalIOErrorInFunction(osPtr()) << "Cannot open file"
150 << exit(FatalIOError);
151 }
152
153 initStream(osPtr());
154 }
155
156 return osPtr;
157}

246bool Foam::functionObjects::writeFile::read(const dictionary& dict)
247{
248 writePrecision_ =
249 dict.getCheckOrDefault
250 (
251 "writePrecision",
252 IOstream::defaultPrecision(),
253 labelMinMax::ge(0)
254 );
255
256 updateHeader_ = dict.getOrDefault("updateHeader", updateHeader_);
257
258 // Only write on master
259 writeToFile_ =
260 Pstream::master() && dict.getOrDefault("writeToFile", writeToFile_);
261
262 // Use user time, e.g. CA deg in preference to seconds
263 useUserTime_ = dict.getOrDefault("useUserTime", true);
264
265 return true;
266}

276Foam::OFstream& Foam::functionObjects::writeFile::file()
277{
278 if (!writeToFile_)
279 {
280 return Snull;
281 }
282
283 if (!filePtr_ && writeToFile_)
284 {
285 filePtr_ = newFileAtStartTime(fileName_);
286 }
287
288 return *filePtr_;
289}
