155inline const T& Foam::UPtrList<T>::at(const label i) const
156{
157 const T* ptr = ptrs_.get(i);
158
159 if (!ptr)

161 FatalErrorInFunction
162 << "Cannot dereference nullptr at index " << i
163 << " in range [0," << size() << ")\n"
164 << abort(FatalError);
165 }
166
167 return *ptr;
