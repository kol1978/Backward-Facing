37template<class GeoField>
38label writeEdgeFields
39(
40 vtkWriterType_edgeMesh& writer,
41 const typename GeoField::Mesh& mesh,
42 const IOobjectList& objects,
43 const bool syncPar
44)
45{
46 label count = 0;
47
48 for (const word& fieldName : objects.sortedNames<GeoField>())
49 {
50 tmp<GeoField> tfield =
51 getField<GeoField>(mesh, objects, fieldName, syncPar);
52
53 if (tfield)
54 {
55 auto tflat = faMeshTools::flattenEdgeField(tfield());
56
57 writer.write(fieldName, tflat());
58
59 tfield.clear();
60 ++count;
61 }
62 }
63
64 return count;
65}

68label writeAllEdgeFields
69(
70 vtkWriterType_edgeMesh& writer,
71 const faMesh& mesh,
72 const IOobjectList& objects,
73 const bool syncPar
74)
75{
76 label count = 0;
77 const label total = objects.size();
78 do
79 {
80 #undef doLocalWriteCode
81 #define doLocalWriteCode(Type) \
82 { \
83 typedef GeometricField<Type, faePatchField, edgeMesh> FieldType; \
84 \
85 count += writeEdgeFields<FieldType> \
86 ( \
87 writer, \
88 mesh, \
89 objects, \
90 syncPar \
91 ); \
92 if (count >= total) break; /* early exit */ \
93 }
94
95 doLocalWriteCode(scalar);
96 doLocalWriteCode(vector);
97 doLocalWriteCode(sphericalTensor);
98 doLocalWriteCode(symmTensor);
99 doLocalWriteCode(tensor);
100
101 #undef doLocalWriteCode
102 }
103 while (false);
104
105 return count;
106}
