55void Foam::patchDistanceToCell::combine(topoSet& set, const bool add) const
56{
57 labelHashSet patchIDs = mesh_.boundaryMesh().patchSet
58 (
59 selectedPatches_,
60 true, // warn if not found
61 true // use patch groups if available
62 );
63
64 if (patchIDs.empty())
65 {
66 WarningInFunction
67 << "Cannot find any patches matching "
68 << flatOutput(selectedPatches_) << nl
69 << "Valid names: " << flatOutput(mesh_.boundaryMesh().names())
70 << endl;
71 return;
72 }
73
74 if (verbose_)
75 {
76 label nFaces = 0;
77 for (const label patchi : patchIDs)
78 {
79 nFaces +=
80 returnReduce
81 (
82 mesh_.boundaryMesh()[patchi].size(),
83 sumOp<label>()
84 );
85 }
86 Info<< " Computing distance from "
87 << patchIDs.size() << " patch(es) with "
88 << nFaces << " faces total ..." << endl;
89 }
90
91 // Compute geometric distance from every cell to the selected patches
92 // using MeshWave (same algorithm as turbulence wall distance).
93 // correctWalls=true gives the exact nearest-point distance for
94 // cells directly adjacent to the patch.
95 const patchWave wave(mesh_, patchIDs, true);
96
97 if (wave.nUnset() > 0 && verbose_)
98 {
99 WarningInFunction
100 << wave.nUnset() << " cell(s) did not receive a valid distance."
101 << " These cells will not be selected." << endl;
102 }
103
104 const scalarField& dist = wave.distance();
105
106 forAll(dist, celli)
107 {
108 if (dist[celli] >= minDistance_ && dist[celli] <= distance_)
109 {
110 addOrDelete(set, celli, add);
111 }
112 }
113}

133Foam::patchDistanceToCell::patchDistanceToCell
134(
135 const polyMesh& mesh,
136 const dictionary& dict
137)
138:
139 topoSetCellSource(mesh, dict),
140 selectedPatches_(),
141 distance_(dict.getCheck<scalar>("distance", scalarMinMax::ge(0))),
142 minDistance_
143 (
144 dict.getCheckOrDefault<scalar>("minDistance", 0, scalarMinMax::ge(0))
145 )
146{
147 if (!dict.readIfPresent("patches", selectedPatches_))
148 {
149 selectedPatches_.resize(1);
150 selectedPatches_.front() = dict.get<wordRe>("patch");
151 }
152
153 if (minDistance_ > distance_)
154 {
155 FatalIOErrorInFunction(dict)
156 << "minDistance (" << minDistance_
157 << ") must not exceed distance (" << distance_ << ")"
158 << exit(FatalIOError);
159 }
160}

178void Foam::patchDistanceToCell::applyToSet
179(
180 const topoSetSource::setAction action,
181 topoSet& set
182) const
183{
184 if (action == topoSetSource::ADD || action == topoSetSource::NEW)
185 {
186 if (verbose_)
187 {
188 Info<< " Adding cells within distance ["
189 << minDistance_ << ", " << distance_
190 << "] from patches: "
191 << flatOutput(selectedPatches_) << " ..." << endl;
192 }
193
194 combine(set, true);
195 }
196 else if (action == topoSetSource::SUBTRACT)
197 {
198 if (verbose_)
199 {
200 Info<< " Removing cells within distance ["
201 << minDistance_ << ", " << distance_
202 << "] from patches: "
203 << flatOutput(selectedPatches_) << " ..." << endl;
204 }
205
206 combine(set, false);
207 }
208}
