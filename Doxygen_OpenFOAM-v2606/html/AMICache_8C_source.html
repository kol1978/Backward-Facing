43Foam::scalar Foam::AMICache::getRotationAngle(const point& globalPoint) const
44{
45 if (!coordSysPtr_)
46 {
47 FatalErrorInFunction
48 << "No co-ordinate system available for theta evaluation"
49 << abort(FatalError);
50 }
51
52
53 scalar theta = coordSysPtr_->localPosition(globalPoint).y();
54
55 // Ensure 0 < theta < 2pi
56 if (mag(theta) < cacheThetaTolerance_)
57 {
58 theta = 0;
59 }
60 else if (theta < 0)
61 {
62 theta += constant::mathematical::twoPi;
63 }
64
65 return theta;
66}

177Foam::AMICache::AMICache
178(
179 const AMICache& cache,
180 const AMIInterpolation& fineAMI,
181 const labelList& sourceRestrictAddressing,
182 const labelList& targetRestrictAddressing
183)
184:
185 size_(cache.size_),
186 rotationAxis_(cache.rotationAxis_),
187 rotationCentre_(cache.rotationCentre_),
188 nThetaStencilMax_(cache.nThetaStencilMax_),
189 forceCache_(cache.forceCache_),
190 restartTime_(cache.restartTime_),
191 complete_(cache.complete_),
192 toSource_(cache.toSource_),
193 index0_(cache.index0_),
194 index1_(cache.index1_),
195 interpWeight_(cache.interpWeight_),
196 coordSysPtr_(nullptr),
197 theta_(cache.theta_),
198 cachedSrcAddress_(cache.size_),
199 cachedSrcWeights_(cache.size_),
200 cachedSrcWeightsSum_(cache.size_),
201 cachedSrcMapPtr_(cache.size_),
202 cachedTgtAddress_(cache.size_),
203 cachedTgtWeights_(cache.size_),
204 cachedTgtWeightsSum_(cache.size_),
205 cachedTgtMapPtr_(cache.size_)
206{
207 if (size_ > 0 && fineAMI.comm() != -1)
208 {
209 for (label cachei : {index0_, index1_})
210 {
211 if (cachei == -1) continue;
212
213 scalarField dummySrcMagSf;
214 labelListList srcAddress;
215 scalarListList srcWeights;
216 scalarField srcWeightsSum;
217 autoPtr<mapDistribute> tgtMapPtr;
218
219 AMIInterpolation::agglomerate
220 (
221 cache.cachedTgtMapPtr()[cachei],
222 fineAMI.srcMagSf(),
223 cache.cachedSrcAddress()[cachei],
224 cache.cachedSrcWeights()[cachei],
225
226 sourceRestrictAddressing,
227 targetRestrictAddressing,
228
229 dummySrcMagSf,
230 srcAddress,
231 srcWeights,
232 srcWeightsSum,
233 tgtMapPtr,
234 fineAMI.comm()
235 );
236
237 scalarField dummyTgtMagSf;
238 labelListList tgtAddress;
239 scalarListList tgtWeights;
240 scalarField tgtWeightsSum;
241 autoPtr<mapDistribute> srcMapPtr;
242
243 AMIInterpolation::agglomerate
244 (
245 cache.cachedSrcMapPtr()[cachei],
246 fineAMI.tgtMagSf(),
247 cache.cachedTgtAddress()[cachei],
248 cache.cachedTgtWeights()[cachei],
249
250 targetRestrictAddressing,
251 sourceRestrictAddressing,
252
253 dummyTgtMagSf,
254 tgtAddress,
255 tgtWeights,
256 tgtWeightsSum,
257 srcMapPtr,
258 fineAMI.comm()
259 );
260
261 cachedSrcAddress_[cachei] = srcAddress;
262 cachedSrcWeights_[cachei] = srcWeights;
263 cachedSrcWeightsSum_[cachei] = srcWeightsSum;
264 cachedSrcMapPtr_[cachei] = srcMapPtr.clone();
265
266 cachedTgtAddress_[cachei] = tgtAddress;
267 cachedTgtWeights_[cachei] = tgtWeights;
268 cachedTgtWeightsSum_[cachei] = tgtWeightsSum;
269 cachedTgtMapPtr_[cachei] = tgtMapPtr.clone();
270 }
271 }
272}

277 size_(readLabel(is)),

279 rotationAxis_(is),
280 rotationCentre_(is),
281 nThetaStencilMax_(readLabel(is)),

286 toSource_(readBool(is)),
287
288 index0_(-1),
289 index1_(-1),
290 interpWeight_(0),
291 coordSysPtr_(nullptr),
292 theta_(),

294 cachedSrcWeights_(),
295 cachedSrcWeightsSum_(),
296 cachedSrcMapPtr_(),
297 cachedTgtAddress_(),
298 cachedTgtWeights_(),
299 cachedTgtWeightsSum_(),
300 cachedTgtMapPtr_()
301{
302 const bitSet goodMap(is);
303
304 if (goodMap.size())
305 {
306 is >> index0_
307 >> index1_
308 >> interpWeight_
309 >> theta_;
310
311 const bool goodCoord(readBool(is));
312 if (goodCoord)
313 {
314 coordSysPtr_.reset(new coordSystem::cylindrical(is));
315 }
316
317 is >> cachedSrcAddress_
318 >> cachedSrcWeights_
319 >> cachedSrcWeightsSum_;
320
321 cachedSrcMapPtr_.setSize(goodMap.size());
322 forAll(goodMap, cachei)
323 {
324 if (goodMap[cachei])
325 {
326 cachedSrcMapPtr_[cachei].reset(new mapDistribute(is));
327 }
328 }
329
330 is >> cachedTgtAddress_
331 >> cachedTgtWeights_
332 >> cachedTgtWeightsSum_;
333
334 cachedTgtMapPtr_.setSize(goodMap.size());
335 forAll(goodMap, cachei)
336 {
337 if (goodMap[cachei])
338 {
339 cachedTgtMapPtr_[cachei].reset(new mapDistribute(is));
340 }
341 }
342 }
343}

348void Foam::AMICache::clearCache()
349{
350 Info<< "-- clearing cache" << endl;
351
352 complete_ = false;
353
354 // Reset the interpolation indices and weight
355 index0_ = -1;
356 index1_ = -1;
357 interpWeight_ = 0;
358
359 // Clear the cached values
360 // Note: preserving the parallel maps
361 for (label cachei = 0; cachei < size_; ++cachei)
362 {
363 theta_[cachei] = GREAT;
364
365 cachedSrcAddress_[cachei].clear();
366 cachedSrcWeights_[cachei].clear();
367 cachedSrcWeightsSum_[cachei].clear();
368 cachedSrcMapPtr_[cachei].clear();
369
370 cachedTgtAddress_[cachei].clear();
371 cachedTgtWeights_[cachei].clear();
372 cachedTgtWeightsSum_[cachei].clear();
373 cachedTgtMapPtr_[cachei].clear();
374 }
375}

378void Foam::AMICache::addToCache
379(
380 const AMIInterpolation& ami,
381 const point& globalPoint
382)
383{
384 DebugPout<< "-- addToCache" << endl;
385
386
387 if (!active())
388 {
389 DebugInfo<< "-- addToCache - deactivated" << endl;
390 return;
391 }
392
393 if (!coordSysPtr_)
394 {
395 DebugInfo
396 << "Creating rotation co-ordinate system:"
397 << " rotationCentre:" << rotationCentre_
398 << " rotationAxis:" << rotationAxis_
399 << " p:" << globalPoint
400 << endl;
401
402 coordSysPtr_.reset
403 (
404 new coordSystem::cylindrical(rotationCentre_, rotationAxis_)
405 );
406 DebugPout<< "Coord sys:" << coordSysPtr_() << endl;
407 }
408
409 // Check if cache is complete
410 if (!complete_)
411 {
412 complete_ = true;
413 forAll(theta_, bini)
414 {
415 if (!validIndex(bini))
416 {
417 complete_ = false;
418 break;
419 }
420 }
421 }
422
423 if (!complete_)
424 {
425 const scalar theta = getRotationAngle(globalPoint);
426 const label bini = thetaIndex(theta);
427
428 DebugPout<< " -- bini:" << bini << " for theta:" << theta << endl;
429
430 // Check if already have entry for this bin
431 if (!validIndex(bini))
432 {
433 DebugPout<< " -- setting cache at index " << bini << endl;
434
435 // New entry
436 theta_[bini] = theta;
437
438 cachedSrcAddress_[bini] = ami.srcAddress();
439 cachedSrcWeights_[bini] = ami.srcWeights();
440 cachedSrcWeightsSum_[bini] = ami.srcWeightsSum();
441
442 if (ami.hasSrcMap())
443 {
444 cachedSrcMapPtr_[bini] = ami.srcMap().clone();
445 }
446
447 cachedTgtAddress_[bini] = ami.tgtAddress();
448 cachedTgtWeights_[bini] = ami.tgtWeights();
449 cachedTgtWeightsSum_[bini] = ami.tgtWeightsSum();
450
451 if (ami.hasTgtMap())
452 {
453 cachedTgtMapPtr_[bini] = ami.tgtMap().clone();
454 }
455 }
456 }
457}

460bool Foam::AMICache::restoreCache(const point& globalPoint, const Time& runTime)
461{
462 DebugPout<< "-- restoreCache" << endl;
463
464 if (runTime.value() >= restartTime_)
465 {
466 clearCache();
467 restartTime_ = GREAT;
468 return false;
469 }
470
471 index0_ = -1;
472 index1_ = -1;
473 interpWeight_ = -1;
474
475 if (!coordSysPtr_ || size_ == -1)
476 {
477 return false;
478 }
479
480 const scalar theta = getRotationAngle(globalPoint);
481 const label bini = thetaIndex(theta);
482
483 DebugPout<< " -- bini:" << bini << " for theta:" << theta << endl;
484
485 if (!validIndex(bini) && forceCache_)
486 {
487 DebugPout
488 << " -- no cache available at bini - forcing evaluation" << endl;
489 return false;
490 }
491
492 // Maximum angle in radians for which to search for cached bins
493 const scalar twoPi = constant::mathematical::twoPi;
494
495 if
496 (
497 validIndex(bini)
498 && (
499 mag(theta - theta_[bini]) < cacheThetaTolerance_
500 || mag(theta - twoPi - theta_[bini]) < cacheThetaTolerance_
501 )
502 )
503 {
504 // Hit cached value - no interpolation needed
505 // index1_ = -1 indicates no interpolation
506 index0_ = bini;
507 index1_ = -1;
508 interpWeight_ = 0;
509
510 DebugInfo
511 << " -- t0:" << theta_[index0_] << " theta:" << theta
512 << " i0:" << index0_ << " i1:" << index1_
513 << " w:" << interpWeight_ << endl;
514 return true;
515 }
516 else
517 {
518 // Find indices and values bracketing theta
519 const label nBin = theta_.size();
520
521 // Participating theta values and bin addresses
522 // - Note we add wrap-around values at start and end
523 DynamicList<scalar> thetap(nBin+2);
524 DynamicList<label> binAddresses(nBin+2);
525
526 // Initialise wrap-around values
527 thetap.push_back(0);
528 binAddresses.push_back(-1);
529
530 // Add valid cached values
531 // - Note if the complete_ flag is set we could skip the validIndex
532 // check or avoid the need to copy the values across
533 forAll(theta_, thetai)
534 {
535 if (validIndex(thetai))
536 {
537 thetap.push_back(theta_[thetai]);
538 binAddresses.push_back(thetai);
539 }
540 }
541
542 // Check that we have enough data points for interpolation
543 // - We added storage for lower wrap-around value, and we then need
544 // at least 2 additional values for the interpolation
545 if (thetap.size() < 3)
546 {
547 DebugPout<< " -- no cache available" << endl;
548 return false;
549 }
550
551 // Set wrap-around values if we have sufficient data
552 thetap[0] = thetap.last() - twoPi;
553 binAddresses[0] = binAddresses.last();
554 thetap.push_back(thetap[1] + twoPi);
555 binAddresses.push_back(binAddresses[1]);
556
557 // Find the lower and upper indices in the thetap list bracketing theta
558 label loweri = labelMax;
559 label upperi = labelMax;
560 forAll(thetap, i)
561 {
562 if (thetap[i] <= theta)
563 {
564 loweri = i;
565 }
566 else
567 {
568 upperi = i;
569 break;
570 }
571 }
572
573 if (loweri == labelMax || upperi == labelMax)
574 {
575 DebugPout<< " -- no cache available" << endl;
576 return false;
577 }
578
579 // Ensure distances are valid
580 if (upperi == loweri)
581 {
582 // Should not get here - go back to direct hit?
583 DebugPout
584 << " -- no cache available: theta:" << theta
585 << " lower:" << loweri << " upper:" << upperi << endl;
586 return false;
587 }
588
589 // Check that bin distances are valid
590 // - Note: bin addresses wrap around
591 label lowerBin = binAddresses[loweri];
592 label upperBin = binAddresses[upperi];
593 if (bini > binAddresses[upperi])
594 {
595 upperBin += size_;
596 }
597 if (bini < binAddresses[loweri])
598 {
599 lowerBin -= size_;
600 }
601
602 if
603 (
604 (bini - lowerBin) > nThetaStencilMax_
605 || (upperBin - bini) > nThetaStencilMax_
606 )
607 {
608 DebugPout
609 << " -- no cache available within nThetaStencilMax_:"
610 << " bini:" << bini << " lower:" << lowerBin
611 << " upper:" << upperBin << endl;
612 return false;
613 }
614
615 index0_ = binAddresses[loweri];
616 index1_ = binAddresses[upperi];
617 interpWeight_ =
618 (theta - thetap[loweri])/(thetap[upperi] - thetap[loweri]);
619
620 DebugInfo
621 << theta_.size()
622 << " -- t0:" << theta_[index0_] << " theta:" << theta
623 << " t1:" << theta_[index1_]
624 << " i0:" << index0_ << " i1:" << index1_
625 << " w:" << interpWeight_ << endl;
626
627 if (debug > 1)
628 {
629 DebugPout
630 << "Cached values for index0" << nl
631 << " - src addressing:" << cachedSrcAddress_[index0_] << nl
632 << " - src weights:" << cachedSrcWeights_[index0_] << nl
633 << " - tgt addressing:" << cachedTgtAddress_[index0_] << nl
634 << " - tgt weights:" << cachedTgtWeights_[index0_] << nl
635 << "Cached values for index1" << nl
636 << " - src addressing:" << cachedSrcAddress_[index1_] << nl
637 << " - src weights:" << cachedSrcWeights_[index1_] << nl
638 << " - tgt addressing:" << cachedTgtAddress_[index1_] << nl
639 << " - tgt weights:" << cachedTgtWeights_[index1_] << nl
640 << endl;
641 }
642
643
644 return true;
645 }
646
647 // If we get here then no valid cache found within stencil
648 DebugPout<< " -- no cache available" << endl;
649 return false;
650}

667bool Foam::AMICache::writeData(Ostream& os) const
668{
669 os << token::SPACE<< size_
670 << token::SPACE<< rotationAxis_
671 << token::SPACE<< rotationCentre_
672 << token::SPACE<< nThetaStencilMax_
673 << token::SPACE<< forceCache_
674 << token::SPACE<< restartTime_
675 << token::SPACE<< complete_
676 << token::SPACE<< toSource_;
677
678 bitSet goodMap(cachedSrcMapPtr_.size());
679 forAll(goodMap, cachei)
680 {
681 goodMap.set(cachei, cachedSrcMapPtr_[cachei].good());
682 }
683 os << token::SPACE << goodMap;
684
685 if (goodMap.size())
686 {
687 os << token::SPACE << index0_
688 << token::SPACE << index1_
689 << token::SPACE << interpWeight_
690 << token::SPACE << theta_;
691
692 os << token::SPACE << coordSysPtr_.good();
693
694 if (coordSysPtr_.good())
695 {
696 os << token::SPACE << coordSysPtr_();
697 }
698
699 os << token::SPACE << cachedSrcAddress_
700 << token::SPACE << cachedSrcWeights_
701 << token::SPACE << cachedSrcWeightsSum_;
702
703 for (const auto& index : goodMap)
704 {
705 os << token::SPACE << cachedSrcMapPtr_[index]();
706 }
707
708 os << token::SPACE << cachedTgtAddress_
709 << token::SPACE << cachedTgtWeights_
710 << token::SPACE << cachedTgtWeightsSum_;
711
712 for (const auto& index : goodMap)
713 {
714 os << token::SPACE << cachedTgtMapPtr_[index]();
715 }
716 }
717
718 return true;
719}
