37Foam::IOstreamOption Foam::IOobject::parseHeader(const dictionary& headerDict)
38{
39 IOstreamOption streamOpt; // == (ASCII, currentVersion)
40
41 // Treat "version" as optional
42 if (token tok; headerDict.readIfPresent("version", tok))
43 {
44 streamOpt.version(tok);
45 }
46
47 // Treat "format" as mandatory for now
48 streamOpt.format(headerDict.get<word>("format"));
49
50 // The "class" entry is mandatory
51 headerClassName_ = headerDict.get<word>("class");
52
53 // The "object" entry is mandatory, but not actually used here
54 (void)headerDict.get<word>("object");
55
56 // The "note" entry is optional
57 headerDict.readIfPresent("note", note_);
58
59 // The "arch" information may be missing
60 sizeofLabel_ = static_cast<unsigned char>(sizeof(label));
61 sizeofScalar_ = static_cast<unsigned char>(sizeof(scalar));
62
63 if (string arch; headerDict.readIfPresent("arch", arch))
64 {
65 if (auto val = foamVersion::labelByteSize(arch); val > 0)
66 {
67 sizeofLabel_ = static_cast<unsigned char>(val);
68 }
69 if (auto val = foamVersion::scalarByteSize(arch); val > 0)
70 {
71 sizeofScalar_ = static_cast<unsigned char>(val);
72 }
73 }
74
75 return streamOpt;
76}

79bool Foam::IOobject::readHeader(dictionary& headerDict, Istream& is)
80{
81 if (IOobject::debug)
82 {
83 InfoInFunction << "Reading header for file " << is.name() << endl;
84 }
85
86 // Check Istream not already bad
87 if (!is.good())
88 {
89 if (isReadRequired())
90 {
91 FatalIOErrorInFunction(is)
92 << " stream not open for reading essential object from file "
93 << is.relativeName()
94 << exit(FatalIOError);
95 }
96
97 if (IOobject::debug)
98 {
99 SeriousIOErrorInFunction(is)
100 << " stream not open for reading from file "
101 << is.relativeName() << endl;
102 }
103
104 return false;
105 }
106
107 token firstToken(is);
108
109 if (is.good() && firstToken.isWord("FoamFile"))
110 {
111 headerDict.read(is, false); // Read sub-dictionary content
112
113 IOstreamOption streamOpt = parseHeader(headerDict);
114
115 is.format(streamOpt.format());
116 is.version(streamOpt.version());
117 is.setLabelByteSize(sizeofLabel_);
118 is.setScalarByteSize(sizeofScalar_);
119 }
120 else
121 {
122 IOWarningInFunction(is)
123 << "First token could not be read or is not 'FoamFile'"
124 << nl << nl
125 << "Check header is of the form:" << nl << endl;
126
127 writeHeader(Info);
128
129 // Mark as not read
130 headerClassName_.clear();
131
132 return false;
133 }
134
135 // Check stream is still OK
136 objState_ = (is.good() ? objectState::GOOD : objectState::BAD);
137
138 if (IOobject::debug)
139 {
140 Info<< " .... read - state: "
141 << (objState_ == objectState::GOOD ? "good" : "bad")
142 << endl;
143
144 }
145
146 if (objState_ == objectState::BAD)
147 {
148 if (isReadRequired())
149 {
150 FatalIOErrorInFunction(is)
151 << " stream failure while reading header"
152 << " on line " << is.lineNumber()
153 << " of file " << is.relativeName()
154 << " for essential object:" << name()
155 << exit(FatalIOError);
156 }
157
158 if (IOobject::debug)
159 {
160 InfoInFunction
161 << "Stream failure while reading header"
162 << " on line " << is.lineNumber()
163 << " of file " << is.relativeName() << endl;
164 }
165
166 return false;
167 }
168
169 return true;
170}

182bool Foam::IOobject::readAndCheckHeader
183(
184 const bool isGlobal,
185 const word& typeName,
186 const bool checkType,
187 const bool search,
188 const bool verbose
189)
190{
191 // Mark as not yet read. cf, IOobject::readHeader()
192 headerClassName_.clear();
193
194 bool ok = false; // Local status
195 bool mismatch = false; // Matching expected vs read type?
196 fileName fName; // The resolved file name
197
198 // Everyone check or just master
199 const bool masterOnly
200 (
201 isGlobal
202 && IOobject::fileModificationChecking_masterOnly()
203 );
204
205 const auto& handler = Foam::fileHandler();
206
207 if (masterOnly)
208 {
209 if (UPstream::master())
210 {
211 // Force master-only header reading
212 const auto oldParRun = UPstream::parRun(false);
213
214 fName = handler.filePath(isGlobal, *this, typeName, search);
215 ok = handler.readHeader(*this, fName, typeName);
216
217 UPstream::parRun(oldParRun);
218
219 mismatch = (ok && checkType && !isHeaderClass(typeName));
220 if (mismatch)
221 {
222 ok = false;
223 }
224 }
225
226 // Make sure all processors know about the read information.
227 // Note: should ideally be inside fileHandler...
228 Pstream::broadcasts
229 (
230 UPstream::worldComm,
231 ok,
232 headerClassName_,
233 note_
234 );
235 }
236 else
237 {
238 // All read header
239 fName = handler.filePath(isGlobal, *this, typeName, search);
240 ok = handler.readHeader(*this, fName, typeName);
241
242 mismatch = (ok && checkType && !isHeaderClass(typeName));
243 if (mismatch)
244 {
245 ok = false;
246 }
247 }
248
249 // Warn if the header class does not match the expected input.
250 // For master-only reading, only reports on the master.
251 if (mismatch && verbose)
252 {
253 WarningInFunction
254 << "Unexpected class name \"" << headerClassName()
255 << "\" expected \"" << typeName
256 << "\" when reading " << fName << endl;
257 }
258
259 return ok;
260}
