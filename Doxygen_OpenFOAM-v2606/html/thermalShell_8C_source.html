92void thermalShell::solveEnergy()
93{
94 DebugInFunction << endl;
95
96 const areaScalarField rhoCph(Cp()*rho()*h_);
97
98 faScalarMatrix TEqn
99 (
100 fam::ddt(rhoCph, T_)
101 - fam::laplacian(kappa()*h_, T_)
102 ==
103 qs_
104 + qr()
105 + faOptions()(h_, rhoCph, T_)
106 );
107
108 TEqn.relax();
109
110 faOptions().constrain(TEqn);
111
112 TEqn.solve();
113
114 faOptions().correct(T_);
115}

170
171void thermalShell::evolveRegion()
172{
173 nNonOrthCorr_ = solution().get<label>("nNonOrthCorr");
174
175 for (int nonOrth = 0; nonOrth <= nNonOrthCorr_; ++nonOrth)
176 {
177 solveEnergy();
178 }
179
180 Info<< T_.name() << " min/max = " << gMinMax(T_) << endl;
181}
