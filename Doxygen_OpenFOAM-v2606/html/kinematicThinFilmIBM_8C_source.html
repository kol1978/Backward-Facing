71void kinematicThinFilmIBM::evolveRegion()
72{
73 DebugInFunction << endl;
74
75 auto& Ustar = immersedBoundary_.Ustar();
76
77 phi2s_ = fac::interpolate(h_)*phif_;
78
79 for (int oCorr=1; oCorr<=nOuterCorr_; oCorr++)
80 {
81 areaVectorField rhs(-fac::grad(pf_*h_)/rho_ + pf_*fac::grad(h_)/rho_);
82
83 // IBM force
84 areaVectorField f((UEqn(Ustar) - rhs) & Ustar);
85
86 // Ensure force is only applied to IBM faces
87 immersedBoundary_.zeroFilter(f);
88
89 pf_.storePrevIter();
90
91 faVectorMatrix UsEqn(UEqn(Uf_) == f);
92
93 UsEqn.relax();
94
95 faOptions().constrain(UsEqn);
96
97 if (momentumPredictor_)
98 {
99 solve(UsEqn == rhs);
100 }
101
102 for (int corr=1; corr<=nCorr_; corr++)
103 {
104 areaScalarField UsA(UsEqn.A());
105
106 Uf_ = UsEqn.H()/UsA;
107 Uf_.correctBoundaryConditions();
108 faOptions().correct(Uf_);
109
110 phif_ =
111 (fac::interpolate(Uf_) & regionMesh().Le())
112 - fac::interpolate(1.0/(rho_*UsA))
113 * fac::lnGrad(pf_*h_)*regionMesh().magLe()
114 + fac::interpolate(pf_/(rho_*UsA))
115 * fac::lnGrad(h_)*regionMesh().magLe();
116
117 for (int nFilm=1; nFilm<=nFilmCorr_; nFilm++)
118 {
119 faScalarMatrix hEqn
120 (
121 fam::ddt(h_)
122 + fam::div(phif_, h_)
123 ==
124 faOptions()(rho_, h_, dimVelocity)
125 + rhoSp_
126 );
127
128 hEqn.relax();
129 faOptions().constrain(hEqn);
130 hEqn.solve();
131 faOptions().correct(h_);
132
133 if (nFilm == nFilmCorr_)
134 {
135 phi2s_ = hEqn.flux();
136 }
137 }
138
139 // Bound h_
140 h_ = max(h_, h0_);
141
142 pf_ = rho_*gn_*h_ - sigma_*fac::laplacian(h_) + pnSp_ + ppf_;
143 pf_.correctBoundaryConditions();
144 pf_.relax();
145
146 Uf_ -= (1.0/(rho_*UsA))*fac::grad(pf_*h_)
147 - (pf_/(rho_*UsA))*fac::grad(h_);
148 Uf_.correctBoundaryConditions();
149 faOptions().correct(Uf_);
150 }
151 }
152
153 Info<< "Film h min/max = " << gMinMax(h_) << nl
154 << "Film mag(U) min/max = " << gMinMaxMag(Uf_) << endl;
155}
