173template<class T>
174inline void Foam::PtrList<T>::push_back(PtrList<T>&& other)
175{
176 if (this == &other)
177 {
178 FatalErrorInFunction
179 << "Attempted push_back to self"
180 << abort(FatalError);
181 }
182
183 const label idx = this->size();
184 const label len = other.size();
185
186 resize(idx + len);
187
188 for (label i = 0; i < len; ++i)
189 {
190 set(idx + i, other.release(i)); // Take pointer ownership
191 }
192
193 other.clear();
194}

284template<class T>
285inline Foam::autoPtr<T> Foam::PtrList<T>::release(const label i)
286{
287 if (i < 0 || i >= this->size())
288 {
289 return nullptr;
290 }
291
292 return autoPtr<T>(UPtrList<T>::set(i, nullptr));
293}

296template<class T>
297inline void Foam::PtrList<T>::transfer(PtrList<T>& list)
298{
299 if (FOAM_UNLIKELY(this == &list))
300 {
301 return; // Self-assignment is a no-op
302 }
303
304 (this->ptrs_).free(); // Free and nullify old pointers
305 UPtrList<T>::transfer(list);
306}
