54//- copy from GAMGAgglomerateLduAddressing but adapted for lduPrimitiveMesh
55// and to create processorGAMGInterfaces instead of GAMGInterfaces.
56Foam::autoPtr<Foam::lduPrimitiveMesh>
57Foam::procFacesGAMGProcAgglomeration::agglomerateLduAddressing
58(
59 const lduPrimitiveMesh& fineMesh,
60 const labelUList& restrictMap,
61 label& nCoarseFaces,
62 labelList& faceRestrictAddr,
63 boolList& faceFlipMap
64)
65{
66 const lduAddressing& fineMeshAddr = fineMesh.lduAddr();
67
68 const labelUList& upperAddr = fineMeshAddr.upperAddr();
69 const labelUList& lowerAddr = fineMeshAddr.lowerAddr();
70
71 const label nFineFaces = upperAddr.size();
72
73 if (min(restrictMap) == -1)
74 {
75 FatalErrorInFunction
76 << "min(restrictMap) == -1" << exit(FatalError);
77 }
78
79 if (restrictMap.size() != fineMeshAddr.size())
80 {
81 FatalErrorInFunction
82 << "restrict map does not correspond to fine level. " << endl
83 << " Sizes: restrictMap: " << restrictMap.size()
84 << " nEqns: " << fineMeshAddr.size()
85 << abort(FatalError);
86 }
87
88
89 // Get the number of coarse cells
90 const label nCoarseCells = (restrictMap.size() ? max(restrictMap) + 1 : 0);
91
92 // Storage for coarse cell neighbours and coefficients
93
94 // Guess initial maximum number of neighbours in coarse cell
95 label maxNnbrs = 10;
96
97 // Number of faces for each coarse-cell
98 labelList cCellnFaces(nCoarseCells, Zero);
99
100 // Setup initial packed storage for coarse-cell faces
101 labelList cCellFaces(maxNnbrs*nCoarseCells);
102
103 // Create face-restriction addressing
104 faceRestrictAddr.resize_nocopy(nFineFaces);
105
106 // Initial neighbour array (not in upper-triangle order)
107 labelList initCoarseNeighb(nFineFaces);
108
109 // Counter for coarse faces
110 nCoarseFaces = 0;
111
112 // Loop through all fine faces
113 forAll(upperAddr, fineFacei)
114 {
115 label rmUpperAddr = restrictMap[upperAddr[fineFacei]];
116 label rmLowerAddr = restrictMap[lowerAddr[fineFacei]];
117
118 if (rmUpperAddr == rmLowerAddr)
119 {
120 // For each fine face inside of a coarse cell keep the address
121 // of the cell corresponding to the face in the faceRestrictAddr
122 // as a negative index
123 faceRestrictAddr[fineFacei] = -(rmUpperAddr + 1);
124 }
125 else
126 {
127 // this face is a part of a coarse face
128
129 label cOwn = rmUpperAddr;
130 label cNei = rmLowerAddr;
131
132 // get coarse owner and neighbour
133 if (rmUpperAddr > rmLowerAddr)
134 {
135 cOwn = rmLowerAddr;
136 cNei = rmUpperAddr;
137 }
138
139 // check the neighbour to see if this face has already been found
140 label* ccFaces = &cCellFaces[maxNnbrs*cOwn];
141
142 bool nbrFound = false;
143 label& ccnFaces = cCellnFaces[cOwn];
144
145 for (int i=0; i<ccnFaces; i++)
146 {
147 if (initCoarseNeighb[ccFaces[i]] == cNei)
148 {
149 nbrFound = true;
150 faceRestrictAddr[fineFacei] = ccFaces[i];
151 break;
152 }
153 }
154
155 if (!nbrFound)
156 {
157 if (ccnFaces >= maxNnbrs)
158 {
159 label oldMaxNnbrs = maxNnbrs;
160 maxNnbrs *= 2;
161
162 cCellFaces.setSize(maxNnbrs*nCoarseCells);
163
164 forAllReverse(cCellnFaces, i)
165 {
166 label* oldCcNbrs = &cCellFaces[oldMaxNnbrs*i];
167 label* newCcNbrs = &cCellFaces[maxNnbrs*i];
168
169 for (int j=0; j<cCellnFaces[i]; j++)
170 {
171 newCcNbrs[j] = oldCcNbrs[j];
172 }
173 }
174
175 ccFaces = &cCellFaces[maxNnbrs*cOwn];
176 }
177
178 ccFaces[ccnFaces] = nCoarseFaces;
179 initCoarseNeighb[nCoarseFaces] = cNei;
180 faceRestrictAddr[fineFacei] = nCoarseFaces;
181 ccnFaces++;
182
183 // new coarse face created
184 nCoarseFaces++;
185 }
186 }
187 } // end for all fine faces
188
189
190 // Renumber into upper-triangular order
191
192 // All coarse owner-neighbour storage
193 labelList coarseOwner(nCoarseFaces);
194 labelList coarseNeighbour(nCoarseFaces);
195 labelList coarseFaceMap(nCoarseFaces);
196
197 label coarseFacei = 0;
198
199 forAll(cCellnFaces, cci)
200 {
201 label* cFaces = &cCellFaces[maxNnbrs*cci];
202 label ccnFaces = cCellnFaces[cci];
203
204 for (int i=0; i<ccnFaces; i++)
205 {
206 coarseOwner[coarseFacei] = cci;
207 coarseNeighbour[coarseFacei] = initCoarseNeighb[cFaces[i]];
208 coarseFaceMap[cFaces[i]] = coarseFacei;
209 coarseFacei++;
210 }
211 }
212
213 forAll(faceRestrictAddr, fineFacei)
214 {
215 if (faceRestrictAddr[fineFacei] >= 0)
216 {
217 faceRestrictAddr[fineFacei] =
218 coarseFaceMap[faceRestrictAddr[fineFacei]];
219 }
220 }
221
222
223 // Create face-flip status
224 faceFlipMap.setSize(nFineFaces, false);
225
226 forAll(faceRestrictAddr, fineFacei)
227 {
228 label coarseFacei = faceRestrictAddr[fineFacei];
229
230 if (coarseFacei >= 0)
231 {
232 // Maps to coarse face
233 label cOwn = coarseOwner[coarseFacei];
234 label cNei = coarseNeighbour[coarseFacei];
235
236 label rmUpperAddr = restrictMap[upperAddr[fineFacei]];
237 label rmLowerAddr = restrictMap[lowerAddr[fineFacei]];
238
239 if (cOwn == rmUpperAddr && cNei == rmLowerAddr)
240 {
241 faceFlipMap[fineFacei] = true;
242 }
243 else if (cOwn == rmLowerAddr && cNei == rmUpperAddr)
244 {
245 //faceFlipMap[fineFacei] = false;
246 }
247 else
248 {
249 FatalErrorInFunction
250 << "problem."
251 << " fineFacei:" << fineFacei
252 << " rmUpperAddr:" << rmUpperAddr
253 << " rmLowerAddr:" << rmLowerAddr
254 << " coarseFacei:" << coarseFacei
255 << " cOwn:" << cOwn
256 << " cNei:" << cNei
257 << exit(FatalError);
258 }
259 }
260 }
261
262
263
264 // Clear the temporary storage for the coarse cell data
265 cCellnFaces.clear();
266 cCellFaces.clear();
267 initCoarseNeighb.clear();
268 coarseFaceMap.clear();
269
270
271 // Create coarse-level interfaces
272
273 // Get reference to fine-level interfaces
274 const auto& fineInterfaces = fineMesh.rawInterfaces();
275
276 const label startOfRequests = UPstream::nRequests();
277
278 // Initialise transfer of restrict addressing on the interface
279 // The finest mesh uses patchAddr from the original lduAdressing.
280 // the coarser levels create their own adressing for faceCells
281 forAll(fineInterfaces, inti)
282 {
283 if (fineInterfaces.set(inti))
284 {
285 fineInterfaces[inti].initInternalFieldTransfer
286 (
287 Pstream::commsTypes::nonBlocking,
288 restrictMap,
289 fineMeshAddr.patchAddr(inti)
290 );
291 }
292 }
293
294 UPstream::waitRequests(startOfRequests);
295
296
297 // Add the coarse level
298 autoPtr<lduPrimitiveMesh> coarseMeshPtr
299 (
300 new lduPrimitiveMesh
301 (
302 nCoarseCells,
303 coarseOwner,
304 coarseNeighbour,
305 fineMesh.comm(),
306 true
307 )
308 );
309
310 lduInterfacePtrsList coarseInterfaces(fineInterfaces.size());
311
312 forAll(fineInterfaces, inti)
313 {
314 if (fineInterfaces.set(inti))
315 {
316 tmp<labelField> restrictMapInternalField;
317
318 restrictMapInternalField =
319 fineInterfaces[inti].interfaceInternalField
320 (
321 restrictMap,
322 fineMeshAddr.patchAddr(inti)
323 );
324
325 tmp<labelField> nbrRestrictMapInternalField =
326 fineInterfaces[inti].internalFieldTransfer
327 (
328 Pstream::commsTypes::nonBlocking,
329 restrictMap
330 );
331
332 coarseInterfaces.set
333 (
334 inti,
335 GAMGInterface::New
336 (
337 inti,
338 fineInterfaces, // fineMesh.rawInterfaces(),
339 fineInterfaces[inti],
340 restrictMapInternalField(),
341 nbrRestrictMapInternalField(),
342 0, // fineLevelIndex,
343 fineMesh.comm()
344 ).ptr()
345 );
346
347 /* Same as below:
348 coarseInterfaces.set
349 (
350 inti,
351 GAMGInterface::New
352 (
353 inti,
354 fineMesh.rawInterfaces(),
355 fineInterfaces[inti],
356 fineInterfaces[inti].interfaceInternalField(restrictMap),
357 fineInterfaces[inti].internalFieldTransfer
358 (
359 Pstream::commsTypes::nonBlocking,
360 restrictMap
361 ),
362 fineLevelIndex,
363 fineMesh.comm()
364 ).ptr()
365 );
366 */
367 }
368 }
369
370 coarseMeshPtr().addInterfaces
371 (
372 coarseInterfaces,
373 lduPrimitiveMesh::nonBlockingSchedule<processorGAMGInterface>
374 (
375 coarseInterfaces
376 )
377 );
378
379 if (debug & 2)
380 {
381 const auto& coarseAddr = coarseMeshPtr().lduAddr();
382
383 Pout<< "GAMGAgglomeration :"
384 // << " agglomerated level " << fineLevelIndex
385 << " from nCells:" << fineMeshAddr.size()
386 << " nFaces:" << upperAddr.size()
387 << " to nCells:" << nCoarseCells
388 << " nFaces:" << nCoarseFaces << nl
389 << " lower:" << flatOutput(coarseAddr.lowerAddr()) << nl
390 << " upper:" << flatOutput(coarseAddr.upperAddr()) << nl
391 << endl;
392 }
393
394 return coarseMeshPtr;
395}

398Foam::autoPtr<Foam::lduPrimitiveMesh>
399Foam::procFacesGAMGProcAgglomeration::singleCellMesh
400(
401 const label singleCellMeshComm,
402 const lduMesh& mesh,
403 scalarField& faceWeights
404) const
405{
406 // Return processor-connectivity as an ldu mesh with the weights
407 // according to the number of faces on the processor interfaces.
408 // Note: valid on master only
409 List<Map<label>> procFaces(UPstream::nProcs(mesh.comm()));
410 Map<label>& myNeighbours = procFaces[UPstream::myProcNo(mesh.comm())];
411
412 {
413 const lduInterfacePtrsList interfaces(mesh.interfaces());
414 forAll(interfaces, intI)
415 {
416 if (interfaces.set(intI))
417 {
418 const processorLduInterface& pp =
419 refCast<const processorLduInterface>
420 (
421 interfaces[intI]
422 );
423 label size = interfaces[intI].faceCells().size();
424 myNeighbours.insert(pp.neighbProcNo(), size);
425 }
426 }
427 }
428
429 Pstream::allGatherList(procFaces, Pstream::msgType(), mesh.comm());
430
431 autoPtr<lduPrimitiveMesh> singleCellMeshPtr;
432
433 if (Pstream::master(mesh.comm()))
434 {
435 // I am master
436 label nCells = Pstream::nProcs(mesh.comm());
437
438 DynamicList<label> l(3*nCells);
439 DynamicList<label> u(3*nCells);
440 DynamicList<scalar> weight(3*nCells);
441
442 DynamicList<label> nbrs;
443 DynamicList<scalar> weights;
444
445 forAll(procFaces, proci)
446 {
447 const Map<label>& neighbours = procFaces[proci];
448
449 // Add all the higher processors
450 nbrs.clear();
451 weights.clear();
452 forAllConstIters(neighbours, iter)
453 {
454 if (iter.key() > proci)
455 {
456 nbrs.append(iter.key());
457 weights.append(iter());
458 }
459 sort(nbrs);
460 forAll(nbrs, i)
461 {
462 l.append(proci);
463 u.append(nbrs[i]);
464 weight.append(weights[i]);
465 }
466 }
467 }
468
469 faceWeights.transfer(weight);
470
471 PtrList<const lduInterface> primitiveInterfaces(0);
472 const lduSchedule ps(0);
473
474 singleCellMeshPtr.reset
475 (
476 new lduPrimitiveMesh
477 (
478 nCells,
479 l,
480 u,
481 primitiveInterfaces,
482 ps,
483 singleCellMeshComm
484 )
485 );
486 }
487 return singleCellMeshPtr;
488}

491Foam::tmp<Foam::labelField>
492Foam::procFacesGAMGProcAgglomeration::processorAgglomeration
493(
494 const lduMesh& mesh,
495 const label nCoarsest
496) const
497{
498 UPstream::communicator singleCellMeshComm
499 (
500 mesh.comm(),
501 labelRange(1) // Processor 0 only
502 );
503
504 // Construct ldu mesh from processor connectivity
505 scalarField faceWeights;
506 autoPtr<lduPrimitiveMesh> singleCellMeshPtr
507 (
508 singleCellMesh
509 (
510 singleCellMeshComm.comm(),
511 mesh,
512 faceWeights
513 )
514 );
515
516 auto tfineToCoarse = tmp<labelField>::New();
517 auto& fineToCoarse = tfineToCoarse.ref();
518
519 if (singleCellMeshPtr)
520 {
521 // On master call the pair-wise agglomerator
522 // Note: no need for renumbering since the cells are processors
523 // and there are no algorithms that rely on the ordering of
524 // the processors.
525 label nCoarseProcs;
526 fineToCoarse = pairGAMGAgglomeration::agglomerate
527 (
528 nCoarseProcs,
529 singleCellMeshPtr(),
530 faceWeights,
531 false // no need for renumber
532 );
533
534 // Keep on agglomerating until we've reached the limit. (nCoarsest is
535 // labelMax it no iteration needed)
536 labelList localFineToCoarse(fineToCoarse);
537 while (nCoarseProcs > nCoarsest)
538 {
539 label nCoarseFaces;
540 labelList faceRestrictAddr;
541 boolList faceFlipMap;
542 singleCellMeshPtr = agglomerateLduAddressing
543 (
544 singleCellMeshPtr(),
545 localFineToCoarse,
546 nCoarseFaces,
547 faceRestrictAddr,
548 faceFlipMap
549 );
550
551 // Accumulate face-weights
552 scalarField coarseFaceWeights(nCoarseFaces, 0.0);
553 forAll(faceWeights, facei)
554 {
555 label coarseFacei = faceRestrictAddr[facei];
556 if (coarseFacei >= 0)
557 {
558 coarseFaceWeights[coarseFacei] += faceWeights[facei];
559 }
560 }
561
562 // Use new face weights to agglomerate cells.
563 localFineToCoarse = pairGAMGAgglomeration::agglomerate
564 (
565 nCoarseProcs,
566 singleCellMeshPtr(),
567 coarseFaceWeights,
568 false // no need for renumber
569 );
570 faceWeights.transfer(coarseFaceWeights);
571
572 // Update ultimate agglomeration
573 forAll(fineToCoarse, celli)
574 {
575 fineToCoarse[celli] = localFineToCoarse[fineToCoarse[celli]];
576 }
577 }
578
579 labelList coarseToMaster(nCoarseProcs, labelMax);
580 forAll(fineToCoarse, celli)
581 {
582 label coarseI = fineToCoarse[celli];
583 coarseToMaster[coarseI] = min(coarseToMaster[coarseI], celli);
584 }
585
586 // Sort according to master and redo restriction
587 labelList newToOld(sortedOrder(coarseToMaster));
588 labelList oldToNew(invert(newToOld.size(), newToOld));
589
590 fineToCoarse = labelUIndList(oldToNew, fineToCoarse)();
591 }
592
593 Pstream::broadcast(fineToCoarse, mesh.comm());
594
595 return tfineToCoarse;
596}

599bool Foam::procFacesGAMGProcAgglomeration::doProcessorAgglomeration
600(
601 const lduMesh& mesh
602) const
603{
604 // Check the need for further agglomeration on any processors
605 const label nCells = mesh.lduAddr().size();
606 bool doAgg =
607 (
608 nMasters_ < labelMax
609 ? UPstream::nProcs(mesh.comm()) > nMasters_
610 : nCells < nAgglomeratingCells_
611 );
612 UPstream::reduceOr(doAgg, mesh.comm());
613 return doAgg;
614}

675bool Foam::procFacesGAMGProcAgglomeration::agglomerate
676(
677 const label fineLevelIndex,
678 const label nCoarsest
679)
680{
681 if (agglom_.hasMeshLevel(fineLevelIndex))
682 {
683 // Get the fine mesh
684 const lduMesh& levelMesh = agglom_.meshLevel(fineLevelIndex);
685 const label levelComm = levelMesh.comm();
686 label nProcs = UPstream::nProcs(levelComm);
687
688 if (nProcs > 1 && doProcessorAgglomeration(levelMesh))
689 {
690 tmp<labelField> tprocAgglomMap
691 (
692 processorAgglomeration(levelMesh, nCoarsest)
693 );
694 const labelField& procAgglomMap = tprocAgglomMap();
695
696 // Master processor
697 labelList masterProcs;
698 // Local processors that agglomerate. agglomProcIDs[0] is in
699 // masterProc.
700 List<label> agglomProcIDs;
701 GAMGAgglomeration::calculateRegionMaster
702 (
703 levelComm,
704 procAgglomMap,
705 masterProcs,
706 agglomProcIDs
707 );
708
709 if (debug)
710 {
711 if (masterProcs.size())
712 {
713 labelListList masterToProcs
714 (
715 invertOneToMany
716 (
717 masterProcs.size(),
718 procAgglomMap
719 )
720 );
721 Info<< typeName << " : agglomerating" << nl
722 << "\tmaster\tnProcs\tprocIDs" << endl;
723 for (const auto& p : masterToProcs)
724 {
725 Info<< '\t' << p[0]
726 << "\t\t" << p.size()
727 << "\t\t"
728 << flatOutput(SubList<label>(p, p.size()-1, 1))
729 << endl;
730 }
731 }
732 }
733
734 // Communicator for the processor-agglomerated matrix
735 comms_.push_back
736 (
737 UPstream::newCommunicator
738 (
739 levelComm,
740 masterProcs
741 )
742 );
743
744 // Use processor agglomeration maps to do the actual
745 // collecting.
746 GAMGProcAgglomeration::agglomerate
747 (
748 fineLevelIndex,
749 procAgglomMap,
750 masterProcs,
751 agglomProcIDs,
752 comms_.back()
753 );
754 }
755 }
756 return true;
757}

760bool Foam::procFacesGAMGProcAgglomeration::agglomerate()
761{
762 if (debug)
763 {
764 Pout<< nl << "Starting mesh overview" << endl;
765 printStats(Pout, agglom_);
766 }
767
768 if (agglom_.size() >= 1)
769 {
770 if (nMasters_ < labelMax)
771 {
772 // Do coarsest level only (see masterCoarsestGAMGProcAgglomeration)
773 const label fineLevelIndex = agglom_.size()-1;
774 // Agglomerate multiple times until the number of master processors
775 // is reached
776 agglomerate(fineLevelIndex, nMasters_);
777 }
778 else
779 {
780 // Apply processor agglomeration on all levels except the finest
781 for
782 (
783 label fineLevelIndex = 2;
784 fineLevelIndex < agglom_.size();
785 fineLevelIndex++
786 )
787 {
788 // Agglomerate only once
789 agglomerate(fineLevelIndex, labelMax);
790 }
791 }
792 }
793
794
795 // Optionally restart local agglomeration
796 // (from masterCoarsest)
797 if (nCellsInMasterLevel_ > 0)
798 {
799 const label levelI = agglom_.size();
800 if (agglom_.hasMeshLevel(levelI))
801 {
802 const lduMesh& fineMesh = agglom_.meshLevel(levelI);
803 const auto& addr = fineMesh.lduAddr();
804 const scalarField weights
805 (
806 addr.lowerAddr().size(),
807 1.0
808 );
809 agglom_.agglomerate
810 (
811 nCellsInMasterLevel_,
812 levelI,
813 weights,
814 false
815 );
816 }
817 }
818
819
820 // Print a bit
821 if (debug)
822 {
823 Pout<< nl << "Agglomerated mesh overview" << endl;
824 printStats(Pout, agglom_);
825 }
826
827 return true;
828}
