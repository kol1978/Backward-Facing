28Class
29 Foam::IndirectListAddressing

31Description
32 A class for storing list addressing (labels, strides etc),
33 which is normally used by IndirectList.
34 Preferable to use with private inheritance by inheriting classes.
35
36Class
37 Foam::IndirectListBase
38
39Description
40 Base for lists with indirect addressing, templated on the list contents
41 type and the addressing type. Storage for both values and addressing
42 is held outside of the class.

316 //- A non-const iterator for an indirect list
317 // Only supports forward prefix increment, since the addressing
318 // may/may not support postfix or decrement.
319 class iterator
320 {
321 T* data_;
322 typename addressing_type::const_iterator iter_;
323
324 public:
325
326 using difference_type = label;
327 using value_type = T;
328 using pointer = T*;
329 using reference = T&;
330 using iterator_category = std::forward_iterator_tag;
331
332 //- Construct from list and addressing iterator
333 iterator
334 (
335 UList<T>& list,
336 typename addressing_type::const_iterator addrIter
337 )
338 :
339 data_(list.data()),
340 iter_(addrIter)
341 {}
342
343 reference operator*() const { return *(data_ + *iter_); }
344
345 iterator& operator++()
346 {
347 ++iter_;
348 return *this;
349 }
350
351 bool operator==(const iterator& rhs) const
352 {
353 return (iter_ == rhs.iter_);
354 }
355
356 bool operator!=(const iterator& rhs) const
357 {
358 return (iter_ != rhs.iter_);
359 }
360 };
