53Foam::DynamicList<Foam::label> Foam::topOVariablesBase::faceFaces
54(
55 const label facei
56) const
57{
58 const fvMesh& mesh = zones_.mesh();
59 const labelListList& edgeFaces = mesh.edgeFaces();
60 DynamicList<label> neighs;
61 if (!mesh.isInternalFace(facei))
62 {
63 const labelList& faceEdges = mesh.faceEdges()[facei];
64 for (const label edgei: faceEdges)
65 {
66 const labelList& edgeIFaces = edgeFaces[edgei];
67 for (const label neiFacei : edgeIFaces)
68 {
69 if (neiFacei != facei && !mesh.isInternalFace(neiFacei))
70 {
71 const label patchi =
72 mesh.boundaryMesh().whichPatch(neiFacei);
73 if (!isA<emptyFvPatch>(mesh.boundary()[patchi]))
74 {
75 neighs.push_back(neiFacei);
76 }
77 }
78 }
79 }
80 }
81 return neighs;
82}

85bool Foam::topOVariablesBase::addCutBoundaryFaceToIsoline
86(
87 const label facei,
88 const cutFaceIso& cutFace,
89 DynamicList<vector>& isoSurfPts,
90 DynamicList<face>& isoSurfFaces,
91 DynamicList<label>& zoneIDs,
92 List<DynamicList<label>>& cuttingFacesPerMeshFace
93) const
94{
95 const fvMesh& mesh = zones_.mesh();
96 // If this is a boundary face being cut, append it to the iso-surface
97 if (!mesh.isInternalFace(facei))
98 {
99 const label cutPatchi = mesh.boundaryMesh().whichPatch(facei);
100 const fvPatch& cutPatch = mesh.boundary()[cutPatchi];
101 if (!isA<coupledFvPatch>(cutPatch) && !isA<emptyFvPatch>(cutPatch))
102 {
103 if
104 (
105 addCuttingFaceToIsoline
106 (
107 cutFace.subFacePoints(),
108 cutPatchi,
109 faceFaces(facei),
110 cuttingFacesPerMeshFace,
111 isoSurfPts,
112 isoSurfFaces,
113 zoneIDs
114 )
115 )
116 {
117 cuttingFacesPerMeshFace[facei].push_back
118 (
119 isoSurfFaces.size() - 1
120 );
121 return true;
122 }
123
124 }
125 }
126 return false;
127}

130bool Foam::topOVariablesBase::addCuttingFaceToIsoline
131(
132 const DynamicList<point>& facePoints,
133 const label nSerialPatches,
134 const DynamicList<label>& cellCutFaces,
135 const List<DynamicList<label>>& cuttingFacesPerMeshFace,
136 DynamicList<vector>& isoSurfPts,
137 DynamicList<face>& isoSurfFaces,
138 DynamicList<label>& zoneIDs
139) const
140{
141 if (facePoints.size() > 2)
142 {
143 // Check whether any of points of the new iso-surface face are already
144 // present in the surface. To reduce the number of comparisons, only
145 // points on iso-surface faces cutting the current cell are checked
146 labelList uniquePointIDs(facePoints.size(), -1);
147 DynamicList<point> uniqueFacePoints(facePoints.size());
148 DynamicList<label> uniqueFacePointEdges(facePoints.size());
149 label addedPoints = Zero;
150 forAll(facePoints, pi)
151 {
152 bool foundInNei = false;
153 for (const label cutMeshFacei : cellCutFaces)
154 {
155 if
156 (
157 isDuplicatePoint
158 (
159 pi,
160 facePoints[pi],
161 cuttingFacesPerMeshFace[cutMeshFacei],
162 isoSurfPts,
163 isoSurfFaces,
164 uniquePointIDs
165 )
166 )
167 {
168 foundInNei = true;
169 break;
170 }
171 }
172 if (!foundInNei)
173 {
174 uniquePointIDs[pi] = isoSurfPts.size() + addedPoints;
175 ++addedPoints;
176 uniqueFacePoints.push_back(facePoints[pi]);
177 }
178 }
179
180 face isoFace(uniquePointIDs);
181 isoSurfPts.append(uniqueFacePoints);
182 isoSurfFaces.append(isoFace);
183 zoneIDs.append(nSerialPatches);
184
185 return true;
186 }
187 return false;
188}

191bool Foam::topOVariablesBase::isDuplicatePoint
192(
193 const label pointID,
194 const vector& pointi,
195 const DynamicList<label>& cuttingFaces,
196 const DynamicList<point>& isoSurfPts,
197 const DynamicList<face>& isoSurfFaces,
198 labelList& uniquePointIDs
199) const
200{
201 for (const label cuttingFacei : cuttingFaces)
202 {
203 const face& cuttingFace = isoSurfFaces[cuttingFacei];
204 for (const label neiPi : cuttingFace)
205 {
206 if (mag(pointi - isoSurfPts[neiPi]) < SMALL)
207 {
208 uniquePointIDs[pointID] = neiPi;
209 return true;
210 }
211 }
212 }
213 return false;
214}

217void Foam::topOVariablesBase::addBoundaryFacesToIsoline
218(
219 const pointScalarField& pointY,
220 const Map<label>& addedFaces,
221 const scalar isoValue,
222 DynamicList<vector>& isoSurfPts,
223 DynamicList<face>& isoSurfFaces,
224 DynamicList<label>& zoneIDs,
225 label& nChangedFaces,
226 labelList& changedFaces,
227 List<wallPointData<label>>& changedFacesInfo,
228 labelList& changedFaceToCutFace,
229 List<DynamicList<label>>& cuttingFacesPerMeshFace
230)
231{
232 const fvMesh& mesh = zones_.mesh();
233 const pointField& points = mesh.points();
234 const faceList& faces = mesh.faces();
235 forAll(mesh.boundary(), patchi)
236 {
237 const fvPatch& patch = mesh.boundary()[patchi];
238 bool isWall = isA<wallFvPatch>(patch);
239 if (!isA<emptyFvPatch>(patch) && !isA<coupledFvPatch>(patch))
240 {
241 const label start = patch.start();
242 forAll(patch, facei)
243 {
244 const label gFacei = start + facei;
245 // On rare occasions, a face value might be negative but all
246 // point values are positive. In such a case, the face won't be
247 // cut and is seen as a fluid face. Hence, the point values are
248 // used to determine the status of the boundary face.
249 bool isFluid(true);
250 for (const label pti : mesh.faces()[gFacei])
251 {
252 isFluid = isFluid && (pointY[pti] >= isoValue);
253 }
254 if (isFluid && !addedFaces.found(gFacei))
255 {
256 // Insert wall faces if they belong to the outer boundary
257 if (isWall)
258 {
259 // Mesh face to changedFace addressing
260 meshFaceToChangedFace_.insert(gFacei, nChangedFaces);
261 // Set origin face for meshWave
262 changedFacesInfo[nChangedFaces] =
263 wallPointData<label>
264 (
265 patch.Cf()[facei],
266 nChangedFaces,
267 0
268 );
269 changedFaces[nChangedFaces] = gFacei;
270 // Origin face-to-cut face
271 changedFaceToCutFace.push_back(isoSurfFaces.size());
272 ++nChangedFaces;
273 }
274
275 if
276 (
277 addCuttingFaceToIsoline
278 (
279 faces[gFacei].points(points),
280 patchi,
281 faceFaces(gFacei),
282 cuttingFacesPerMeshFace,
283 isoSurfPts,
284 isoSurfFaces,
285 zoneIDs
286 )
287 )
288 {
289 cuttingFacesPerMeshFace[gFacei].push_back
290 (
291 isoSurfFaces.size() - 1
292 );
293 }
294 }
295 }
296 }
297 }
298}

301void Foam::topOVariablesBase::writeSurfaceFiles
302(
303 const pointField& pts,
304 const faceList& faces,
305 const labelList& zoneIds,
306 const label nSerialPatches
307) const
308{
309 const fvMesh& mesh = zones_.mesh();
310
311 // Write vtp file
312 const word timeName = mesh.time().timeName();
313 fileName localName("topOIsoSurface" + timeName);
314 fileName fname(isoSurfFolder_/localName);
315 vtk::surfaceWriter vtkFile(pts, faces, fname);
316 vtkFile.beginFile();
317 vtkFile.writeGeometry();
318 vtkFile.beginCellData(1);
319 vtkFile.writeCellData("zoneIds", zoneIds);
320 vtkFile.close();
321
322 // It appears that there is no interface to write a multi-zone stl file (?)
323 // Follow a process similar to proxySurfaceWriter, but also use the zoneIds.
324
325 // Dummy faceIds.
326 // faceMap is built after merging the geometry from all processors, to
327 // be based on the global addressing
328 labelList faceIds;
329
330 autoPtr<meshedSurf> surf(nullptr);
331 if (Pstream::parRun())
332 {
333 surf.reset
334 (
335 autoPtr<meshedSurf>::NewFrom<mergedSurf>
336 (
337 pts, faces, zoneIds, faceIds, surfaceWriter::defaultMergeDim
338 )
339 );
340 }
341 else
342 {
343 surf.reset
344 (
345 autoPtr<meshedSurf>::NewFrom<meshedSurfRef>
346 (
347 pts, faces, zoneIds, faceIds
348 )
349 );
350 }
351
352 if (Pstream::master())
353 {
354 const faceList& surfFaces = surf().faces();
355 const labelList& surfZoneIds = surf().zoneIds();
356
357 // Size per zone
358 labelList zoneSizes(nSerialPatches + 1, 0);
359 forAll(surfFaces, fi)
360 {
361 ++zoneSizes[surfZoneIds[fi]];
362 }
363
364 // Faces passed in previous zones
365 labelList cumulZoneSizes(nSerialPatches + 1, 0);
366 for (label zi = 1; zi < cumulZoneSizes.size(); ++zi)
367 {
368 cumulZoneSizes[zi] = cumulZoneSizes[zi - 1] + zoneSizes[zi - 1];
369 }
370
371 // Construction of the faceMap
372 // ---------------------------
373 labelList faceMap(surfFaces.size(), -1);
374 // Faces visited so far per zone
375 labelList passedFacesPerZone(surfZoneIds.size(), 0);
376
377 forAll(surfFaces, fi)
378 {
379 const label zi = surfZoneIds[fi];
380 faceMap[cumulZoneSizes[zi] + passedFacesPerZone[zi]] = fi;
381 ++passedFacesPerZone[zi];
382 }
383
384 // Construction of the surfZoneList
385 // --------------------------------
386 List<surfZone> surfZones(nSerialPatches + 1);
387 label zi = 0;
388 forAll(mesh.boundary(), pi)
389 {
390 const fvPatch& patch = mesh.boundary()[pi];
391 const word& name = patch.name();
392 if (!isA<coupledFvPatch>(patch) && !isA<emptyFvPatch>(patch))
393 {
394 surfZones[zi] =
395 surfZone(name, zoneSizes[zi], cumulZoneSizes[zi], zi);
396 zi++;
397 }
398 }
399 surfZones[zi] =
400 surfZone
401 (
402 "topOPatch",
403 zoneSizes[nSerialPatches],
404 cumulZoneSizes[nSerialPatches],
405 zi
406 );
407 surfZones.setSize(zi + 1);
408
409 MeshedSurfaceProxy<face>
410 (
411 surf().points(),
412 surfFaces,
413 surfZones,
414 faceMap
415 ).write
416 (
417 fname + ".stl"
418 // BINARY seems to loose the patch names
419 //IOstreamOption
420 //(
421 // IOstreamOption::streamFormat::ASCII,
422 // IOstreamOption::compressionType::COMPRESSED
423 //)
424 );
425 }
426}

464void Foam::topOVariablesBase::sourceTerm
465(
466 DimensionedField<scalar, volMesh>& field,
467 const topOInterpolationFunction& interpolationFunc,
468 const scalar betaMax,
469 const word& interpolationFieldName
470) const
471{
472 // Interpolate based on beta values
473 const scalarField& beta = this->beta().primitiveField();
474 interpolationFunc.interpolate(beta, field.field());
475 field *= betaMax;
476}

479void Foam::topOVariablesBase::sourceTermSensitivities
480(
481 scalarField& sens,
482 const topOInterpolationFunction& interpolationFunc,
483 const scalar betaMax,
484 const word& designVariablesName,
485 const word& interpolationFieldName
486) const
487{
488 if (designVariablesName == "beta")
489 {
490 // Multiply with derivative of the interpolation function
491 const scalarField& beta = this->beta().primitiveField();
492 sens *= betaMax*interpolationFunc.derivative(beta);
493 }

497void Foam::topOVariablesBase::writeFluidSolidInterface
498(
499 const volScalarField& indicator,
500 const scalar isoValue,
501 labelList& changedFaces,
502 List<wallPointData<label>>& changedFacesInfo
503)
504{
505 const fvMesh& mesh = zones_.mesh();
506
507 // Current number of mesh faces being cut
508 label nChangedFaces(0);
509
510 // Indicator at the mesh points, used to compute the iso-surface
511 pointScalarField pointY(volPointInterpolation(mesh).interpolate(indicator));
512 if (debug && mesh.time().writeTime())
513 {
514 pointY.rename("pointY");
515 pointY.write();
516 }
517
518 // Storage for the cutting face/cell mechanism
519 cutCellIso cutCell(mesh, pointY);
520 cutFaceIso cutFace(mesh, pointY);
521 DynamicList<face> isoSurfFaces(mesh.nFaces()/100);
522 DynamicList<point> isoSurfPts(mesh.nPoints()/100);
523 DynamicList<label> zoneIDs(mesh.nFaces()/100);
524
525 // Number of patches before processor ones
526 label nSerialPatches = mesh.boundaryMesh().nNonProcessor();
527
528 // Map between iso-surface and mesh faces (internal and boundary)
529 meshFaceToChangedFace_.clearStorage();
530
531 // Map between changedFace and cutFace
532 // Index is the changedFaceID, output is the cutFaceID
533 DynamicList<label> changedFaceToCutFace;
534
535 // Per mesh face, its cutting faces
536 List<DynamicList<label>> cuttingFacesPerMeshFace(mesh.nFaces());
537
538 // Loop over cells and check whether they are cut by zero iso-surface
539 const cellList& cells = mesh.cells();
540
541 forAll(cells, celli)
542 {
543 if (!cutCell.calcSubCell(celli, isoValue))
544 {
545 const vector& cuttingCf = cutCell.faceCentre();
546 vector cuttingNf = cutCell.faceArea();
547 cuttingNf.normalise();
548 DynamicList<label> cellCutFaces(cells[celli].size());
549
550 // Loop over faces of the cut cell and check whether they are cut
551 // too, setting their distance from the cutting face
552 forAll(cells[celli], fi)
553 {
554 const label facei = cells[celli][fi];
555 if (!cutFace.calcSubFace(facei, isoValue))
556 {
557 const vector& Cf = mesh.Cf()[facei];
558 const scalar distSqr =
559 magSqr((Cf - cuttingCf) & cuttingNf);
560 const vector lsPoint =
561 Cf + ((cuttingCf - Cf) & cuttingNf)*cuttingNf;
562 cellCutFaces.push_back(facei);
563
564 if (!meshFaceToChangedFace_.found(facei))
565 {
566 // If the face being cut is a boundary one, part of
567 // it belongs to the iso-surface
568 bool addedToIsoSurf = addCutBoundaryFaceToIsoline
569 (
570 facei,
571 cutFace,
572 isoSurfPts,
573 isoSurfFaces,
574 zoneIDs,
575 cuttingFacesPerMeshFace
576 );
577 if (mesh.isInternalFace(facei) || addedToIsoSurf)
578 {
579 // Mesh face to changedFace addressing
580 meshFaceToChangedFace_.insert(facei, nChangedFaces);
581 // Set origin face for meshWave
582 changedFacesInfo[nChangedFaces] =
583 wallPointData<label>
584 (
585 lsPoint,
586 nChangedFaces,
587 distSqr
588 );
589 changedFaces[nChangedFaces] = facei;
590 // Origin face-to-cut face
591 changedFaceToCutFace.push_back(isoSurfFaces.size());
592 ++nChangedFaces;
593 }
594 }
595 else
596 {
597 label visitedFace = meshFaceToChangedFace_.at(facei);
598 if (distSqr < changedFacesInfo[visitedFace].distSqr())
599 {
600 // Set origin face for meshWave
601 changedFacesInfo[visitedFace] =
602 wallPointData<label>
603 (
604 lsPoint,
605 visitedFace,
606 distSqr
607 );
608 // Origin face-to-cut face
609 changedFaceToCutFace[visitedFace] =
610 isoSurfFaces.size();
611 }
612 }
613 }
614 }
615
616 if
617 (
618 addCuttingFaceToIsoline
619 (
620 cutCell.facePoints(),
621 nSerialPatches,
622 cellCutFaces,
623 cuttingFacesPerMeshFace,
624 isoSurfPts,
625 isoSurfFaces,
626 zoneIDs
627 )
628 )
629 {
630 for (const label facei : cellCutFaces)
631 {
632 cuttingFacesPerMeshFace[facei].push_back
633 (
634 isoSurfFaces.size() - 1
635 );
636 }
637 }
638 }
639 }
640
641 // Add boundary faces on the initial domain with a positive values on all
642 // points to the zero iso-surface
643 addBoundaryFacesToIsoline
644 (
645 pointY,
646 meshFaceToChangedFace_,
647 isoValue,
648 isoSurfPts,
649 isoSurfFaces,
650 zoneIDs,
651 nChangedFaces,
652 changedFaces,
653 changedFacesInfo,
654 changedFaceToCutFace,
655 cuttingFacesPerMeshFace
656 );
657
658 changedFaces.setSize(nChangedFaces);
659 changedFacesInfo.setSize(nChangedFaces);
660
661 // vtp and multi-region stl files holding the current geometry
662 surfPoints_.transfer(isoSurfPts);
663 surfFaces_.transfer(isoSurfFaces);
664
665 const labelList zoneIds(std::move(zoneIDs));
666 if (debug || mesh.time().writeTime())
667 {
668 writeSurfaceFiles(surfPoints_, surfFaces_, zoneIds, nSerialPatches);
669 }
670
671 // Invert changedFace-to-cuttingFace map for the sensitivity computations
672 //changedFacesPerCuttingFace_ =
673 // invertOneToMany(surfFaces_.size(), changedFaceToCutFace);
674
675 // Transform origin cut faces to a global numbering
676
677 const label changedFacesOffset =
678 globalIndex::calcOffset(changedFaces.size());
679
680 for (auto& info : changedFacesInfo)
681 {
682 info.data() += changedFacesOffset;
683 }
684}
