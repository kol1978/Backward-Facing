102 //- Inplace expand occurrences of variables according to the mapping.
103 //- Does \b not use environment values.
104 //
105 // Expansion includes:
106 // -# Variables
107 // - \c $VAR
108 // - \c ${VAR}
109 //
110 // -# Default and alternative values as per the POSIX shell:
111 // - \c ${parameter:-defValue}
112 // If parameter is unset or null, the \c defValue is substituted.
113 // Otherwise, the value of parameter is substituted.
114 // - \c ${parameter:+altValue}
115 // If parameter is unset or null, nothing is substituted.
116 // Otherwise the \c altValue is substituted.
117 // .
118 //
119 // General behaviour:
120 // - Unknown entries are removed silently.
121 // - Malformed entries (eg, brace mismatch, sigil followed by unknown
122 // characters) are left as is.
123 //
124 // \param[in,out] s The string to modify inplace.
125 // \param mapping The lookup table
126 // \param sigil The leading sigil. Can be changed to avoid conflict
127 // with other string expansions. (default: '$')
128 //
129 // \par Expansion behaviour
130 // - alternatives = True
131 // - environment = False
132 // - allow empty = True
133 // - subDict = Not applicable
134 // .
135 //
136 // \note Currently only used by Foam::dynamicCode.
137 void inplaceExpand
138 (
139 std::string& s,
140 const HashTable<string>& mapping,
141 const char sigil = '$'
142 );

145 //- Inplace expand occurrences of variables according to the dictionary
146 //- and (optionally) environment variables.
147 //
148 // Expansion includes:
149 // -# Dictionary variables and (optionally) environment variables
150 // - \c $VAR
151 // - \c ${VAR}
152 // - \c ${VAR:-defValue}
153 // - \c ${VAR:+altValue}
154 // -# Mathematical evaluation using stringOps::evaluate
155 // - \c ${{EXPR}}
156 // -# Current directory
157 // - leading "./"
158 // : the current directory - Foam::cwd()
159 // -# Leading tag expansion for commonly used directories
160 // - <b> <home>/ </b>
161 // : home directory
162 // - <b> <etc>/ </b>
163 // : user/group/other OpenFOAM etc directory
164 // - <b> <etc:</b><em>[ugoa]+</em>)<b>>/ </b>
165 // : user/group/other etc with specified location mode
166 // - <b> <case>/ </b>
167 // : The \c $FOAM_CASE directory
168 // - <b> <constant>/ </b>
169 // : The \c $FOAM_CASE/constant directory
170 // - <b> <system>/ </b>
171 // : The \c $FOAM_CASE/system directory
172 // -# Tilde expansion
173 // - leading "~/" : home directory
174 // - leading "~user" : home directory for specified user
175 // -# Default and alternative values as per the POSIX shell:
176 // - \c ${parameter:-defValue}
177 // If parameter is unset or null, the \c defValue is substituted.
178 // Otherwise, the value of parameter is substituted.
179 // - \c ${parameter:+altValue}
180 // If parameter is unset or null, nothing is substituted.
181 // Otherwise the \c altValue is substituted.
182 // .
183 //
184 // General behaviour:
185 // - Malformed entries (eg, brace mismatch, sigil followed by unknown
186 // characters) are left as is.
187 // - Supports recursive variable expansions.
188 // For example, "${var${num}}" and "${{100 + ${var}}}"
189 //
190 // \param[in,out] s The string to modify inplace
191 // \param dict The dictionary context for the expansion
192 // \param allowEnv Allow use of environment values as fallback
193 // \param allowEmpty Allow empty expansions, or treat as Fatal
194 // \param allowSubDict Allow expansion of subDict entries as well as
195 // primitive entries (default: false)
196 // \param sigil The leading sigil. Can be changed to avoid conflict
197 // with other string expansions. (default: '$')
198 //
199 // \sa Foam::findEtcEntry(), Foam::findEtcEntries(), stringOps::evaluate()
200 //
201 // \par Expansion behaviour
202 // - alternatives = True
203 // - environment = Given by parameter
204 // - allow empty = Given by parameter
205 // - subDict = Given by parameter (default: False)
206 // .
207 //
208 // \note This function has too many parameters and should generally
209 // be avoided in user coding.
210 void inplaceExpand
211 (
212 std::string& s,
213 const dictionary& dict,
214 const bool allowEnv,
215 const bool allowEmpty,
216 const bool allowSubDict = false,
217 const char sigil = '$'
218 );
