46bool Foam::faBoundaryMesh::hasGroupIDs() const
47{
48 if (groupIDsPtr_)
49 {
50 // Use existing cache
51 return !groupIDsPtr_->empty();
52 }
53
54 const faPatchList& patches = *this;
55
56 for (const faPatch& p : patches)
57 {
58 if (!p.inGroups().empty())
59 {
60 return true;
61 }
62 }
63
64 return false;
65}

67
68void Foam::faBoundaryMesh::calcGroupIDs() const
69{
70 if (groupIDsPtr_)
71 {
72 return; // Or FatalError
73 }
74
75 groupIDsPtr_.emplace(16);
76 auto& groupLookup = *groupIDsPtr_;
77
78 const faPatchList& patches = *this;
79
80 forAll(patches, patchi)
81 {
82 for (const word& groupName : patches[patchi].inGroups())
83 {
84 groupLookup(groupName).push_back(patchi);
85 }
86 }
87
88 // Remove groups that clash with patch names
89 forAll(patches, patchi)
90 {
91 if (groupLookup.empty())
92 {
93 break; // Early termination
94 }
95 else if (groupLookup.erase(patches[patchi].name()))
96 {
97 WarningInFunction
98 << "Removed group '" << patches[patchi].name()
99 << "' which clashes with patch " << patchi
100 << " of the same name."
101 << endl;
102 }
103 }
104}

107void Foam::faBoundaryMesh::populate(PtrList<entry>&& entries)
108{
109 clearLocalAddressing();
110
111 faPatchList& patches = *this;
112
113 patches.resize_null(entries.size());
114
115 // Transcribe.
116 // Does not handle nullptr at all (what could possibly be done?)
117 forAll(patches, patchi)
118 {
119 patches.set
120 (
121 patchi,
122 faPatch::New
123 (
124 entries[patchi].keyword(),
125 entries[patchi].dict(),
126 patchi,
127 *this
128 )
129 );
130 }
131
132 entries.clear();
133}

136bool Foam::faBoundaryMesh::readIOcontents(const bool allowOptionalRead)
137{
138 bool updated = false;
139 PtrList<entry> entries;
140
141 if
142 (
143 this->isReadRequired()
144 || (allowOptionalRead && this->isReadOptional() && this->headerOk())
145 )
146 {
147 // Warn for MUST_READ_IF_MODIFIED
148 warnNoRereading<faBoundaryMesh>();
149
150 // Read entries
151 Istream& is = readStream(typeName);
152
153 is >> entries;
154
155 is.check(FUNCTION_NAME);
156 close();
157 updated = true;
158 }
159 // Future: support master-only and broadcast?
160
161 if (updated)
162 {
163 populate(std::move(entries));
164 }
165
166 return updated;
167}

273void Foam::faBoundaryMesh::calcGeometry()
274{
275 // processor initGeometry send/recv the following:
276 // - edgeCentres() : faMesh::edgeCentres()
277 // - edgeLengths() : faMesh::Le()
278 // - edgeFaceCentres() : faMesh::areaCentres()
279 //
280 // faMesh::Le() has its own point-to-point communication (OK) but
281 // triggers either/or edgeAreaNormals(), pointAreaNormals()
282 // with their own communication that can block.
283
284 // This uses parallel comms and hence will not be trigggered
285 // on processors that do not have a processorFaPatch so instead
286 // force construction.
287
288 (void)mesh_.edgeAreaNormals();
289 (void)mesh_.pointAreaNormals();
290
291 (void)mesh_.areaCentres();
292 (void)mesh_.faceAreaNormals();
293
294
295 PstreamBuffers pBufs(Pstream::defaultCommsType);
296
297 if
298 (
299 pBufs.commsType() == Pstream::commsTypes::buffered
300 || pBufs.commsType() == Pstream::commsTypes::nonBlocking
301 )
302 {
303 forAll(*this, patchi)
304 {
305 operator[](patchi).initGeometry(pBufs);
306 }
307
308 pBufs.finishedSends();
309
310 forAll(*this, patchi)
311 {
312 operator[](patchi).calcGeometry(pBufs);
313 }
314 }
315 else if (pBufs.commsType() == Pstream::commsTypes::scheduled)
316 {
317 const lduSchedule& patchSchedule = mesh().globalData().patchSchedule();
318
319 // Dummy.
320 pBufs.finishedSends();
321
322 for (const auto& patchEval : patchSchedule)
323 {
324 const label patchi = patchEval.patch;
325
326 if (patchEval.init)
327 {
328 operator[](patchi).initGeometry(pBufs);
329 }
330 else
331 {
332 operator[](patchi).calcGeometry(pBufs);
333 }
334 }
335 }
336}

483void Foam::faBoundaryMesh::setGroup
484(
485 const word& groupName,
486 const labelUList& patchIDs
487)
488{
489 groupIDsPtr_.reset(nullptr);
490
491 faPatchList& patches = *this;
492
493 boolList pending(patches.size(), true);
494
495 // Add to specified patches
496 for (const label patchi : patchIDs)
497 {
498 if (pending.test(patchi))
499 {
500 pending.unset(patchi);
501 patches[patchi].addGroup(groupName);
502 }
503 }
504
505 // Remove from other patches
506 forAll(patches, patchi)
507 {
508 if (pending.test(patchi))
509 {
510 patches[patchi].removeGroup(groupName);
511 }
512 }
513}

599Foam::labelList Foam::faBoundaryMesh::indices
600(
601 const wordRe& matcher,
602 const bool useGroups
603) const
604{
605 if (matcher.empty())
606 {
607 return labelList();
608 }
609
610 // Only check groups if requested and they exist
611 const bool checkGroups = (useGroups && this->hasGroupIDs());
612
613 labelHashSet ids;
614
615 if (matcher.isPattern())
616 {
617 if (checkGroups)
618 {
619 ids.reserve(this->size());
620
621 const auto& groupLookup = groupPatchIDs();
622 forAllConstIters(groupLookup, iter)
623 {
624 if (matcher(iter.key()))
625 {
626 // Add ids associated with the group
627 ids.insert(iter.val());
628 }
629 }
630 }
631
632 if (ids.empty())
633 {
634 return PtrListOps::findMatching(*this, matcher);
635 }
636 else
637 {
638 ids.insert(PtrListOps::findMatching(*this, matcher));
639 }
640 }
641 else
642 {
643 // Literal string.
644 // Special version of above for reduced memory footprint
645
646 const label patchId = PtrListOps::firstMatching(*this, matcher);
647
648 if (patchId >= 0)
649 {
650 return labelList(one{}, patchId);
651 }
652 else if (checkGroups)
653 {
654 const auto iter = groupPatchIDs().cfind(matcher);
655
656 if (iter.good())
657 {
658 // Add ids associated with the group
659 ids.insert(iter.val());
660 }
661 }
662 }
663
664 return ids.sortedToc();
665}

802 }
803
804 // Not found, return -1

826Foam::label Foam::faBoundaryMesh::whichPatch(const label edgeIndex) const
827{
828 if (edgeIndex < mesh().nInternalEdges())
829 {
830 // Internal edge
831 return -1;
832 }
833 else if (edgeIndex >= mesh().nEdges())
834 {
835 // Bounds error: abort
836 FatalErrorInFunction
837 << "Edge " << edgeIndex
838 << " out of bounds. Number of geometric edges " << mesh().nEdges()
839 << abort(FatalError);
840
841 return -1;
842 }
843
844 // Find patch that the edgeIndex belongs to.
845
846 forAll(*this, patchi)
847 {
848 label start = mesh_.patchStarts()[patchi];
849 label size = operator[](patchi).faPatch::size();
850
851 if (edgeIndex >= start && edgeIndex < start + size)
852 {
853 return patchi;
854 }
855 }
856
857 // If not in any of above, it's trouble!
858 FatalErrorInFunction
859 << "Error in patch search algorithm"
860 << abort(FatalError);
861
862 return -1;
863}

866bool Foam::faBoundaryMesh::checkParallelSync(const bool report) const
867{
868 if (!Pstream::parRun())
869 {
870 return false;
871 }
872
873 const faBoundaryMesh& bm = *this;
874
875 bool hasError = false;
876
877 // Collect non-proc patches and check proc patches are last.
878 wordList localNames(bm.size());
879 wordList localTypes(bm.size());
880
881 label nonProci = 0;
882
883 forAll(bm, patchi)
884 {
885 if (!isA<processorFaPatch>(bm[patchi]))
886 {
887 if (nonProci != patchi)
888 {
889 // A processor patch in between normal patches!
890 hasError = true;
891
892 if (debug || report)
893 {
894 Pout<< " ***Problem with boundary patch " << patchi
895 << " name:" << bm[patchi].name()
896 << " type:" << bm[patchi].type()
897 << " - seems to be preceeded by processor patches."
898 << " This is usually a problem." << endl;
899 }
900 }
901 else
902 {
903 localNames[nonProci] = bm[patchi].name();
904 localTypes[nonProci] = bm[patchi].type();
905 ++nonProci;
906 }
907 }
908 }
909 localNames.resize(nonProci);
910 localTypes.resize(nonProci);
911
912 // Check and report error(s) on master
913 // - don't need indexing on master itself
914
915 const globalIndex procAddr(globalIndex::gatherNonLocal{}, nonProci);
916
917 const wordList allNames(procAddr.gather(localNames));
918 const wordList allTypes(procAddr.gather(localTypes));
919
920 // Automatically restricted to master
921 for (const int proci : procAddr.subProcs())
922 {
923 const auto procNames(allNames.slice(procAddr.range(proci)));
924 const auto procTypes(allTypes.slice(procAddr.range(proci)));
925
926 if (procNames != localNames || procTypes != localTypes)
927 {
928 hasError = true;
929
930 if (debug || report)
931 {
932 Info<< " ***Inconsistent patches across processors, "
933 "processor0 has patch names:" << localNames
934 << " patch types:" << localTypes
935 << " processor" << proci
936 << " has patch names:" << procNames
937 << " patch types:" << procTypes
938 << endl;
939 }
940 }
941 }
942
943 // Reduce (not broadcast) to respect local out-of-order errors (first loop)
944 return returnReduceOr(hasError);
945}

948bool Foam::faBoundaryMesh::checkDefinition(const bool report) const
949{
950 label nextPatchStart = mesh().nInternalEdges();
951 const faBoundaryMesh& bm = *this;
952
953 bool hasError = false;
954
955 forAll(bm, patchi)
956 {
957 if (bm[patchi].start() != nextPatchStart && !hasError)
958 {
959 hasError = true;
960
961 InfoInFunction
962 << " ****Problem with boundary patch " << patchi
963 << " named " << bm[patchi].name()
964 << " of type " << bm[patchi].type()
965 << ". The patch should start on face no " << nextPatchStart
966 << " and the patch specifies " << bm[patchi].start()
967 << "." << endl
968 << "Possibly consecutive patches have this same problem."
969 << " Suppressing future warnings." << endl;
970 }
971
972 // Warn about duplicate boundary patches?
973
974 nextPatchStart += bm[patchi].faPatch::size();
975 }
976
977 if (hasError)
978 {
979 SeriousErrorInFunction
980 << "This mesh is not valid: boundary definition is in error."
981 << endl;
982 }
983 else
984 {
985 if (debug || report)
986 {
987 Info << "Boundary definition OK." << endl;
988 }
989 }
990
991 return hasError;
992}

995void Foam::faBoundaryMesh::movePoints(const pointField& p)
996{
997 // See comments in calcGeometry()
