260 // Comparison
261
262 //- Test for equality of start/size values
263 template<class IntT2>
264 constexpr bool operator==(const IntRange<IntT2>& other) const noexcept
265 {

267 }

269 //- Compare lexicographically: compares start first, then size
270 template<class IntT2>
271 constexpr bool operator<(const IntRange<IntT2>& other) const noexcept
272 {
273 return

276 || (start() == other.start() && size() < other.size())
277 );
278 }
