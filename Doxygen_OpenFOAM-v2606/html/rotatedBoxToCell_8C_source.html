72void Foam::rotatedBoxToCell::combine(topoSet& set, const bool add) const
73{
74 // Check vector handedness: (i ^ j) & k should be positive
75 const scalar vol = (i_ ^ j_) & k_;
76
77 if (mag(vol) < SMALL)
78 {
79 WarningInFunction
80 << "Spanning vectors are coplanar or near-zero." << nl
81 << "(i ^ j) & k = " << vol << nl
82 << "Rotated box has zero volume; will not select any cells." << nl
83 << endl;
84 }
85 else if (vol < 0)
86 {
87 WarningInFunction
88 << "Spanning vectors do not form a right-handed system." << nl
89 << "(i ^ j) & k = " << vol << nl
90 << "Rotated box has zero volume; will not select any cells." << nl
91 << "Reorder the vectors to satisfy (i ^ j) & k > 0." << nl
92 << endl;
93 }
94
95 // Define a cell for the box
96 pointField boxPoints(8, origin_);
97 // boxPoints[0] = origin_;
98 boxPoints[1] += i_;
99 boxPoints[2] += i_ + j_;
100 boxPoints[3] += j_;
101 boxPoints[4] += k_;
102 boxPoints[5] += k_ + i_;
103 boxPoints[6] += k_ + i_ + j_;
104 boxPoints[7] += k_ + j_;
105
106 // Get outwards pointing faces.
107 faceList boxFaces(hexCell::modelFaces());
108
109 // Precalculate normals
110 vectorField boxFaceNormals(boxFaces.size());
111 forAll(boxFaces, i)
112 {
113 boxFaceNormals[i] = boxFaces[i].areaNormal(boxPoints);
114
115 //Pout<< "Face:" << i << " position:" << boxFaces[i].centre(boxPoints)
116 // << " normal:" << boxFaceNormals[i] << endl;
117 }
118
119 // Check whether cell centre is inside all faces of box.
120
121 const tmp<pointField> tctrs(this->transform(mesh_.cellCentres()));
122 const pointField& ctrs = tctrs();
123
124 forAll(ctrs, celli)
125 {
126 bool inside = true;
127
128 forAll(boxFaces, i)
129 {
130 const face& f = boxFaces[i];
131
132 if (((ctrs[celli] - boxPoints[f[0]]) & boxFaceNormals[i]) > 0)
133 {
134 inside = false;
135 break;
136 }
137 }
138
139 if (inside)
140 {
141 addOrDelete(set, celli, add);
142 }
143 }
144}

192void Foam::rotatedBoxToCell::applyToSet
193(
194 const topoSetSource::setAction action,
195 topoSet& set
196) const
197{
198 if (action == topoSetSource::ADD || action == topoSetSource::NEW)
199 {
200 if (verbose_)
201 {
202 Info<< " Adding cells with centre within rotated box"
203 << endl;
204 }
205
206 combine(set, true);
207 }
208 else if (action == topoSetSource::SUBTRACT)
209 {
210 if (verbose_)
211 {
212 Info<< " Removing cells with centre within rotated box"
213 << endl;
214 }
215
216 combine(set, false);
217 }
218}
