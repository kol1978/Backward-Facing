36// The maps (labelListList) are not human-modifiable but if we need to
37// inspect them in ASCII, it is much more convenient if each sub-list
38// is flattened on a single line. Also reduce some extraneous newlines
39// for binary output.
40static Ostream& printMaps(Ostream& os, const labelListList& maps)
41{
42 if (maps.empty())
43 {
44 os << maps;
45 }
46 else
47 {
48 // Leading space (newline) handled by the caller
49 os << maps.size() << nl << token::BEGIN_LIST << nl;
50
51 // Compact single-line output for each labelList (ascii)
52 // or with less whitespace (binary) - labelList is contiguous
53
54 for (const auto& list : maps)
55 {
56 if (os.format() == IOstreamOption::BINARY)
57 {
58 const label len = list.size();
59
60 os << len;
61 if (len)
62 {
63 // write(...) includes surrounding start/end delimiters
64 os.write(list.cdata_bytes(), list.size_bytes());
65 }
66 os << nl;
67 }
68 else
69 {
70 list.writeList(os) << nl;
71 }
72 }
73 os << token::END_LIST;
74 }
75
76 return os;
77}

80// Read from sub-dictionary
81static void readMapEntries
82(
83 const dictionary& dict,
84 const word& mapName,
85 bool& hasFlip,
86 labelListList& maps
87)
88{
89 const auto& subdict = dict.subDict(mapName);
90
91 subdict.readEntry("flip", hasFlip);
92 subdict.readEntry("maps", maps);
93}
94
95
96// Write as sub-dictionary content
97static void writeMapEntries
98(
99 Ostream& os,
100 const word& mapName,
101 const bool hasFlip,
102 const labelListList& maps
103)
104{
105 os << nl;
106 os.beginBlock(mapName);
107 os.writeEntry("flip", hasFlip);
108 if (maps.empty())
109 {
110 os.writeEntry("maps", maps);
111 }
112 else
113 {
114 os << indent << "maps" << nl;
115 printMaps(os, maps);
116 os.endEntry();
117 }
118 os.endBlock();
119}

202Foam::Ostream& Foam::operator<<(Ostream& os, const mapDistributeBase& map)
203{
204 os << map.constructSize() << nl;
205
206 map.writeSubMap(os);
207 map.writeConstructMap(os);
208
209 // Flips as y/n instead of 1/0 (readability)
210 os << (map.subHasFlip() ? 'y' : 'n') << token::SPACE
211 << (map.constructHasFlip() ? 'y' : 'n') << token::SPACE
212 << map.comm() << nl;
213
214 return os;
215}

234
235 os << indent << "remote { flip " << (map.constructHasFlip() ? 'y' : 'n')
