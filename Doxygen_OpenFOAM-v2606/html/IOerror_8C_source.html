169void Foam::IOerror::SafeFatalIOError
170(
171 std::string_view functionName,
172 std::string_view sourceFileName,
173 const int sourceFileLineNumber,
174 const IOstream& ioStream,
175 const std::string& msg
176)
177{
178 if (JobInfo::constructed)
179 {
180 FatalIOError
181 (
182 functionName,
183 sourceFileName,
184 sourceFileLineNumber,
185 ioStream
186 ) << msg.c_str() << Foam::exit(FatalIOError);
187 }
188 else
189 {
190 // Without (openfoam=API patch=NN) since it is rarely used
191 std::cerr
192 << nl
193 << "--> FOAM FATAL IO ERROR:" << nl
194 << msg << nl
195 << "file: " << ioStream.relativeName()
196 << " at line " << ioStream.lineNumber() << '.' << nl << nl
197 << " From " << functionName << nl
198 << " in file " << sourceFileName
199 << " at line " << sourceFileLineNumber << '.' << std::endl;
200 std::exit(1);
201 }
202}

220void Foam::IOerror::exiting(const int errNo, const bool isAbort)
221{
222 if (throwing_)
223 {
224 if (!isAbort)
225 {
226 // Make a copy of the error to throw
227 IOerror errorException(*this);
228
229 // Reset the message buffer for the next error message
230 error::clear();
231
232 throw errorException;
233 return;
234 }
235 }
236 else if (JobInfo::constructed)
237 {
238 jobInfo.add("FatalIOError", operator dictionary());
239 JobInfo::shutdown(isAbort || error::useAbort());
240 }
241
242 simpleExit(errNo, isAbort);
243}

260void Foam::IOerror::write(Ostream& os, const bool withTitle) const
261{
262 if (os.bad())
263 {
264 return;
265 }
266
267 os << nl;
268 if (withTitle && !title().empty())
269 {
270 os << title().c_str()
271 << "(openfoam-" << foamVersion::api;
272
273 if (foamVersion::patched())
274 {
275 // Patch-level, when defined
276 os << " patch=" << foamVersion::patch.c_str();
277 }
278 os << ')' << nl;
279 }
280 os << message().c_str();
281
282
283 if (!ioFileName().empty())
284 {
285 os << nl << nl
286 << "file: " << ioFileName().c_str();
287
288 if (ioStartLineNumber() >= 0)
289 {
290 os << " at line " << ioStartLineNumber();
291 if (ioStartLineNumber() < ioEndLineNumber())
292 {
293 os << " to " << ioEndLineNumber();
294 }
295 os << '.';
296 }
297 }
298
299
300 const auto lineNo = sourceFileLineNumber();
301
302 if (messageStream::level >= 2 && lineNo && !functionName().empty())
303 {
304 os << nl << nl
305 << " From " << functionName().c_str() << nl;
306
307 if (!sourceFileName().empty())
308 {
309 os << " in file " << sourceFileName().c_str();
310
311 if (lineNo > 0)
312 {
313 os << " at line " << lineNo << '.';
314 }
315 }
316 }
317}
