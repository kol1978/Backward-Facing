56
57Foam::scalar Foam::humidityTemperatureCoupledMixedFvPatchScalarField::Sh
58(
59 const scalar Re,
60 const scalar Sc
61) const
62{
63 if (Re < 5.0E+05)
64 {
65 return 0.664*sqrt(Re)*cbrt(Sc);
66 }
67 else
68 {
69 return 0.037*pow(Re, 0.8)*cbrt(Sc);
70 }
71}

74Foam::scalar
75Foam::humidityTemperatureCoupledMixedFvPatchScalarField::htcCondensation
76(
77 const scalar Tsat,
78 const scalar Re
79) const
80{
81 // (BLID:Eq. 10.52-10.53)
82 if (Tsat > 295.15 && Tsat < 373.15)
83 {
84 return 51104 + 2044*(Tsat - 273.15);
85 }
86 else
87 {
88 return 255510;
89 }
90}

208Foam::humidityTemperatureCoupledMixedFvPatchScalarField::
209humidityTemperatureCoupledMixedFvPatchScalarField
210(
211 const fvPatch& p,
212 const DimensionedField<scalar, volMesh>& iF,
213 const dictionary& dict
214)
215:
216 parent_bctype(p, iF),
217 temperatureCoupledBase(patch(), dict),
218 mode_(mtCondensationAndEvaporation),
219 pName_(dict.getOrDefault<word>("p", "p")),
220 UName_(dict.getOrDefault<word>("U", "U")),
221 rhoName_(dict.getOrDefault<word>("rho", "rho")),
222 muName_(dict.getOrDefault<word>("mu", "thermo:mu")),
223 TnbrName_(dict.getOrDefault<word>("Tnbr", "T")),
224 qrNbrName_(dict.getOrDefault<word>("qrNbr", "none")),
225 qrName_(dict.getOrDefault<word>("qr", "none")),
226 specieName_(dict.getOrDefault<word>("specie", "none")),
227 liquid_(nullptr),
228 liquidDict_(),
229 mass_(patch().size(), Zero),
230 Tvap_(0.0),
231 myKDelta_(patch().size(), Zero),
232 dmHfg_(patch().size(), Zero),
233 mpCpTp_(patch().size(), Zero),
234 Mcomp_(0.0),
235 L_(0.0),
236 fluid_(false),
237 cp_(patch().size(), Zero),
238 thickness_(patch().size(), Zero),
239 rho_(patch().size(), Zero),
240 thicknessLayers_(0),
241 kappaLayers_(0)
242{
243 if (!isA<mappedPatchBase>(this->patch().patch()))
244 {
245 FatalIOErrorInFunction(dict)
246 << "\n patch type '" << p.type()
247 << "' not type '" << mappedPatchBase::typeName << "'"
248 << "\n for patch " << p.name()
249 << " of field " << internalField().name()
250 << " in file " << internalField().objectPath()
251 << exit(FatalIOError);
252 }
253
254 this->readValueEntry(dict, IOobjectOption::MUST_READ);
255
256 if (massModeTypeNames_.readIfPresent("mode", dict, mode_))
257 {
258 fluid_ = true;
259 }
260
261 if (dict.readIfPresent("thicknessLayers", thicknessLayers_))
262 {
263 dict.readEntry("kappaLayers", kappaLayers_);
264 }
265
266 if (fluid_)
267 {
268 switch(mode_)
269 {
270 case mtConstantMass:
271 {
272 thickness_ = scalarField("thickness", dict, p.size());
273 cp_ = scalarField("cp", dict, p.size());
274 rho_ = scalarField("rho", dict, p.size());
275
276 break;
277 }
278 case mtCondensation:
279 case mtEvaporation:
280 case mtCondensationAndEvaporation:
281 {
282 dict.readEntry("carrierMolWeight", Mcomp_);
283 dict.readEntry("L", L_);
284 dict.readEntry("Tvap", Tvap_);
285 liquidDict_ = dict.subDict("liquid");
286 liquid_ =
287 liquidProperties::New(liquidDict_.subDict(specieName_));
288
289 if (dict.found("thickness"))
290 {
291 scalarField& Tp = *this;
292 const scalarField& magSf = patch().magSf();
293
294 // Assume initially standard pressure for rho calculation
295 scalar pf = 1e5;
296 thickness_ = scalarField("thickness", dict, p.size());
297 forAll(thickness_, i)
298 {
299 mass_[i] =
300 thickness_[i]*liquid_->rho(pf, Tp[i])*magSf[i];
301 }
302 }
303 fluid_ = true;
304
305 break;
306 }
307 default:
308 {
309 FatalIOErrorInFunction(dict)
310 << "Did not find mode " << mode_
311 << " on patch " << patch().name()
312 << nl
313 << "Please set 'mode' to one of "
314 << massModeTypeNames_.sortedToc()
315 << exit(FatalIOError);
316 }
317 }
318 }
319
320
321 if (this->readMixedEntries(dict))
322 {
323 // Full restart
324 }
325 else
326 {
327 // Start from user entered data. Assume fixedValue.
328 refValue() = *this;
329 refGrad() = 0.0;
330 valueFraction() = 1.0;
331 }
332}

418void Foam::humidityTemperatureCoupledMixedFvPatchScalarField::updateCoeffs()
419{
420 if (updated())
421 {
422 return;
423 }
424
425 // Get the coupling information from the mappedPatchBase
426 const auto& mpp = refCast<const mappedPatchBase>(patch().patch());
427
428 const scalarField& magSf = patch().magSf();
429
430 const label nbrPatchI = mpp.samplePolyPatch().index();
431 const polyMesh& mesh = patch().boundaryMesh().mesh();
432 const polyMesh& nbrMesh = mpp.sampleMesh();
433 const fvPatch& nbrPatch =
434 refCast<const fvMesh>(nbrMesh).boundary()[nbrPatchI];
435
436 const auto& nbrField =
437 refCast<const this_bctype>
438 (
439 nbrPatch.lookupPatchField<volScalarField>(TnbrName_)
440 );
441
442 // Swap to obtain full local values of neighbour internal field
443 scalarField nbrIntFld(nbrField.patchInternalField());
444 mpp.distribute(nbrIntFld);
445
446
447 scalarField& Tp = *this;
448
449 const volScalarField& T =
450 static_cast<const volScalarField&>(internalField());
451
452 const scalarField TpOld(T.oldTime().boundaryField()[patch().index()]);
453
454 scalarField Tin(patchInternalField());
455
456 const scalarField K(this->kappa(*this));
457
458 // Neighbour kappa done separately because we need kappa solid for the
459 // htc correlation
460 scalarField nbrK(nbrField.kappa(*this));
461 mpp.distribute(nbrK);
462
463 scalarField nrbDeltaCoeffs(nbrPatch.deltaCoeffs());
464 mpp.distribute(nrbDeltaCoeffs);
465
466 scalarField KDeltaNbr(nbrField.kappa(*this)*nbrPatch.deltaCoeffs());
467 mpp.distribute(KDeltaNbr);
468
469 myKDelta_ = K*patch().deltaCoeffs();
470
471 if (thicknessLayers_.size() > 0)
472 {
473 myKDelta_ = 1.0/myKDelta_;
474 forAll(thicknessLayers_, iLayer)
475 {
476 myKDelta_ += thicknessLayers_[iLayer]/kappaLayers_[iLayer];
477 }
478 myKDelta_ = 1.0/myKDelta_;
479 }
480
481 scalarField dm(patch().size(), Zero);
482
483 // Fluid Side
484 if (fluid_)
485 {
486 scalarField Yvp(patch().size(), Zero);
487 const scalar dt = mesh.time().deltaTValue();
488
489 const scalarField myDelta(patch().deltaCoeffs());
490
491 if (mode_ != mtConstantMass)
492 {
493 scalarField cp(patch().size(), Zero);
494 scalarField hfg(patch().size(), Zero);
495 scalarField htc(patch().size(), GREAT);
496 scalarField liquidRho(patch().size(), Zero);
497 scalarField Tdew(patch().size(), Zero);
498 scalarField RH(patch().size(), Zero);
499
500 auto& Yp =
501 refConstCast<fixedGradientFvPatchField<scalar>>
502 (
503 patch().lookupPatchField<volScalarField>(specieName_)
504 );
505
506 const auto& pp =
507 patch().lookupPatchField<volScalarField>(pName_);
508
509 const auto& Up =
510 patch().lookupPatchField<volVectorField>(UName_);
511
512 const auto& rhop =
513 patch().lookupPatchField<volScalarField>(rhoName_);
514
515 const auto& mup =
516 patch().lookupPatchField<volScalarField>(muName_);
517
518 const vectorField Ui(Up.patchInternalField());
519 const scalarField Yi(Yp.patchInternalField());
520
521 forAll(Tp, faceI)
522 {
523 const scalar Tf = Tp[faceI];
524 const scalar Tint = Tin[faceI];
525 const vector Uf = Ui[faceI];
526 const scalar pf = pp[faceI];
527
528 const scalar muf = mup[faceI];
529 const scalar rhof = rhop[faceI];
530 const scalar nuf = muf/rhof;
531 const scalar pSat = liquid_->pv(pf, Tint);
532 const scalar Mv = liquid_->W();
533 const scalar TSat = liquid_->pvInvert(pSat);
534 const scalar Re = mag(Uf)*L_/nuf;
535
536 cp[faceI] = liquid_->Cp(pf, Tf);
537 hfg[faceI] = liquid_->hl(pf, Tf);
538
539 // Calculate relative humidity
540 const scalar invMwmean =
541 Yi[faceI]/Mv + (1.0 - Yi[faceI])/Mcomp_;
542 const scalar Xv = Yi[faceI]/invMwmean/Mv;
543 RH[faceI] = min(Xv*pf/pSat, 1.0);
544
545 scalar RHmin = 0.01;
546 Tdew[faceI] = 0.0;
547
548 if (RH[faceI] > RHmin)
549 {
550 scalar b = 243.5;
551 scalar c = 17.65;
552 scalar TintDeg = Tint - 273;
553 Tdew[faceI] =
554 b*(log(RH[faceI]) + (c*TintDeg)/(b + TintDeg))
555 /(c - log(RH[faceI]) - ((c*TintDeg)/(b + TintDeg)))
556 + 273;
557 }
558
559 if
560 (
561 Tf < Tdew[faceI]
562 && RH[faceI] > RHmin
563 && (
564 mode_ == mtCondensation
565 || mode_ == mtCondensationAndEvaporation
566 )
567 )
568 {
569 htc[faceI] = htcCondensation(TSat, Re)*nbrK[faceI]/L_;
570
571 scalar htcTotal =
572 1.0/((1.0/myKDelta_[faceI]) + (1.0/htc[faceI]));
573
574 // Heat flux [W] (>0 heat is converted into mass)
575 const scalar q = (Tint - Tf)*htcTotal*magSf[faceI];
576
577 // Mass flux rate [Kg/s/m2]
578 dm[faceI] = q/hfg[faceI]/magSf[faceI];
579
580 mass_[faceI] += q/hfg[faceI]*dt;
581
582 // -dYp/dn = q/Dab (fixedGradient)
583 const scalar Dab = liquid_->D(pf, Tf);
584 Yvp[faceI] =
585 -min(dm[faceI]/Dab/rhof, Yi[faceI]*myDelta[faceI]);
586 }
587 else if
588 (
589 Tf > Tvap_
590 && mass_[faceI] > 0.0
591 && (
592 mode_ == mtEvaporation
593 || mode_ == mtCondensationAndEvaporation
594 )
595 )
596 {
597 const scalar Dab = liquid_->D(pf, Tf);
598
599 const scalar Sc = nuf/Dab;
600 const scalar Sh = this->Sh(Re, Sc);
601
602 const scalar Ys = Mv*pSat/(Mv*pSat + Mcomp_*(pf - pSat));
603
604 // Mass transfer coefficient [m/s]
605 const scalar hm = Dab*Sh/L_;
606
607 const scalar Yinf = max(Yi[faceI], 0.0);
608
609 // Mass flux rate [Kg/s/m2]
610 dm[faceI] = -rhof*hm*max((Ys - Yinf), 0.0)/(1.0 - Ys);
611
612 // Set fixedGradient for carrier species.
613 Yvp[faceI] = -dm[faceI]/Dab/rhof;
614
615 // Total mass accumulated [Kg]
616 mass_[faceI] += dm[faceI]*magSf[faceI]*dt;
617
618 htc[faceI] = htcCondensation(TSat, Re)*nbrK[faceI]/L_;
619 }
620 else if (Tf > Tdew[faceI] && Tf < Tvap_ && mass_[faceI] > 0.0)
621 {
622 htc[faceI] = htcCondensation(TSat, Re)*nbrK[faceI]/L_;
623 }
624 else if (mass_[faceI] == 0.0)
625 {
626 // Do nothing
627 }
628
629 liquidRho[faceI] = liquid_->rho(pf, Tf);
630 }
631
632 mass_ = max(mass_, scalar(0));
633
634 Yp.gradient() = Yvp;
635
636 // Output film delta (e.g. H2OThickness) [m]
637 const word fieldName(specieName_ + "Thickness");
638
639 scalarField& pDelta =
640 thicknessField
641 (
642 fieldName,
643 refCast<const fvMesh>(mesh),
644 dimLength
645 ).boundaryFieldRef()[patch().index()];
646
647
648 pDelta = mass_/liquidRho/magSf;
649
650 // Weight myKDelta and htc
651 myKDelta_ = 1.0/((1.0/myKDelta_) + (1.0/htc));
652
653 mpCpTp_ = mass_*cp/dt/magSf;
654
655 // Heat flux due to change of phase [W/m2]
656 dmHfg_ = dm*hfg;
657
658 if (debug)
659 {
660 // Output RH and Tdew
661 scalarField& bRH =
662 thicknessField
663 (
664 "RH",
665 refCast<const fvMesh>(mesh)
666 ).boundaryFieldRef()[patch().index()];
667 bRH = RH;
668
669 scalarField& bTdew =
670 thicknessField
671 (
672 "Tdew",
673 refCast<const fvMesh>(mesh),
674 dimTemperature
675 ).boundaryFieldRef()[patch().index()];
676 bTdew = Tdew;
677 }
678 }
679 else
680 {
681 // Inertia term [W/K/m2]
682 mpCpTp_ = thickness_*rho_*cp_/dt;
683 }
684 }
685
686 scalarField mpCpTpNbr(patch().size(), Zero);
687 scalarField dmHfgNbr(patch().size(), Zero);
688
689 if (!fluid_)
690 {
691 mpCpTpNbr = nbrField.mpCpTp();
692 mpp.distribute(mpCpTpNbr);
693
694 dmHfgNbr = nbrField.dmHfg();
695 mpp.distribute(dmHfgNbr);
696 }
697
698 // Obtain Rad heat (qr)
699 scalarField qr(Tp.size(), Zero);
700 if (qrName_ != "none")
701 {
702 qr = patch().lookupPatchField<volScalarField>(qrName_);
703 }
704
705 scalarField qrNbr(Tp.size(), Zero);
706 if (qrNbrName_ != "none")
707 {
708 qrNbr = nbrPatch.lookupPatchField<volScalarField>(qrNbrName_);
709 mpp.distribute(qrNbr);
710 }
711
712 const scalarField dmHfg(dmHfgNbr + dmHfg_);
713
714 const scalarField mpCpdt(mpCpTpNbr + mpCpTp_);
715
716 // qr > 0 (heat up the wall)
717 scalarField alpha(KDeltaNbr + mpCpdt - (qr + qrNbr)/Tp);
718
719 valueFraction() = alpha/(alpha + myKDelta_);
720
721 refValue() = (KDeltaNbr*nbrIntFld + mpCpdt*TpOld + dmHfg)/alpha;
722
723 this->parent_bctype::updateCoeffs();
724
725 if (debug && fluid_)
726 {
727 scalar Qdm = gWeightedSum(magSf, dm);
728 scalar QMass = gSum(mass_);
729 scalar Qt = gSum(myKDelta_*(Tp - Tin)*magSf);
730 scalar QtSolid = gSum(KDeltaNbr*(Tp - nbrIntFld)*magSf);
731
732 MinMax<scalar> limits = gMinMax(*this);
733
734 Info<< mesh.name() << ':'
735 << patch().name() << ':'
736 << internalField().name() << " <- "
737 << nbrMesh.name() << ':'
738 << nbrPatch.name() << ':'
739 << internalField().name() << " :" << nl
740 << " Total mass flux [Kg/s] : " << Qdm << nl
741 << " Total mass on the wall [Kg] : " << QMass << nl
742 << " Total heat (>0 leaving the wall to the fluid) [W] : "
743 << Qt << nl
744 << " Total heat (>0 leaving the wall to the solid) [W] : "
745 << QtSolid << nl
746 << " wall temperature "
747 << " min:" << limits.min()
748 << " max:" << limits.max()
749 << " avg:" << gAverage(*this)
750 << endl;
751 }
752}

755void Foam::humidityTemperatureCoupledMixedFvPatchScalarField::write
756(
757 Ostream& os
758) const
759{
760 mixedFvPatchField<scalar>::write(os);
