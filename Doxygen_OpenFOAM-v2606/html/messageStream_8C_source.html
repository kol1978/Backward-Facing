108Foam::OSstream& Foam::messageStream::stream
109(
110 OSstream* alternative,
111 int communicator
112)
113{
114 if (communicator < 0)
115 {
116 communicator = UPstream::worldComm;
117 }
118
119 if (level)
120 {
121 // Master-only output?
122 const bool masterOnly
123 (
124 !UPstream::parRun()
125 || ((severity_ & ~errorSeverity::USE_STDERR) == errorSeverity::INFO)
126 || ((severity_ & ~errorSeverity::USE_STDERR) == errorSeverity::WARNING)
127 );
128
129 if
130 (
131 masterOnly
132 && (UPstream::parRun() && !UPstream::master(communicator))
133 )
134 {
135 // Requested master-only output but is non-master (in parallel)
136 // -> early exit
137 return Snull;
138 }
139
140
141 // Use stderr instead of stdout:
142 // - requested via static <redirect> variable
143 // - explicit: with USE_STDERR mask
144 // - inferred: WARNING -> stderr when infoDetailLevel == 0
145 const bool use_stderr =
146 (
147 (redirect == 2)
148 || (severity_ & errorSeverity::USE_STDERR)
149 || (severity_ == errorSeverity::WARNING && Foam::infoDetailLevel == 0)
150 );
151
152
153 OSstream* osptr;
154
155 if (masterOnly)
156 {
157 // Use supplied alternative? Valid for master-only output
158 osptr =
159 (
160 alternative
161 ? alternative
162 : (use_stderr ? &Serr : &Sout)
163 );
164 }
165 else
166 {
167 osptr = (use_stderr ? &Perr : &Pout);
168 }
169
170 if (!title_.empty())
171 {
172 (*osptr) << title_.c_str();
173 }
174
175 if (maxErrors_ && (++errorCount_ >= maxErrors_))
176 {
177 FatalErrorInFunction
178 << "Too many errors..."
179 << abort(FatalError);
180 }
181
182 return *osptr;
183 }
184
185 return Snull;
186}

189Foam::OSstream& Foam::messageStream::masterStream(int communicator)
190{
191 if (communicator < 0)
192 {
193 communicator = UPstream::worldComm;
194 }
195
196 if (UPstream::warnComm >= 0 && communicator != UPstream::warnComm)
197 {
198 Perr<< "** messageStream with comm:" << communicator << endl;
199 error::printStack(Perr);
200 }
201
202 if (communicator == UPstream::worldComm || UPstream::master(communicator))
203 {
204 return this->stream(nullptr, communicator);
205 }
206
207 return Snull;
208}

220Foam::OSstream& Foam::messageStream::deprecated
221(
222 const int afterVersion,
223 std::string_view functionName,
224 std::string_view sourceFileName,
225 const int sourceFileLineNumber
226)
227{
228 OSstream& os = this->stream();
229
230 // No warning for 0 (unversioned) or -ve values (silent versioning).
231 // Also no warning for (version >= foamVersion::api), which
232 // can be used to denote future expiry dates of transition features.
233
234 if (afterVersion > 0 && afterVersion < foamVersion::api)
235 {
236 const int months =
237 (
238 // YYMM -> months
239 (12 * (foamVersion::api/100) + (foamVersion::api % 100))
240 - (12 * (afterVersion/100) + (afterVersion % 100))
241 );
242
243 os << nl
244 << ">>> DEPRECATED after version " << afterVersion;
245
246 if (afterVersion < 1000)
247 {
248 // Predates YYMM versioning (eg, 240 for version 2.4)
249 os << ". This is very old! <<<" << nl;
250 }
251 else
252 {
253 os << ". This is about " << months << " months old. <<<" << nl;
254 }
255 }
256
257 os << nl;
258 if (!functionName.empty())
259 {
260 os << " From ";
261 // Emit function name (unquoted) and newline
262 os.writeQuoted(functionName.data(), functionName.size(), false);
263 os << nl;
264
265 if (!sourceFileName.empty())
266 {
267 os << " in file ";
268 // Emit source file name (unquoted)
269 os.writeQuoted(sourceFileName.data(), sourceFileName.size(), false);
270
271 if (sourceFileLineNumber >= 0)
272 {
273 os << " at line " << sourceFileLineNumber;
274 }
275 os << nl;
276 }
277 }
278 os << " ";
279
280 return os;
281}
