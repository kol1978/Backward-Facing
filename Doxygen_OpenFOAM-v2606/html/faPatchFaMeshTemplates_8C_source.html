44template<class GeometricField>
45const typename GeometricField::Patch*
46Foam::faPatch::cfindPatchField(const word& fldName) const
47{
48 const auto* fldptr =
49 boundaryMesh().mesh().thisDb().template
50 cfindObject<GeometricField>(fldName);
51
52 if (fldptr)
53 {
54 return &(fldptr->boundaryField()[this->index()]);

57 {
58 return nullptr;
