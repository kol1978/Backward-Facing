63// Per-proc edge output (debugging)
64if (args.found("write-edges-vtk"))
65{
66 // The base name for output file
67 word outputName(vtkBaseFileName + "-edges");
68
69 if (UPstream::parRun())
70 {
71 outputName += ("-proc" + Foam::name(UPstream::myProcNo()));
72 }
73
74 // finiteArea - edges.
75 // NB: edges are with reordered boundaries (not primitive patch order)
76 vtk::lineWriter writer
77 (
78 aMesh.points(),
79 aMesh.edges(),
80 (aMesh.time().globalPath() / outputName),
81 false // parallel = false. every rank writes separately
82 );
83
84 writer.writeGeometry();
85
86 // CellData
87 writer.beginCellData(UPstream::parRun() ? 4 : 3);
88 if (UPstream::parRun())
89 {
90 // DIY for procID since each writer is non-parallel
91 writer.writeUniform("procID", label(UPstream::myProcNo()));
92 }
93
94 // Edge ids in rank-local numbering (useful for comparing to edgeMap etc)
95 writer.writeLocalIDs("edgeID");
96 // Edge ids (patch order) in rank-local numbering
97 writer.writeCellData("edgeLabel", faMeshTools::edgeLabels(aMesh));
98
99 // Patch ids
100 {
101 labelList values(aMesh.nEdges());
102
103 // Internal
104 label patchId = -1;
105 values = patchId;
106
107 // Boundaries
108 label start = aMesh.nInternalEdges();
109 for (const auto& p : aMesh.boundary())
110 {
111 const auto count = p.nEdges();
112 values.slice(start, count) = ++patchId;
113 start += count;
114 }
115
116 writer.writeCellData("patchID", values);
117 }
118
119 Pout<< " " << writer.output().name() << nl;
120}
