53void Foam::ensightOutputSurface::write(ensightGeoFile& os) const
54{
55 const bool hasGeometry =
56 (
57 vertexOutput_ ? !points_.empty() : (total() > 0)
58 );
59
60 if (!hasGeometry)
61 {
62 return;
63 }
64
65 // Coordinates
66 ensightOutput::Detail::writeCoordinates
67 (
68 os,
69 index(),
70 name(),
71 points_.size(),
72 points_,
73 false // serial
74 );
75
76 if (vertexOutput_)
77 {
78 if
79 (
80 const label nTotalVerts = points_.size();
81 (nTotalVerts && UPstream::master()) // serial only
82 )
83 {
84 os.writeKeyword(ensightFaces::kw_vertex());
85 os.write(nTotalVerts);
86 os.newline();
87
88 for (label pointi = 0; pointi < nTotalVerts; ++pointi)
89 {
90 os.write(pointi+1); // From 0-based to 1-based index
91 os.newline();
92 }
93 }
94 }
95 else
96 {
97 // Faces
98 ensightOutput::writeFaceConnectivity
99 (
100 os,
101 *this,
102 faces_,
103 false // serial
104 );
105 }
106}
