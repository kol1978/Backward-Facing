104//- A functor that returns its argument unchanged (cf. C++20 std::identity)
105//- Should \em never be specialized.
106struct identityOp
107{
108 using is_transparent = void;
109
110 template<class T>
111 constexpr T&& operator()(T&& val) const noexcept
112 {
113 return std::forward<T>(val);
114 }
115
116 // Allow use as an identity array/map
117 template<class T>
118 constexpr T&& operator[](T&& val) const noexcept
119 {
120 return std::forward<T>(val);
121 }
122};

127//- Swap non-array types as per std::swap, but in Foam namespace.
128// \sa http://www.cplusplus.com/reference/utility/swap/
129//
130// \note For complex structures, it is usual to provide a swap() member
131// function and specialize Swap()
132template<class T>
133void Swap(T& a, T& b)
134{
135 std::swap(a, b);
136}
137
138
139//- Swap array types as per std::swap example, but in Foam namespace.
140// \sa http://www.cplusplus.com/reference/utility/swap/
141template<class T, size_t N>
142void Swap(T (&a)[N], T (&b)[N])
143{
144 for (size_t i = 0; i < N; ++i)
145 {
146 Foam::Swap(a[i], b[i]);
147 }
148}

208//- Test for a \c is_base_of relationship for templated base classes
209template<template<class...> class Base, class Derived>
210struct is_template_base_of
211{
212private:
213
214 // Derived from Base => true
215 template<class... Args>
216 static std::true_type test(const Base<Args...>*);
217
218 // Fallback => false
219 static std::false_type test(...);
220
221 using decayed = std::remove_cv_t<std::remove_reference_t<Derived>>;
222
223public:
224
225 // OR: decltype(test(std::declval<decayed*>()))::value
226 static constexpr bool value
227 = decltype(test(static_cast<decayed*>(nullptr)))::value;
228};
229

416//- Rudimentary functionality similar to std::span for holding memory view
417template<class Type>
418class span
419{
420 // Private Data
421
422 //- The data pointer
423 Type* data_;
424
425 //- The size of the span
426 std::size_t size_;
427
428
429public:
430
431 // STL type definitions
432
433 using element_type = Type;
434 using value_type = std::remove_cv_t<Type>;
435 using size_type = std::size_t;
436 using difference_type = std::ptrdiff_t;
437 using pointer = Type*;
438 using const_pointer = const Type*;
439 using reference = element_type&;
440 using const_reference = const element_type&;
441 using iterator = Type*;
442
443
444 // Generated Methods
445
446 //- Copy construct
447 span(const span& other) noexcept = default;
448
449 //- Copy assignment
450 span& operator=(const span& other) noexcept = default;
451
452
453 // Constructors
454
455 //- Default construct
456 constexpr span() noexcept
457 :
458 data_(nullptr),
459 size_(0)
460 {}
461
462 //- Construct from pointer and size
463 constexpr span(pointer ptr, size_type count) noexcept
464 :
465 data_(ptr),
466 size_(count)
467 {}
468
469 //- Construct from begin/end pointers
470 constexpr span(pointer first, pointer last) noexcept
471 :
472 data_(first),
473 size_(last - first)
474 {}
475
476
477 //- Destructor
478 ~span() noexcept = default;
479
480
481 // Member Functions
482
483 // Iterators
484
485 //- Iterator to begin of span
486 constexpr iterator begin() const noexcept { return data_; }
487
488 //- Iterator to one-past end of span
489 constexpr iterator end() const noexcept { return (data_ + size_); }
490
491
492 // Element access
493
494 //- Access the first element. Undefined if span is empty
495 constexpr reference front() const { return *(data_); }
496
497 //- Access the last element. Undefined if span is empty
498 constexpr reference back() const { return *(data_ + size_ - 1); }
499
500 //- Access an element of the sequence
501 constexpr reference operator[](size_type idx) const
502 {
503 return *(data_ + idx);
504 }
505
506 //- Return a pointer to the beginning of the sequence
507 constexpr pointer data() const noexcept { return data_; }
508
509
510 // Observers
511
512 //- Number of elements in the sequence
513 constexpr size_type size() const noexcept { return size_; }
514
515 //- The size of the sequence in bytes
516 constexpr size_type size_bytes() const noexcept
517 {
518 return (size_*sizeof(Type));
519 }
520
521 //- True if the sequence is empty
522 constexpr bool empty() const noexcept { return !size_; }
523
524
525 // Subview
526
527 //- Obtains a span of the first count elements
528 span<Type> first(size_type count) const noexcept
529 {
530 return span<Type>(data_, count);
531 }
532
533 //- Obtains a span of the last count elements
534 span<Type> last(size_type count) const noexcept
535 {
536 return span<Type>(data_ + (size_ - count), count);
537 }
538
539 //- Obtains a sub-span starting at pos until end of the sequence
540 span<Type> subspan(size_type pos) const noexcept
541 {
542 return span<Type>(data_ + pos, size_ - pos);
543 }
544
545 //- Obtains a sub-span of length len starting at pos.
546 // Graciously handles excess lengths.
547 span<Type> subspan(size_type pos, size_type len) const noexcept
548 {
549 return span<Type>(data_ + pos, std::min(size_ - pos, len));
550 }
551
552
553 // Additional members, similar to UList etc.
554 // The std::span has as_bytes() and as_writeable_bytes() as free functions
555
556 //- A readonly view as byte content
557 constexpr const char* cdata_bytes() const noexcept
558 {
559 return reinterpret_cast<const char*>(data_);
560 }
561
562 //- A writable view as byte content (if the pointer type is non-const).
563 //- Like data(), the access itself is const.
564 template
565 <
566 class TypeT = Type,
567 class = std::enable_if_t<!std::is_const_v<TypeT>>
568 >
569 char* data_bytes() const noexcept
570 {
571 return reinterpret_cast<char*>(data_);
572 }
573};
