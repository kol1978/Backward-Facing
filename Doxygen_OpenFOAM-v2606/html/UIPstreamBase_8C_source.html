49inline static void processFlags(Istream& is, int flagMask)
50{

80// Return the position with specified byte alignment (eg, word boundary)
81template<typename IntType>
82inline IntType byteAlign(IntType pos, size_t align) noexcept
83{
84 return
85 (
86 (align > 1)
87 ? (align + ((pos - 1) & ~(align - 1)))
88 : pos
89 );
90}

112template<class T>
113inline void Foam::UIPstreamBase::readFromBuffer(T& val)
114{
115 prepareBuffer(sizeof(T));
116
117 val = reinterpret_cast<T&>(recvBuf_[recvBufPos_]);
118 recvBufPos_ += sizeof(T);
119 checkEof();
120}

123inline void Foam::UIPstreamBase::readFromBuffer
124(
125 void* data,
126 const size_t count
127)
128{
129 if (!count)
130 {
131 // Nothing to do, don't even do any alignment
132 return;
133 }
134
135 if (data)
136 {
137 // Only read into non-null pointers.
138 // With nullptr just behaves like a forward seek
139
140 const char* const __restrict__ buf = &recvBuf_[recvBufPos_];
141 char* const __restrict__ output = reinterpret_cast<char*>(data);
142
143 for (size_t i = 0; i < count; ++i)
144 {
145 output[i] = buf[i];
146 }
147 }
148
149 recvBufPos_ += count;
150 checkEof();
151}

154inline Foam::Istream& Foam::UIPstreamBase::readString(std::string& str)
155{
156 // Use std::string::assign() to copy content, including embedded nul chars.
157 // Stripping (when desired) is the responsibility of the sending side.
158
159 size_t len;
160 readFromBuffer(len);
161
162 if (len)
163 {
164 str.assign(&recvBuf_[recvBufPos_], len);
165 recvBufPos_ += len;
166 checkEof();
167 }
168 else
169 {
170 str.clear();
171 }
172
173 return *this;
174}

289Foam::Istream& Foam::UIPstreamBase::read(token& t)
290{
291 // Return the put back token if it exists
292 // - with additional handling for special stream flags
293 if (Istream::getBack(t))
294 {
295 if (t.isFlag())
296 {
297 processFlags(*this, t.flagToken());
298 }
299 else
300 {
301 return *this;
302 }
303 }
304
305
306 // Reset token, adjust its line number according to the stream
307 t.reset();
308 t.lineNumber(this->lineNumber());
309
310
311 // Read character, return on error
312 // - with additional handling for special stream flags
313
314 char c;
315 do
316 {
317 if (!read(c))
318 {
319 t.setBad(); // Error
320 return *this;
321 }
322
323 if (c == token::FLAG)
324 {
325 if (char flagVal; read(flagVal))
326 {
327 processFlags(*this, flagVal);
328 }
329 else
330 {
331 t.setBad(); // Error
332 return *this;
333 }
334 }
335 }
336 while (c == token::FLAG);
337
338
339 // Analyse input starting with this character.
340 switch (c)
341 {
342 // Punctuation
343 case token::END_STATEMENT :
344 case token::BEGIN_LIST :
345 case token::END_LIST :
346 case token::BEGIN_SQR :
347 case token::END_SQR :
348 case token::BEGIN_BLOCK :
349 case token::END_BLOCK :
350 case token::COLON :
351 case token::COMMA :
352 case token::ASSIGN :
353 case token::PLUS :
354 case token::MINUS :
355 case token::MULTIPLY :
356 case token::DIVIDE :
357 {
358 t = token::punctuationToken(c);
359 return *this;
360 }
361
362 // The word-variants
363 case token::tokenType::WORD :
364 case token::tokenType::DIRECTIVE :
365 {
366 if (word val; readString(val))
367 {
368 if
369 (
370 !token::compound::isCompound(val)
371 || !readCompoundToken(t, val, *this)
372 )
373 {
374 t = std::move(val);
375 t.setType(token::tokenType(c));
376 }
377 }
378 else
379 {
380 t.setBad();
381 }
382 return *this;
383 }
384
385 // The string-variants
386 case token::tokenType::STRING :
387 case token::tokenType::EXPRESSION :
388 case token::tokenType::VARIABLE :
389 case token::tokenType::VERBATIM :
390 case token::tokenType::CHAR_DATA :
391 {
392 if (string val; readString(val))
393 {
394 t = std::move(val);
395 t.setType(token::tokenType(c));
396 }
397 else
398 {
399 t.setBad();
400 }
401 return *this;
402 }
403
404 // (signed) int32
405 case token::tokenType::INTEGER_32 :
406 {
407 if (int32_t val; read(val))
408 {
409 t.int32Token(val);
410 }
411 else
412 {
413 t.setBad();
414 }
415 return *this;
416 }
417
418 // (signed) int64
419 case token::tokenType::INTEGER_64 :
420 {
421 if (int64_t val; read(val))
422 {
423 t.int64Token(val);
424 }
425 else
426 {
427 t.setBad();
428 }
429 return *this;
430 }
431
432 // (unsigned) int32
433 case token::tokenType::UNSIGNED_INTEGER_32 :
434 {
435 if (uint32_t val; read(val))
436 {
437 t.uint32Token(val);
438 }
439 else
440 {
441 t.setBad();
442 }
443 return *this;
444 }
445
446 // (unsigned) int64
447 case token::tokenType::UNSIGNED_INTEGER_64 :
448 {
449 if (uint64_t val; read(val))
450 {
451 t.uint64Token(val);
452 }
453 else
454 {
455 t.setBad();
456 }
457 return *this;
458 }
459
460 // Float
461 case token::tokenType::FLOAT :
462 {
463 if (float val; read(val))
464 {
465 t = val;
466 }
467 else
468 {
469 t.setBad();
470 }
471 return *this;
472 }
473
474 // Double
475 case token::tokenType::DOUBLE :
476 {
477 if (double val; read(val))
478 {
479 t = val;
480 }
481 else
482 {
483 t.setBad();
484 }
485 return *this;
486 }
487
488 // Character (returned as a single character word) or error
489 default:
490 {
491 if (isalpha(c))
492 {
493 t = charToWord(c);
494 return *this;
495 }
496
497 setBad();
498 t.setBad();
499
500 return *this;
501 }
502 }
503}

569Foam::Istream& Foam::UIPstreamBase::read(char* data, std::streamsize count)
570{
571 if (count > 0)
572 {
573 // For count=0 (no-op, also no alignment)
574 // Further handling (eg, data=nullptr) decided in readRaw()
575 // - see UOPstream::write(const char*, streamsize)
576
577 beginRawRead(); // Check binary, enforce alignment
578 readRaw(data, count);
579 endRawRead();
580 }
581
582 return *this;
583}

586Foam::Istream& Foam::UIPstreamBase::readRaw(char* data, std::streamsize count)
587{
588 // No check for IOstreamOption::BINARY since this is either done in the
589 // beginRawRead() method, or the caller knows what they are doing.
590
591 // Any alignment must have been done prior to this call
592 if (count > 0)
593 {
594 readFromBuffer(data, count);
595 }
596 return *this;

600bool Foam::UIPstreamBase::beginRawRead()
601{
602 if (format() != IOstreamOption::BINARY)
603 {
604 FatalErrorInFunction
605 << "stream format not binary"
606 << Foam::abort(FatalError);
607 }
608
609 // Align on word boundary (64-bit)
610 // - as per read(const char*, streamsize)
611 // The check for zero-size will have been done by the caller
612 prepareBuffer(8);
613
614 return true;
615}

657void Foam::UIPstreamBase::rewind()
658{
659 Istream::rewind(); // Drop any putback
660
661 recvBufPos_ = 0; // Assume the entire buffer is for us to read from
662 setOpened();
663 setGood();
664 if (recvBuf_.empty() || !messageSize_)
665 {
666 setEof();
667 }
