41// General blocking/non-blocking MPI receive
42std::streamsize Foam::UPstream::mpi_receive
43(
44 const UPstream::commsTypes commsType,
45 void* buf, // Type checking done by caller
46 std::streamsize count,
47 const UPstream::dataTypes dataTypeId, // Proper type passed by caller
48 const int fromProcNo,
49 const int tag,
50 const int communicator,
51 UPstream::Request* req
52)
53{
54 MPI_Datatype datatype = PstreamGlobals::getDataType(dataTypeId);
55
56 PstreamGlobals::reset_request(req);
57
58 // Could check if nonBlocking and request are consistently specified...
59
60
61 // TODO: some corrective action, at least when not nonBlocking
62 #if 0
63 // No warnings here, just on the sender side.
64 if (count > std::streamsize(INT_MAX))
65 {
66 Perr<< "[mpi_recv] from rank " << fromProcNo
67 << " exceeds INT_MAX values of "
68 << PstreamGlobals::dataType_name(datatype)
69 << Foam::endl;
70
71 error::printStack(Perr);
72 }
73 #endif
74
75 if (FOAM_UNLIKELY(PstreamGlobals::warnCommunicator(communicator)))
76 {
77 Perr<< "[mpi_recv] : starting recv from:" << fromProcNo
78 << " type:" << int(dataTypeId)
79 << " count:" << label(count)
80 << " tag:" << tag << " comm:" << communicator
81 << " commsType:" << UPstream::commsTypeNames[commsType]
82 << " warnComm:" << UPstream::warnComm
83 << Foam::endl;
84 error::printStack(Perr);
85 }
86 else if (FOAM_UNLIKELY(UPstream::debug))
87 {
88 Perr<< "[mpi_recv] : starting recv from:" << fromProcNo
89 << " type:" << int(dataTypeId)
90 << " count:" << label(count)
91 << " tag:" << tag << " comm:" << communicator
92 << " commsType:" << UPstream::commsTypeNames[commsType]
93 << Foam::endl;
94 }
95
96 int returnCode = MPI_ERR_UNKNOWN;
97
98 profilingPstream::beginTiming();
99
100 if
101 (
102 commsType == UPstream::commsTypes::buffered
103 || commsType == UPstream::commsTypes::scheduled
104 )
105 {
106 // Not UPstream::commsTypes::nonBlocking
107
108 MPI_Status status;
109
110 {
111 returnCode = MPI_Recv
112 (
113 buf,
114 count,
115 datatype,
116 fromProcNo,
117 tag,
118 PstreamGlobals::MPICommunicators_[communicator],
119 &status
120 );
121 }
122
123 profilingPstream::addGatherTime();
124
125 if (FOAM_UNLIKELY(returnCode != MPI_SUCCESS))
126 {
127 FatalErrorInFunction
128 << "[mpi_recv] : cannot receive message from:"
129 << fromProcNo
130 << " type:" << int(dataTypeId)
131 << " count:" << label(count) << " tag:" << tag
132 << Foam::abort(FatalError);
133 return 0;
134 }
135 else if (FOAM_UNLIKELY(UPstream::debug))
136 {
137 Perr<< "[mpi_recv] : finished recv from:"
138 << fromProcNo
139 << " type:" << int(dataTypeId)
140 << " count:" << label(count) << " tag:" << tag
141 << Foam::endl;
142 }
143
144 // Check size of message read (number of basic elements)
145 MPI_Count num_recv(0);
146 MPI_Get_elements_x(&status, datatype, &num_recv);
147
148 // Errors
149 if (FOAM_UNLIKELY(num_recv == MPI_UNDEFINED || int64_t(num_recv) < 0))
150 {
151 FatalErrorInFunction
152 << "[mpi_recv] : receive from:" << fromProcNo
153 << " type:" << int(dataTypeId)
154 << " received count is undefined or negative value"
155 << Foam::abort(FatalError);
156 }
157 else
158 {
159 // From number of basic elements to number of 'datatype'
160 num_recv /= PstreamGlobals::dataTypesCount_[int(dataTypeId)];
161 }
162
163 if (FOAM_UNLIKELY(int64_t(num_recv) > int64_t(UList<char>::max_size())))
164 {
165 FatalErrorInFunction
166 << "[mpi_recv] : receive from:" << fromProcNo
167 << " type:" << int(dataTypeId)
168 << " received count is larger than UList<T>::max_size()"
169 << Foam::abort(FatalError);
170 }
171 else if (FOAM_UNLIKELY(count < std::streamsize(num_recv)))
172 {
173 FatalErrorInFunction
174 << "[mpi_recv] : receive from:" << fromProcNo
175 << " type:" << int(dataTypeId)
176 << " count:" << label(count)
177 << " buffer is too small for incoming message ("
178 << label(num_recv) << ')'
179 << Foam::abort(FatalError);
180 }
181
182 return std::streamsize(num_recv);
183 }
184 else if (commsType == UPstream::commsTypes::nonBlocking)
185 {
186 MPI_Request request;
187
188 {
189 returnCode = MPI_Irecv
190 (
191 buf,
192 count,
193 datatype,
194 fromProcNo,
195 tag,
196 PstreamGlobals::MPICommunicators_[communicator],
197 &request
198 );
199 }
200
201 if (FOAM_UNLIKELY(returnCode != MPI_SUCCESS))
202 {
203 FatalErrorInFunction
204 << "[mpi_recv] : cannot start non-blocking receive from:"
205 << fromProcNo
206 << " type:" << int(dataTypeId)
207 << " count:" << label(count)
208 << Foam::abort(FatalError);
209
210 return 0;
211 }
212
213 PstreamGlobals::push_request(request, req);
214 profilingPstream::addRequestTime();
215
216
217 if (FOAM_UNLIKELY(UPstream::debug))
218 {
219 Perr<< "[mpi_recv] : started non-blocking recv from:"
220 << fromProcNo
221 << " type:" << int(dataTypeId)
222 << " count:" << label(count) << " tag:" << tag
223 << " request:" <<
224 (req ? label(-1) : PstreamGlobals::outstandingRequests_.size())
225 << Foam::endl;
226 }
227
228 // Assume the message will be completely received.
229 return count;
230 }
231
232 FatalErrorInFunction
233 << "Unsupported communications type " << int(commsType)
234 << Foam::abort(FatalError);
235
236 return 0;
237}

242void Foam::UIPstream::bufferIPCrecv()
243{
244 // Called by constructor
245 if (FOAM_UNLIKELY(UPstream::debug))
246 {
247 Perr<< "UIPstream IPC read buffer :"
248 << " from:" << fromProcNo_
249 << " tag:" << tag_ << " comm:" << comm_
250 << " wanted size:" << recvBuf_.capacity()
251 << Foam::endl;
252 }
253
254 // Fallback value
255 messageSize_ = recvBuf_.capacity();
256
257 if (commsType() == UPstream::commsTypes::nonBlocking)
258 {
259 // Non-blocking
260 // ~~~~~~~~~~~~
261 // No chance of probing for size nor relying on the returned message
262 // size (since it returns immediately without any further checks)
263 //
264 // Fortunately there are not many (any?) places that are using
265 // a non-blocking IPstream with streaming anyhow.
266
267 messageSize_ = recvBuf_.size();
268 }
269 else if (!recvBuf_.capacity())
270 {
271 // No buffer size allocated/specified - probe size of incoming message
272 profilingPstream::beginTiming();
273
274 const int source = (fromProcNo_ < 0) ? MPI_ANY_SOURCE : fromProcNo_;
275 // Supporting MPI_ANY_TAG is not particularly useful...
276
277 MPI_Status status;
278
279 MPI_Probe
280 (
281 source,
282 tag_,
283 PstreamGlobals::MPICommunicators_[comm_],
284 &status
285 );
286
287 if (fromProcNo_ < 0)
288 {
289 // Use the correct (probed) source
290 fromProcNo_ = status.MPI_SOURCE;
291 }
292
293 profilingPstream::addProbeTime();
294
295 // Buffer of characters (bytes)
296 MPI_Count num_recv(0);
297 MPI_Get_elements_x(&status, MPI_BYTE, &num_recv);
298
299 // Errors
300 if (FOAM_UNLIKELY(num_recv == MPI_UNDEFINED || int64_t(num_recv) < 0))
301 {
302 FatalErrorInFunction
303 << "UIPstream IPC read buffer from:" << fromProcNo_
304 << " received count is undefined or negative value"
305 << Foam::abort(FatalError);
306 }
307
308 // Count is already in basic elements, no need to scale the result
309
310 if (FOAM_UNLIKELY(int64_t(num_recv) > int64_t(UList<char>::max_size())))
311 {
312 FatalErrorInFunction
313 << "UIPstream IPC read buffer from:" << fromProcNo_
314 << " received count is larger than UList<T>::max_size()"
315 << Foam::abort(FatalError);
316 }
317
318 if (FOAM_UNLIKELY(UPstream::debug))
319 {
320 Perr<< "UIPstream::bufferIPCrecv : probed size:"
321 << label(num_recv) << Foam::endl;
322 }
323
324 recvBuf_.resize(label(num_recv));
325 messageSize_ = label(num_recv);
326 }
327
328 // Errors
329 if (FOAM_UNLIKELY(fromProcNo_ < 0))
330 {
331 FatalErrorInFunction
332 << "UIPstream IPC failed with ANY_SOURCE probing?"
333 << " cannot have proc=" << fromProcNo_
334 << Foam::abort(FatalError);
335 }
336
337 std::streamsize count = UPstream::mpi_receive
338 (
339 commsType(),
340 recvBuf_.data(), // buffer
341 messageSize_, // expected size
342 UPstream::dataTypes::type_byte, // MPI_BYTE
343 fromProcNo_,
344 tag_,
345 comm_,
346 nullptr // UPstream::Request
347 );
348
349 // Errors
350 if (FOAM_UNLIKELY(count < 0))
351 {
352 FatalErrorInFunction
353 << "UIPstream IPC read buffer from:" << fromProcNo_
354 << " with negative size?"
355 << Foam::abort(FatalError);
356 }
357 else if (FOAM_UNLIKELY(int64_t(count) > int64_t(UList<char>::max_size())))
358 {
359 FatalErrorInFunction
360 << "UIPstream IPC read buffer from:" << fromProcNo_
361 << " received size is larger than UList<T>::max_size()"
362 << Foam::abort(FatalError);
363 }
364
365 // Set addressed size. Leave actual allocated memory intact.
366 recvBuf_.resize(label(count));
367 messageSize_ = label(count);
368
369 if (recvBuf_.empty())
370 {
371 setEof();
372 }
373}
