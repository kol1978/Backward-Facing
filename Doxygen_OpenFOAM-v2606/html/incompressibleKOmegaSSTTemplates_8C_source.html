68template<class CDType>
69auto incompressibleKOmegaSST::F1(const CDType& CDkOmega) const -> decltype(auto)
70{
71 // Field for sizing constants
72 const volScalarField& dummy = this->omega_;
73
74 // Constants in volScalarField form
75 const volConstant one(dummy, scalar(1));
76 const volConstant hundred(dummy, scalar(100));
77 const volConstant fiveHundred(dummy, scalar(500));
78 const volConstant betaStar(dummy, betaStar_);
79 const volConstant twoOverBetaStar(dummy, scalar(2)/betaStar_);
80 const volConstant rho(dummy, 1.0);
81
82 // Wrap some fields
83 const auto wk = volExpr(k_);
84 const auto wy = volExpr(y_);
85 const auto womega = volExpr(omega_);
86 const auto wmu = volExpr(this->mu());
87
88 auto CDkOmegaPlus = max
89 (
90 CDkOmega,
91 volConstant(dummy, dimensionedScalar(dimless/sqr(dimTime), 1.0e-10))
92 );
93
94 auto arg1 = min
95 (
96 min
97 (
98 max
99 (
100 (one/betaStar)*sqrt(wk)/(womega*wy),
101 fiveHundred*(wmu/rho)/(sqr(wy)*womega)
102 ),
103 volConstant(dummy, 4*alphaOmega2_)*wk/(CDkOmegaPlus*sqr(wy))
104 ),
105 volConstant(dummy, 10)
106 );
107
108 return tanh(pow4(arg1));
109}

112auto incompressibleKOmegaSST::F2() const -> decltype(auto)
113{
114 // Field for sizing constants. Only sizes used, not values. Should be of
115 // same type as constant and persist until after evaluation.
116 const volScalarField& dummy = this->omega_;
117
118 // Constants in volScalarField form
119 const volConstant hundred(dummy, scalar(100));
120 const volConstant fiveHundred(dummy, scalar(500));
121 const volConstant twoOverBetaStar(dummy, scalar(2)/betaStar_);
122
123 const volConstant rho(dummy, 1.0);
124
125 // Wrap some fields
126 const auto wk = volExpr(k_);
127 const auto wy = volExpr(y_);
128 const auto womega = volExpr(omega_);
129 const auto wmu = volExpr(this->mu()());
130
131 auto arg2 = min
132 (
133 max
134 (
135 twoOverBetaStar * sqrt(wk) / (womega*wy),
136 fiveHundred
137 * (wmu / rho)
138 / (sqr(wy)*womega)
139 ),
140 hundred
141 );
142
143 return tanh(sqr(arg2));
144}

147auto incompressibleKOmegaSST::F3() const -> decltype(auto)
148{
149 // Field for sizing constants. Only sizes used, not values. Should be of
150 // same type as constant and persist until after evaluation.
151 const volScalarField& dummy = this->omega_;
152
153 // Constants in volScalarField form
154 const volConstant c150(dummy, scalar(150));
155 const volConstant c10(dummy, scalar(10));
156 const volConstant rho(dummy, 1.0);
157
158 // Wrap some fields
159 const auto wy = volExpr(y_);
160 const auto womega = volExpr(omega_);
161 const auto wmu = volExpr(this->mu()());
162
163 auto arg3 = min
164 (
165 c150*(wmu/rho)/(womega*sqr(wy)),
166 c10
167 );
168
169 return volConstant(dummy, 1) - tanh(pow4(arg3));
170}

173template<class S2Type>
174void incompressibleKOmegaSST::correctNut(const S2Type& S2)
175{
176 // Field for sizing constants
177 const volScalarField& dummy = this->omega_;
178
179 const volConstant a1(dummy, a1_);
180 const volConstant b1(dummy, b1_);
181
182 // Wrap some fields
183 const auto wk = volExpr(k_);
184 const auto womega = volExpr(omega_);
185 auto wnut = volEvaluator(this->nut_);
186
187 // Correct the turbulence viscosity
188 if (F3_)
189 {
190 wnut = a1*wk/max(a1*womega, b1*F2()*F3()*sqrt(S2));
191 }
192 else
193 {
194 wnut = a1*wk/max(a1*womega, b1*F2()*sqrt(S2));
195 }
196 this->nut_.correctBoundaryConditions();
197 fv::options::New(this->mesh_).correct(this->nut_);
198}

201template<class E1>
202auto incompressibleKOmegaSST::Pk(const E1& G) const -> decltype(auto)
203{
204 // Field for sizing constants. Only sizes used, not values. Should be of
205 // same type as constant and persist until after evaluation.
206 const label nCells = this->omega_.size();
207
208 // Constants in scalarList form
209 const constant c1(nCells, c1_);
210 const constant betaStar(nCells, betaStar_);
211
212 // Wrap some fields
213 const auto wk = expr(k_.internalField());
214 const auto womega = expr(omega_.internalField());
215
216 return min(G, (c1*betaStar)*wk*womega);
217}

220template<class E1, class E2, class E3>
221auto incompressibleKOmegaSST::GbyNu
222(
223 const E1& GbyNu0,
224 const E2& F2,
225 const E3& S2
226) const -> decltype(auto)
227{
228 // Field for sizing constants. Only sizes used, not values. Should be of
229 // same type as constant and persist until after evaluation.
230 const label nCells = this->omega_.size();
231
232 // Constants in scalarList form
233 const constant a1(nCells, a1_);
234 const constant b1(nCells, b1_);
235 const constant c1(nCells, c1_);
236 const constant betaStar(nCells, betaStar_);
237
238 // Wrap some fields
239 const auto womega = expr(omega_.internalField());
240
241 return min
242 (
243 GbyNu0,
244 (c1/a1)*betaStar*womega*max(a1*womega, b1*F2*sqrt(S2))
245 );
246}

250template<class E1>
251auto incompressibleKOmegaSST::blend
252(
253 const E1& F1,
254 const scalar& p1,
255 const scalar& p2
256) const -> decltype(auto)
257{
258 // Field for sizing constants
259 const volScalarField& dummy = this->omega_;
260 const volConstant psi1(dummy, p1);
261 const volConstant psi2(dummy, p2);
262
263 return F1*(psi1 - psi2) + psi2;
264}
