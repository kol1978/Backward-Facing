80bool Foam::resolutionIndexModel::read(const dictionary& dict)
81{
82 resultName_ = dict.getOrDefault("result", type());
83
84 auto* indexPtr = mesh_.getObjectPtr<volScalarField>(resultName_);
85
86 if (!indexPtr)
87 {
88 indexPtr = new volScalarField
89 (
90 IOobject
91 (
92 resultName_,
93 mesh_.time().timeName(),
94 mesh_.thisDb(),
95 IOobject::LAZY_READ,
96 IOobject::NO_WRITE,
97 IOobject::REGISTER
98 ),
99 mesh_,
100 dimensionedScalar(dimless, Zero),
101 fvPatchFieldBase::zeroGradientType()
102 );
103
104 regIOobject::store(indexPtr);
105 }
106
107 return true;
108}
