33template<class T, int SizeMin>
34Foam::label Foam::DynamicList<T, SizeMin>::removeElements
35(
36 const labelRange& slice
37)
38{
39 if (!slice.good())
40 {
41 // No-op
42 return 0;
43 }
44
45 // Note: already checked for valid begin_value before
46 if (slice.end_value() >= this->size())
47 {
48 // Remove entire tail
49 this->resize(slice.begin_value());
50 }
51 else
52 {
53 // Move all trailing elements down into space previously
54 // occupied by the slice. Truncate after
55 std::move
56 (
57 this->begin(slice.end_value()),
58 this->end(),
59 this->begin(slice.begin_value())
60 );
61
62 this->resize(this->size() - slice.size());
63 }
64
65 return slice.size();
66}

69template<class T, int SizeMin>
70Foam::label Foam::DynamicList<T, SizeMin>::subsetElements
71(
72 const labelRange& slice
73)
74{
75 if (slice.begin_value() > 0)
76 {
77 // Move elements down.
78 // Since begin_value > 0, the initial destination is non-overlapping
79 std::move
80 (
81 this->begin(slice.begin_value()),
82 this->begin(slice.end_value()),
83 this->begin()
84 );
85 }
86
87 // Don't need min size, since slice size was already checked before
88 resize(slice.size());
89 return this->size();
90}
