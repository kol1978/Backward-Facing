36bool Foam::TimePaths::detectProcessorCase()
37{
38 if (processorCase_)
39 {
40 return processorCase_;
41 }
42
43 // Look for "processor", but should really check for following digits too
44 const auto sep = globalCaseName_.rfind('/');
45 const auto pos = globalCaseName_.find
46 (
47 "processor",
48 (sep == string::npos ? 0 : sep)
49 );
50
51 if (pos == 0)
52 {
53 globalCaseName_ = ".";
54 processorCase_ = true;
55 }
56 else if (pos != string::npos && sep != string::npos && sep == pos-1)
57 {
58 globalCaseName_.resize(sep);
59 processorCase_ = true;
60 }
61
62 return processorCase_;
63}

131Foam::label Foam::TimePaths::findClosestTimeIndex
132(
133 const instantList& timeDirs,
134 const scalar t,
135 const word& constantDirName
136)
137{
138 const label nTimes = timeDirs.size();
139
140 label nearestIndex = -1;
141 scalar deltaT = GREAT;
142
143 for (label timei=0; timei < nTimes; ++timei)
144 {
145 if (timeDirs[timei].name() == constantDirName) continue;
146
147 const scalar diff = mag(timeDirs[timei].value() - t);
148 if (diff < deltaT)
149 {
150 deltaT = diff;
151 nearestIndex = timei;
152 }
153 }
154
155 return nearestIndex;
156}

159Foam::instant Foam::TimePaths::findClosestTime(const scalar t) const
160{
161 instantList timeDirs = findTimes(path(), constant());
162
163 const label nTimes = timeDirs.size();
164
165 if (nTimes == 0)
166 {
167 // Cannot really fail at this point, but for some safety...
168 return instant(0, constant());
169 }
170 else if (nTimes == 1)
171 {
172 // Only one time (likely "constant") so return it
173 return timeDirs.front();
174 }
175 else if (t < timeDirs[1].value())
176 {
177 return timeDirs[1];
178 }
179 else if (t > timeDirs.back().value())
180 {
181 return timeDirs.back();
182 }
183
184 label nearestIndex = 0; // Failsafe value
185 scalar deltaT = GREAT;
186
187 for (label timei=1; timei < nTimes; ++timei)
188 {
189 const scalar diff = mag(timeDirs[timei].value() - t);
190 if (diff < deltaT)
191 {
192 deltaT = diff;
193 nearestIndex = timei;
194 }
195 }
196
197 return timeDirs[nearestIndex];
198}
