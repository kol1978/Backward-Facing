-------------------------------------------------------------Запуск
Режим 1: Pure MPI (базовая линия — 12 процессов, 1 поток)

export OMP_NUM_THREADS=1
numactl --interleave=all mpirun -np 12 \
    test-parallel-hybrid -parallel -N 100000 -mode pure_mpi -repeat 3

Режим 2: Hybrid (2 процесса по сокетам, 6 потоков на каждый)

export OMP_NUM_THREADS=6
numactl --cpunodebind=0 --membind=0 mpirun -np 1 \
    test-parallel-hybrid -N 100000 -mode pure_omp -repeat 3 &
numactl --cpunodebind=1 --membind=1 mpirun -np 1 \
    test-parallel-hybrid -N 100000 -mode pure_omp -repeat 3 &
wait

Режим 2b: Hybrid (упрощённый запуск, 2 процесса, 6 потоков)

export OMP_NUM_THREADS=6
mpirun -np 2 --map-by ppr:1:socket --bind-to core \
    test-parallel-hybrid -parallel -N 100000 -mode hybrid -repeat 3

Режим 3: Pure OpenMP (1 процесс, 12 потоков)

export OMP_NUM_THREADS=12
numactl --interleave=all \
    test-parallel-hybrid -N 100000 -mode pure_omp -repeat 3
