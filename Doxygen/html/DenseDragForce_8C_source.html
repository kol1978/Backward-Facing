85template<class CloudType>
86void Foam::DenseDragForce<CloudType>::cacheFields(const bool store)
87{
88 if (store)
89 {
90 if (!this->mesh().template foundObject<volScalarField>(alphacName_))
91 {
92 alphacPtr_.reset
93 (
94 new volScalarField(alphacName_, 1 - this->owner().alpha())
95 );
96 }
97
98 const volScalarField& alphac =
99 this->mesh().template lookupObject<volScalarField>(alphacName_);
100
101 alphacInterpPtr_.reset
102 (
103 interpolation<scalar>::New
104 (
105 this->owner().solution().interpolationSchemes(),
106 alphac
107 ).ptr()
108 );
109 }
110 else
111 {
112 alphacInterpPtr_.clear();
113 alphacPtr_.clear();
114 }
115}
