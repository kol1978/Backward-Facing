146Foam::label Foam::nonConformalCoupledPolyPatch::errorPatchIndex() const
147{
148 if (errorPatchIndex_ == -1)
149 {
150 forAll(patch_.boundaryMesh(), patchi)
151 {
152 const polyPatch& p = patch_.boundaryMesh()[patchi];
153
154 if
155 (
156 isA<nonConformalErrorPolyPatch>(p)
157 && refCast<const nonConformalErrorPolyPatch>(p).origPatchIndex()
158 == origPatchIndex()
159 )
160 {
161 errorPatchIndex_ = patchi;
162 break;
163 }
164 }
165 }
166
167 if (errorPatchIndex_ == -1)
168 {
169 FatalErrorInFunction
170 << "No error patch of type "
171 << nonConformalErrorPolyPatch::typeName
172 << " defined for patch " << origPatchName()
173 << exit(FatalError);
174 }
175
176 return errorPatchIndex_;
177}
