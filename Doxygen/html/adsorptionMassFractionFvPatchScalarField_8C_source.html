73Foam::tmp<Foam::scalarField>
74Foam::adsorptionMassFractionFvPatchScalarField::calcPhiYp() const
75{
76 if (c_ == scalar(0))
77 {
78 return tmp<scalarField>(new scalarField(patch().size(), Zero));
79 }
80
81 const word& YName = internalField().name();
82
83 const fluidThermo& thermo =
84 db().lookupObject<fluidThermo>(physicalProperties::typeName);
85
86 // Get the cell-mass fraction
87 const scalarField Yc(patchInternalField());
88
89 // Get the patch delta coefficients multiplied by the diffusivity
90 const fluidThermophysicalTransportModel& ttm =
91 db().lookupType<fluidThermophysicalTransportModel>();
92
93 const volScalarField& Yi = refCast<const volScalarField>(internalField());
94
95 const scalarField DEffDeltap
96 (
97 ttm.DEff(Yi, patch().index())*patch().deltaCoeffs()
98 );
99
100 // Get the specie molecular weight, if needed
101 scalar Wi = NaN;
102 if (property_ != massFraction)
103 {
104 const fluidMulticomponentThermo& thermo = this->thermo(db());
105 Wi = thermo.Wi(thermo.species()[YName]).value();
106 }
107
108 // Get the mixture molecular weights, if needed
109 tmp<scalarField> tW;
110 if (property_ == moleFraction || property_ == partialPressure)
111 {
112 tW = thermo.W(patch().index());
113 }
114
115 // Construct coefficients that convert mass fraction to the property that
116 // drives the transfer
117 scalarField k(patch().size(), 1);
118 switch(property_)
119 {
120 case massFraction:
121 break;
122
123 case moleFraction:
124 k *= tW/Wi;
125 break;
126
127 case molarConcentration:
128 k *= thermo.rho(patch().index())/Wi;
129 break;
130
131 case partialPressure:
132 k *= thermo.p().boundaryField()[patch().index()]*tW/Wi;
133 break;
134 }
135
136 // The transport is limited by both the difference in the amount of the
137 // specie available and the rate of diffusion to the wall. The coefficients
138 // associated with these two transfer process are harmonically averaged to
139 // represent this limiting.
140 return
141 patch().magSf()
142 /(1/c_ + k/DEffDeltap)
143 *k*Yc;
144}
