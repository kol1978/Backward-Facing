51void combineReduce

54 T& Value,
55 const CombineOp& cop,
56 const int tag,

65template<class T, class CombineOp>
66void combineReduce
67(
68 T& Value,
69 const CombineOp& cop,
70 const int tag = Pstream::msgType(),
71 const label comm = Pstream::worldComm
72)
73{
74 if (UPstream::nProcs(comm) < UPstream::nProcsSimpleSum)
75 {
76 Pstream::combineGather
77 (
78 UPstream::linearCommunication(comm),
79 Value,
80 cop,
81 tag,
82 comm
83 );
84 Pstream::combineScatter
85 (
86 UPstream::linearCommunication(comm),
87 Value,
88 tag,
89 comm
90 );
91 }
92 else
93 {
94 Pstream::combineGather
95 (
96 UPstream::treeCommunication(comm),
97 Value,
98 cop,
99 tag,
100 comm
101 );
102 Pstream::combineScatter
103 (
104 UPstream::treeCommunication(comm),
105 Value,
106 tag,
107 comm
108 );
109 }
110}
