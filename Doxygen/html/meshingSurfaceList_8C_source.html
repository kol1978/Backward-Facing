30void Foam::meshingSurfaceList::mergeBoundingBoxes
31(
32 boundBox& bb1,
33 const boundBox& bb2
34)
35{
36 if (bb1.volume() == 0)
37 {
38 bb1 = bb2;
39 return;
40 }
41
42 point& min1 = bb1.min();
43 point& max1 = bb1.max();
44 const point& min2 = bb2.min();
45 const point& max2 = bb2.max();
46
47 forAll(min1, i)
48 {
49 min1[i] = Foam::min(min1[i], min2[i]);
50 max1[i] = Foam::max(max1[i], max2[i]);
51 }
52}

70bool Foam::meshingSurfaceList::regionsValid
71(
72 const wordList& specifiedRegions,
73 const wordList& regions,
74 const word& opt
75)
76{
77 if (specifiedRegions.empty())
78 {
79 return false;
80 }
81
82 forAll(specifiedRegions, s)
83 {
84 bool match(false);
85
86 forAll(regions, r)
87 {
88 match = (specifiedRegions[s] == regions[r]) || match;
89 if (match)
90 {
91 break;
92 }
93 }
94
95 if (!match)
96 {
97 FatalErrorInFunction
98 << "Region '"<< specifiedRegions[s]
99 << "' specified with the '" << opt
100 << "' option" << nl
101 << "does not match any regions in the external surface"
102 << exit(FatalError);
103 }
104 }
105
106 return true;
107}

110void Foam::meshingSurfaceList::setSurfaceTypes
111(
112 const List<word>& surfaces,
113 const surfaceType& type
114)
115{
116 forAll(surfaces, i)
117 {
118 const word surface = fileName(surfaces[i]).lessExt();
119 const word surfaceType = meshingSurface::surfaceTypeNames[type];
120 bool match = false;
121
122 // Check if the surface name matches a surface name
123 forAll(*this, surfi)
124 {
125 if (surface == operator[](surfi).name())
126 {
127 // For cellZone and rotatingZone, ensure surface is closed
128 if
129 (
130 !operator[](surfi).closed()
131 && (
132 type == meshingSurface::surfaceType::rotatingZone
133 || type == meshingSurface::surfaceType::cellZone
134 )
135 )
136 {
137 FatalErrorInFunction
138 << "Argument to '-" << surfaceType
139 << "' contains the " << surfaceType << " '"
140 << surface << "'" << nl
141 << "which is not a closed surface."
142 << exit(FatalError);
143 }
144
145 operator[](surfi).type() = type;
146 match = true;
147 break;
148 }
149 }
150
151 if (!match)
152 {
153 FatalErrorInFunction
154 << "Argument to '-" << surfaceType
155 << "' contains the " << surfaceType << " '"
156 << surface << "'" << nl
157 << "which does not correspond to any surfaces."
158 << exit(FatalError);
159 }
160 }
161}

176void Foam::meshingSurfaceList::identifyCellZones()
177{
178 forAll(*this, i)
179 {
180 if
181 (
182 operator[](i).type() == surfaceType::external
183 || operator[](i).type() == surfaceType::rotatingZone
184 || operator[](i).nParts() != 1
185 || !operator[](i).closed()
186 )
187 {
188 continue;
189 }
190
191 forAll(*this, j)
192 {
193 if (i == j)
194 {
195 continue;
196 }
197
198 if (operator[](i).bb().contains(operator[](j).bb()))
199 {
200 operator[](i).type() = surfaceType::cellZone;
201 continue;
202 }
203 }
204 }
205}

223void Foam::meshingSurfaceList::reportSurfaces()
224{
225 Info<< "Case contains the following surface geometry files:"
226 << endl;
227
228 forAll(*this, i)
229 {
230 Info<< "+ " << operator[](i).name() << nl
231 << " + File: " << operator[](i).file() << nl
232 << " + Bounding box: " << operator[](i).bb() << nl
233 << " + " << (operator[](i).closed() ? "Closed" : "Open")
234 << " surface" << endl;
235
236 switch (operator[](i).type())
237 {
238 case surfaceType::external:
239 {
240 Info << " + External boundary surface" << endl;
241
242 if (!operator[](i).inletRegions().empty())
243 {
244 Info<< " + Inlet regions: ";
245 reportWordList(operator[](i).inletRegions());
246 }
247
248 if (!operator[](i).outletRegions().empty())
249 {
250 Info << " + Outlet regions: ";
251 reportWordList(operator[](i).outletRegions());
252 }
253
254 break;
255 }
256
257 case surfaceType::wall:
258 {
259 Info << " + Wall boundary surface" << endl;
260 break;
261 }
262
263 case surfaceType::cellZone:
264 {
265 Info << " + Cell zone surface" << endl;
266 break;
267 }
268
269 case surfaceType::rotatingZone:
270 {
271 Info << " + Rotating zone surface" << endl;
272 break;
273 }
274
275 case surfaceType::baffle:
276 {
277 Info << " + Baffle wall surface" << endl;
278 break;
279 }
280 }
281 }
282
283 Info<< endl;
284}

287void Foam::meshingSurfaceList::setBounds(const boundBox& bb)
288{
289 if (bb.contains(bb_))
290 {
291 Info<< "Specified bounding box contains the overall bounding box"
292 << endl;
293 }
294 else
295 {
296 WarningInFunction
297 << "Specified bounding box does not contain the overall "
298 << "bounding box"
299 << endl;
300 }
301
302 if (bb.mag() > 1.5*bb_.mag())
303 {
304 Info<< "Specified bounding box is > (1.5 * overall "
305 << "bounding box)" << nl
306 << "**Assuming this is an external flow**"
307 << endl;
308
309 operator[](0).type() = surfaceType::wall;
310 }
311
312 Info<< endl;
313
314 bb_ = bb;
315}

320Foam::meshingSurfaceList::meshingSurfaceList
321(
322 const Time& time,
323 const fileNameList& surfaces,
324 const wordList& cellZones,
325 const wordList& rotatingZones,
326 const wordList& baffles,
327 const boundBox& bb,
328 const wordList& specifiedInletRegions,
329 const wordList& specifiedOutletRegions,
330 const bool closedDomain
331)
332:
333 PtrList<meshingSurface>(),
334 bb_(),
335 rzbb_()
336{
337 // Load all the surfaces and construct the bounding box
338 forAll(surfaces, i)
339 {
340 append(new meshingSurface(surfaces[i], time));
341
342 mergeBoundingBoxes(bb_, last().bb());
343 }
344
345 label externalID = 0;
346 forAll(surfaces, i)
347 {
348 // Test for external surface inflates the bounding box by a small factor
349 // to account for imprecise surface geometry files
350 boundBox bbInflate = operator[](i).bb();
351 bbInflate.inflate(1e-4);
352
353 if
354 (
355 operator[](i).closed()
356 && operator[](i).nParts() == 1
357 && (closedDomain || operator[](i).regions().size() != 1)
358 && bbInflate.contains(bb_)
359 )
360 {
361 externalID = i;
362 operator[](i).type() = surfaceType::external;
363
364 // Override any inlet and outlet regions on external boundary
365 // specified by the '-inletRegions' and '-outletRegions' options
366 const wordList& regions = operator[](i).regions();
367
368 wordList& inletRegions = operator[](i).inletRegions();
369 if (regionsValid(specifiedInletRegions, regions, "-inletRegions"))
370 {
371 inletRegions = specifiedInletRegions;
372 }
373
374 wordList& outletRegions = operator[](i).outletRegions();
375 if (regionsValid(specifiedOutletRegions, regions, "-outletRegions"))
376 {
377 outletRegions = specifiedOutletRegions;
378 }
379
380 // If inletRegions and outletRegions are both empty, set "template"
381 // names, unless "closedDomain" is specified
382 if
383 (
384 inletRegions.empty()
385 && outletRegions.empty()
386 && !closedDomain
387 )
388 {
389 inletRegions.append("<inletRegion>");
390 outletRegions.append("<outletRegion>");
391 }
392 }
393 }
394
395 swapExternalIndexZero(externalID);
396
397 if (!rotatingZones.empty())
398 {
399 setSurfaceTypes(rotatingZones, surfaceType::rotatingZone);
400 }
401
402 if (!baffles.empty())
403 {
404 setSurfaceTypes(baffles, surfaceType::baffle);
405 }
406
407 setRotatingZoneBounds();
408
409 if (cellZones.empty())
410 {
411 identifyCellZones();
412 }
413 else
414 {
415 setSurfaceTypes(cellZones, surfaceType::cellZone);
416 }
417
418 reportSurfaces();
419
420 if (bb.volume() != 0)
421 {
422 setBounds(bb);
423 }
424}
