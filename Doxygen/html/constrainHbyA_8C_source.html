145Foam::tmp<Foam::surfaceScalarField> Foam::constrainPhid
146(
147 const tmp<surfaceScalarField>& tphid,
148 const volVectorField& U,
149 const volScalarField& p,
150 surfaceScalarField& rhorAUf,
151 surfaceScalarField& rhorAAtUf
152)
153{
154 surfaceScalarField& phid = tphid.ref();
155 surfaceScalarField::Boundary& phidBf = phid.boundaryFieldRef();
156
157 const volScalarField::Boundary& pBf = p.boundaryField();
158
159 forAll(phidBf, patchi)
160 {
161 // Fixed flux
162 if (isA<fixedFluxPressureFvPatchScalarField>(pBf[patchi]))
163 {
164 phidBf[patchi] = 0;
165 }
166 // Fixed velocity and fixed pressure (supersonic inlet)
167 else if
168 (
169 !U.boundaryField()[patchi].assignable()
170 && !pBf[patchi].assignable()
171 )
172 {
173 rhorAUf.boundaryFieldRef()[patchi] = 0;
174 rhorAAtUf.boundaryFieldRef()[patchi] = 0;
175 }
176 }
177
178 return tphid;
179}
