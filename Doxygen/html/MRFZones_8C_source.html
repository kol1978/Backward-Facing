38Foam::IOobject Foam::MRFZones::createIOobject(const fvMesh& mesh)
39{
40 typeIOobject<IOdictionary> io
41 (
42 "MRFProperties",
43 mesh.time().constant(),
44 mesh,
45 IOobject::MUST_READ,
46 IOobject::NO_WRITE
47 );
48
49 if (io.headerOk())
50 {
51 Info<< indentOrNl
52 << "Constructing MRF zones from " << io.name()
53 << endl;
54
55 io.readOpt() = IOobject::MUST_READ_IF_MODIFIED;
56 return io;
57 }
58 else
59 {
60 io.readOpt() = IOobject::NO_READ;
61 return io;
62 }
63}
