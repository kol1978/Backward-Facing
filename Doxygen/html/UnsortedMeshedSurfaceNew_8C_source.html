30template<class Face>
31Foam::autoPtr<Foam::UnsortedMeshedSurface<Face>>
32Foam::UnsortedMeshedSurface<Face>::New(const fileName& name, const word& ext)
33{
34 if (debug)
35 {
36 InfoInFunction << "Constructing UnsortedMeshedSurface" << endl;
37 }
38
39 typename fileExtensionConstructorTable::iterator cstrIter =
40 fileExtensionConstructorTablePtr_->find(ext);
41
42 if (cstrIter == fileExtensionConstructorTablePtr_->end())
43 {
44 // no direct reader, use the parent if possible
45 wordHashSet supported = ParentType::readTypes();
46 if (supported.found(ext))
47 {
48 // create indirectly
49 autoPtr<UnsortedMeshedSurface<Face>> surf
50 (
51 new UnsortedMeshedSurface<Face>
52 );
53 surf().transfer(ParentType::New(name, ext)());
54
55 return surf;
56 }
57
58 // nothing left but to issue an error
59 supported += readTypes();
60
61 FatalErrorInFunction
62 << "Unknown file extension " << ext << nl << nl
63 << "Valid types are:" << nl
64 << supported
65 << exit(FatalError);
66 }
67
68 return autoPtr<UnsortedMeshedSurface<Face>>(cstrIter()(name));
69}
