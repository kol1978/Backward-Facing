85
86Foam::tmp<Foam::volScalarField> Foam::XiEqModels::instability::Db() const
87{
88 const volScalarField& rho = momentumTransport_.rho();
89
90 const objectRegistry& db = Su_.db();
91 const volScalarField& Xi = db.lookupObject<volScalarField>("Xi");
92 const volScalarField& mgb = db.lookupObject<volScalarField>("mgb");
93
94 return XiEqModel_->Db()
95 + rho*Su_*(Xi - 1.0)*mgb*(0.5*lambdaIn_)/(mgb + 1.0/lambdaIn_);
96}
