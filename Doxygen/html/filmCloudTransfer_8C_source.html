98template<class Type>
99Foam::tmp<Foam::VolInternalField<Type>>
100inline Foam::fv::filmCloudTransfer::CloudToFilmTransferRate
101(
102 const Field<Type>& prop,
103 const dimensionSet& dimProp
104) const
105{
106 tmp<VolInternalField<Type>> tSu
107 (
108 VolInternalField<Type>::New
109 (
110 "Su",
111 mesh(),
112 dimensioned<Type>(dimProp/dimVolume/dimTime, Zero)
113 )
114 );
115
116 if (cloudFieldsTransferred_)
117 {
118 const fvMesh& cloudMesh = film_.surfacePatchMap().nbrMesh();
119 const label cloudPatchi = film_.surfacePatchMap().nbrFvPatch().index();
120
121 UIndirectList<Type>(tSu.ref(), film_.surfacePatch().faceCells()) =
122 film_.surfacePatchMap().fromNeighbour
123 (
124 prop/cloudMesh.boundary()[cloudPatchi].magSf()
125 );
126
127 tSu.ref().primitiveFieldRef() /= film_.VbyA;
128 tSu.ref().primitiveFieldRef() /= mesh().time().deltaTValue();
129 }
130
131 return tSu;
132}

135void Foam::fv::filmCloudTransfer::addSup
136(
137 const volScalarField& rho,
138 const volScalarField& alpha,
139 fvMatrix<scalar>& eqn
140) const
141{
142 if (debug)
143 {
144 Info<< type() << ": applying source to " << eqn.psi().name() << endl;
145 }
146
147 if (&alpha == &film_.alpha && &eqn.psi() == &film_.alpha)
148 {
149 eqn += CloudToFilmTransferRate<scalar>(massFromCloud_, dimMass);
150
151 if (ejection_.valid())
152 {
153 eqn -= fvm::Sp(ejection_->rate()*rho(), eqn.psi());
154 }
155 }
156 else
157 {
158 FatalErrorInFunction
159 << "Support for field " << alpha.name() << " is not implemented"
160 << exit(FatalError);
161 }
162}

231void Foam::fv::filmCloudTransfer::resetFromCloudFields()
232{
233 const fvMesh& cloudMesh = film_.surfacePatchMap().nbrMesh();
234 const label cloudPatchi = film_.surfacePatchMap().nbrFvPatch().index();
235 const label nCloudPatchFaces = cloudMesh.boundary()[cloudPatchi].size();
236
237 if (massFromCloud_.size() != nCloudPatchFaces)
238 {
239 massFromCloud_.setSize(nCloudPatchFaces);
240 momentumFromCloud_.setSize(nCloudPatchFaces);
241 energyFromCloud_.setSize(nCloudPatchFaces);
242 }
243
244 massFromCloud_ = 0;
245 momentumFromCloud_ = Zero;
246 energyFromCloud_ = 0;
247
248 cloudFieldsTransferred_ = true;
249
250 // Enable ejection correction on next call to correct()
251 correctEjection_ = true;
252}
