84bool Foam::patchDistMethods::advectionDiffusion::correct
85(
86 volScalarField& y,
87 volVectorField& n
88)
89{
90 if (!predicted_)
91 {
92 pdmPredictor_->correct(y);
93 predicted_ = true;
94 }
95
96 volVectorField ny
97 (
98 IOobject
99 (
100 "ny",
101 mesh_.time().name(),
102 mesh_,
103 IOobject::NO_READ,
104 IOobject::NO_WRITE,
105 false
106 ),
107 mesh_,
108 dimensionedVector(dimless, Zero),
109 patchTypes<vector>(mesh_, patchIndices_)
110 );
111
112 const fvPatchList& patches = mesh_.boundary();
113 volVectorField::Boundary& nybf = ny.boundaryFieldRef();
114
115 forAllConstIter(labelHashSet, patchIndices_, iter)
116 {
117 label patchi = iter.key();
118 nybf[patchi] == -patches[patchi].nf();
119 }
120
121 int iter = 0;
122 scalar initialResidual = 0;
123
124 do
125 {
126 ny = fvc::grad(y);
127 ny /= (mag(ny) + small);
128
129 surfaceVectorField nf(fvc::interpolate(ny));
130 nf /= (mag(nf) + small);
131
132 surfaceScalarField yPhi("yPhi", mesh_.Sf() & nf);
133
134 fvScalarMatrix yEqn
135 (
136 fvm::div(yPhi, y)
137 - fvm::Sp(fvc::div(yPhi), y)
138 - epsilon_*y*fvm::laplacian(y)
139 ==
140 dimensionedScalar(dimless, 1.0)
141 );
142
143 yEqn.relax();
144 initialResidual = yEqn.solve().initialResidual();
145
146 } while (initialResidual > tolerance_ && ++iter < maxIter_);
147
148 // Only calculate n if the field is defined
149 if (notNull(n))
150 {
151 n = -ny;
152 }
153
154 return true;
155}
