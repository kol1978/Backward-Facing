61Foam::tmp<Foam::scalarField>
62Foam::contactAngleModels::gravitational::cosTheta
63(
64 const fvPatchVectorField& Up,
65 const vectorField& nHat
66) const
67{
68 const uniformDimensionedVectorField& g =
69 Up.db().lookupObject<uniformDimensionedVectorField>("g");
70
71 const scalarField uCoeff(tanh((nHat & g.value())/gTheta_));
72
73 return cos
74 (
75 theta0_
76 + (thetaRec_ - theta0_)*max(uCoeff, scalar(0))
77 - (thetaAdv_ - theta0_)*min(uCoeff, scalar(0))
78 );
79}
