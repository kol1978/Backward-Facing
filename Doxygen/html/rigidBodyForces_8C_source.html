47Foam::vector Foam::functionObjects::rigidBodyForces::CofR() const
48{
49 const fvMeshMovers::pointMeshMover& mover =
50 refCast<const fvMeshMovers::pointMeshMover>(mesh_.mover());
51
52 const RBD::rigidBodyMotion& motion =
53 refCast<const RBD::rigidBodyMotion>(mover.mover());
54
55 const label bodyID = motion.bodyIndex(body_);
56
57 return motion.X0(bodyID).r();
58}
