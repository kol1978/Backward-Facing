117Foam::tmp<Foam::volScalarField>
118Foam::radiationModels::absorptionEmissionModels::cloud::EDisp
119(
120 const label bandI
121) const
122{
123 tmp<volScalarField> tE
124 (
125 volScalarField::New
126 (
127 "E",
128 mesh_,
129 dimensionedScalar(dimMass/dimLength/pow3(dimTime), 0)
130 )
131 );
132
133 forAll(cloudNames_, i)
134 {
135 const parcelCloud& c =
136 mesh_.objectRegistry::lookupObject<parcelCloud>(cloudNames_[i]);
137
138 tE.ref() += c.Ep();
139 }
140
141 // Total emission is 4 times the projected emission
142 return 4*tE;
143}
