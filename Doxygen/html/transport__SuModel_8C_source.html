90void Foam::SuModels::transport::correct()
91{
92 const fvMesh& mesh(thermo_.mesh());
93
94 const Foam::fvModels& fvModels(Foam::fvModels::New(mesh));
95 const Foam::fvConstraints& fvConstraints(Foam::fvConstraints::New(mesh));
96
97 const volScalarField& rho = momentumTransport_.rho();
98 const volScalarField& b = thermo_.Y("b");
99 const volScalarField& mgb = mesh.lookupObject<volScalarField>("mgb");
100 const volScalarField& Xi = mesh.lookupObject<volScalarField>("Xi");
101 const surfaceScalarField& phiSt =
102 mesh.lookupObject<surfaceScalarField>("phiSt");
103 const volScalarField& Db = mesh.lookupObject<volScalarField>("Db");
104 const volVectorField& n = mesh.lookupObject<volVectorField>("n");
105 const surfaceScalarField& nf = mesh.lookupObject<surfaceScalarField>("nf");
106
107 const surfaceScalarField phiXi
108 (
109 "phiXi",
110 phiSt - fvc::interpolate(fvc::laplacian(Db, b)/mgb)*nf
111 );
112
113 const surfaceScalarField& phi = momentumTransport_.alphaRhoPhi();
114 const volVectorField& U(momentumTransport_.U());
115
116 const volScalarField sigmas
117 (
118 ((n & n)*fvc::div(U) - (n & fvc::grad(U) & n))/Xi
119 + (
120 (n & n)*fvc::div(Su_*n)
121 - (n & fvc::grad(Su_*n) & n)
122 )*(Xi + scalar(1))/(2*Xi)
123 );
124
125 const volScalarField Su0(Su0_->Su());
126
127 const volScalarField SuInf
128 (
129 Su0*max(scalar(1) - sigmas/sigmaExt_, scalar(0.01))
130 );
131
132 const volScalarField Rc
133 (
134 (sigmas*SuInf*(Su0 - SuInf) + sqr(SuMin_)*sigmaExt_)
135 /(sqr(Su0 - SuInf) + sqr(SuMin_))
136 );
137
138 fvScalarMatrix SuEqn
139 (
140 fvm::ddt(rho, Su_)
141 + fvm::div(phi + phiXi, Su_, "div(phiXi,Su)")
142 - fvm::Sp(fvc::div(phiXi), Su_)
143 ==
144 - fvm::SuSp(-rho*Rc*Su0/Su_, Su_)
145 - fvm::SuSp(rho*(sigmas + Rc), Su_)
146 + fvModels.source(rho, Su_)
147 );
148
149 SuEqn.relax();
150
151 fvConstraints.constrain(SuEqn);
152
153 SuEqn.solve();
154
155 fvConstraints.constrain(Su_);
156
157 // Limit the maximum and minimum Su
158 Su_.min(SuMax_);
159 Su_.max(SuMin_);
160}
