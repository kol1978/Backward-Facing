300template<class Type>
301Type average(const UList<Type>& f)
302{
303 if (f.size())
304 {
305 Type avrg = sum(f)/f.size();
306
307 return avrg;
308 }
309 else
310 {
311 WarningInFunction
312 << "empty field, returning zero" << endl;
313
314 return Zero;
315 }
316}

318template<class Type>
319Type gAverage
320(
321 const UList<Type>& f,
322 const label comm
323)
324{
325 label n = f.size();
326 Type s = sum(f);
327 sumReduce(s, n, Pstream::msgType(), comm);
328
329 if (n > 0)
330 {
331 Type avrg = s/n;
332
333 return avrg;
334 }
335 else
336 {
337 WarningInFunction
338 << "empty field, returning zero." << endl;
339
340 return Zero;
341 }
342}
