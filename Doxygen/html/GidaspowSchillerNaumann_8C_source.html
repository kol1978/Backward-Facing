62Foam::tmp<Foam::volScalarField>
63Foam::dragModels::GidaspowSchillerNaumann::CdRe() const
64{
65 const volScalarField alpha2
66 (
67 max(interface_.continuous(), interface_.continuous().residualAlpha())
68 );
69
70 const volScalarField Res(alpha2*interface_.Re());
71
72 const volScalarField CdsRes
73 (
74 neg(Res - 1000)*24*(1.0 + 0.15*pow(Res, 0.687))/alpha2
75 + pos0(Res - 1000)*0.44*Res
76 );
77
78 return CdsRes*pow(alpha2, -1.65);
79}
