107void Foam::interstitialInletVelocityFvPatchVectorField::updateCoeffs()
108{
109 if (updated())
110 {
111 return;
112 }
113
114 const fvPatchField<scalar>& alphap =
115 patch().lookupPatchField<volScalarField, scalar>(alphaName_);
116
117 operator==(inletVelocity_/alphap);
118 fixedValueFvPatchVectorField::updateCoeffs();
119}
