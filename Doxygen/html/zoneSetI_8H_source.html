28template<class ZoneType>
29inline void Foam::zoneSet::checkValid(const tmpNrc<ZoneType>& zone) const
30{
31 if (!zone.valid())
32 {
33 FatalErrorInFunction
34 << "zoneSet does not contain a " << ZoneType::typeName << nl;
35
36 if (pZone_.valid())
37 {
38 FatalError
39 << " zoneSet contains pointZone " << pZone_().name() << nl;
40 }
41
42 if (cZone_.valid())
43 {
44 FatalError
45 << " zoneSet contains cellZone " << cZone_().name() << nl;
46 }
47
48 if (fZone_.valid())
49 {
50 FatalError
51 << " zoneSet contains facesZone " << fZone_().name() << nl;
52 }
53
54 FatalError << exit(FatalError);
55 }
56}
