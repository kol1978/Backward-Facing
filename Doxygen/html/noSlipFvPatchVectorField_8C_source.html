93void Foam::noSlipFvPatchVectorField::updateCoeffs()
94{
95 if (updated())
96 {
97 return;
98 }
99
100 const fvMesh& mesh = patch().mesh();
101
102 if (mesh.moving())
103 {
104 const fvPatch& p = patch();
105
106 const volVectorField& U =
107 static_cast<const volVectorField&>(internalField());
108
109 const vectorField n(p.nf());
110 tmp<scalarField> Un = fvc::meshPhi(U, p.index())/(p.magSf() + vSmall);
111
112 vectorField::operator=(n*Un);
113 }
114
115 fixedValueFvPatchVectorField::updateCoeffs();
116}
