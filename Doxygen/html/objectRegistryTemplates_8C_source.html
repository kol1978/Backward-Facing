53{
54 wordList objectNames(size());
55
56 label count = 0;
57 forAllConstIter(HashTable<regIOobject*>, *this, iter)

59 if (isA<Type>(*iter()))

141template<class Type>
142bool Foam::objectRegistry::foundObject(const word& name) const
143{
144 const_iterator iter = find(name);
145
146 if (iter != end())
147 {
148 const Type* vpsiPtr_ = dynamic_cast<const Type*>(iter());
149
150 if (vpsiPtr_)
151 {
152 return true;
153 }
154 }
155 else if (this->parentNotTime())
156 {
157 return parent_.foundObject<Type>(name);
158 }
159
160 return false;
161}

164template<class Type>
165const Type& Foam::objectRegistry::lookupObject(const word& name) const
166{
167 const_iterator iter = find(name);
168
169 if (iter != end())
170 {
171 const Type* vpsiPtr_ = dynamic_cast<const Type*>(iter());
172
173 if (vpsiPtr_)
174 {
175 return *vpsiPtr_;
176 }
177
178 FatalErrorInFunction
179 << nl
180 << " lookup of " << name << " from objectRegistry "
181 << this->name()
182 << " successful\n but it is not a " << Type::typeName
183 << ", it is a " << iter()->type()
184 << abort(FatalError);
185 }
186 else
187 {
188 if (this->parentNotTime())

190 return parent_.lookupObject<Type>(name);

196 << " " << name << " from objectRegistry " << this->name()

203 FatalErrorInFunction

211 FatalErrorInFunction

214
215 return NullObjectRef<Type>();

240template<class Object>
241bool Foam::objectRegistry::cacheTemporaryObject(Object& ob) const
242{
243 if (cacheTemporaryObjectsState_ == -1) return false;
244
245 const objectRegistry& root = time_;
246
247 root.readCacheTemporaryObjects();
248 if (root.cacheTemporaryObjects_.empty()) return false;
249
250 temporaryObjects_.insert(ob.name());
251
252 HashTable<Pair<bool>>::iterator rootIter =
253 root.cacheTemporaryObjects_.find(ob.name());
254 if (rootIter == root.cacheTemporaryObjects_.end()) return false;
255
256 HashTable<Pair<bool>>::iterator iter =
257 cacheTemporaryObjects_.find(ob.name());
258 if (iter == cacheTemporaryObjects_.end())
259 {
260 cacheTemporaryObjects_.insert(rootIter.key(), rootIter());
261 iter = cacheTemporaryObjects_.find(ob.name());
262 }
263
264 // Cache object ob if it hasn't been cached yet
265 if (iter().first() == false)
266 {
267 rootIter().first() = true;
268 rootIter().second() = true;
269 iter().first() = true;
270 iter().second() = true;
271
272 if (ob.db().template foundObject<Object>(ob.name()))
273 {
274 Object& cachedOb =
275 ob.db().template lookupObjectRef<Object>(ob.name());
276
277 // If the object is already cached in the database delete it
278 if (&cachedOb != &ob && cachedOb.ownedByRegistry())
279 {
280 deleteCachedObject(cachedOb);
281 }
282 }
283
284 if (debug)
285 {
286 Info<< "Caching " << ob.name()
287 << " of type " << ob.type() << endl;
288 }
289
290 ob.release();
291 ob.checkOut();
292 store(new Object(move(ob)));
293
294 return true;
295 }
296
297 return false;
298}
