258{ \
259 if (!ds.dimensions().dimensionless()) \

261 FatalErrorInFunction \
262 << "ds not dimensionless" \
263 << abort(FatalError); \
264 } \
265 \
266 return dimensionedScalar \

268 #func "(" + ds.name() + ')', \
