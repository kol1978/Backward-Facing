32(
33 DynamicList<filePos>& stack,
34 const dictionary& contextDict,
35 Context& context,
36 Istream& is
37) const
38{
39 const label nNested = stack.size();
40
41 stack.append(filePos(is.name(), is.lineNumber()));
42
43 const string arg
44 (
45 operator[](1).stringToken() + char(token::END_STATEMENT)
46 );
47 IStringStream argStream(arg);
48 argStream.lineNumber() = operator[](1).lineNumber();
49 const primitiveEntry e("ifEntry", contextDict, argStream);
50 const Switch doIf(e.stream());
51
52 bool ok = ifeqEntry::execute(doIf, stack, contextDict, context, is);
53
54 if (stack.size() != nNested)

56 FatalIOErrorInFunction(contextDict)

61 }
62
63 return ok;
64}
