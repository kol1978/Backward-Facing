49Foam::pointMeshMovers::list::list
50(
51 const polyMesh& mesh,
52 const dictionary& dict
53)
54:
55 pointMeshMover(mesh, typeName),
56 movers_(0)
57{
58 const dictionary& solversDict = dict.optionalSubDict("list");
59
60 forAllConstIter(dictionary, solversDict, iter)
61 {
62 if (iter().isDict())
63 {
64 const word& name = iter().keyword();
65 const dictionary& dict = iter().dict();
66
67 movers_.append
68 (
69 name,
70 pointMeshMover::New(mesh, dict).ptr()
71 );
72 }
73 }
74}

85Foam::tmp<Foam::pointField> Foam::pointMeshMovers::list::newPoints()
86{
87 if (movers_.size())
88 {
89 // Accumulated displacement
90 pointField disp(poly().nPoints(), Zero);
91
92 forAllIter(PtrListDictionary<pointMeshMover>, movers_, iter)
93 {
94 disp += iter().newPoints() - poly().points();
95 }
96
97 return poly().points() + disp;
98 }
99 else
100 {
101 return poly().points();
102 }
103}
