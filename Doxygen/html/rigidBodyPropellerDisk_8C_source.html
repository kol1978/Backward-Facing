50void Foam::fv::rigidBodyPropellerDisk::readCoeffs(const dictionary& dict)
51{
52 propellerDisk::readCoeffs(dict);
53
54 if (!dict.found("centre"))
55 {
56 FatalIOErrorInFunction(dict)
57 << "Required entry 'centre' not found"
58 << exit(FatalIOError);

60
61 dict.lookup("body") >> body_;

63}
