40Foam::multicomponentLagrangianThermo::implementation::implementation
41(
42 const dictionary& dict,
43 const speciesTable& species,
44 const LagrangianMesh& mesh,
45 const word& phaseName
46)
47:
48 defaultSpecieName_
49 (
50 species.size()
51 ? dict.lookup<word>("defaultSpecie")
52 : "undefined"
53 ),
54 defaultSpeciei_
55 (
56 species.size()
57 ? species[defaultSpecieName_]
58 : -1
59 ),
60 Y_(species.size())
61{
62 tmp<LagrangianScalarField> Ydefault;
63
64 bool Yset = false;
65
66 // Read the species' mass fractions
67 forAll(species, i)
68 {
69 typeIOobject<LagrangianScalarDynamicField> io
70 (
71 IOobject::groupName(species[i], phaseName),
72 mesh.time().name(),
73 mesh,
74 IOobject::NO_READ
75 );
76
77 if (io.headerOk())
78 {
79 Y_.set
80 (
81 i,
82 new LagrangianScalarDynamicField
83 (
84 IOobject
85 (
86 IOobject::groupName(species[i], phaseName),
87 mesh.time().name(),
88 mesh,
89 IOobject::MUST_READ,
90 IOobject::AUTO_WRITE
91 ),
92 mesh
93 )
94 );
95
96 Yset = true;
97 }
98 else
99 {
100 if (!Ydefault.valid())
101 {
102 Ydefault = new LagrangianScalarField
103 (
104 IOobject
105 (
106 IOobject::groupName("Ydefault", phaseName),
107 mesh.time().name(),
108 mesh,
109 IOobject::MUST_READ,
110 IOobject::NO_WRITE
111 ),
112 mesh
113 );
114 }
115
116 Y_.set
117 (
118 i,
119 new LagrangianScalarDynamicField
120 (
121 IOobject
122 (
123 IOobject::groupName(species[i], phaseName),
124 mesh.time().name(),
125 mesh,
126 IOobject::NO_READ,
127 IOobject::AUTO_WRITE
128 ),
129 Ydefault()
130 )
131 );
132 }
133 }
134
135 // If the mass fractions have been specified check and normalise
136 if (Yset)
137 {
138 // Scale the mass fractions so that they sum to one in case the input is
139 // not exact
140 tmp<LagrangianScalarField> tYt
141 (
142 new LagrangianScalarField
143 (
144 IOobject::groupName("Yt", phaseName),
145 Y_[0]
146 )
147 );
148 LagrangianScalarField& Yt = tYt.ref();
149
150 for (label i = 1; i < Y_.size(); ++ i)
151 {
152 Yt += Y_[i];
153 }
154
155 if (min(Yt.primitiveField()) == 0 && max(Yt.primitiveField()) == 0)
156 {
157 FatalErrorInFunction
158 << "Sum of specie mass fractions = 0"
159 << exit(FatalError);
160 }
161
162 if (min(Yt.primitiveField()) < 0.999)
163 {
164 FatalErrorInFunction
165 << "Min sum of specie mass fractions "
166 << min(Yt.primitiveField())
167 << " < 0.999"
168 << exit(FatalError);
169 }
170
171 if (max(Yt.primitiveField()) > 1.001)
172 {
173 FatalErrorInFunction
174 << "Max sum of specie mass fractions "
175 << max(Yt.primitiveField())
176 << " > 1.001"
177 << exit(FatalError);
178 }
179
180 forAll(Y_, i)
181 {
182 Y_[i] /= Yt;
183 }
184 }
185}

239void Foam::multicomponentLagrangianThermo::implementation::normaliseY
240(
241 const LagrangianSubMesh& subMesh
242)
243{
244 if (defaultSpeciei_ == -1 || !species().size()) return;
245
246 tmp<LagrangianSubScalarField> tYt
247 (
248 LagrangianSubScalarField::New
249 (
250 IOobject::groupName("Yt", phaseName()),
251 subMesh,
252 dimensionedScalar(dimless, 0)
253 )
254 );
255 LagrangianSubScalarField& Yt = tYt.ref();
256
257 forAll(Y(), i)
258 {
259 if (i == defaultSpeciei_) continue;
260
261 LagrangianSubScalarSubField subYi(subMesh.sub(Y_[i]));
262 subYi = max(subYi, scalar(0));
263 Yt += subYi;
264 }
265
266 LagrangianSubScalarSubField subYdefault(subMesh.sub(Y_[defaultSpeciei_]));
267 subYdefault = max(scalar(1) - Yt, scalar(0));
268}
