72void ShihQuadraticKE::correctNonlinearStress(const volTensorField& gradU)
73{
74 volSymmTensorField S(symm(gradU));
75 volTensorField W(skew(gradU));
76
77 volScalarField sBar((k_/epsilon_)*sqrt(2.0)*mag(S));
78 volScalarField wBar((k_/epsilon_)*sqrt(2.0)*mag(W));
79
80 volScalarField Cmu((2.0/3.0)/(Cmu1_ + sBar + Cmu2_*wBar));
81
82 boundEpsilon();
83 nut_ = Cmu*sqr(k_)/epsilon_;
84 nut_.correctBoundaryConditions();
85
86 nonlinearStress_ =
87 k_*sqr(k_/epsilon_)/(Cbeta_ + pow3(sBar))
88 *(
89 Cbeta1_*dev(innerSqr(S))
90 + Cbeta2_*twoSymm(S&W)
91 + Cbeta3_*dev(symm(W&W))
92 );
93}

192void ShihQuadraticKE::correct()
193{
194 if (!turbulence_)
195 {
196 return;
197 }
198
199 nonlinearEddyViscosity<incompressible::RASModel>::correct();
200
201 tmp<volTensorField> tgradU = fvc::grad(U_);
202 const volTensorField& gradU = tgradU();
203
204 volScalarField G
205 (
206 GName(),
207 (nut_*twoSymm(gradU) - nonlinearStress_) && gradU
208 );
209
210
211 // Update epsilon and G at the wall
212 epsilon_.boundaryFieldRef().updateCoeffs();
213
214 // Dissipation equation
215 tmp<fvScalarMatrix> epsEqn
216 (
217 fvm::ddt(epsilon_)
218 + fvm::div(phi_, epsilon_)
219 - fvm::laplacian(DepsilonEff(), epsilon_)
220 ==
221 Ceps1_*G*epsilon_/k_
222 - fvm::Sp(Ceps2_*epsilon_/k_, epsilon_)
223 );
224
225 epsEqn.ref().relax();
226 epsEqn.ref().boundaryManipulate(epsilon_.boundaryFieldRef());
227 solve(epsEqn);
228 boundEpsilon();
229
230
231 // Turbulent kinetic energy equation
232 tmp<fvScalarMatrix> kEqn
233 (
234 fvm::ddt(k_)
235 + fvm::div(phi_, k_)
236 - fvm::laplacian(DkEff(), k_)
237 ==
238 G
239 - fvm::Sp(epsilon_/k_, k_)
240 );
241
242 kEqn.ref().relax();
243 solve(kEqn);
244 bound(k_, kMin_);
245
246
247 // Re-calculate viscosity and non-linear stress
248 correctNonlinearStress(gradU);
249}
