123template<class DimensionedFieldType>
124void Foam::DimensionedFieldFunctions::Surfaces<DimensionedFieldType>::evaluate()
125{
126 this->field_ = defaultValue_;
127
128 forAll(surfacesPtr_(), i)
129 {
130 this->field_ +=
131 DimensionedField
132 <
133 scalar,
134 typename DimensionedFieldType::GeoMesh_
135 >::New
136 (
137 IOobject::groupName("alpha", surfacesPtr_()[i].name()),
138 this->field_.mesh(),
139 dimless,
140 searchableSurfaces::insideFraction
141 (
142 surfacesPtr_()[i],
143 this->field_.mesh().poly()
144 )
145 )
146 *(values_[i] - this->field_);
147 }
148}
