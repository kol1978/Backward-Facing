33template<class ZoneType, class ZonesType>
34template<class Type>
35void Foam::Zone<ZoneType, ZonesType>::select(const Type& zone)
36{
37 const pointField& ctrs = zone.meshCentres();
38
39 labelList& indices = *this;
40 indices.setSize(ctrs.size());
41
42 label nInZone = 0;
43 forAll(ctrs, i)
44 {
45 if (zone.contains(ctrs[i]))
46 {
47 indices[nInZone++] = i;
48 }
49 }
50
51 indices.setSize(nInZone);
52}

55template<class ZoneType, class ZonesType>
56void Foam::Zone<ZoneType, ZonesType>::calcLookupMap() const
57{
58 if (lookupMapPtr_)
59 {
60 FatalErrorInFunction
61 << "Lookup map already calculated" << nl
62 << abort(FatalError);
63 }
64
65 const labelList& indices = *this;
66
67 lookupMapPtr_ = new Map<label>(2*indices.size());
68 Map<label>& lm = *lookupMapPtr_;
69
70 forAll(indices, i)
71 {
72 lm.insert(indices[i], i);
73 }
74}

77template<class ZoneType, class ZonesType>
78void Foam::Zone<ZoneType, ZonesType>::topoChange
79(
80 const labelList& map,
81 const labelList& reverseMap
82)
83{
84 clearAddressing();
85
86 labelHashSet indices;
87
88 forAll(map, i)
89 {
90 if (map[i] >= 0 && localIndex(map[i]) != -1)
91 {
92 indices.insert(i);
93 }
94 }
95
96 forAll(reverseMap, i)
97 {
98 if (reverseMap[i] >= 0 && localIndex(i) != -1)
99 {
100 indices.insert(reverseMap[i]);
101 }
102 }
103
104 labelList::operator=(indices.sortedToc());
105}

219template<class ZoneType, class ZonesType>
220Foam::label Foam::Zone<ZoneType, ZonesType>::localIndex
221(
222 const label globalIndex
223) const
224{
225 const Map<label>& lm = lookupMap();
226
227 Map<label>::const_iterator lmIter = lm.find(globalIndex);
228
229 if (lmIter == lm.end())
230 {
231 return -1;
232 }
233 else
234 {
235 return lmIter();
236 }

259template<class ZoneType, class ZonesType>
260bool Foam::Zone<ZoneType, ZonesType>::checkDefinition
261(
262 const label maxSize,
263 const bool report
264) const
265{
266 const labelList& indices = *this;
267
268 bool hasError = false;
269
270 // To check for duplicate entries
271 labelHashSet elems(size());
272
273 forAll(indices, i)
274 {
275 if (indices[i] < 0 || indices[i] >= maxSize)
276 {
277 hasError = true;
278
279 if (report)
280 {
281 SeriousErrorInFunction
282 << "Zone " << name_
283 << " contains invalid index label " << indices[i] << nl
284 << "Valid index labels are 0.."
285 << maxSize-1 << endl;
286 }
287 else
288 {
289 // w/o report - can stop checking now
290 break;
291 }
292 }
293 else if (!elems.insert(indices[i]))
294 {
295 if (report)
296 {
297 WarningInFunction
298 << "Zone " << name_
299 << " contains duplicate index label " << indices[i] << endl;
300 }
301 }
302 }
303
304 return hasError;
305}
