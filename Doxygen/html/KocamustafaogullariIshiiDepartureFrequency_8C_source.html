53template<class ScalarFieldType>
54Foam::tmp<ScalarFieldType>
55Foam::wallBoilingModels::departureFrequencyModels::
56KocamustafaogullariIshiiDepartureFrequency::calculate
57(
58 const fvMesh& mesh,
59 const ScalarFieldType& dDep,
60 const ScalarFieldType& rhoLiquid,
61 const ScalarFieldType& rhoVapour,
62 const ScalarFieldType& sigma
63) const
64{
65 auto g = coefficient<ScalarFieldType>::value
66 (
67 mesh.lookupObject<uniformDimensionedVectorField>("g")
68 );
69
70 auto Cf = coefficient<ScalarFieldType>::value(Cf_);
71
72 return
73 (Cf/dDep)*pow025(sigma*mag(g)*(rhoLiquid - rhoVapour)/sqr(rhoLiquid));
74}
