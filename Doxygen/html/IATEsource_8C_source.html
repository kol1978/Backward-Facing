45Foam::autoPtr<Foam::diameterModels::IATEsource>
46Foam::diameterModels::IATEsource::New
47(
48 const word& type,
49 const IATE& iate,
50 const dictionary& dict
51)
52{
53 dictionaryConstructorTable::iterator cstrIter =
54 dictionaryConstructorTablePtr_->find(type);
55
56 if (cstrIter == dictionaryConstructorTablePtr_->end())
57 {
58 FatalErrorInFunction
59 << "Unknown IATE source type "
60 << type << nl << nl
61 << "Valid IATE source types : " << endl
62 << dictionaryConstructorTablePtr_->sortedToc()
63 << exit(FatalError);
64 }
65
66 printDictionary print(dict);
67
68 return autoPtr<IATEsource>(cstrIter()(iate, dict));
69}

74Foam::tmp<Foam::volScalarField> Foam::diameterModels::IATEsource::Ur() const
75{
76 const uniformDimensionedVectorField& g =
77 phase().db().lookupObject<uniformDimensionedVectorField>("g");
78
79 return
80 sqrt(2.0)
81 *pow025
82 (
83 sigma()*mag(g)
84 *(otherPhase().rho() - phase().rho())
85 /sqr(otherPhase().rho())
86 )
87 *pow(max(1 - phase(), scalar(0)), 1.75);
88}

98{

103Foam::tmp<Foam::volScalarField> Foam::diameterModels::IATEsource::CD() const
104{
105 const volScalarField Eo(this->Eo());
106 const volScalarField Re(this->Re());
107
108 return
109 max
110 (
111 min
112 (
113 (16/Re)*(1 + 0.15*pow(Re, 0.687)),
114 48/Re
115 ),
116 8*Eo/(3*(Eo + 4))
117 );
118}

121Foam::tmp<Foam::volScalarField> Foam::diameterModels::IATEsource::Mo() const
122{
123 const uniformDimensionedVectorField& g =
124 phase().db().lookupObject<uniformDimensionedVectorField>("g");
125
126 return
127 mag(g)*pow4(otherPhase().fluidThermo().nu())*sqr(otherPhase().rho())
128 *(otherPhase().rho() - phase().rho())
129 /pow3(sigma());
130}

133Foam::tmp<Foam::volScalarField> Foam::diameterModels::IATEsource::Eo() const
134{
135 const uniformDimensionedVectorField& g =
136 phase().db().lookupObject<uniformDimensionedVectorField>("g");
137
138 return
139 mag(g)*sqr(phase().d())
140 *(otherPhase().rho() - phase().rho())
141 /sigma();
142}

145Foam::tmp<Foam::volScalarField> Foam::diameterModels::IATEsource::We() const
146{
