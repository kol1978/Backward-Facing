60template<class GeoField>
61bool Foam::functionObjects::tr::CalcTr<Foam::direction(2)>::calc(tr& fo)
62{
63 if (!fo.foundObject<GeoField>(fo.fieldName_)) return false;
64
65 const GeoField& geoField =
66 fo.lookupObject<GeoField>(fo.fieldName_);
67
68 fo.store(fo.resultName_, Foam::tr(geoField));
69
70 return true;
71}

74bool Foam::functionObjects::tr::calc()
75{
76 bool processed = false;
77
78 #define calcTrType(Type, GeoField) \
79 processed = \
80 processed \
81 || CalcTr<pTraits<Type>::rank>::calc<GeoField<Type>>(*this);
82 FOR_ALL_FIELD_TYPES(calcTrType, VolField);
83 FOR_ALL_FIELD_TYPES(calcTrType, VolInternalField);
84 FOR_ALL_FIELD_TYPES(calcTrType, SurfaceField);
85 #undef calcTrType
86
87 if (!processed)
88 {
89 cannotFindObject(fieldName_);
90 }
91
92 return processed;
93}
