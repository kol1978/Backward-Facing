51Foam::Ostream& Foam::Ostream::write(const token& t)
52{
53 switch (t.type())
54 {
55 case token::UNDEFINED:
56 write("UNDEFINED");
57 WarningInFunction << "Undefined token" << Foam::endl;
58 break;
59
60 case token::PUNCTUATION:
61 // Cast to char required to work around a bug in gcc versions < 10
62 write(static_cast<char>(t.pToken()));
63 // write(t.pToken());
64 break;
65
66 case token::WORD:
67 write(t.wordToken());
68 break;
69
70 case token::FUNCTIONNAME:
71 write(t.functionNameToken());
72 break;
73
74 case token::VARIABLE:
75 write(t.variableToken());
76 break;
77
78 case token::STRING:
79 write(t.stringToken());
80 break;
81
82 case token::VERBATIMSTRING:
83 write(t.verbatimStringToken());
84 break;
85
86 case token::INTEGER_32:
87 write(t.integer32Token());
88 break;
89
90 case token::INTEGER_64:
91 write(t.integer64Token());
92 break;
93
94 case token::UNSIGNED_INTEGER_32:
95 write(t.unsignedInteger32Token());
96 break;
97
98 case token::UNSIGNED_INTEGER_64:
99 write(t.unsignedInteger64Token());
100 break;
101
102 case token::FLOAT_SCALAR:
103 write(t.floatScalarToken());
104 break;
105
106 case token::DOUBLE_SCALAR:
107 write(t.doubleScalarToken());
108 break;
109
110 case token::LONG_DOUBLE_SCALAR:
111 write(t.longDoubleScalarToken());
112 break;
113
114 case token::COMPOUND:
115 *this << t.compoundToken();
116 break;
117
118 case token::ERROR:
119 write("ERROR");
120 WarningInFunction << "Error token" << Foam::endl;
121 break;
122
123 default:
124 write("UNKNOWN");
125 SeriousErrorInFunction << "Unknown token" << Foam::endl;
126 }
127
128 // Check state of stream
129 check("Ostream& Ostream::write(const token&)");
130
131 return *this;
132}

135Foam::Ostream& Foam::Ostream::writeCompoundTag(const word& typeName)
136{
137 if (token::compound::isCompound(typeName))
138 {
139 write(typeName);
140 // Cast to char required to work around a bug in gcc versions < 10
141 write(static_cast<char>(token::SPACE));
142 // write(token::SPACE);
143 }
144
145 return *this;
146}
