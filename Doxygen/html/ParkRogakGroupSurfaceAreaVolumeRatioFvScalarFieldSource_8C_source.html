33Foam::ParkRogakGroupSurfaceAreaVolumeRatioFvScalarFieldSource::value
34(
35 const label j,
36 const fvSource& source
37) const
38{
39 const populationBalanceModel& popBal = this->popBal();
40 const label i = this->i();
41
42 const populationBalance::shapeModels::fractal& fractal =
43 refCast<const populationBalance::shapeModels::fractal>(popBal.shape());
44
45 const volScalarField::Internal& kappaj = fractal.fld(j);
46
47 const dimensionedScalar& vi = popBal.v(i);
48 const dimensionedScalar& vj = popBal.v(j);
49
50 const volScalarField::Internal a(kappaj*vj);
51
52 const volScalarField::Internal da1
53 (
54 (2.0/3.0)
55 *(vi - vj)
56 *(kappaj + fractal.Df(j)*(1/fractal.d(j)()() - kappaj/6))
57 );
58
59 const volScalarField::Internal dp
60 (
61 6/kappaj + 6*((vi - vj)*a - vj*da1)/sqr(a)
62 );
63
64 tmp<volScalarField::Internal> np
65 (
66 6*vi/constant::mathematical::pi/pow3(dp)
67 );
68
69 tmp<volScalarField::Internal> dc
70 (
71 dp*pow(np/fractal.alphaC(i), 1/fractal.Df(i))
72 );
73
74 tmp<volScalarField::Internal> da2
75 (
76 (vi - vj)*(4/dp + 2*fractal.Df(i)/3*(1/dc - 1/dp))
77 );
78
79 return (a + 0.5*da1 + 0.5*da2)/vi;
80}
