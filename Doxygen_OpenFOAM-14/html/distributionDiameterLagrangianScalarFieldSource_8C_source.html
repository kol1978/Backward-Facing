79Foam::tmp<Foam::LagrangianSubScalarField>
80Foam::distributionDiameterLagrangianScalarFieldSource::value
81(
82 const LagrangianInjection& injection,
83 const LagrangianSubMesh& subMesh
84) const
85{
86 // Obtain up the distribution number source
87 const uniformSizeNumberLagrangianScalarFieldSource& uniformSizeNumber =
88 cloud<clouds::grouped>(injection)
89 .number
90 .sources()[injection.name()]
91 .fieldSourceCast<uniformSizeNumberLagrangianScalarFieldSource>
92 (injection);
93
94 // The sample size exponent of the distribution is obtained from the number
95 // source. This wasn't available during construction, so it is set here if
96 // this is the first execution.
97 if (timeIndex_ == -1)
98 {
99 distribution_ =
100 distribution::New(distribution_, uniformSizeNumber.sampleQ());
101 }
102
103 // Restart the distribution if the time index has not changed. This ensures
104 // multiple evaluations from different conditions and/or iterations
105 // generate the same values
106 distribution_->start(timeIndex_ == time().timeIndex());
107 timeIndex_ = time().timeIndex();
108
109 // Sample the distribution and return the result as a sub-field
110 return
111 LagrangianSubScalarField::New
112 (
113 internalField().name() + ":" + injection.name(),
114 subMesh,
115 internalDimensions(),
116 distribution_->sample(subMesh.size())
117 );
118}
