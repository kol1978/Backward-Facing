87template<class BasicMomentumTransportModel>
88void kEpsilonLopesdaCosta<BasicMomentumTransportModel>::
89setPorosityCoefficients()
90{
91 const PtrListDictionary<fvModel>& fvModels
92 (
93 Foam::fvModels::New(this->mesh_)
94 );
95
96 forAll(fvModels, i)
97 {
98 if (isA<fv::porosityForce>(fvModels[i]))
99 {
100 const fv::porosityForce& eps =
101 refCast<const fv::porosityForce>(fvModels[i]);
102
103 if (isA<porosityModels::powerLawLopesdaCosta>(eps.model()))
104 {
105 const porosityModels::powerLawLopesdaCosta& pm =
106 refCast<const porosityModels::powerLawLopesdaCosta>
107 (
108 eps.model()
109 );
110
111 setPorosityCoefficient(Cmu_, pm);
112 setPorosityCoefficient(C1_, pm);
113 setPorosityCoefficient(C2_, pm);
114 setPorosityCoefficient(sigmak_, pm);
115 setPorosityCoefficient(sigmaEps_, pm);
116
117 setCdAv(CdAv_, pm);
118 setPorosityCoefficient(betap_, pm);
119 setPorosityCoefficient(betad_, pm);
120 setPorosityCoefficient(C4_, pm);
121 setPorosityCoefficient(C5_, pm);
122 }
123 }
124 }

128template<class BasicMomentumTransportModel>
129tmp<volScalarField>
130kEpsilonLopesdaCosta<BasicMomentumTransportModel>::boundEpsilon()
131{
132 tmp<volScalarField> tCmuk2(Cmu_*sqr(k_));
133 epsilon_ = max(epsilon_, tCmuk2()/(this->nutMaxCoeff_*this->nu()));
134 return tCmuk2;
135}

148tmp<fvScalarMatrix> kEpsilonLopesdaCosta<BasicMomentumTransportModel>::kSource
149(
150 const volScalarField::Internal& magU,
151 const volScalarField::Internal& magU3

153{

361template<class BasicMomentumTransportModel>
362void kEpsilonLopesdaCosta<BasicMomentumTransportModel>::correct()
363{
364 if (!this->turbulence_)
365 {
366 return;
367 }
368
369 // Local references
370 const alphaField& alpha = this->alpha_;
371 const rhoField& rho = this->rho_;
372 const surfaceScalarField& alphaRhoPhi = this->alphaRhoPhi_;
373 const volVectorField& U = this->U_;
374 volScalarField& nut = this->nut_;
375 const Foam::fvModels& fvModels(Foam::fvModels::New(this->mesh_));
376 const Foam::fvConstraints& fvConstraints
377 (
378 Foam::fvConstraints::New(this->mesh_)
379 );
380
381 eddyViscosity<RASModel<BasicMomentumTransportModel>>::correct();
382
383 volScalarField::Internal divU
384 (
385 fvc::div(fvc::absolute(this->phi(), U))().v()
386 );
387
388 tmp<volTensorField> tgradU = fvc::grad(U);
389 volScalarField::Internal G
390 (
391 this->GName(),
392 nut.v()*(dev(twoSymm(tgradU().v())) && tgradU().v())
393 );
394 tgradU.clear();
395
396 // Update epsilon and G at the wall
397 epsilon_.boundaryFieldRef().updateCoeffs();
398
399 volScalarField::Internal magU(mag(U));
400 volScalarField::Internal magU3(pow3(magU));
401
402 // Dissipation equation
403 tmp<fvScalarMatrix> epsEqn
404 (
405 fvm::ddt(alpha, rho, epsilon_)
406 + fvm::div(alphaRhoPhi, epsilon_)
407 - fvm::laplacian(alpha*rho*DepsilonEff(), epsilon_)
408 ==
409 C1_*alpha()*rho()*G*epsilon_()/k_()
410 - fvm::SuSp(((2.0/3.0)*C1_)*alpha()*rho()*divU, epsilon_)
411 - fvm::Sp(C2_*alpha()*rho()*epsilon_()/k_(), epsilon_)
412 + epsilonSource(magU, magU3)
413 + fvModels.source(alpha, rho, epsilon_)
414 );
415
416 epsEqn.ref().relax();
417 fvConstraints.constrain(epsEqn.ref());
418 epsEqn.ref().boundaryManipulate(epsilon_.boundaryFieldRef());
419 solve(epsEqn);
420 fvConstraints.constrain(epsilon_);
421 boundEpsilon();
422
423 // Turbulent kinetic energy equation
424 tmp<fvScalarMatrix> kEqn
425 (
426 fvm::ddt(alpha, rho, k_)
427 + fvm::div(alphaRhoPhi, k_)
428 - fvm::laplacian(alpha*rho*DkEff(), k_)
429 ==
430 alpha()*rho()*G
431 - fvm::SuSp((2.0/3.0)*alpha()*rho()*divU, k_)
432 - fvm::Sp(alpha()*rho()*epsilon_()/k_(), k_)
433 + kSource(magU, magU3)
434 + fvModels.source(alpha, rho, k_)
435 );
436
437 kEqn.ref().relax();
438 fvConstraints.constrain(kEqn.ref());
439 solve(kEqn);
440 fvConstraints.constrain(k_);
441 bound(k_, this->kMin_);
442
443 correctNut();
444}
