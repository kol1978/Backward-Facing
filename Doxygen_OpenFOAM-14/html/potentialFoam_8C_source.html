27Description
28 Potential flow solver which solves for the velocity potential, to
29 calculate the flux-field, from which the velocity field is obtained by
30 reconstructing the flux.
31
32 This application is particularly useful to generate starting fields for
33 Navier-Stokes codes.

54int main(int argc, char *argv[])
55{
56 argList::addOption
57 (
58 "pName",
59 "pName",
60 "Name of the pressure field"
61 );
62
63 argList::addBoolOption
64 (
65 "initialiseUBCs",
66 "Initialise U boundary conditions"
67 );
68
69 argList::addBoolOption
70 (
71 "writePhi",
72 "Write the velocity potential field"
73 );
74
75 argList::addBoolOption
76 (
77 "writep",
78 "Calculate and write the pressure field"
79 );
80
81 #include "setRootCaseFunctionObjects.H"
82 #include "createTime.H"
83 #include "createMesh.H"
84
85 nonOrthogonalSolutionControl potentialFlow(mesh, "potentialFlow");
86
87 #include "createFields.H"
88
89 // * * * * * * * * * * * * * * * * * * * * * * * * * * * * * * * * * * * //
90
91 Info<< nl << "Calculating potential flow" << endl;
92
93 // Since solver contains no time loop it would never execute
94 // function objects so do it ourselves
95 runTime.functionObjects().start();
96
97 MRF.makeRelative(phi);
98 adjustPhi(phi, U, p);
99
100 // Non-orthogonal velocity potential corrector loop
101 while (potentialFlow.correctNonOrthogonal())
102 {
103 fvScalarMatrix PhiEqn
104 (
105 fvm::laplacian(dimensionedScalar(dimless, 1), Phi)
106 ==
107 fvc::div(phi)
108 );
109
110 PhiEqn.setReference(PhiRefCell, PhiRefValue);
111 PhiEqn.solve();
112
113 if (potentialFlow.finalNonOrthogonalIter())
114 {
115 phi -= PhiEqn.flux();
116 }
117 }
118
119 Info<< "Continuity error = "
120 << mag(fvc::div(phi))().weightedAverage(mesh.V()).value()
121 << endl;
122
123 U = fvc::reconstruct(MRF.absolute(phi));
124 U.correctBoundaryConditions();
125
126 Info<< "Interpolated velocity error = "
127 << (
128 sqrt(sum(sqr(fvc::flux(U) - MRF.absolute(phi))))
129 /sum(mesh.magSf())
130 ).value()
131 << endl;
132
133 // Write U and phi
134 U.write();
135 phi.write();
136
137 // Optionally write Phi
138 if (args.optionFound("writePhi"))
139 {
140 Phi.write();
141 }
142
143 // Calculate the pressure field
144 if (args.optionFound("writep"))
145 {
146 Info<< nl << "Calculating approximate pressure field" << endl;
147
148 label pRefCell = 0;
149 scalar pRefValue = 0.0;
150 setRefCell
151 (
152 p,
153 potentialFlow.dict(),
154 pRefCell,
155 pRefValue
156 );
157
158 // Calculate the flow-direction filter tensor
159 volScalarField magSqrU(magSqr(U));
160 volSymmTensorField F(sqr(U)/(magSqrU + small*average(magSqrU)));
161
162 // Calculate the divergence of the flow-direction filtered div(U*U)
163 // Filtering with the flow-direction generates a more reasonable
164 // pressure distribution in regions of high velocity gradient in the
165 // direction of the flow
166 volScalarField divDivUU
167 (
168 fvc::div
169 (
170 F & fvc::div(phi, U),
171 "div(div(phi,U))"
172 )
173 );
174
175 // Solve a Poisson equation for the approximate pressure
176 while (potentialFlow.correctNonOrthogonal())
177 {
178 fvScalarMatrix pEqn
179 (
180 fvm::laplacian(p) + divDivUU
181 );
182
183 pEqn.setReference(pRefCell, pRefValue);
184 pEqn.solve();
185 }
186
187 p.write();
188 }
189
190 runTime.functionObjects().end();
191
192 Info<< "ExecutionTime = " << runTime.elapsedCpuTime() << " s"
193 << " ClockTime = " << runTime.elapsedClockTime() << " s"
194 << nl << endl;
195
196 Info<< "End\n" << endl;
197
198 return 0;
199}
