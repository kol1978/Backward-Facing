30Foam::autoPtr<Foam::dragModel> Foam::dragModel::New
31(
32 const dictionary& modelDict,
33 const phaseInterface& interface,
34 const bool registerObject
35)
36{
37 const word dragModelType(modelDict.lookup("type"));
38
39 Info<< indentOrNl << "Selecting " << typeName << ' ' << dragModelType;
40 if (registerObject) Info << " for " << interface.name();
41 Info<< endl;
42
43 dictionaryConstructorTable::iterator cstrIter =
44 dictionaryConstructorTablePtr_->find(dragModelType);
45
46 if (cstrIter == dictionaryConstructorTablePtr_->end())
47 {
48 FatalIOErrorInFunction(modelDict)
49 << "Unknown dragModel type "
50 << dragModelType << endl << endl
51 << "Valid dragModel types are : " << endl
52 << dictionaryConstructorTablePtr_->sortedToc()
53 << exit(FatalIOError);
54 }
55
56 printDictionary print(modelDict);
57
58 return cstrIter()(modelDict, interface, registerObject);
59}
