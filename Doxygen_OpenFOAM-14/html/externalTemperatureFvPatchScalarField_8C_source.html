97externalTemperatureFvPatchScalarField
98(
99 const fvPatch& p,
100 const DimensionedField<scalar, fvMesh>& iF,
101 const dictionary& dict

104 mixedFvPatchScalarField(p, iF, dict, false),

106 Q_

108 haveQ_

121 dict

137 ),

143 : NaN
144 ),
145 Ta_
146 (
147 h_.valid() || haveEmissivity_
148 ? Function1<scalar>::New
149 (
150 "Ta",
151 time().userUnits(),
152 dimTemperature,
153 dict
154 ).ptr()
155 : nullptr
156 ),
157 relax_(dict.lookupOrDefault<scalar>("relaxation", units::fraction, 1)),
158 qrName_(dict.lookupOrDefault<word>("qr", word::null)),
159 qrRelax_(dict.lookupOrDefault<scalar>("qrRelaxation", units::fraction, 1)),
160 qrPrevious_
161 (
162 qrName_ != word::null
163 ? dict.found("qrPrevious")
164 ? scalarField("qrPrevious", dimPower/dimArea, dict, p.size())
165 : scalarField(p.size(), 0)
166 : scalarField()
167 )
168{
169 fvPatchScalarField::operator=
170 (
171 scalarField("value", iF.dimensions(), dict, p.size())
172 );
173
174 if (haveEmissivity_ && (emissivity_ < 0 || emissivity_ > 1))
175 {
176 FatalIOErrorInFunction(dict)
177 << "Emissivity must be in the range 0 to 1"
178 << exit(FatalIOError);
179 }
180
181 if (dict.found("refValue"))
182 {
183 // Full restart
184 refValue() = scalarField("refValue", iF.dimensions(), dict, p.size());
185 refGrad() =
186 scalarField
187 (
188 "refGradient",
189 iF.dimensions()/dimLength,
190 dict,
191 p.size()
192 );
193 valueFraction() =
194 scalarField("valueFraction", units::fraction, dict, p.size());
195 }
196 else
197 {
198 // Start from user entered data. Assume fixedValue.
199 refValue() = *this;
200 refGrad() = 0;
201 valueFraction() = 1;
202 }
203}

345void Foam::externalTemperatureFvPatchScalarField::updateCoeffs()
346{
347 if (updated())
348 {
349 return;
350 }
351
352 const scalar t = time().value();
353
354 // Get thermal conductivities, the patch temperature and any explicit
355 // correction heat flux
356 scalarField kappa;
357 tmp<scalarField> sumKappaTcByDelta, sumKappaByDelta;
358 tmp<scalarField> T;
359 tmp<scalarField> sumq;
360 getKappa(kappa, sumKappaTcByDelta, sumKappaByDelta, T, sumq);
361
362 // Add any user specified heat fluxes
363 if (haveQ_)
364 {
365 plusEqOp(sumq, Q_->value(t)/gSum(patch().magSf()));
366 }
367 if (haveq_)
368 {
369 plusEqOp(sumq, q_->value(t));
370 }
371
372 // Add the (relaxed) radiative heat flux
373 if (qrName_ != word::null)
374 {
375 const fvPatchScalarField& qrCurrent =
376 patch().lookupPatchField<volScalarField, scalar>(qrName_);
377
378 const scalarField qr(qrRelax_*qrCurrent + (1 - qrRelax_)*qrPrevious_);
379
380 qrPrevious_ = qr;
381
382 plusEqOp(sumq, qr);
383 }
384
385 // Evaluate the ambient temperature
386 const scalar Ta = h_.valid() || haveEmissivity_ ? Ta_->value(t) : NaN;
387
388 // Evaluate the combined convective and radiative heat transfer coefficient
389 tmp<scalarField> hEff;
390 if (h_.valid())
391 {
392 h_->update();
393 plusEqOp(hEff, h_());
394 }
395 if (haveEmissivity_)
396 {
397 plusEqOp
398 (
399 hEff,
400 emissivity_
401 *constant::physicoChemical::sigma.value()
402 *(sqr(Ta) + sqr(T()))
403 *(Ta + T())
404 );
405 }
406
407 // If we have a heat transfer coefficient then add it to the kappa sums
408 if (hEff.valid())
409 {
410 plusEqOp(sumKappaByDelta, hEff());
411 plusEqOp(sumKappaTcByDelta, hEff*Ta);
412 }
413
414 // Set the mixed parameters
415 const scalarField kappaByDelta(kappa*patch().deltaCoeffs());
416 tmp<scalarField> kappaPlusSumKappaByDelta
417 (
418 sumKappaByDelta.valid()
419 ? kappaByDelta + sumKappaByDelta()
420 : tmp<scalarField>(kappaByDelta)
421 );
422
423 // ... value fraction
424 if (sumKappaByDelta.valid())
425 {
426 valueFraction() = sumKappaByDelta()/kappaPlusSumKappaByDelta();
427 }
428 else
429 {
430 valueFraction() = Zero;
431 }
432
433 // ... reference value
434 tmp<scalarField> trefValue;
435 if (sumKappaByDelta.valid())
436 {
437 plusEqOp
438 (
439 trefValue,
440 max(sumKappaTcByDelta, small*kappaByDelta*patchInternalField())
441 /max(sumKappaByDelta, small*kappaByDelta)
442 );
443 }
444 if (sumq.valid())
445 {
446 plusEqOp(trefValue, sumq()/kappaPlusSumKappaByDelta());
447 }
448 if (trefValue.valid())
449 {
450 refValue() = trefValue;
451 }
452 else
453 {
454 refValue() = Zero;
455 }
456
457 // ... and reference gradient
458 if (sumq.valid())
459 {
460 refGrad() = sumq*patch().deltaCoeffs()/kappaPlusSumKappaByDelta();
461 }
462 else
463 {
464 refGrad() = Zero;
465 }
466
467 // Modify the mixed parameters for under-relaxation
468 if (relax_ != 1)
469 {
470 const scalarField f(valueFraction());
471 valueFraction() = 1 - relax_*(1 - f);
472 refValue() = (f*relax_*refValue() + (1 - relax_)*T)/valueFraction();
473 }
474
475 mixedFvPatchScalarField::updateCoeffs();
476
477 if (debug)
478 {
479 const scalar Q = gSum(kappa*patch().magSf()*snGrad());
480
481 Info<< patch().mesh().name() << ':'
482 << patch().name() << ':'
483 << this->internalField().name() << " :"
484 << " heat transfer rate:" << Q
485 << " walltemperature "
486 << " min:" << gMin(*this)
487 << " max:" << gMax(*this)
488 << " avg:" << gAverage(*this)
489 << endl;
490 }
491}

494void Foam::externalTemperatureFvPatchScalarField::write
495(
496 Ostream& os
497) const
498{
499 fvPatchScalarField::write(os);
500
501 if (haveQ_)
502 {
503 writeEntry(os, time().userUnits(), dimPower, Q_());
504 }
505
506 if (haveq_)
507 {
508 writeEntry(os, time().userUnits(), dimPower/dimArea, q_());
509 }
510
511 if (h_.valid())
512 {
513 writeEntry(os, h_());
514 }
515
516 if (haveEmissivity_)
517 {
518 writeEntry(os, "emissivity", emissivity_);
519 }
520
521 if (h_.valid() || haveEmissivity_)
522 {
523 writeEntry(os, time().userUnits(), dimTemperature, Ta_());
524 }
525
526 writeEntryIfDifferent(os, "relaxation", scalar(1), relax_);
527
528 if (qrName_ != word::null)
529 {
530 writeEntry(os, "qr", qrName_);
531 writeEntry(os, "qrRelaxation", qrRelax_);
532 writeEntry(os, "qrPrevious", qrPrevious_);
533 }
534
535 writeEntry(os, "refValue", refValue());
536 writeEntry(os, "refGradient", refGrad());
537 writeEntry(os, "valueFraction", valueFraction());
538 writeEntry(os, "value", *this);
539}
