391void kkLOmega::correct()
392{
393 eddyViscosity<incompressible::RASModel>::correct();
394
395 if (!turbulence_)
396 {
397 return;
398 }
399
400 volScalarField y(this->y());
401
402 const volScalarField lambdaT(sqrt(kt_)/(omega_ + omegaMin_));
403
404 const volScalarField lambdaEff(min(Clambda_*y, lambdaT));
405
406 const volScalarField fw
407 (
408 pow
409 (
410 lambdaEff
411 /(lambdaT + dimensionedScalar(dimLength, rootVSmall)),
412 2.0/3.0
413 )
414 );
415
416 tmp<volTensorField> tgradU(fvc::grad(U_));
417 const volTensorField& gradU = tgradU();
418
419 const volScalarField Omega(sqrt(2.0)*mag(skew(gradU)));
420
421 const volScalarField S2(2*magSqr(dev(symm(gradU))));
422
423 const volScalarField ktS(fSS(Omega)*fw*kt_);
424
425 const volScalarField nuts
426 (
427 fv(sqr(fw)*kt_/nu()/(omega_ + omegaMin_))
428 *fINT()
429 *Cmu(sqrt(S2))*sqrt(ktS)*lambdaEff
430 );
431 const volScalarField Pkt(this->GName(), nuts*S2);
432
433 const volScalarField ktL(kt_ - ktS);
434 const volScalarField ReOmega(sqr(y)*Omega/nu());
435
436 const volScalarField nutl
437 (
438 min
439 (
440 C11_*fTaul(lambdaEff, ktL, Omega)*Omega*sqr(lambdaEff)
441 *sqrt(ktL)*lambdaEff/nu()
442 + C12_*BetaTS(ReOmega)*sqr(sqr(lambdaEff/Clambda_)*Omega)/nu()
443 , 0.5*(kl_ + ktL)/(sqrt(S2) + omegaMin_)
444 )
445 );
446
447 const volScalarField Pkl(nutl*S2);
448
449 const volScalarField alphaTEff
450 (
451 alphaT(lambdaEff, fv(sqr(fw)*kt_/nu()/(omega_ + omegaMin_)), ktS)
452 );
453
454 // By pass source term divided by kl_
455
456 const dimensionedScalar fwMin("small", dimless, rootVSmall);
457
458 const volScalarField Rbp
459 (
460 CR_*(1 - exp(-phiBP(Omega)()/Abp_))*omega_
461 /(fw + fwMin)
462 );
463
464 const volScalarField fNatCrit(1 - exp(-Cnc_*sqrt(kl_)*y/nu()));
465
466 // Natural source term divided by kl_
467 const volScalarField Rnat
468 (
469 CrNat_*(1 - exp(-phiNAT(ReOmega, fNatCrit)/Anat_))*Omega
470 );
471
472
473 omega_.boundaryFieldRef().updateCoeffs();
474
475 // Turbulence specific dissipation rate equation
476 tmp<fvScalarMatrix> omegaEqn
477 (
478 fvm::ddt(omega_)
479 + fvm::div(phi_, omega_)
480 - fvm::laplacian(DomegaEff(alphaTEff), omega_)
481 ==
482 Cw1_*Pkt*omega_/(kt_ + kMin_)
483 - fvm::SuSp
484 (
485 (1 - CwR_/(fw + fwMin))*kl_*(Rbp + Rnat)/(kt_ + kMin_)
486 , omega_
487 )
488 - fvm::Sp(Cw2_*sqr(fw)*omega_, omega_)
489 + (
490 Cw3_*fOmega(lambdaEff, lambdaT)*alphaTEff*sqr(fw)*sqrt(kt_)
491 )()()/pow3(y())
492 );
493
494 omegaEqn.ref().relax();
495 omegaEqn.ref().boundaryManipulate(omega_.boundaryFieldRef());
496
497 solve(omegaEqn);
498 bound(omega_, omegaMin_);
499
500
501 const volScalarField Dl(D(kl_));
502
503 // Laminar kinetic energy equation
504 tmp<fvScalarMatrix> klEqn
505 (
506 fvm::ddt(kl_)
507 + fvm::div(phi_, kl_)
508 - fvm::laplacian(nu(), kl_)
509 ==
510 Pkl
511 - fvm::Sp(Rbp + Rnat + Dl/(kl_ + kMin_), kl_)
512 );
513
514 klEqn.ref().relax();
515 klEqn.ref().boundaryManipulate(kl_.boundaryFieldRef());
516
517 solve(klEqn);
518 bound(kl_, kMin_);
519
520
521 const volScalarField Dt(D(kt_));
522
523 // Turbulent kinetic energy equation
524 tmp<fvScalarMatrix> ktEqn
525 (
526 fvm::ddt(kt_)
527 + fvm::div(phi_, kt_)
528 - fvm::laplacian(DkEff(alphaTEff), kt_)
529 ==
530 Pkt
531 + (Rbp + Rnat)*kl_
532 - fvm::Sp(omega_ + Dt/(kt_+ kMin_), kt_)
533 );
534
535 ktEqn.ref().relax();
536 ktEqn.ref().boundaryManipulate(kt_.boundaryFieldRef());
537
538 solve(ktEqn);
539 bound(kt_, kMin_);
540
541
542 // Update total fluctuation kinetic energy dissipation rate
543 epsilon_ = kt_*omega_ + Dl + Dt;
544
545
546 // Re-calculate turbulent viscosity
547 nut_ = nuts + nutl;
548 nut_.correctBoundaryConditions();
549}
