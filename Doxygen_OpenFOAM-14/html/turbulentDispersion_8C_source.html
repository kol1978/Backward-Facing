192void Foam::Lagrangian::turbulentDispersion::postConstruct()
193{
194 LagrangianModels& modelList = cloud().LagrangianModels();
195
196 dragPtr_ = nullptr;
197
198 forAll(modelList, i)
199 {
200 if (!isA<drag>(modelList[i])) continue;
201
202 if (dragPtr_ != nullptr)
203 {
204 FatalErrorInFunction
205 << "Multiple drag models found. Turbulent dispersion "
206 << "requires exactly one drag model."
207 << exit(FatalError);
208 }
209
210 dragPtr_ = &refCast<const drag>(modelList[i]);
211 }
212
213 if (dragPtr_ == nullptr)
214 {
215 FatalErrorInFunction
216 << "No drag models found. Turbulent dispersion "
217 << "requires exactly one drag model."
218 << exit(FatalError);
219 }
220
221 completeTurbField<maxLagrangianScalarFieldSource>(fractionTurb_);
222 completeTurbField<NaNLagrangianScalarFieldSource>(tTurb_);
223 completeTurbField<NaNLagrangianVectorFieldSource>(Uturb_);
224}

227void Foam::Lagrangian::turbulentDispersion::preAddSup
228(
229 const LagrangianSubScalarField& deltaT,
230 const bool final
231)
232{
233 const LagrangianSubMesh& subMesh = deltaT.mesh();
234
235 // References to the evolving fields
236 LagrangianSubScalarSubField& fractionTurb = fractionTurb_.ref(subMesh);
237 LagrangianSubScalarSubField& tTurb = tTurb_.ref(subMesh);
238 LagrangianSubVectorSubField& Uturb = Uturb_.ref(subMesh);
239 const LagrangianSubScalarField& fractionTurb0 = fractionTurb.oldTime();
240 const LagrangianSubVectorField& Uturb0 = Uturb.oldTime();
241
242 // Update the eddy time-scale
243 const LagrangianSubScalarField magUrel
244 (
245 mag(cloud().U(subMesh) - cloud<clouds::carried>().Uc(subMesh))
246 );
247 static const dimensionedScalar rootVSmallEpsilon
248 (
249 dimensions::turbulentEpsilon,
250 rootVSmall
251 );
252 static const dimensionedScalar rootVSmallEpsilonU
253 (
254 dimensions::turbulentEpsilon*dimensions::velocity,
255 rootVSmall
256 );
257 tTurb =
258 min
259 (
260 kc_(subMesh)/max(epsilonc_(subMesh), rootVSmallEpsilon),
261 Cmu75_*kc_(subMesh)*sqrt(kc_(subMesh))
262 /max(epsilonc_(subMesh)*magUrel, rootVSmallEpsilonU)
263 );
264
265 // Velocity fluctuation magnitude
266 LagrangianSubScalarField magUturb(sqrt(2.0/3.0*kc_(subMesh)));
267
268 // Modify particles on the walls to account for the turbulent kinetic
269 // energy being constrained to zero. This prevents a turbulent velocity
270 // fluctuation from pointing through a wall and causing a particle to
271 // vibrate as the drag model and the rebound model fight each other.
272 const PackedBoolList patchIsWall
273 (
274 mesh().poly().boundary().findIndices<wallPolyPatch>().toc()
275 );
276 forAll(subMesh, subi)
277 {
278 const label i = subi + subMesh.start();
279
280 if (mesh().state(i) < LagrangianState::onPatchZero) continue;
281
282 const label patchi =
283 static_cast<label>(mesh().state(i))
284 - static_cast<label>(LagrangianState::onPatchZero);
285
286 if (!patchIsWall[patchi]) continue;
287
288 tTurb[subi] = rootVSmall;
289
290 magUturb[subi] = 0;
291 }
292
293 // Initialise the average turbulent velocities
294 avgUturbPtr_.set
295 (
296 LagrangianSubVectorField::New
297 (
298 "avgUturb",
299 subMesh,
300 dimensionedVector(dimensions::velocity, Zero)
301 ).ptr()
302 );
303
304 // Consider each particle in turn
305 forAll(subMesh, subi)
306 {
307 // Create independent generators for each particle. That way if the
308 // sequence grows or shrinks across iterations, that effect doesn't
309 // propagate and affect other particles' sequences
310 randomGenerator rndGen
311 (
312 rndGen_.sampleAB(labelMin, labelMax),
313 false
314 );
315 distributions::standardNormal stdNormal
316 (
317 rndGen_.sampleAB(labelMin, labelMax),
318 false
319 );
320
321 // Generate a random direction with a normally distributed magnitude
322 auto rndDir = [&rndGen, &stdNormal]()
323 {
324 const scalar theta =
325 rndGen.scalar01()*constant::mathematical::twoPi;
326 const scalar z = 2*rndGen.scalar01() - 1;
327 const scalar r = sqrt(1 - sqr(z));
328 return stdNormal.sample()*vector(r*cos(theta), r*sin(theta), z);
329 };
330
331 // Set up sub-stepping
332 const scalar Dt = max(deltaT[subi], rootVSmall);
333 scalar dt = 0;
334
335 // Continue/complete the previous eddy
336 if (fractionTurb0[subi] < 1)
337 {
338 dt += tTurb[subi]*(1 - fractionTurb0[subi]);
339
340 avgUturbPtr_()[subi] += min(dt/Dt, 1)*Uturb0[subi];
341 }
342
343 // Add new eddies across the time-step
344 const scalar nEddies = Dt/tTurb[subi];
345 if (nEddies < maxDiscreteEddies_)
346 {
347 while (dt < Dt)
348 {
349 // Create a turbulent velocity fluctuation for the new eddy
350 Uturb[subi] = rndDir()*magUturb[subi];
351
352 // Add this eddy to the average
353 avgUturbPtr_()[subi] +=
354 (min(dt + tTurb[subi], Dt) - dt)/Dt*Uturb[subi];
355
356 // Increment the time
357 dt += tTurb[subi];
358 }
359
360 // Set the fraction to where we got to in the current eddy
361 fractionTurb[subi] = 1 - (dt - Dt)/tTurb[subi];
362 }
363 else
364 {
365 // Create a turbulent velocity fluctuation for the new eddy
366 Uturb[subi] = rndDir()*magUturb[subi];
367
368 // Add the combined effect of all the eddies to the average
369 avgUturbPtr_()[subi] +=
370 (Dt - dt)/Dt
371 *sqrt(nEddies/3)
372 *stdNormal.sample<vector>()
373 *magUturb[subi];
374
375 // Set the fraction to indicate that the eddies are finished
376 fractionTurb[subi] = 1;
377 }
378 }
379
380 // If this is not the final iteration then rewind the generator so that the
381 // same numbers are generated in the next iteration
382 rndGen_.start(!final);
383}
