30Foam::autoPtr<Foam::distribution> Foam::distribution::New
31(
32 const unitSet& units,
33 const dictionary& dict,
34 const label sampleQ,
35 randomGenerator&& rndGen,
36 const bool report
37)
38{
39 const word distributionType(dict.lookup("type"));
40
41 if (report)
42 {
43 Info<< indentOrNl << "Selecting " << typeName
44 << " type " << distributionType << endl;
45 }
46
47 dictionaryConstructorTable::iterator cstrIter =
48 dictionaryConstructorTablePtr_->find(distributionType);
49
50 if (cstrIter == dictionaryConstructorTablePtr_->end())
51 {
52 FatalIOErrorInFunction(dict)
53 << "Unknown " << typeName << " type " << distributionType
54 << nl << nl << "Valid " << typeName << " types are:" << nl
55 << dictionaryConstructorTablePtr_->sortedToc()
56 << exit(FatalIOError);
57 }
58
59 const dictionary& distributionDict =
60 dict.optionalSubDict(distributionType + "Distribution");
61
62 printDictionary print(distributionDict);
63
64 autoPtr<distribution> distributionPtr
65 (
66 cstrIter()
67 (
68 units,
69 distributionDict,
70 sampleQ,
71 std::move(rndGen)
72 )
73 );
74
75 if (report)
76 {
77 Info<< indent << "min/average/max value = "
78 << distributionPtr->min() << '/'
79 << distributionPtr->mean() << '/'
80 << distributionPtr->max() << endl;
81 }
82
83 return distributionPtr;
84}

105)
106{
107 if (dPtr->sampleQ_ == sampleQ)
108 {
109 return autoPtr<distribution>(dPtr.ptr());
