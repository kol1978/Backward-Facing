31template<class LListBase, class T>
32Foam::Ostream& Foam::operator<<(Ostream& os, const ULPtrList<LListBase, T>& lst)
33{
34 // Write size
35 os << nl << lst.size();
36
37 // Write beginning of contents
38 os << nl << token::BEGIN_LIST << nl;
39
40 // Write contents
41 for
42 (
43 typename ULPtrList<LListBase, T>::const_iterator iter = lst.begin();
44 iter != lst.end();
45 ++iter
46 )
47 {
48 os << iter() << nl;
49 }
50
51 // Write end of contents
52 os << token::END_LIST;
53
54 // Check state of IOstream
55 os.check("Ostream& operator<<(Ostream&, const ULPtrList<LListBase, T>&)");
56
57 return os;
58}
