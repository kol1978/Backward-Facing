70Foam::zoneSet Foam::zoneGenerators::normal::generate() const
71{
72 const zoneSet zs(zoneGenerator_->generate());
73 const faceZone& fZone = zs.fZone();
74
75 labelList faceIndices(fZone);
76 boolList flipMap
77 (
78 fZone.oriented()
79 ? fZone.flipMap()
80 : boolList(faceIndices.size(), false)
81 );
82
83 const vectorField& faceAreas = mesh_.faceAreas();
84
85 label fj = 0;
86 forAll(faceIndices, fi)
87 {
88 const label facei = faceIndices[fi];
89 const bool flip = flipMap[fi];
90 const vector n(normalised(faceAreas[facei]));
91 const vector fn(flip ? -n : n);
92
93 if (mag(1 - (fn & normal_)) < tol_)
94 {
95 faceIndices[fj] = facei;
96 flipMap[fj] = flip;
97 fj++;
98 }
99 }
100
101 faceIndices.setSize(fj);
102 flipMap.setSize(fj);
103
104 return zoneSet
105 (
106 new faceZone
107 (
108 zoneName_,
109 faceIndices,
110 flipMap,
111 mesh_.faceZones(),
112 moveUpdate_,
113 true
114 )
115 );
116}
