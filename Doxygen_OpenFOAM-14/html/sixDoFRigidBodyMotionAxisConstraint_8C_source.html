88bool Foam::sixDoFRigidBodyMotionConstraints::axis::read
89(
90 const dictionary& sDoFRBMCDict
91)
92{
93 sixDoFRigidBodyMotionConstraint::read(sDoFRBMCDict);
94
95 sDoFRBMCCoeffs_.lookup("axis") >> axis_;
96
97 scalar magFixedAxis(mag(axis_));
98
99 if (magFixedAxis > vSmall)
100 {
101 axis_ /= magFixedAxis;
102 }
103 else
104 {
105 FatalErrorInFunction
106 << "axis has zero length"
107 << abort(FatalError);
108 }
109
110 return true;
111}
