45template<class ThermoType>
46const ThermoType& Foam::UInhomogeneousMixture<ThermoType>::mixture
47(
48 const scalar fu
49) const
50{
51 if (fu < 0.0001)
52 {
53 return oxidant_;
54 }
55 else
56 {
57 const scalar ox = 1 - fu;
58
59 mixture_ = fu*fuel_;
60 mixture_ += ox*oxidant_;
61
62 return mixture_;
63 }
64}
