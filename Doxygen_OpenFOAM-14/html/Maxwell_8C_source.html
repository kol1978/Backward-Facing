52template<class BasicMomentumTransportModel>
53PtrList<dimensionedScalar>
54Maxwell<BasicMomentumTransportModel>::readModeCoefficients
55(
56 const word& type,
57 const word& name,
58 const dimensionSet& dims
59) const
60{
61 PtrList<dimensionedScalar> modeCoeffs(nModes_);
62
63 if (modeCoefficients_.size())
64 {
65 if (this->typeDict(type).found(name))
66 {
67 IOWarningInFunction(this->typeDict(type))
68 << "Using 'modes' list, '" << name << "' entry will be ignored."
69 << endl;
70 }
71
72 label modei = 0;
73 forAllConstIter(dictionary, modeCoefficients_, iter)
74 {
75 modeCoeffs.set
76 (
77 modei++,
78 new dimensioned<scalar>
79 (
80 name,
81 dims,
82 iter().dict().lookup(name)
83 )
84 );
85 }
86 }
87 else
88 {
89 modeCoeffs.set
90 (
91 0,
92 new dimensioned<scalar>
93 (
94 name,
95 dims,
96 this->typeDict(type).lookup(name)
97 )
98 );
99 }
100
101 return modeCoeffs;
102}

250template<class BasicMomentumTransportModel>
251bool Maxwell<BasicMomentumTransportModel>::read()
252{
253 if (laminarModel<BasicMomentumTransportModel>::read())
254 {
255 if (modeCoefficients_.size())
256 {
257 this->typeDict().lookup("modes") >> modeCoefficients_;
258 }
259
260 nuM_ = readModeCoefficients("nuM", dimKinematicViscosity);
261 lambdas_ = readModeCoefficients("lambda", dimTime);
262 nuMSum_ = sumNuM(nuM_);
263
264 return true;
265 }
266 else
267 {
268 return false;
269 }
270}

294template<class BasicMomentumTransportModel>
295tmp<surfaceVectorField> Maxwell<BasicMomentumTransportModel>::devTau() const
296{
297 const surfaceVectorField nf(this->mesh().nf());
298
299 return surfaceVectorField::New
300 (
301 this->groupName("devTau"),
302 fvc::dotInterpolate
303 (
304 nf,
305 this->alpha_*this->rho_*this->nuMSum_*fvc::grad(this->U_)
306 )
307 + fvc::dotInterpolate
308 (
309 nf,
310 this->alpha_*this->rho_*sigma_
311 )
312 - fvc::dotInterpolate
313 (
314 nf,
315 this->alpha_*this->rho_*this->nu()*dev2(T(fvc::grad(this->U_)))
316 )
317 - fvc::interpolate(this->alpha_*this->rho_*nu0())*fvc::snGrad(this->U_)
318 );
319}

322template<class BasicMomentumTransportModel>
323template<class RhoFieldType>
324tmp<fvVectorMatrix>
325Maxwell<BasicMomentumTransportModel>::DivDevTau
326(
327 const RhoFieldType& rho,
328 volVectorField& U
329) const
330{
331 return
332 (
333 fvc::div(this->alpha_*rho*this->nuMSum_*fvc::grad(U))
334 + fvc::div(this->alpha_*rho*sigma_)
335 - fvc::div(this->alpha_*rho*this->nu()*dev2(T(fvc::grad(U))))
336 - fvm::laplacian(this->alpha_*rho*nu0(), U)
337 );
338}

363template<class BasicMomentumTransportModel>
364void Maxwell<BasicMomentumTransportModel>::correct()
365{
366 // Local references
367 const alphaField& alpha = this->alpha_;
368 const rhoField& rho = this->rho_;
369 const surfaceScalarField& alphaRhoPhi = this->alphaRhoPhi_;
370 const volVectorField& U = this->U_;
371 const Foam::fvModels& fvModels(Foam::fvModels::New(this->mesh_));
372 const Foam::fvConstraints& fvConstraints
373 (
374 Foam::fvConstraints::New(this->mesh_)
375 );
376
377 laminarModel<BasicMomentumTransportModel>::correct();
378
379 tmp<volTensorField> tgradU(fvc::grad(U));
380 const volTensorField& gradU = tgradU();
381
382 forAll(lambdas_, modei)
383 {
384 volSymmTensorField& sigma = nModes_ == 1 ? sigma_ : sigmas_[modei];
385
386 uniformDimensionedScalarField rLambda
387 (
388 IOobject
389 (
390 this->groupName
391 (
392 "rLambda"
393 + (nModes_ == 1 ? word::null : name(modei))
394 ),
395 this->runTime_.constant(),
396 this->mesh_
397 ),
398 1/lambdas_[modei]
399 );
400
401 // Note sigma is positive on lhs of momentum eqn
402 const volSymmTensorField P
403 (
404 twoSymm(sigma & gradU)
405 - nuM_[modei]*rLambda*twoSymm(gradU)
406 );
407
408 // Viscoelastic stress equation
409 fvSymmTensorMatrix sigmaEqn
410 (
411 fvm::ddt(alpha, rho, sigma)
412 + fvm::div
413 (
414 alphaRhoPhi,
415 sigma,
416 "div(" + alphaRhoPhi.name() + ',' + sigma_.name() + ')'
417 )
418 ==
419 alpha*rho*P
420 + sigmaSource(modei, sigma)
421 + fvModels.source(alpha, rho, sigma)
422 );
423
424 sigmaEqn.relax();
425 fvConstraints.constrain(sigmaEqn);
426 sigmaEqn.solve("sigma");
427 fvConstraints.constrain(sigma);
428 }
429
430 if (sigmas_.size())
431 {
432 volSymmTensorField sigmaSum("sigmaSum", sigmas_[0]);
433
434 for (label modei = 1; modei<sigmas_.size(); modei++)
435 {
436 sigmaSum += sigmas_[modei];
437 }
438
439 sigma_ == sigmaSum;
440 }
441}
