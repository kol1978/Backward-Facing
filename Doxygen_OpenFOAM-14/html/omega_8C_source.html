31Foam::omega::omega(const dictionary& dict)
32:
33 dimensionedScalar("omega", dimless/dimTime, NaN)
34{
35 const bool foundOmega = dict.found("omega");
36 const bool foundRpm = dict.found("rpm");
37
38 if (foundOmega && foundRpm)
39 {
40 FatalIOErrorInFunction(dict)
41 << "Rotational speeds rpm and omega both defined in dictionary "
42 << dict.name() << exit(FatalIOError);
43 }
44
45 if (!foundOmega && !foundRpm)
46 {
47 FatalIOErrorInFunction(dict)
48 << "Neither rotational speed rpm nor omega defined in dictionary "
49 << dict.name() << exit(FatalIOError);
50 }
51
52 value() =
53 foundOmega
54 ? dict.lookup<scalar>("omega", units::radians/dimTime)
55 : dict.lookup<scalar>("rpm", units::lookup("rpm"));
56}
