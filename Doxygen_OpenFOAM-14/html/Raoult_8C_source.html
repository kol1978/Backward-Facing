99void Foam::interfaceCompositionModels::Raoult::update(const volScalarField& Tf)
100{
101 YNonVapour_ = scalar(1);
102
103 forAllIter
104 (
105 HashPtrTable<interfaceCompositionModel>,
106 speciesModels_,
107 iter
108 )
109 {
110 iter()->update(Tf);
111
112 YNonVapour_ -=
113 otherMulticomponentThermo().Y(iter.key())
114 *iter()->Yf(iter.key(), Tf);
115
116 YNonVapourPrime_ -=
117 otherMulticomponentThermo().Y(iter.key())
118 *iter()->YfPrime(iter.key(), Tf);
119 }
120}

123Foam::tmp<Foam::volScalarField> Foam::interfaceCompositionModels::Raoult::Yf
124(
125 const word& speciesName,
126 const volScalarField& Tf
127) const
128{
129 if (species().found(speciesName))
130 {
131 return
132 otherMulticomponentThermo().Y(speciesName)
133 *speciesModels_[speciesName].Yf(speciesName, Tf);
134 }
135 else
136 {
137 return thermo().Y(speciesName)*YNonVapour_;
138 }
139}
