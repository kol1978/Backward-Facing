32template<class TrackingData>
33inline bool Foam::wallPoint::update
34(
35 const point& pt,
36 const wallPoint& w2,
37 const scalar tol,
38 TrackingData& td
39)
40{
41 const scalar dist2 = magSqr(pt - w2.origin());
42
43 if (valid(td))
44 {
45 const scalar diff = distSqr() - dist2;
46
47 if (diff < 0)
48 {
49 // already nearer to pt
50 return false;
51 }
52
53 if ((diff < small) || ((distSqr() > small) && (diff/distSqr() < tol)))
54 {
55 // don't propagate small changes
56 return false;
57 }
58 }
59
60 // Either *this is not yet valid or w2 is closer
61 distSqr() = dist2;
62 origin() = w2.origin();
63
64 return true;
65}

141template<class TrackingData>
142inline bool Foam::wallPoint::sameGeometry
143(
144 const wallPoint& w2,
145 const scalar tol,
146 TrackingData& td
147) const
148{
149 const scalar diff = mag(distSqr() - w2.distSqr());
150
151 return
152 diff < small
153 || ((distSqr() > small) && (diff/distSqr() < tol));
154}
