42void Foam::processorFvPatch::makeWeights(scalarField& w) const
43{
44 if (Pstream::parRun())
45 {
46 if (!mesh().conformal())
47 {
48 coupledFvPatch::makeWeights
49 (
50 w,
51 - mesh().Sf().boundaryField()[index()],
52 mesh().Cf().boundaryField()[index()]
53 - mesh().C().boundaryField()[index()]
54 );
55 }
56 else
57 {
58 coupledFvPatch::makeWeights
59 (
60 w,
61 processorPoly_.neighbFaceAreas(),
62 processorPoly_.neighbFaceCentres()
63 - processorPoly_.neighbFaceCellCentres()
64 );
65 }
66 }
67 else
68 {
69 w = 1;
70 }
71}

74Foam::tmp<Foam::vectorField> Foam::processorFvPatch::delta() const
75{
76 if (Pstream::parRun())
77 {
78 if (!mesh().conformal())
79 {
80 return
81 coupledFvPatch::delta()
82 - (
83 mesh().Cf().boundaryField()[index()]
84 - mesh().C().boundaryField()[index()]
85 );
86 }
87 else
88 {
89 return
90 coupledFvPatch::delta()
91 - transform().transform
92 (
93 processorPoly_.neighbFaceCentres()
94 - processorPoly_.neighbFaceCellCentres()
95 );
96 }
97 }
98 else
99 {
100 return coupledFvPatch::delta();
101 }
102}
