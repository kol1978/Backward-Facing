111Foam::scalar Foam::distributions::exponential::sampleForZeroQ() const
112{
113 const scalar s = rndGen_.sample01<scalar>();
114 const Pair<scalar>& Phi01 = this->Phi01();
115 const scalar PhiS = (1 - s)*Phi01[0] + s*Phi01[1];
116 return - 1/lambda_*log(- PhiS);
117}
