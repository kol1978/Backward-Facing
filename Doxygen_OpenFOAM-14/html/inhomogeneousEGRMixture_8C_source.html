46template<class ThermoType>
47Foam::scalar Foam::inhomogeneousEGRMixture<ThermoType>::fres
48(
49 const scalarFieldListSlice& Y
50) const
51{
52 return max(Y[FT] - (1 - Y[FT] - Y[EGR])/stoicRatio_, 0);
53}

56template<class ThermoType>
57Foam::scalar Foam::inhomogeneousEGRMixture<ThermoType>::Phi
58(
59 const scalarFieldListSlice& Y
60) const
61{
62 const scalar ftu = Y[FT] + Y[EGR]/(stoicRatio_ + 1);
63 return stoicRatio_*ftu/max(1 - ftu, small);
64}

67template<class ThermoType>
68const ThermoType& Foam::inhomogeneousEGRMixture<ThermoType>::mixture
69(
70 const scalar ft,
71 const scalar fu,
72 const scalar egr
73) const
74{
75 if (ft < 0.0001 && egr < 0.0001)
76 {
77 return oxidant_;
78 }
79 else
80 {
81 const scalar ox = 1 - ft - egr - (ft - fu)*stoicRatio_;
82 const scalar pr = 1 - fu - ox;
83
84 mixture_ = fu*fuel_;
85 mixture_ += ox*oxidant_;
86 mixture_ += pr*products_;
87
88 return mixture_;
89 }
90}

149template<class ThermoType>
150void Foam::inhomogeneousEGRMixture<ThermoType>::reset
151(
152 PtrList<volScalarField>& Y
153) const
154{
155 const volScalarField& fu = Y[FU];
156 volScalarField& ft = Y[FT];
157 volScalarField& b = Y[B];
158 volScalarField& egr = Y[EGR];
159
160 for (label i=0; i<=fu.nOldTimes(); i++)
161 {
162 egr.oldTimeRef(i) += (ft.oldTime(i) - fu.oldTime(i))*(1 + stoicRatio_);
163 ft.oldTimeRef(i) = fu.oldTime(i);
164 b.oldTimeRef(i) = 1;
165 }
166}
