Detailed Description

Cavitation fvModel.

Usage
Example usage: 
VoFCavitation
{
    type    VoFCavitation;

    model   SchnerrSauer;

    Kunz
    {
        pSat    2300;   // Saturation pressure

        UInf    20.0;
        tInf    0.005; // L = 0.1 m
        Cc      1000;
        Cv      1000;
    }

    Merkle
    {
        pSat    2300;   // Saturation pressure

        UInf    20.0;
        tInf    0.005;  // L = 0.1 m
        Cc      80;
        Cv      1e-03;
    }

    SchnerrSauer
    {
        pSat    2300;   // Saturation pressure

        n       1.6e+13;
        dNuc    2.0e-06;
        Cc      1;
        Cv      1;
    }
}
