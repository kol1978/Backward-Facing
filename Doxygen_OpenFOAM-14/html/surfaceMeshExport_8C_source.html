24Application
25 surfaceMeshExport
26
27Description
28 Export from surfMesh to various third-party surface formats with
29 optional scaling or transformations (rotate/translate) on a
30 coordinateSystem.
31
32Usage
33 \b surfaceMeshExport outputFile [OPTION]
34
35 Options:
36 - \par -clean
37 Perform some surface checking/cleanup on the input surface.
38
39 - \par -name <name>
40 Specify an alternative surface name when writing.
41
42 - \par -scaleIn <scale>
43 Specify a scaling factor when reading files.
44
45 - \par -scaleOut <scale>
46 Specify a scaling factor when writing files.
47
48 - \par -from <coordinateSystem>
49 Specify a coordinate system when reading files.
50
51 - \par -to <coordinateSystem>
52 Specify a coordinate system when writing files.
53
54 Note:
55 The filename extensions are used to determine the file format type.
56

68
69int main(int argc, char *argv[])
70{
71 argList::addNote
72 (
73 "export from surfMesh to various third-party surface formats"
74 );
75
76 argList::noParallel();
77 argList::validArgs.append("output surface file");
78
79 argList::addBoolOption
80 (
81 "clean",
82 "perform some surface checking/cleanup on the input surface"
83 );
84 argList::addOption
85 (
86 "name",
87 "name",
88 "specify an alternative surface name when reading - "
89 "default is 'default'"
90 );
91 argList::addOption
92 (
93 "scaleIn",
94 "factor",
95 "geometry scaling factor on input - default is 1"
96 );
97 argList::addOption
98 (
99 "scaleOut",
100 "factor",
101 "geometry scaling factor on output - default is 1"
102 );
103 argList::addOption
104 (
105 "from",
106 "coordinateSystem",
107 "specify the source coordinate system, applied after '-scaleIn'"
108 );
109 argList::addOption
110 (
111 "to",
112 "coordinateSystem",
113 "specify the target coordinate system, applied before '-scaleOut'"
114 );
115
116 argList args(argc, argv);
117 Time runTime(args.rootPath(), args.caseName());
118
119 const fileName exportName = args[1];
120 const word importName = args.optionLookupOrDefault<word>("name", "default");
121
122 // check that writing is supported
123 if (!MeshedSurface<face>::canWriteType(exportName.ext(), true))
124 {
125 return 1;
126 }
127
128
129 // Get the coordinate transformations
130 autoPtr<coordinateSystem> fromCsys;
131 autoPtr<coordinateSystem> toCsys;
132
133 if (args.optionFound("from") || args.optionFound("to"))
134 {
135 const coordinateSystems::coordinateSystems& csLst
136 (
137 coordinateSystems::coordinateSystems::New(runTime)
138 );
139
140 if (args.optionFound("from"))
141 {
142 const word csName = args["from"];
143 fromCsys = csLst[csName].clone();
144 }
145
146 if (args.optionFound("to"))
147 {
148 const word csName = args["to"];
149 toCsys = csLst[csName].clone();
150 }
151 }
152
153
154 surfMesh smesh
155 (
156 IOobject
157 (
158 importName,
159 runTime.constant(),
160 runTime,
161 IOobject::MUST_READ_IF_MODIFIED,
162 IOobject::NO_WRITE
163 )
164 );
165
166 Info<< "read surfMesh:\n " << smesh.relativeObjectPath() << endl;
167
168
169 // Simply copy for now, but really should have a separate write method
170
171 MeshedSurface<face> surf(smesh);
172
173 if (args.optionFound("clean"))
174 {
175 surf.cleanup(true);
176 }
177
178 scalar scaleIn = 0;
179 if (args.optionReadIfPresent("scaleIn", scaleIn) && scaleIn > 0)
180 {
181 Info<< " -scaleIn " << scaleIn << endl;
182 surf.scalePoints(scaleIn);
183 }
184
185 if (fromCsys.valid())
186 {
187 Info<< " -from " << fromCsys().name() << endl;
188 tmp<pointField> tpf = fromCsys().localPosition(surf.points());
189 surf.setPoints(tpf());
190 }
191
192 if (toCsys.valid())
193 {
194 Info<< " -to " << toCsys().name() << endl;
195 tmp<pointField> tpf = toCsys().globalPosition(surf.points());
196 surf.setPoints(tpf());
197 }
198
199 scalar scaleOut = 0;
200 if (args.optionReadIfPresent("scaleOut", scaleOut) && scaleOut > 0)
201 {
202 Info<< " -scaleOut " << scaleOut << endl;
203 surf.scalePoints(scaleOut);
204 }
205
206
207 surf.writeStats(Info);
208 Info<< endl;
209
210 Info<< "writing " << exportName << endl;
211 surf.write(exportName);
212
213 Info<< "\nEnd\n" << endl;
214
215 return 0;
216}
