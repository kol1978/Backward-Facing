69Foam::volScalarField& Foam::functionObjects::phaseScalarTransport::Phi()
70{
71 if (!PhiPtr_.valid())
72 {
73 const surfaceScalarField& phi =
74 mesh_.lookupObject<surfaceScalarField>(phiName_);
75 const volScalarField& p =
76 mesh_.lookupObject<volScalarField>(pName_);
77
78 wordList PhiPatchFieldTypes(mesh_.poly().boundary().size());
79 forAll(p.boundaryField(), patchi)
80 {
81 PhiPatchFieldTypes[patchi] =
82 p.boundaryField()[patchi].fixesValue()
83 ? fixedValueFvPatchField<scalar>::typeName
84 : zeroGradientFvPatchField<scalar>::typeName;
85 }
86
87 PhiPtr_.set
88 (
89 new volScalarField
90 (
91 IOobject
92 (
93 typedName(IOobject::groupName("Phi", phaseName_)),
94 time_.name(),
95 mesh_,
96 IOobject::READ_IF_PRESENT,
97 IOobject::NO_WRITE
98 ),
99 mesh_,
100 dimensionedScalar(phi.dimensions()/dimLength, Zero),
101 PhiPatchFieldTypes
102 )
103 );
104
105 mesh_.schemes().setFluxRequired(PhiPtr_->name());
106 }
107
108 return PhiPtr_();
109}

112Foam::tmp<Foam::surfaceScalarField>
113Foam::functionObjects::phaseScalarTransport::alphaPhi()
114{
115 if (!solveAlphaPhi_)
116 {
117 return mesh_.lookupObject<surfaceScalarField>(alphaPhiName_);
118 }
119
120 const volScalarField& alpha =
121 mesh_.lookupObject<volScalarField>(alphaName_);
122 const surfaceScalarField& phi =
123 mesh_.lookupObject<surfaceScalarField>(phiName_);
124
125 // Make a crude guess of the phase flux using default interpolation
126 tmp<surfaceScalarField> tAlphaPhi
127 (
128 new surfaceScalarField
129 (
130 alphaPhiName_,
131 phi*fvc::interpolate(alpha)
132 )
133 );
134 surfaceScalarField& alphaPhi = tAlphaPhi.ref();
135
136 // Get the potential field
137 volScalarField& Phi(this->Phi());
138
139 // Construct the Phi Laplacian scheme names
140 const word PhiLaplacianScheme = "laplacian(" + pName_ + ")";
141
142 // Debug writing. Write the material derivative of alpha, before and after
143 // the solution of the potential and the correction of alphaPhi. Before
144 // correction the field should be non-zero, and after it should be
145 // comparable to the solution tolerance.
146 auto writeDDt = [&](const label i)
147 {
148 const volScalarField DDtAlpha
149 (
150 "DDt("
151 + IOobject::groupName
152 (
153 IOobject::member(alpha.name()) + Foam::name(i),
154 IOobject::group(alpha.name())
155 )
156 + ")",
157 fvc::ddt(alpha) + fvc::div(alphaPhi)
158 );
159 Info<< type() << ": Writing " << DDtAlpha.name() << endl;
160 DDtAlpha.write();
161 };
162 if (debug && time_.writeTime())
163 {
164 writeDDt(0);
165 }
166
167 // Lookup the non-orthogonal solution control
168 nonOrthogonalSolutionControl& control =
169 mesh_.lookupObjectRef<nonOrthogonalSolutionControl>
170 (
171 solutionControl::typeName
172 );
173
174 // Solve for the potential and correct alphaPhi with the resulting flux
175 if (phi.dimensions() == dimVolume/dimTime)
176 {
177 while (control.correctNonOrthogonal())
178 {
179 fvScalarMatrix PhiEqn
180 (
181 fvm::laplacian(Phi, PhiLaplacianScheme)
182 + fvc::ddt(alpha)
183 + fvc::div(alphaPhi)
184 );
185
186 PhiEqn.solve(pName_);
187
188 if (control.finalNonOrthogonalIter())
189 {
190 alphaPhi += PhiEqn.flux();
191 }
192 }
193 }
194 else if (phi.dimensions() == dimMass/dimTime)
195 {
196 const volScalarField& rho =
197 mesh_.lookupObject<volScalarField>(rhoName_);
198
199 while (control.correctNonOrthogonal())
200 {
201 fvScalarMatrix PhiEqn
202 (
203 fvm::laplacian(Phi, PhiLaplacianScheme)
204 + fvc::ddt(rho, alpha)
205 + fvc::div(alphaPhi)
206 );
207
208 PhiEqn.solve(pName_);
209
210 if (control.finalNonOrthogonalIter())
211 {
212 alphaPhi += PhiEqn.flux();
213 }
214 }
215 }
216 else
217 {
218 PhiDimensionErrorInFunction(phi);
219 }
220
221 // Debug writing
222 if (debug && time_.writeTime())
223 {
224 writeDDt(1);
225 }
226
227 return tAlphaPhi;
228}

231Foam::tmp<Foam::volScalarField>
232Foam::functionObjects::phaseScalarTransport::D
233(
234 const surfaceScalarField& alphaPhi
235) const
236{
237 const word Dname("D" + s_.name());
238
239 if (diffusivity_ == scalarTransport::diffusivityType::constant)
240 {
241 return volScalarField::New
242 (
243 Dname,
244 mesh_,
245 dimensionedScalar(dimKinematicViscosity, D_)
246 );
247 }
248 else
249 {
250 const word& nameNoPhase = momentumTransportModel::typeName;
251 const word namePhase = IOobject::groupName(nameNoPhase, phaseName_);
252
253 // Try looking up the phase transport model, then try the mixture
254 // transport model, then fail with an error relating to the phase
255 // transport model
256 const momentumTransportModel& turbulence =
257 mesh_.foundObject<momentumTransportModel>(namePhase)
258 ? mesh_.lookupObject<momentumTransportModel>(namePhase)
259 : mesh_.foundObject<momentumTransportModel>(nameNoPhase)
260 ? mesh_.lookupObject<momentumTransportModel>(nameNoPhase)
261 : mesh_.lookupObject<momentumTransportModel>(namePhase);
262
263 return volScalarField::New
264 (
265 Dname,
266 alphal_*turbulence.nu() + alphat_*turbulence.nut()
267 );
268 }
269}

319bool Foam::functionObjects::phaseScalarTransport::read(const dictionary& dict)
320{
321 fvMeshFunctionObject::read(dict);
322
323 solveAlphaPhi_ = dict.lookupOrDefault<bool>("solveAlphaPhi", false);
324
325 alphaName_ =
326 dict.lookupOrDefault<word>
327 (
328 "alpha",
329 IOobject::groupName("alpha", phaseName_)
330 );
331 const word defaultAlphaPhiName =
332 IOobject::groupName("alphaPhi", phaseName_);
333 alphaPhiName_ =
334 solveAlphaPhi_
335 ? typedName(defaultAlphaPhiName)
336 : dict.lookupOrDefault<word>("alphaPhi", defaultAlphaPhiName);
337 phiName_ = dict.lookupOrDefault<word>("phi", "phi");
338 rhoName_ =
339 dict.lookupOrDefault<word>
340 (
341 "rho",
342 IOobject::groupName("rho", phaseName_)
343 );
344 pName_ = dict.lookupOrDefault<word>("p", "p");
345
346 schemesField_ = dict.lookupOrDefault<word>("schemesField", fieldName_);
347 solverField_ = dict.lookupOrDefault<word>("solverField", fieldName_);
348
349 diffusivity_ =
350 scalarTransport::diffusivityTypeNames_.read(dict.lookup("diffusivity"));
351
352 switch(diffusivity_)
353 {
354 case scalarTransport::diffusivityType::none:
355 break;
356
357 case scalarTransport::diffusivityType::constant:
358 dict.lookup("D") >> D_;
359 break;
360
361 case scalarTransport::diffusivityType::viscosity:
362 alphal_ =
363 dict.lookupBackwardsCompatible<scalar>({"alphal", "alphaD"});
364 alphat_ =
365 dict.lookupBackwardsCompatible<scalar>({"alphat", "alphaDt"});
366 break;
367 }
368
369 residualAlpha_ = dict.lookupOrDefault<scalar>("residualAlpha", rootSmall);
370 writeAlphaField_ = dict.lookupOrDefault<bool>("writeAlphaField", true);
371
372 return true;
373}

382bool Foam::functionObjects::phaseScalarTransport::execute()
383{
384 Info<< type() << ": Executing" << endl;
385
386 const volScalarField& alpha =
387 mesh_.lookupObject<volScalarField>(alphaName_);
388
389 // Get the phase flux
390 tmp<surfaceScalarField> tAlphaPhi(this->alphaPhi());
391 const surfaceScalarField& alphaPhi = tAlphaPhi();
392
393 const int nCorr =
394 mesh_.solution().solverDict(solverField_)
395 .lookupOrDefaultBackwardsCompatible<label>
396 (
397 {"nCorrectors", "nCorr"},
398 0
399 );
400
401 // Get the relaxation coefficient
402 const scalar relaxCoeff =
403 mesh_.solution().relaxEquation(solverField_)
404 ? mesh_.solution().equationRelaxationFactor(solverField_)
405 : 0;
406
407 // Models and constraints
408 const Foam::fvModels& fvModels = Foam::fvModels::New(mesh_);
409 const Foam::fvConstraints& fvConstraints = Foam::fvConstraints::New(mesh_);
410
411 // Solve
412 if (alphaPhi.dimensions() == dimVolume/dimTime)
413 {
414 for (int i=0; i<=nCorr; i++)
415 {
416 fvScalarMatrix fieldEqn
417 (
418 fvm::ddt(alpha, s_)
419 + fvm::div
420 (
421 alphaPhi,
422 s_,
423 "div(" + alphaPhi.name() + "," + schemesField_ + ")"
424 )
425 ==
426 fvModels.source(alpha, s_)
427 - fvm::ddt(residualAlpha_, s_)
428 + fvc::ddt(residualAlpha_, s_)
429 );
430
431 if (diffusivity_ != scalarTransport::diffusivityType::none)
432 {
433 const volScalarField D(this->D(alphaPhi));
434
435 fieldEqn -=
436 fvm::laplacian
437 (
438 fvc::interpolate(alpha)*fvc::interpolate(D),
439 s_,
440 "laplacian("
441 + alphaPhi.name() + ","
442 + D.name() + ","
443 + schemesField_ + ")"
444 );
445 }
446
447 fieldEqn.relax(relaxCoeff);
448 fvConstraints.constrain(fieldEqn);
449 fieldEqn.solve(solverField_);
450 fvConstraints.constrain(s_);
451 }
452 }
453 else if (alphaPhi.dimensions() == dimMass/dimTime)
454 {
455 const volScalarField& rho =
456 mesh_.lookupObject<volScalarField>(rhoName_);
457
458 for (int i=0; i<=nCorr; i++)
459 {
460 fvScalarMatrix fieldEqn
461 (
462 fvm::ddt(alpha, rho, s_)
463 + fvm::div
464 (
465 alphaPhi,
466 s_,
467 "div(" + alphaPhi.name() + "," + schemesField_ + ")"
468 )
469 ==
470 fvModels.source(alpha, rho, s_)
471 - fvm::ddt(residualAlpha_*rho, s_)
472 + fvc::ddt(residualAlpha_*rho, s_)
473 );
474
475 if (diffusivity_ != scalarTransport::diffusivityType::none)
476 {
477 const volScalarField D(this->D(alphaPhi));
478
479 fieldEqn -=
480 fvm::laplacian
481 (
482 fvc::interpolate(alpha)*fvc::interpolate(rho*D),
483 s_,
484 "laplacian("
485 + alphaPhi.name() + ","
486 + D.name() + ","
487 + schemesField_ + ")"
488 );
489 }
490
491 fieldEqn.relax(relaxCoeff);
492 fvConstraints.constrain(fieldEqn);
493 fieldEqn.solve(solverField_);
494 fvConstraints.constrain(s_);
495 }
496 }
497 else
498 {
499 PhiDimensionErrorInFunction(alphaPhi);
500 }
501
502 // Update the alpha*S field
503 if (writeAlphaField_)
504 {
505 if (!alphaSPtr_.valid())
506 {
507 alphaSPtr_.set
508 (
509 new volScalarField
510 (
511 IOobject
512 (
513 "alpha"
514 + word(toupper(fieldName_[0]))
515 + fieldName_(1, fieldName_.size() - 1),
516 time_.name(),
517 mesh_,
518 IOobject::NO_READ,
519 IOobject::NO_WRITE
520 ),
521 mesh_,
522 dimensionedScalar(s_.dimensions(), Zero)
523 )
524 );
525 }
526
527 alphaSPtr_() = alpha*s_;
528 }
529 else
530 {
531 if (alphaSPtr_.valid())
532 {
533 alphaSPtr_().clear();
534 }
535 }
536
537 Info<< endl;
538
539 return true;
540}
