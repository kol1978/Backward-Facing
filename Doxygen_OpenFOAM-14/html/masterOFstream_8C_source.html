35void Foam::masterOFstream::checkWrite
36(
37 const fileName& fName,
38 const string& str
39)
40{
41 mkDir(fName.path());
42
43 OFstream os
44 (
45 fName,
46 IOstream::BINARY,
47 version(),
48 compression_,
49 append_
50 );
51
52 if (!os.good())
53 {
54 FatalIOErrorInFunction(os)
55 << "Could not open file " << fName
56 << exit(FatalIOError);
57 }
58
59 os.writeQuoted(str, false);
60 if (!os.good())
61 {
62 FatalIOErrorInFunction(os)
63 << "Failed writing to " << fName
64 << exit(FatalIOError);
65 }
66}

91Foam::masterOFstream::~masterOFstream()
92{
93 if (Pstream::parRun())
94 {
95 List<fileName> filePaths(Pstream::nProcs());
96 filePaths[Pstream::myProcNo()] = filePath_;
97 Pstream::gatherList(filePaths);
98
99 bool uniform =
100 fileOperations::masterUncollatedFileOperation::uniformFile
101 (
102 filePaths
103 );
104
105 Pstream::scatter(uniform);
106
107 if (uniform)
108 {
109 if (Pstream::master() && write_)
110 {
111 checkWrite(filePath_, str());
112 }
113 return;
114 }
115 boolList write(Pstream::nProcs());
116 write[Pstream::myProcNo()] = write_;
117 Pstream::gatherList(write);
118
119
120 // Different files
121 PstreamBuffers pBufs(Pstream::commsTypes::nonBlocking);
122
123 // Send my buffer to master
124 if (!Pstream::master())
125 {
126 UOPstream os(Pstream::masterNo(), pBufs);
127 string s(this->str());
128 os.write(&s[0], s.size());
129 }
130
131 labelList recvSizes;
132 pBufs.finishedSends(recvSizes);
133
134 if (Pstream::master())
135 {
136 // Write my own data
137 {
138 if (write[Pstream::myProcNo()])
139 {
140 checkWrite(filePaths[Pstream::myProcNo()], str());
141 }
142 }
143
144 for (label proci = 1; proci < Pstream::nProcs(); proci++)
145 {
146 UIPstream is(proci, pBufs);
147 List<char> buf(recvSizes[proci]);
148
149 is.read(buf.begin(), buf.size());
150
151 if (write[proci])
152 {
153 checkWrite
154 (
155 filePaths[proci],
156 string(buf.begin(), buf.size())
157 );
158 }
159 }
160 }
161 }
162 else
163 {
164 checkWrite(filePath_, str());
165 }
166}
