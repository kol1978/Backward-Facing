52void Foam::fv::heatSource::readCoeffs(const dictionary& dict)
53{
54 if (!dict.found("q") && !dict.found("Q"))
55 {
56 FatalIOErrorInFunction(coeffs(dict))
57 << "Neither heat source per unit volume, q, nor total heat source, "
58 << "Q, has been specified. One is required." << exit(FatalIOError);
59 }
60
61 if (dict.found("q") && dict.found("Q"))
62 {
63 FatalIOErrorInFunction(coeffs(dict))
64 << "Both heat source per unit volume, q, and total heat source, "
65 << "Q, have been specified. One is required."
66 << exit(FatalIOError);
67 }
68
69 zone_.read(coeffs(dict));
70
71 if (dict.found("q"))
72 {
73 q_.reset
74 (
75 Function1<scalar>::New
76 (
77 "q",
78 mesh().time().userUnits(),
79 dimPower/dimVolume,
80 dict
81 ).ptr()
82 );
83 }
84 else
85 {
86 q_.reset
87 (
88 new Function1s::Scale<scalar>
89 (
90 "q",
91 Function1s::Constant<scalar>("1/V", 1/zone_.V()),
92 Function1s::Constant<scalar>("1", 1),
93 Function1<scalar>::New
94 (
95 "Q",
96 mesh().time().userUnits(),
97 dimPower,
98 dict
99 )()
100 )
101 );
102 }
103}

141void Foam::fv::heatSource::addSup
142(
143 const volScalarField& he,
144 fvMatrix<scalar>& eqn
145) const
146{
147 zone_.regenerate();
148 const labelList& cells = zone_.zone();
149
150 const scalar q = q_->value(mesh().time().value());
151
152 scalarField& eqnSource = eqn.source();
153 forAll(cells, i)
154 {
155 eqnSource[cells[i]] -= mesh().V()[cells[i]]*q;
156 }
157}
