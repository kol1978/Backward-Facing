34void Foam::lagrangianWriter::writeIOField(const wordList& objects)
35{
36 forAll(objects, i)
37 {
38 const word& object = objects[i];
39
40 IOobject header
41 (
42 object,
43 vMesh_.mesh().time().name(),
44 lagrangian::cloud::prefix/cloudName_,
45 vMesh_.mesh(),
46 IOobject::MUST_READ,
47 IOobject::NO_WRITE,
48 false
49 );
50
51 IOField<Type> fld(header);
52
53 os_ << object << ' ' << pTraits<Type>::nComponents << ' '
54 << fld.size() << " float" << std::endl;
55
56 DynamicList<floatScalar> fField(pTraits<Type>::nComponents*fld.size());
57
58 vtkWriteOps::insert(fld, fField);
59
