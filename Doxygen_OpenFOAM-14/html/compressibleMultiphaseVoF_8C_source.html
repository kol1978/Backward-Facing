122void Foam::solvers::compressibleMultiphaseVoF::prePredictor()
123{
124 multiphaseVoFSolver::prePredictor();
125
126 contErr = fvc::ddt(rho)()() + fvc::div(rhoPhi)()();
127
128 forAll(mixture.phases(), phasei)
129 {
130 const volScalarField& rho = phases[phasei].thermo().rho();
131 contErr.ref() -= (fvModels().source(phases[phasei], rho)&rho)();
132 }
133}
