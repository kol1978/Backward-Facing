43void Foam::solvers::multiphaseVoFSolver::correctCoNum()
44{
45 VoFSolver::correctCoNum();
46
47 const scalarField sumPhi
48 (
49 mixture.nearInterface()().primitiveField()
50 *fvc::surfaceSum(mag(phi))().primitiveField()
51 );
52
53 alphaCoNum =
54 0.5*gMax(sumPhi/mesh.V().primitiveField())*runTime.deltaTValue();
55
56 const scalar meanAlphaCoNum =
57 0.5
58 *(gSum(sumPhi)/gSum(mesh.V().primitiveField()))
59 *runTime.deltaTValue();
60
61 Info<< "Interface Courant Number mean: " << meanAlphaCoNum
62 << " max: " << alphaCoNum << endl;
63}

68bool Foam::solvers::multiphaseVoFSolver::read()
69{
70 VoFSolver::read();
71
72 const dictionary& alphaControls = mesh.solution().solverDict("alpha");
73
74 cAlpha = alphaControls.lookup<scalar>("cAlpha");
75
76 nAlphaSubCyclesPtr =
77 Function1<scalar>::New
78 (
79 alphaControls.found("nAlphaSubCycles")
80 ? "nAlphaSubCycles"
81 : "nSubCycles",
82 dimless,
83 dimless,
84 alphaControls
85 );
86
87 MULEScontrols.read(alphaControls);
88
89 return true;
90}
