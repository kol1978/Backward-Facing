31void Foam::solvers::compressibleVoF::alphaSuSp
32(
33 tmp<volScalarField::Internal>& tSu,
34 tmp<volScalarField::Internal>& tSp
35)
36{
37 const dimensionedScalar Szero(vDot.dimensions(), 0);
38
39 tSp = volScalarField::Internal::New("Sp", mesh, Szero);
40 tSu = volScalarField::Internal::New("Su", mesh, Szero);
41
42 volScalarField::Internal& Sp = tSp.ref();
43 volScalarField::Internal& Su = tSu.ref();
44
45 if (fvModels().addsSupToField(mixture.rho1().name()))
46 {
47 const volScalarField::Internal alpha2ByRho1(alpha2()/mixture.rho1()());
48 const fvScalarMatrix alphaRho1Sup
49 (
50 fvModels().sourceProxy(alpha1, mixture.rho1(), alpha1)
51 );
52
53 Su += alpha2ByRho1*alphaRho1Sup.Su();
54 Sp += alpha2ByRho1*alphaRho1Sup.Sp();
55 }
56
57 if (fvModels().addsSupToField(mixture.rho2().name()))
58 {
59 const volScalarField::Internal alpha1ByRho2(alpha1()/mixture.rho2()());
60 const fvScalarMatrix alphaRho2Sup
61 (
62 fvModels().sourceProxy(alpha2, mixture.rho2(), alpha2)
63 );
64
65 Su -= alpha1ByRho2*(alphaRho2Sup.Su() + alphaRho2Sup.Sp());
66 Sp += alpha1ByRho2*alphaRho2Sup.Sp();
67 }
68
69 forAll(vDot, celli)
70 {
71 if (vDot[celli] > 0.0)
72 {
73 Sp[celli] -=
74 vDot[celli]/max(1.0 - alpha1[celli], vDotResidualAlpha);
75 Su[celli] +=
76 vDot[celli]/max(1.0 - alpha1[celli], vDotResidualAlpha);
77 }
78 else if (vDot[celli] < 0.0)
79 {
80 Sp[celli] += vDot[celli]/max(alpha1[celli], vDotResidualAlpha);
81 }
82 }
83}
