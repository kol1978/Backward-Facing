154inline specie operator==(const specie& st1, const specie& st2)
155{

159 diffY = small;

164
165 scalar molWeight = great;
166 if (mag(diffRW) > small)
167 {
168 molWeight = diffY/diffRW;
169 }
170
171 return specie(diffY, molWeight);
