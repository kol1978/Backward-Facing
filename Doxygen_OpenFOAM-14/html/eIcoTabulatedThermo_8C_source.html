62template<class EquationOfState>
63void Foam::eIcoTabulatedThermo<EquationOfState>::write
64(
65 Ostream& os
66) const
67{
68 EquationOfState::write(os);
69
70 dictionary dict("thermodynamics");
71 dict.add("hf", hf_);
72 dict.add("sf", sf_);
73 dict.add("Cv", Cv_.values());
74 os << indent << dict.dictName() << dict;
75}
