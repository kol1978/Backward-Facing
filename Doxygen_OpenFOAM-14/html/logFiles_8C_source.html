32void Foam::functionObjects::logFiles::createFiles()
33{
34 if (Pstream::master())
35 {
36 const word timeName = fileObr_.time().name();
37
38 forAll(names_, i)
39 {
40 if (!filePtrs_.set(i))
41 {
42 const fileName outputDir(baseFileDir()/prefix_/timeName);
43 mkDir(outputDir);
44 filePtrs_.set(i, new OFstream(outputDir/(names_[i] + ".dat")));
45 initStream(filePtrs_[i]);
46 writeFileHeader(i);
47 }
48 }
49 }
50}

113Foam::OFstream& Foam::functionObjects::logFiles::file()
114{
115 if (!Pstream::master())
116 {
117 FatalErrorInFunction
118 << "Request for file() can only be done by the master process"
119 << abort(FatalError);
120 }
121
122 if (filePtrs_.size() != 1)
123 {
124 WarningInFunction
125 << "Requested single file, but multiple files are present"
126 << endl;
127 }
128
129 if (!filePtrs_.set(0))
130 {
131 FatalErrorInFunction
132 << "File pointer at index " << 0 << " not allocated"
133 << abort(FatalError);
134 }
135
136 return filePtrs_[0];
137}
