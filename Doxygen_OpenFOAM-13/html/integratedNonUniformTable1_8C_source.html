43Foam::Function1s::integratedNonUniformTable::integratedNonUniformTable
44(
45 const word& name,
46 const unitConversions& units,
47 const dictionary& dict
48)
49:
50 NonUniformTable<scalar>(name, units, dict),
51 intf_(values().size()),
52 intfByT_(values().size())
53{
54 assertNoConvertUnits(typeName, units, dict);
55
56 intf_[0] = 0;
57 intfByT_[0] = 0;
58
59 for(label i = 0; i<intf_.size() - 1; i++)
60 {
61 intf_[i + 1] = intfdT(values()[i + 1].first());
62 intfByT_[i + 1] = intfByTdT(values()[i + 1].first());
63 }
64
65 const scalar intfStd = intfdT(Tstd);
66 const scalar intfByTStd = intfByTdT(Tstd);
67
68 forAll(intf_, i)
69 {
70 intf_[i] -= intfStd;
71 intfByT_[i] -= intfByTStd;
72 }
73}

78Foam::scalar Foam::Function1s::integratedNonUniformTable::intfdT
79(
80 scalar T
81) const
82{
83 const label i = index(T);
84 const scalar Ti = values()[i].first();
85 const scalar fi = values()[i].second();
86 const scalar dT = T - Ti;
87 const scalar lambda = dT/(values()[i + 1].first() - Ti);
88
89 return
90 intf_[i]
91 + (fi + 0.5*lambda*(values()[i + 1].second() - fi))*dT;
92}

95Foam::scalar Foam::Function1s::integratedNonUniformTable::intfByTdT
96(
97 scalar T
98) const
99{
100 const label i = index(T);
101 const scalar Ti = values()[i].first();
102 const scalar fi = values()[i].second();
103 const scalar gradf =
104 (values()[i + 1].second() - fi)/(values()[i + 1].first() - Ti);
105
106 return
107 intfByT_[i] + ((fi - gradf*Ti)*log(T/Ti) + gradf*(T - Ti));
108}
