62void Foam::solidParticle::readFields(lagrangian::Cloud<solidParticle>& c)
63{
64 bool valid = c.size();
65
66 particle::readFields(c);
67
68 IOField<scalar> d(c.fieldIOobject("d", IOobject::MUST_READ), valid);
69 c.checkFieldIOobject(c, d);
70
71 IOField<vector> U(c.fieldIOobject("U", IOobject::MUST_READ), valid);
72 c.checkFieldIOobject(c, U);
73
74 label i = 0;
75 forAllIter(lagrangian::Cloud<solidParticle>, c, iter)
76 {
77 solidParticle& p = iter();
78
79 p.d_ = d[i];
80 p.U_ = U[i];
81 i++;
82 }
83}

86void Foam::solidParticle::writeFields(const lagrangian::Cloud<solidParticle>& c)
87{
88 particle::writeFields(c);
89
90 label np = c.size();
91
92 IOField<scalar> d(c.fieldIOobject("d", IOobject::NO_READ), np);
93 IOField<vector> U(c.fieldIOobject("U", IOobject::NO_READ), np);
94
95 label i = 0;
96 forAllConstIter(lagrangian::Cloud<solidParticle>, c, iter)
97 {
98 const solidParticle& p = iter();
99
100 d[i] = p.d_;
101 U[i] = p.U_;
102 i++;
103 }
104
105 d.write(np > 0);
106 U.write(np > 0);
107}

112Foam::Ostream& Foam::operator<<(Ostream& os, const solidParticle& p)
113{
114 if (os.format() == IOstream::ASCII)

126 solidParticle::sizeofFields_

129
130 // Check state of Ostream
131 os.check("Ostream& operator<<(Ostream&, const solidParticle&)");
132
133 return os;
134}
