84void writeDecomposition(const domainDecomposition& meshes)
85{
86 // Write as volScalarField::Internal for postprocessing.
87 volScalarField::Internal cellProc
88 (
89 IOobject
90 (
91 "cellProc",
92 meshes.completeMesh().time().name(),
93 meshes.completeMesh(),
94 IOobject::NO_READ,
95 IOobject::AUTO_WRITE
96 ),
97 meshes.completeMesh(),
98 dimless,
99 scalarField(scalarList(meshes.cellProc()))
100 );
101
102 cellProc.write();
103
104 Info<< "Wrote decomposition as volScalarField::Internal to "
105 << cellProc.name() << " for use in postprocessing"
106 << endl;
107}
108
109}
