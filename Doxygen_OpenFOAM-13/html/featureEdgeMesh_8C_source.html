38Foam::featureEdgeMesh::featureEdgeMesh(const IOobject& io)
39:
40 regIOobject(io),
41 edgeMesh(pointField(0), edgeList(0))
42{
43 if
44 (
45 io.readOpt() == IOobject::MUST_READ
46 || io.readOpt() == IOobject::MUST_READ_IF_MODIFIED
47 || (io.readOpt() == IOobject::READ_IF_PRESENT && headerOk())
48 )
49 {
50 readStream(typeName) >> *this;
51 close();
52 }
53
54 if (debug)
55 {
56 Pout<< "featureEdgeMesh::featureEdgeMesh :"
57 << " constructed from IOobject :"
58 << " points:" << points().size()
59 << " edges:" << edges().size()
60 << endl;
61 }
62}
