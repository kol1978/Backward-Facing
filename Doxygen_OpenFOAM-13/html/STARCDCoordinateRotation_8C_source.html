52void Foam::STARCDCoordinateRotation::calcTransform
53(
54 const scalar z,
55 const scalar x,
56 const scalar y
57)
58{
59 R_ =
60 (
61 tensor
62 (
63 cos(y)*cos(z) - sin(x)*sin(y)*sin(z),
64 -cos(x)*sin(z),
65 sin(x)*cos(y)*sin(z) + sin(y)*cos(z),
66
67 cos(y)*sin(z) + sin(x)*sin(y)*cos(z),
68 cos(x)*cos(z),
69 sin(y)*sin(z) - sin(x)*cos(y)*cos(z),
70
71 -cos(x)*sin(y),
72 sin(x),
73 cos(x)*cos(y)
74 )
75 );
76
77 Rtr_ = R_.T();
78}
