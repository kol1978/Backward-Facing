58int main(int argc, char *argv[])
59{
60 argList::noParallel();
61 argList::addOption
62 (
63 "file",
64 "name",
65 "specify alternative input file name - default is otape17"
66 );
67 argList::addOption
68 (
69 "version",
70 "version",
71 "specify kiva version [kiva3|kiva3v] - default is '3v'"
72 );
73 argList::addOption
74 (
75 "zHeadMin",
76 "scalar",
77 "minimum z-height for transferring liner faces to cylinder-head"
78 );
79
80 #include "setRootCase.H"
81 #include "createTimeNoFunctionObjects.H"
82
83 const fileName kivaFileName =
84 args.optionLookupOrDefault<fileName>("file", "otape17");
85
86 kivaVersions kivaVersion = kiva3v;
87 if (args.optionFound("version"))
88 {
89 const word versionName = args["version"];
90
91 if (versionName == "kiva3")
92 {
93 kivaVersion = kiva3;
94 }
95 else if (versionName == "kiva3v")
96 {
97 kivaVersion = kiva3v;
98 }
99 else
100 {
101 args.printUsage();
102
103 FatalErrorInFunction
104 << "KIVA file version " << versionName << " not supported"
105 << exit(FatalError);
106 }
107 }
108
109 scalar zHeadMin = -great;
110 args.optionReadIfPresent("zHeadMin", zHeadMin);
111
112 #include "readKivaGrid.H"
113
114 Info<< "End\n" << endl;
115
116 return 0;
117}
