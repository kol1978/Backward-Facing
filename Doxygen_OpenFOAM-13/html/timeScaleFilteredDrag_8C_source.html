76Foam::tmp<Foam::volScalarField>
77Foam::dragModels::timeScaleFilteredDrag::CdRe() const
78{
79 const volScalarField limit
80 (
81 sqr(interface_.dispersed().d())
82 *interface_.dispersed().rho()
83 /0.75
84 /swarmCorrection_->Cs()
85 /interface_.continuous().rho()
86 /interface_.continuous().fluidThermo().nu()
87 /minRelaxTime_
88 );
89
90 return min(refCast<const dispersedDragModel>(dragModel_()).CdRe(), limit);
91}
