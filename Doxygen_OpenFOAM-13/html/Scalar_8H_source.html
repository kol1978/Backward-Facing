350inline Scalar sqrtSumSqr(const Scalar a, const Scalar b)
351{
352 Scalar maga = mag(a);
353 Scalar magb = mag(b);
354
355 if (maga > magb)
356 {
357 return maga*sqrt(1 + sqr(magb/maga));
358 }
359 else
360 {
361 return magb < ScalarVSmall ? 0 : magb*sqrt(1 + sqr(maga/magb));
362 }
363}

366//- Stabilisation around zero for division
367inline Scalar stabilise(const Scalar s, const Scalar small)
368{
369 if (s >= 0)
370 {
371 return s + small;
372 }
373 else
374 {
375 return s - small;
376 }
377}

380//- Divide two numbers and protect the result from overflowing
381inline Scalar protectedDivide(const Scalar a, const Scalar b)
382{
383 return mag(a)/ScalarVGreat < mag(b) ? a/b : sign(a)*sign(b)*ScalarVGreat;
384}
385
386
387//- Divide two numbers, protect the result from overflowing, and clip the
388// result between 0 and 1
389inline Scalar protectedDivideAndClip01(const Scalar a, const Scalar b)
390{
391 const Scalar bStar = max(mag(a), mag(b));
392
393 return bStar == 0 ? 0 : max(a*sign(b), 0)/bStar;
394}
