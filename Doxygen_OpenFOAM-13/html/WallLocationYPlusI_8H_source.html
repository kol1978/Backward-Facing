31template<class TrackingYPlus>
32inline bool Foam::WallLocationYPlus<WallLocation>::update
33(
34 const point& pt,
35 const WallLocationYPlus<WallLocation>& w2,
36 const scalar tol,
37 TrackingYPlus& td
38)
39{
40 WallLocationYPlus<WallLocation> copy(*this);
41
42 bool result =
43 WallLocationData<WallLocation, scalar>::update(pt, w2, tol, td);
44
45 if (result && sqrt(this->distSqr())/w2.data() >= td.yPlusCutOff)
46 {
47 *this = copy;
48 result = false;
49 }
50
51 return result;
52}
