102Foam::fv::heatTransferCoefficientModels::function1::htc() const
103{
104 const volVectorField& U = mesh_.lookupObject<volVectorField>(UName_);
105
106 tmp<volScalarField> tHtc =

109 typedName("htc"),
110 mesh_,
111 dimPower/dimTemperature/dimArea,

113 );
114
