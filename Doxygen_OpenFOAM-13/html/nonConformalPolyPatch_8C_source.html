53void Foam::nonConformalPolyPatch::rename(const wordList& newNames)

152Foam::label Foam::nonConformalPolyPatch::origPatchIndex() const
153{
154 if (origPatchIndex_ == -1)
155 {
156 origPatchIndex_ = patch_.boundaryMesh().findIndex(origPatchName());
157
158 if (origPatchIndex_ == -1)
159 {
160 FatalErrorInFunction
161 << "Illegal neighbourPatch name " << origPatchName()
162 << endl << "Valid patch names are "
163 << patch_.boundaryMesh().names()
164 << exit(FatalError);
165 }
166
167 const polyPatch& p = patch_.boundaryMesh()[origPatchIndex_];
168
169 if (isA<nonConformalPolyPatch>(p))
170 {
171 FatalErrorInFunction
172 << "The originalPatch for the "
173 << patch_.type() << " patch " << patch_.name() << " is "
174 << p.name() << " which is also of "
175 << nonConformalPolyPatch::typeName << " type. This is not "
176 << "allowed. The originalPatch must be of a non-"
177 << nonConformalPolyPatch::typeName << " type."
178 << exit(FatalError);
179 }
180 }
181
182 return origPatchIndex_;
183}
