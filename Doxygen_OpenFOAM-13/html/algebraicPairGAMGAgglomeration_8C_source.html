47Foam::algebraicPairGAMGAgglomeration::algebraicPairGAMGAgglomeration
48(
49 const lduMatrix& matrix,
50 const dictionary& controlDict
51)
52:
53 pairGAMGAgglomeration(matrix.mesh(), controlDict)
54{
55 const lduMesh& mesh = matrix.mesh();
56
57 if (matrix.hasLower())
58 {
59 agglomerate(mesh, max(mag(matrix.upper()), mag(matrix.lower())));
