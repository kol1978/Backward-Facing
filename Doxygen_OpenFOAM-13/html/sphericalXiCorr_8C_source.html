74Foam::dimensionedScalar Foam::XiCorrModels::spherical::Ak
75(
76 const dimensionedScalar& Vk
77) const
78{
79 // Radius of the ignition kernel
80 dimensionedScalar rk
81 (
82 pow
83 (
84 (3.0/4.0)*Vk
85 /(sphereFraction_*constant::mathematical::pi),
86 1.0/3.0

88 );
