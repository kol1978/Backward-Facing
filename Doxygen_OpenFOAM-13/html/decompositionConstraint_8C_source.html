48Foam::autoPtr<Foam::decompositionConstraint>
49Foam::decompositionConstraint::New
50(
51 const dictionary& dict,
52 const word& modelType
53)
54{
55 Info<< "Selecting decompositionConstraint " << modelType << endl;
56
57 dictionaryConstructorTable::iterator cstrIter =
58 dictionaryConstructorTablePtr_->find(modelType);
59
60 if (cstrIter == dictionaryConstructorTablePtr_->end())
61 {
62 FatalIOErrorInFunction(dict)
63 << "Unknown decompositionConstraint type "
64 << modelType << nl << nl
65 << "Valid decompositionConstraint types:" << endl
66 << dictionaryConstructorTablePtr_->sortedToc()
67 << exit(FatalIOError);
68 }
69
70 return autoPtr<decompositionConstraint>
71 (
72 cstrIter()(dict, modelType)
73 );
74}
