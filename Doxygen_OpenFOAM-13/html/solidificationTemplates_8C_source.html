31template<class AlphaFieldType, class RhoFieldType>
32void Foam::porosityModels::solidification::apply
33(
34 scalarField& Udiag,
35 const scalarField& V,
36 const AlphaFieldType& alpha,
37 const RhoFieldType& rho,
38 const volVectorField& U
39) const
40{
41 const labelList& cells = mesh_.cellZones()[zoneName_];
42
43 const volScalarField& T = mesh_.lookupObject<volScalarField>
44 (
45 IOobject::groupName(TName_, U.group())
46 );
47
48 forAll(cells, i)
49 {
50 const label celli = cells[i];
51 Udiag[celli] +=
52 V[celli]*alpha[celli]*rho[celli]*D_->value(T[celli]);
