76Foam::distribution::distribution
77(
78 const word& name,
79 const unitConversion& units,

81 const label sampleQ,
82 randomGenerator&& rndGen
83)
84:
85 Q_(dict.lookup<scalar>("Q")),
86 sampleQ_(sampleQ),
87 rndGen_("rndGen", dict, std::move(rndGen))
88{
89 if (!dict.lookupOrDefault<bool>("allowNegativeQ", false) && Q_ < 0)
90 {
91 FatalIOErrorInFunction(dict)

95
96 if (sampleQ_ < 0)
97 {
98 FatalErrorInFunction
99 << name << ": Sampling size exponent cannot be negative" << nl
100 << " sampleQ = " << sampleQ_ << abort(FatalError);
101 }
102}

166Foam::tmp<Foam::scalarField> Foam::distribution::plotX(const label n) const
167{
168 const scalar x0 = min(), x1 = max(), d = 0.1*(x1 - x0);
169
170 tmp<scalarField> tResult(new scalarField(n));
171 scalarField& result = tResult.ref();
172
173 result[0] = x0 - d;
174 result[1] = x0*(1 - sign(x0)*small);
175 result[2] = x0*(1 + sign(x0)*small);
176
177 for (label i = 3; i < n - 3; ++ i)
178 {
179 const scalar f = scalar(i - 2)/(n - 5);
180
181 result[i] = (1 - f)*x0 + f*x1;
182 }
183
184 result[n - 3] = x1*(1 - sign(x1)*small);
185 result[n - 2] = x1*(1 + sign(x1)*small);
186 result[n - 1] = x1 + d;
187
188 return tResult;
189}

194void Foam::writeEntry
195(
196 Ostream& os,
197 const word& entryName,
198 const unitConversion& units,
199 const distribution& d,
200 const bool write,
201 const bool writeState
202)
203{
204 writeKeyword(os, entryName);
205
206 os << nl << indent << token::BEGIN_BLOCK << nl << incrIndent;
207
208 if (write) d.write(os, units);
209 if (writeState) d.writeState(os);
210
211 os << decrIndent << indent << token::END_BLOCK << endl;
212}
