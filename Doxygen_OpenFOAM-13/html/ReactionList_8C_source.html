109template<class ThermoType>
110void Foam::ReactionList<ThermoType>::write(Ostream& os) const
111{
112 os << "reactions" << nl;
113 os << token::BEGIN_BLOCK << incrIndent << nl;
114
115 forAll(*this, i)
116 {
117 const Reaction<ThermoType>& r = this->operator[](i);
118
119 os << indent << r.name() << nl
120 << indent << token::BEGIN_BLOCK << incrIndent << nl;
121
122 writeEntry(os, "type", r.type());
123
124 r.write(os);
125
126 os << decrIndent << indent << token::END_BLOCK << nl;
127 }
128
129 os << decrIndent << token::END_BLOCK << nl;
130
131 writeEntry(os, "Tlow", Reaction<ThermoType>::TlowDefault);
132 writeEntry(os, "Thigh", Reaction<ThermoType>::ThighDefault);
133
134 os << nl;
135}
