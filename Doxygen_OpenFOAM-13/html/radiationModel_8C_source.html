45Foam::IOobject Foam::radiationModel::createIOobject(const fvMesh& mesh) const
46{
47 typeIOobject<IOdictionary> io
48 (
49 "radiationProperties",
50 mesh.time().constant(),
51 mesh,
52 IOobject::MUST_READ,
53 IOobject::NO_WRITE
54 );
55
56 if (io.headerOk())
57 {
58 io.readOpt() = IOobject::MUST_READ_IF_MODIFIED;
59 return io;
60 }
61 else
62 {
63 io.readOpt() = IOobject::NO_READ;
64 return io;
65 }
66}

69void Foam::radiationModel::initialise()
70{
71 solverFreq_ = max(1, lookupOrDefault<label>("solverFreq", 1));
72
73 absorptionEmission_.reset
74 (

76 );
77

169void Foam::radiationModel::correct()
170{
171 if (firstIter_ || (time_.timeIndex() % solverFreq_ == 0))
172 {
173 calculate();
174 firstIter_ = false;
175 }
176
177 if (!soot_.empty())
178 {
179 soot_->correct();
180 }
181}

184bool Foam::radiationModel::read()
185{

187 {
188 coeffs_ = subOrEmptyDict(type() + "Coeffs");
189
190 solverFreq_ = lookupOrDefault<label>("solverFreq", 1);
191 solverFreq_ = max(1, solverFreq_);
192
193 return true;
194 }
195 else
196 {
197 return false;

202Foam::tmp<Foam::fvScalarMatrix> Foam::radiationModel::Sh
203(
204 const basicThermo& thermo,
205 const volScalarField& he
206) const
207{
208 const volScalarField Cpv(thermo.Cpv());
209 const volScalarField T3(pow3(T_));
210
211 return
212 (
213 Ru()
214 - fvm::Sp(4.0*Rp()*T3/Cpv, he)
215 - Rp()*T3*(T_ - 4.0*he/Cpv)
216 );
217}

220Foam::tmp<Foam::fvScalarMatrix> Foam::radiationModel::ST
221(
222 const dimensionedScalar& rhoCp,
223 volScalarField& T
224) const
225{
226 return
227 (
228 Ru()/rhoCp
229 - fvm::Sp(Rp()*pow3(T)/rhoCp, T)
230 );
231}
