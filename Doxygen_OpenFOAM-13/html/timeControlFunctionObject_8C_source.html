85bool Foam::functionObjects::timeControl::execute()
86{
87 if
88 (
89 executeControl_.active()
90 && (
91 postProcess
92 || executeControl_.execute()
93 || (executeAtStart() && time_.timeIndex() == time_.startTimeIndex())
94 )
95 )
96 {
97 foPtr_->execute();
98 }
99
100 return true;
101}

104bool Foam::functionObjects::timeControl::write()
105{
106 if
107 (
108 writeControl_.active()
109 && (
110 postProcess
111 || writeControl_.execute()
112 || (executeAtStart() && time_.timeIndex() == time_.startTimeIndex())
113 )
114 )
115 {
116 foPtr_->write();
117 }
118
119 return true;
120}

123bool Foam::functionObjects::timeControl::end()
