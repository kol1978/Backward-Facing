58Foam::fvMesh::readUpdateState Foam::vtkMesh::readUpdate()
59{
60 fvMesh::readUpdateState meshState = baseMesh_.readUpdate();
61
62 if (meshState != fvMesh::UNCHANGED)
63 {
64 // Note: since fvMeshSubset has no setPoints() functionality,
65 // reconstruct the subset even if only movement.
66
67 topoPtr_.clear();
68
69 if (setName_.size())
70 {
71 Info<< "Subsetting mesh based on cellSet " << setName_ << endl;
72
73 // Read cellSet using whole mesh
74 cellSet currentSet(baseMesh_, setName_);
75
76 subsetter_.setLargeCellSubset(currentSet);
77 }
78 }
79
80 return meshState;
81}
