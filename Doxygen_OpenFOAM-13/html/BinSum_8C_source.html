74template<class IndexType, class List, class CombineOp>
75void Foam::BinSum<IndexType, List, CombineOp>::add
76(
77 const IndexType& indexVal,
78 const typename List::const_reference val,
79 const CombineOp& cop
80)
81{
82 if (indexVal < min_)
83 {
84 cop(lowSum_, val);
85 }
86 else if (indexVal >= max_)
87 {
88 cop(highSum_, val);
89 }
90 else
91 {
92 label index = (indexVal-min_)/delta_;
93 cop(this->operator[](index), val);
94 }
95}
