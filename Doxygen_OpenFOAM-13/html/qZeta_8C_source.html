59void qZeta::boundZeta()
60{
61 tmp<volScalarField> tCmuk2(Cmu_*sqr(k_));
62 zeta_ = max(zeta_, Cmu_*pow3(q_)/((2*nutMaxCoeff_)*nu()));
63}

66tmp<volScalarField> qZeta::fMu() const
67{
68 const volScalarField Rt(q_*k_/(2.0*nu()*zeta_));
69
70 if (anisotropic_)
71 {
72 return exp((-scalar(2.5) + Rt/20.0)/pow3(scalar(1) + Rt/130.0));
73 }
74 else
75 {
76 return
77 exp(-6.0/sqr(scalar(1) + Rt/50.0))
78 *(scalar(1) + 3.0*exp(-Rt/10.0));
79 }
80}

83tmp<volScalarField> qZeta::f2() const
84{
85 tmp<volScalarField> Rt = q_*k_/(2.0*nu()*zeta_);
86 return scalar(1) - 0.3*exp(-sqr(Rt));
87}

209void qZeta::correct()
210{
211 if (!turbulence_)
212 {
213 return;
214 }
215
216 eddyViscosity<incompressible::RASModel>::correct();
217
218 volScalarField G(GName(), nut_/(2.0*q_)*2*magSqr(symm(fvc::grad(U_))));
219 const volScalarField E(nu()*nut_/q_*fvc::magSqrGradGrad(U_));
220
221 // Zeta equation
222 tmp<fvScalarMatrix> zetaEqn
223 (
224 fvm::ddt(zeta_)
225 + fvm::div(phi_, zeta_)
226 - fvm::laplacian(DzetaEff(), zeta_)
227 ==
228 (2.0*C1_ - 1)*G*zeta_/q_
229 - fvm::SuSp((2.0*C2_*f2() - dimensionedScalar(1.0))*zeta_/q_, zeta_)
230 + E
231 );
232
233 zetaEqn.ref().relax();
234 solve(zetaEqn);
235 boundZeta();
236
237
238 // q equation
239 tmp<fvScalarMatrix> qEqn
240 (
241 fvm::ddt(q_)
242 + fvm::div(phi_, q_)
243 - fvm::laplacian(DqEff(), q_)
244 ==
245 G - fvm::Sp(zeta_/q_, q_)
246 );
247
248 qEqn.ref().relax();
249 solve(qEqn);
250 bound(q_, qMin_);
251 boundZeta();
252
253 // Re-calculate k and epsilon
254 k_ = sqr(q_);
255 k_.correctBoundaryConditions();
256
257 epsilon_ = 2*q_*zeta_;
258 epsilon_.correctBoundaryConditions();
259
260 correctNut();
261}
