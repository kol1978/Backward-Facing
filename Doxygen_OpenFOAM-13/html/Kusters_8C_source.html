69void Foam::diameterModels::breakupModels::Kusters::setBreakupRate
70(
71 volScalarField::Internal& breakupRate,
72 const label i
73)
74{
75 const sizeGroup& fi = popBal_.sizeGroups()[i];
76
77 tmp<volScalarField> tdi = fi.d();
78 const volScalarField::Internal& di = tdi();
79
80 tmp<volScalarField> tepsilonc(popBal_.continuousTurbulence().epsilon());
81 const volScalarField::Internal& epsilonc = tepsilonc();
82 tmp<volScalarField> tnu(popBal_.continuousPhase().fluidThermo().nu());
83 const volScalarField::Internal nuc = tnu();
84
85 breakupRate =
86 sqrt(4*epsilonc/(15*pi*nuc))
87 *exp(- B_/(dP_*0.5*pow(pow(di/dP_, Df_)/kc_, 1/Df_))/epsilonc);
88}
