131//- Modify a fixed list for reordering a tri (does nothing)
132template<class Type, unsigned Size>
133inline FixedList<Type, 3> triReorder
134(
135 const FixedList<Type, Size>& x,
136 const FixedList<label, 3>& indices
137)
138{
139 FixedList<Type, 3> result;
140 for (unsigned i = 0; i < 3; ++ i)
141 {
142 result[i] = x[indices[i]];
143 }
144 return result;
145}

148//- Modify a list for cutting a tri from a tri
149template<class Type>
150inline FixedList<Type, 3> triCutTri
151(
152 const FixedList<Type, 3>& x,
153 const Pair<scalar>& f
154)
155{
156 FixedList<Type, 3> result;
157 result[0] = x[0];
158 for (label i = 0; i < 2; ++ i)
159 {
160 result[i+1] = (1 - f[i])*x[0] + f[i]*x[i+1];
161 }
162 return result;
163}

167template<class Type>
168inline FixedList<Type, 4> triCutQuad
169(
170 const FixedList<Type, 3>& x,
171 const Pair<scalar>& f
172)
173{
174 FixedList<Type, 4> result;
175 for (label i = 0; i < 2; ++ i)
176 {
177 result[i] = x[i+1];
178 result[3-i] = (1 - f[i])*x[0] + f[i]*x[i+1];
179 }
180 return result;

237//- Modify a list for cutting prism01 from a tet
238template<class Type>
239inline FixedList<Type, 6> tetCutPrism01
240(
241 const FixedList<Type, 4>& x,
242 const FixedList<scalar, 4>& f
243)
244{
245 FixedList<Type, 6> result;
246 for (label i = 0; i < 2; ++ i)
247 {
248 result[3*i] = x[i];
249 for (label j = 0; j < 2; ++ j)
250 {
251 result[3*i+j+1] = (1 - f[2*i+j])*x[i] + f[2*i+j]*x[j+2];
252 }
253 }
254
255 return result;
256}

275//- Cut a tri from a tri and apply an operation to the result. The cut is made
276// along the two edges connected to vertex 0, and the cut locations are given
277// as factors along these edges. The result is the side connected to vertex 0.
278template<class Op, class Point>
279inline typename Op::result triCutTri
280(
281 const Op& op,
282 const FixedList<Point, 3>& p,
283 const Pair<scalar>& f
284)

290//- Apply an operation to a quad. Splits the quad into two tris.
291template<class Op, class OpData, class Point>
292inline typename Op::result quadOp
293(
294 const OpData& opData,
295 const FixedList<Point, 4>& p
296)
297{
298 static const Pair<FixedList<label, 3>> i = {{0, 1, 2}, {0, 2, 3}};
299 return
300 Op(triReorder(opData, i[0]))(triReorder(p, i[0]))
301 + Op(triReorder(opData, i[1]))(triReorder(p, i[1]));
302}

335//- Apply an operation to a prism. Splits the prism into three tets.
336template<class Op, class OpData, class Point>
337inline typename Op::result prismOp
338(
339 const OpData& opData,
340 const FixedList<Point, 6>& p
341)
342{
343 static const FixedList<FixedList<label, 4>, 3> i =
344 {{0, 1, 2, 4}, {0, 2, 5, 4}, {0, 4, 5, 3}};
345 return
346 Op(tetReorder(opData, i[0]))(tetReorder(p, i[0]))
347 + Op(tetReorder(opData, i[1]))(tetReorder(p, i[1]))
348 + Op(tetReorder(opData, i[2]))(tetReorder(p, i[2]));
349}
