99template<class BasicThermophysicalTransportModel>
100tmp<surfaceScalarField> Fourier<BasicThermophysicalTransportModel>::q() const
101{
102 const thermoModel& thermo = this->thermo();
103
104 return surfaceScalarField::New
105 (
106 IOobject::groupName
107 (
108 "q",
109 this->momentumTransport().alphaRhoPhi().group()
110 ),
111 -fvc::interpolate(this->alpha()*thermo.kappa())*fvc::snGrad(thermo.T())
112 );
113}

116template<class BasicThermophysicalTransportModel>
117tmp<scalarField> Fourier<BasicThermophysicalTransportModel>::q
118(
119 const label patchi
120) const
121{
122 return
123 - (
124 this->alpha().boundaryField()[patchi]
125 *this->thermo().kappa().boundaryField()[patchi]
126 *this->thermo().T().boundaryField()[patchi].snGrad()
127 );
128}

131template<class BasicThermophysicalTransportModel>
132tmp<fvScalarMatrix>
133Fourier<BasicThermophysicalTransportModel>::divq(volScalarField& he) const
134{
135 const thermoModel& thermo = this->thermo();
136
137 // Return heat flux source as an implicit energy correction
138 // to the temperature gradient flux
139 return
140 -fvc::laplacian(this->alpha()*thermo.kappa(), thermo.T())
141 -fvm::laplacianCorrection
142 (
143 this->alpha()*thermo.kappa()/thermo.Cpv(),
144 he
145 );
146}
