163void Foam::solvers::incompressibleFluid::preSolve()
164{
165 if ((mesh.dynamic() || MRF.size()) && !Uf.valid())
166 {
167 Info<< "Constructing face momentum Uf" << endl;
168
169 // Ensure the U BCs are up-to-date before constructing Uf
170 U_.correctBoundaryConditions();
171
172 Uf = new surfaceVectorField
173 (
174 IOobject
175 (
176 "Uf",
177 runTime.name(),
178 mesh,
179 IOobject::READ_IF_PRESENT,
180 IOobject::AUTO_WRITE
181 ),
182 fvc::interpolate(U)
183 );
184 }
185
186 fvModels().preUpdateMesh();
187
188 if (transient())
189 {
190 correctCoNum();
191 }
192 else if (LTS)
193 {
194 setRDeltaT();
195 }
196
197 // Update the mesh for topology change, mesh to mesh mapping
198 mesh_.update();
199}
