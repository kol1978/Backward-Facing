30Foam::autoPtr<Foam::solidBodyMotionFunction> Foam::solidBodyMotionFunction::New
31(
32 const dictionary& SBMFCoeffs,
33 const Time& runTime
34)
35{
36 const word motionType(SBMFCoeffs.lookup("solidBodyMotionFunction"));
37
38 Info<< "Selecting solid-body motion function " << motionType << endl;
39
40 dictionaryConstructorTable::iterator cstrIter =
41 dictionaryConstructorTablePtr_->find(motionType);
42
43 if (cstrIter == dictionaryConstructorTablePtr_->end())
44 {
45 FatalIOErrorInFunction(SBMFCoeffs)
46 << "Unknown solidBodyMotionFunction type "
47 << motionType << nl << nl
48 << "Valid solidBodyMotionFunctions are : " << endl
49 << dictionaryConstructorTablePtr_->sortedToc()
50 << exit(FatalIOError);
51 }
52
53 return autoPtr<solidBodyMotionFunction>(cstrIter()(SBMFCoeffs, runTime));
