41Foam::IOobject Foam::zonesGenerator::io(const polyMesh& mesh) const
42{
43 typeIOobject<IOdictionary> result
44 (
45 typeName,
46 mesh.time().constant(),
47 mesh,
48 IOobject::MUST_READ,
49 IOobject::NO_WRITE
50 );
51
52 if (!result.headerOk())
53 {
54 result.readOpt() = IOobject::NO_READ;
55 }
56 else
57 {
58 result.readOpt() = IOobject::MUST_READ_IF_MODIFIED;
59 }
60
61 return result;
62}
