41void Foam::boxToFace::combine(topoSet& set, const bool add) const
42{
43 const pointField& ctrs = mesh_.faceCentres();
44
45 forAll(ctrs, facei)
46 {
47 forAll(bbs_, i)
48 {
49 if (bbs_[i].contains(ctrs[facei]))
50 {
51 addOrDelete(set, facei, add);
52 break;
53 }
54 }
55 }
56}

96void Foam::boxToFace::applyToSet
97(
98 const topoSetSource::setAction action,
99 topoSet& set
100) const
101{
102 if ((action == topoSetSource::NEW) || (action == topoSetSource::ADD))
103 {
104 Info<< " Adding faces with centre within boxes " << bbs_ << endl;
105
106 combine(set, true);
107 }
108 else if (action == topoSetSource::DELETE)
109 {
110 Info<< " Removing faces with centre within boxes " << bbs_ << endl;
111
112 combine(set, false);
113 }
