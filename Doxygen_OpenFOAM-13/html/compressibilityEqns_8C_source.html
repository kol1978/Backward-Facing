36Foam::PtrList<Foam::fvScalarMatrix>
37Foam::solvers::multiphaseEuler::compressibilityEqns
38(
39 const PtrList<volScalarField::Internal>& dmdts
40) const
41{
42 volScalarField& p_rgh = p_rgh_;
43
44 PtrList<fvScalarMatrix> pEqnComps(phases.size());
45
46 forAll(phases_, phasei)
47 {
48 phaseModel& phase = phases_[phasei];
49 const volScalarField& alpha = phase;
50 volScalarField& rho = phase.rho();
51
52 pEqnComps.set(phasei, new fvScalarMatrix(p_rgh, dimVolume/dimTime));
53 fvScalarMatrix& pEqnComp = pEqnComps[phasei];
54
55 // Density variation
56 if (!phase.isochoric() || !phase.pure())
57 {
58 pEqnComp +=
59 (
60 fvc::ddt(alpha, rho) + fvc::div(phase.alphaRhoPhi())
61 - fvc::Sp(fvc::ddt(alpha) + fvc::div(phase.alphaPhi()), rho)
62 )/rho;
63 }
64
65 // Mesh dilatation correction
66 if (mesh.moving())
67 {
68 pEqnComp += fvc::div(mesh.phi())*alpha;
69 }
70
71 // Compressibility
72 if (!phase.incompressible())
73 {
74 if (pimple.transonic())
75 {
76 const surfaceScalarField phid
77 (
78 IOobject::groupName("phid", phase.name()),
79 fvc::interpolate(phase.fluidThermo().psi())*phase.phi()
80 );
81
82 pEqnComp +=
83 correction
84 (
85 (alpha/rho)*
86 (
87 phase.fluidThermo().psi()*fvm::ddt(p_rgh)
88 + fvm::div(phid, p_rgh)
89 - fvm::Sp(fvc::div(phid), p_rgh)
90 )
91 );
92
93 pEqnComps[phasei].relax();
94 }
95 else
96 {
97 pEqnComp +=
98 (alpha*phase.fluidThermo().psi()/rho)
99 *correction(fvm::ddt(p_rgh));
100 }
101 }
102
103 // Option sources
104 if (fvModels().addsSupToField(rho.name()))
105 {
106 pEqnComp -= fvModels().sourceProxy(alpha, rho, p_rgh)/rho;
107 }
108
109 // Mass transfer
110 if (dmdts.set(phasei))
111 {
112 pEqnComp -= dmdts[phasei]/rho;
113 }
114 }
115
116 return pEqnComps;
117}
