56template<class Thermo>
57void Foam::tabulatedTransport<Thermo>::write(Ostream& os) const
58{
59 os << this->name() << endl;
60 os << token::BEGIN_BLOCK << incrIndent << nl;
61
62 Thermo::write(os);
63
64 dictionary dict("transport");
65
66 dictionary muDict("mu");
67 muDict.add("values", mu_.values());
68 dict.add("mu", muDict);
69
70 dictionary kappaDict("kappa");
71 kappaDict.add("values", kappa_.values());
72 dict.add("kappa", kappaDict);
73
74 os << indent << dict.dictName() << dict;
75
76 os << decrIndent << token::END_BLOCK << nl;
77}
