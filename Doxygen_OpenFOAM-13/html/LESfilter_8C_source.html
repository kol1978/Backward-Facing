39Foam::autoPtr<Foam::LESfilter> Foam::LESfilter::New
40(
41 const fvMesh& mesh,
42 const dictionary& dict,
43 const word& filterDictName
44)
45{
46 const word filterType(dict.lookup(filterDictName));
47
48 dictionaryConstructorTable::iterator cstrIter =
49 dictionaryConstructorTablePtr_->find(filterType);
50
51 if (cstrIter == dictionaryConstructorTablePtr_->end())
52 {
53 FatalErrorInFunction
54 << "Unknown LESfilter type "
55 << filterType << nl << nl
56 << "Valid LESfilter types are :" << endl
57 << dictionaryConstructorTablePtr_->sortedToc()
58 << exit(FatalError);
59 }
60
61 Info << incrIndent;
62 autoPtr<LESfilter> modelPtr(cstrIter()(mesh, dict));
63 Info << decrIndent;
64 return modelPtr;
65}
