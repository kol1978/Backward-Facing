34template<class T>
35void Foam::UList<T>::deepCopy(const UList<T>& a)
36{
37 if (a.size_ != this->size_)
38 {
39 FatalErrorInFunction
40 << "ULists have different sizes: "
41 << this->size_ << " " << a.size_
42 << abort(FatalError);
43 }
44
45 if (this->size_)
46 {
47 #ifdef USEMEMCPY
48 if (contiguous<T>())
49 {
50 memcpy(this->v_, a.v_, this->byteSize());
51 }
52 else
53 #endif
54 {
55 List_ACCESS(T, (*this), vp);
56 List_CONST_ACCESS(T, a, ap);
57 List_FOR_ALL((*this), i)
58 List_ELEM((*this), vp, i) = List_ELEM(a, ap, i);
59 List_END_FOR_ALL
60 }
61 }
62}

99template<class T>
100std::streamsize Foam::UList<T>::byteSize() const
101{
102 if (!contiguous<T>())
103 {
104 FatalErrorInFunction
105 << "Cannot return the binary size of a list of "
106 "non-primitive elements"
107 << abort(FatalError);
108 }
109
110 return this->size_*sizeof(T);
111}

144template<class T>
145bool Foam::UList<T>::operator==(const UList<T>& a) const
146{
147 if (this->size_ != a.size_)
148 {
149 return false;
150 }
151
152 bool equal = true;
153
154 List_CONST_ACCESS(T, (*this), vp);
155 List_CONST_ACCESS(T, (a), ap);
156
157 List_FOR_ALL((*this), i)
158 equal = equal && (List_ELEM((*this), vp, i) == List_ELEM((a), ap, i));
159 List_END_FOR_ALL
160
161 return equal;
162}

172template<class T>
173bool Foam::UList<T>::operator<(const UList<T>& a) const
174{
175 for
176 (
177 const_iterator vi = begin(), ai = a.begin();
178 vi < end() && ai < a.end();
179 vi++, ai++
180 )
181 {
182 if (*vi < *ai)
183 {
184 return true;
185 }
186 else if (*vi > *ai)
187 {
188 return false;
189 }
190 }
191
192 if (this->size_ < a.size_)
193 {
194 return true;
195 }
196 else
197 {
198 return false;
199 }
200}
