32void Foam::solvers::incompressibleFluid::setRDeltaT()
33{
34 volScalarField& rDeltaT = trDeltaT.ref();
35
36 const dictionary& pimpleDict = pimple.dict();
37
38 const scalar maxCo
39 (
40 pimpleDict.lookupOrDefault<scalar>("maxCo", 0.8)
41 );
42
43 const scalar rDeltaTSmoothingCoeff
44 (
45 pimpleDict.lookupOrDefault<scalar>("rDeltaTSmoothingCoeff", 0.02)
46 );
47
48 const scalar rDeltaTDampingCoeff
49 (
50 pimpleDict.lookupOrDefault<scalar>("rDeltaTDampingCoeff", 1.0)
51 );
52
53 const volScalarField rDeltaT0("rDeltaT0", rDeltaT);
54
55 // Set the reciprocal time-step from the local Courant number
56 rDeltaT.internalFieldRef() = fvc::surfaceSum(mag(phi))/((2*maxCo)*mesh.V());
57
58 // Clip to user-defined maximum and minimum time-steps
59 scalar minRDeltaT = gMin(rDeltaT.primitiveField());
60 scalar maxRDeltaT = gMax(rDeltaT.primitiveField());
61 if (pimpleDict.found("maxDeltaT") || minRDeltaT < rootVSmall)
62 {
63 const scalar clipRDeltaT = 1/pimpleDict.lookup<scalar>("maxDeltaT");
64 rDeltaT.max(clipRDeltaT);
65 minRDeltaT = max(minRDeltaT, clipRDeltaT);
66 maxRDeltaT = max(maxRDeltaT, clipRDeltaT);
67 }
68 if (pimpleDict.found("minDeltaT") || maxRDeltaT > rootVGreat)
69 {
70 const scalar clipRDeltaT = 1/pimpleDict.lookup<scalar>("minDeltaT");
71 rDeltaT.min(clipRDeltaT);
72 minRDeltaT = min(minRDeltaT, clipRDeltaT);
73 maxRDeltaT = min(maxRDeltaT, clipRDeltaT);
74 }
75
76 Info<< "Flow time scale min/max = "
77 << 1/maxRDeltaT << ", " << 1/minRDeltaT << endl;
78
79 // Update the boundary values of the reciprocal time-step
80 rDeltaT.correctBoundaryConditions();
81
82 if (rDeltaTSmoothingCoeff < 1.0)
83 {
84 fvc::smooth(rDeltaT, rDeltaTSmoothingCoeff);
85 }
86
87 Info<< "Smoothed flow time scale min/max = "
88 << gMin(1/rDeltaT.primitiveField())
89 << ", " << gMax(1/rDeltaT.primitiveField()) << endl;
90
91 // Limit rate of change of time scale
92 // - reduce as much as required
93 // - only increase at a fraction of old time scale
94 if
95 (
96 rDeltaTDampingCoeff < 1.0
97 && runTime.timeIndex() > runTime.startTimeIndex() + 1
98 )
99 {
100 rDeltaT =
101 rDeltaT0
102 *max(rDeltaT/rDeltaT0, scalar(1) - rDeltaTDampingCoeff);
103
104 Info<< "Damped flow time scale min/max = "
105 << gMin(1/rDeltaT.primitiveField())
106 << ", " << gMax(1/rDeltaT.primitiveField()) << endl;
107 }
108}
