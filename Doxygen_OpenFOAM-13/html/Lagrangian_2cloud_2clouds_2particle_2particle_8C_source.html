68Foam::tmp<Foam::LagrangianSubVectorField> Foam::clouds::particle::dUdt
69(
70 const LagrangianSubMesh& subMesh
71) const
72{
73 const LagrangianSubScalarSubField& m = this->m.ref(subMesh);
74 const LagrangianSubVectorSubField& U = this->U.ref(subMesh);
75
76 return
77 LagrangianModels().addsSupToField(m)
78 ? (Lagrangianc::Ddt(m, U) - Lagrangianc::Ddt(m)*U)/m
79 : Lagrangianc::Ddt(U);
80}

83bool Foam::clouds::particle::reCalculateModified()
84{
85 const bool dUdt = tracking == trackingType::parabolic;
86
87 const LagrangianSubMesh subMesh = this->mesh().subNone();
88
89 LagrangianSubScalarSubField& m = this->m.ref(subMesh);
90 LagrangianSubVectorSubField& U = this->U.ref(subMesh);
91
92 bool result = false;
93
94 if (LagrangianModels().addsSupToField(m))
95 {
96 result = Lagrangianm::initDdt(dimless, m, dUdt) || result;
97
98 if (context != contextType::functionObject)
99 {
100 result = Lagrangianm::initDdt(dimVolume, rhoc(subMesh)) || result;
101 }
102 }
103
104 result = Lagrangianm::initDdt(dimMass, U, dUdt) || result;
105
106 if (context != contextType::functionObject)
107 {
108 result = Lagrangianm::initDdt(dimMass, Uc(subMesh)) || result;
109 }
110
111 return result;
112}

115void Foam::clouds::particle::calculate
116(
117 const LagrangianSubScalarField& deltaT,
118 const bool final
119)
120{
121 const LagrangianSubMesh& subMesh = deltaT.mesh();
122
123 LagrangianSubScalarSubField& m = this->m.ref(subMesh);
124 const LagrangianSubScalarSubField rho(subMesh.sub(this->rho));
125 LagrangianSubVectorSubField& U = this->U.ref(subMesh);
126
127 // Solve the mass equation if a model provides a mass source
128 if (LagrangianModels().addsSupToField(m))
129 {
130 LagrangianEqn<scalar> mEqn
131 (
132 Lagrangianm::Ddt(deltaT, m)
133 ==
134 LagrangianModels().source(deltaT, m)
135 );
136
137 mEqn.solve(final);
138
139 // Correct the diameter, assuming the density remains constant
140 spherical::correct(toSubField(m/rho));
141
142 if (context != contextType::functionObject && final)
143 {
144 LagrangianEqn<scalar> mcEqn
145 (
146 Lagrangianm::noDdt(deltaT, dimVolume, rhoc(subMesh))
147 ==
148 LagrangianModels().sourceProxy(deltaT, m, rhoc(subMesh))
149 );
150
151 carrierEqn(rhoc) += mcEqn;
152 }
153 }
154
155 // Solve the momentum equation
156 {
157 LagrangianEqn<vector> UEqn
158 (
159 Lagrangianm::Ddt(deltaT, m, U)
160 ==
161 LagrangianModels().source(deltaT, m, U)
162 );
163
164 UEqn.solve(final);
165
166 if (context != contextType::functionObject && final)
167 {
168 LagrangianEqn<vector> UcEqn
169 (
170 Lagrangianm::noDdt(deltaT, dimMass, Uc(subMesh))
171 ==
172 LagrangianModels().sourceProxy(deltaT, m, U, Uc(subMesh))
173 );
174
175 carrierEqn(Uc) += UcEqn;
176 }
177 }
178}
