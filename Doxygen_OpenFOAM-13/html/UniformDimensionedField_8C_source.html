30template<class Type>
31Foam::UniformDimensionedField<Type>::UniformDimensionedField
32(
33 const IOobject& io,
34 const dimensioned<Type>& dt
35)
36:
37 regIOobject(io),
38 dimensioned<Type>(dt),
39 OldTimeField<UniformDimensionedField>(this->time().timeIndex())
40{
41 // Read value
42 if
43 (
44 (
45 io.readOpt() == IOobject::MUST_READ
46 || io.readOpt() == IOobject::MUST_READ_IF_MODIFIED
47 )
48 || (io.readOpt() == IOobject::READ_IF_PRESENT && headerOk())
49 )
50 {
51 dictionary dict(readStream(typeName));
52
53 this->dimensions().read(dict.lookup("dimensions"));
54
55 this->value() = dict.lookup<Type>("value", this->dimensions());
56 }
57}

124template<class Type>
125bool Foam::UniformDimensionedField<Type>::writeData(Ostream& os) const
126{
127 writeKeyword(os, "dimensions");
128 this->dimensions().write(os) << token::END_STATEMENT << nl;
129 writeEntry(os, "value", this->value());
130 os << nl;
131
132 return (os.good());
133}
