32Foam::NamedEnum<Enum, nEnum>::NamedEnum(const FixedList<word, nEnum>& names)

53template<class Enum, unsigned int nEnum>
54Enum Foam::NamedEnum<Enum, nEnum>::read(Istream& is) const
55{
56 const word name(is);
57
58 HashTable<unsigned int>::const_iterator iter = table_.find(name);
59
60 if (iter == HashTable<unsigned int>::end())
61 {
62 FatalIOErrorInFunction(is)
63 << name << " is not in enumeration: "
64 << *this << exit(FatalIOError);
65 }
66
67 return Enum(iter());
68}

85template<class Enum, unsigned int nEnum>
86Enum Foam::NamedEnum<Enum, nEnum>::select(const dictionary& dict) const
87{
88 const FixedList<word, nEnum>& names = *this;
89
90 Enum selection = Enum(0);
91 unsigned int nSelections = 0;
92 forAll(names, ei)
93 {
94 if (dict.found(names[ei]))
95 {
96 selection = Enum(ei);
97 nSelections++;
98 }
99 }
100
101 if (nSelections == 0)
102 {
103 FatalIOErrorInFunction(dict)
104 << "None of the options selected, please specify one of: "
105 << names << exit(FatalIOError);
106 }
107 else if (nSelections > 1)
108 {
109 FatalIOErrorInFunction(dict)
110 << "More than one option selected, please specify one of: "
111 << names << exit(FatalIOError);
112 }
113
114 return selection;
115}
