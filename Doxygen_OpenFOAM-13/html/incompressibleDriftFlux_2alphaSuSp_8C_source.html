48void Foam::solvers::incompressibleDriftFlux::alphaSuSp
49(
50 tmp<volScalarField::Internal>& tSu,
51 tmp<volScalarField::Internal>& tSp
52)
53{
54 if (!divergent()) return;
55
56 const dimensionedScalar Szero(dimless/dimTime, 0);
57
58 tSp = volScalarField::Internal::New("Sp", mesh, Szero);
59 tSu = volScalarField::Internal::New("Su", mesh, Szero);
60
61 volScalarField::Internal& Sp = tSp.ref();
62 volScalarField::Internal& Su = tSu.ref();
63
64 if (fvModels().addsSupToField(alpha1.name()))
65 {
66 const fvScalarMatrix alpha1Sup(fvModels().source(alpha1));
67
68 Su += alpha2()*alpha1Sup.Su();
69 Sp += alpha2()*alpha1Sup.Sp();
70 }
71
72 if (fvModels().addsSupToField(alpha2.name()))
73 {
74 const fvScalarMatrix alpha2Sup(fvModels().source(alpha2));
75
76 Su -= alpha1()*(alpha2Sup.Su() + alpha2Sup.Sp());
77 Sp += alpha1()*alpha2Sup.Sp();
78 }
79}
