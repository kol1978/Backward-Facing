49void Foam::solidBodyMotionSolver::updateSetPointIndices()
50{
51 if (zone_.all())
52 {
53 setPointIndices_.clear();
54 return;
55 }
56
57 boolList pointInSet(mesh().nPoints(), false);
58
59 forAll(zone_.zone(), setCelli)
60 {
61 const cell& c = mesh().cells()[zone_.zone()[setCelli]];
62 forAll(c, cfi)
63 {
64 const face& f = mesh().faces()[c[cfi]];
65 forAll(f, fpi)
66 {
67 pointInSet[f[fpi]] = true;
68 }
69 }
70 }
71
72 syncTools::syncPointList(mesh(), pointInSet, orEqOp<bool>(), false);
73
74 setPointIndices_.resize(count(pointInSet, true));
75
76 label setPointi = 0;
77 forAll(pointInSet, pointi)
78 {
79 if (pointInSet[pointi])
80 {
81 setPointIndices_[setPointi ++] = pointi;
82 }
83 }
84}

119Foam::tmp<Foam::pointField> Foam::solidBodyMotionSolver::curPoints() const
120{
121 transform_ = SBMFPtr_().transformation();
122
123 if (zone_.all())
124 {
125 return transformPoints(transform_, points0_);
126 }
127 else
128 {
129 tmp<pointField> ttransformedPts(new pointField(mesh().points()));
130 pointField& transformedPts = ttransformedPts.ref();
131
132 UIndirectList<point>(transformedPts, setPointIndices_) = transformPoints
133 (
134 transform_,
135 pointField(points0_, setPointIndices_)
136 );
137
138 return ttransformedPts;
139 }
140}

143void Foam::solidBodyMotionSolver::topoChange(const polyTopoChangeMap& map)
144{
145 zone_.topoChange(map);
146 updateSetPointIndices();
147
148 boolList pointInSet(mesh().nPoints(), zone_.all());
149 UIndirectList<bool>(pointInSet, setPointIndices_) = true;
150
151 // pointMesh already updates pointFields
152
153 // Get the new points either from the map or the mesh
154 const pointField& points = mesh().points();
155
156 pointField newPoints0(map.pointMap().size());
157
158 forAll(newPoints0, pointi)
159 {
160 const label oldPointi = map.pointMap()[pointi];
161
162 if (oldPointi < 0)
163 {
164 FatalErrorInFunction
165 << "Cannot determine co-ordinates of introduced vertices."
166 << " New vertex " << pointi << " at co-ordinate "
167 << points[pointi] << exit(FatalError);
168 }
169
170 if (map.reversePointMap()[oldPointi] == pointi)
171 {
172 newPoints0[pointi] = points0_[oldPointi];
173 }
174 else
175 {
176 newPoints0[pointi] =
177 pointInSet[pointi]
178 ? transform_.invTransformPoint(points[pointi])
179 : points[pointi];
180 }
181 }
182
183 twoDCorrectPoints(newPoints0);
184
185 // Move into base class storage and mark as to-be-written
186 points0_.transfer(newPoints0);
187 points0_.writeOpt() = IOobject::AUTO_WRITE;
188 points0_.instance() = mesh().time().name();
189}
