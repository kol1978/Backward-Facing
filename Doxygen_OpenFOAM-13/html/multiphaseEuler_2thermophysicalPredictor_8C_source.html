36void Foam::solvers::multiphaseEuler::compositionPredictor()
37{
38 autoPtr<HashPtrTable<fvScalarMatrix>> popBalSpecieTransferPtr =
39 populationBalanceSystem_.specieTransfer();
40 HashPtrTable<fvScalarMatrix>& popBalSpecieTransfer =
41 popBalSpecieTransferPtr();
42
43 fluid_.correctReactions();
44
45 forAll(fluid.multicomponentPhases(), multicomponentPhasei)
46 {
47 phaseModel& phase = fluid_.multicomponentPhases()[multicomponentPhasei];
48
49 UPtrList<volScalarField>& Y = phase.YRef();
50 const volScalarField& alpha = phase;
51 const volScalarField& rho = phase.rho();
52
53 forAll(Y, i)
54 {
55 if (phase.solveSpecie(i))
56 {
57 fvScalarMatrix YiEqn
58 (
59 phase.YiEqn(Y[i])
60 ==
61 *popBalSpecieTransfer[Y[i].name()]
62 + fvModels().source(alpha, rho, Y[i])
63 );
64
65 YiEqn.relax();
66
67 fvConstraints().constrain(YiEqn);
68
69 YiEqn.solve("Yi");
70
71 fvConstraints().constrain(Y[i]);
72 }
73 else
74 {
75 Y[i].correctBoundaryConditions();
76 }
77 }
78 }
79
80 fluid_.correctSpecies();
81}

84void Foam::solvers::multiphaseEuler::energyPredictor()
85{
86 autoPtr<HashPtrTable<fvScalarMatrix>> heatTransferPtr =
87 heatTransferSystem_.heatTransfer();
88 HashPtrTable<fvScalarMatrix>& heatTransfer =
89 heatTransferPtr();
90
91 autoPtr<HashPtrTable<fvScalarMatrix>> popBalHeatTransferPtr =
92 populationBalanceSystem_.heatTransfer();
93 HashPtrTable<fvScalarMatrix>& popBalHeatTransfer =
94 popBalHeatTransferPtr();
95
96 forAll(fluid.thermalPhases(), thermalPhasei)
97 {
98 phaseModel& phase = fluid_.thermalPhases()[thermalPhasei];
99
100 const volScalarField& alpha = phase;
101 const volScalarField& rho = phase.rho();
102
103 fvScalarMatrix EEqn
104 (
105 phase.heEqn()
106 ==
107 *heatTransfer[phase.name()]
108 + *popBalHeatTransfer[phase.name()]
109 + fvModels().source(alpha, rho, phase.thermo().he())
110 );
111
112 EEqn.relax();
113 fvConstraints().constrain(EEqn);
114 EEqn.solve();
115 fvConstraints().constrain(phase.thermo().he());
116 }
117
118 fluid_.correctThermo();
119 fluid_.correctContinuityError(populationBalanceSystem_.dmdts());
120}
