69bool Foam::functionObjects::uniform::read(const dictionary& dict)
70{
71 fvMeshFunctionObject::read(dict);
72
73 fieldType_ = dict.lookup<word>("fieldType");
74 fieldName_ = dict.lookupBackwardsCompatible<word>({"field", "name"});
75 dimensions_.reset(dict.lookup<dimensionSet>("dimensions"));
76
77 bool ok = false;
78 #define readValueType(Type, GeoField) \
79 if (GeoField<Type>::typeName == fieldType_) \
80 { \
81 ok = true; \
82 Type##Value_ = dict.lookup<Type>("value"); \
83 }
84 FOR_ALL_FIELD_TYPES(readValueType, VolField);
85 FOR_ALL_FIELD_TYPES(readValueType, VolInternalField);
86 FOR_ALL_FIELD_TYPES(readValueType, SurfaceField);
87 #undef readValueType
88
89 if (!ok)
90 {
91 FatalErrorInFunction
92 << "Field type " << fieldType_ << " not recognised" << endl << endl;
93
94 DynamicList<word> fieldTypes;
95 #define getFieldType(Type, GeoField) \
96 fieldTypes.append(GeoField<Type>::typeName);
97 FOR_ALL_FIELD_TYPES(getFieldType, VolField);
98 FOR_ALL_FIELD_TYPES(getFieldType, VolInternalField);
99 FOR_ALL_FIELD_TYPES(getFieldType, SurfaceField);
100 #undef getFieldType
101
102 FatalErrorInFunction
103 << "Available field types are : " << endl
104 << fieldTypes
105 << exit(FatalError);
106 }
107
108 return true;
109}

112bool Foam::functionObjects::uniform::execute()
113{
114 #define calcType(Type, GeoField) \
115 if (GeoField<Type>::typeName == fieldType_) \
116 { \
117 store \
118 ( \
119 fieldName_, \
120 GeoField<Type>::New \
121 ( \
122 fieldName_, \
123 mesh_, \
124 dimensioned<Type>(dimensions_, Type##Value_) \
125 ) \
126 ); \
127 }
128 FOR_ALL_FIELD_TYPES(calcType, VolField);
129 FOR_ALL_FIELD_TYPES(calcType, VolInternalField);
130 FOR_ALL_FIELD_TYPES(calcType, SurfaceField);
131 #undef calcType
132
133 return true;
134}
