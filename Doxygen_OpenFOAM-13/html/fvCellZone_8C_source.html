31void Foam::fvCellZone::update()
32{
33 const labelList& cells(this->zone());
34
35 nGlobalCells_ = cells.size();
36 reduce(nGlobalCells_, sumOp<label>());
37
38 V_ = 0;
39 forAll(cells, i)
40 {
41 V_ += mesh_.V()[cells[i]];
42 }
43 reduce(V_, sumOp<scalar>());
44}
